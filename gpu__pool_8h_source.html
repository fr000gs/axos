3// GPUMemoryPool: per-thread caching allocator for device memory, plus the
4// per-thread cuSPARSE / cuDSS / cuBLAS / cuSOLVER handles
5// (docs/TENSOR_SPEC.md §9). Obtain it with GPUMemoryPool::get().
6//
7// allocate(bytes) device memory, 256-byte aligned (512-byte granules);
8// nullptr for 0 bytes. A freed block is reused for a
9// later request that fits (smallest first) as long as
10// it is at most twice the request, so a loop that
11// allocates the same sizes calls cudaMalloc only in its
12// first iteration. On out-of-memory the cached blocks
13// go back to the driver and the allocation is retried
14// once; after that it throws std::runtime_error with
15// the requested size and the MB the pool held.
16// deallocate(p, n) returns a block to the cache (the pool tracks block
17// sizes itself, so n is informational). A pointer the
18// pool did not hand out (adopted memory, or a block
19// from another thread's pool) is freed directly.
20// clear() returns all cached blocks to the driver.
21// get_cusparse() ... lazily created, one per thread, reused for the
22// thread's lifetime; creation failure throws.
23//
24// The allocation mode is fixed at compile time by AXOS_GPU_ALLOC_MODE:
25// 0 = Pool (default, described above), 1 = Raw (cudaMalloc/cudaFree per
26// call), 2 = Async (cudaMallocAsync/cudaFreeAsync on the default stream).
27// A thread's pool and handles are released when the thread exits; objects
28// destroyed after that (static storage) free their memory directly.

81 struct Stats {

84 size_t reuses = 0; // requests served from the cache
85 size_t bytes_in_use = 0; // handed out, not yet returned

92 {

94 if (s.pool) return *s.pool;
95 s.pool = new GPUMemoryPool();

106 void *
107 allocate(size_t bytes)
108 {
109 if (bytes == 0) return nullptr;
110 const size_t need = (bytes + kGranule - 1) / kGranule * kGranule;
111 if (mode == Mode::Pool) {
112 auto it = free_.lower_bound(need);
113 if (it != free_.end() && it->first <= 2 * need) {
114 const size_t sz = it->first;
115 void *p = it->second;
116 free_.erase(it);
117 st_.bytes_cached -= sz;
118 track(p, sz);
119 ++st_.reuses;
120 return p;
121 }
122 }
123 void *p = device_malloc(need);
124 track(p, need);
125 return p;
126 }
127
128 void
129 deallocate(void *p, size_t bytes) noexcept
130 {
131 (void)bytes;
132 if (!p) return;
133 auto it = used_.find(p);
134 if (it == used_.end()) {
135 device_free(p);
136 return;
137 }
138 const size_t sz = it->second;
139 used_.erase(it);
140 st_.bytes_in_use -= sz;
141 if (mode == Mode::Pool) {
142 try {
143 free_.emplace(sz, p);
144 st_.bytes_cached += sz;
145 return;
146 } catch (...) {
147 }
148 }
149 device_free(p);
150 }
151
152 void
153 clear() noexcept
154 {
155 for (auto &kv : free_)
156 device_free(kv.second);
157 free_.clear();
158 st_.bytes_cached = 0;
159 }

162 stats() const

240 void
241 track(void *p, size_t sz)
242 {
243 auto r = used_.emplace(p, sz);
244 if (!r.second) { // stale entry: freed directly by another thread
245 st_.bytes_in_use -= r.first->second;
246 r.first->second = sz;
247 }
248 st_.bytes_in_use += sz;
249 }
250
251 bool
252 try_malloc(void **p, size_t need)
253 {
254 if (limit_ && st_.bytes_in_use + st_.bytes_cached + need > limit_)
255 return false;
256 const cudaError_t e = mode == Mode::Async ? cudaMallocAsync(p, need, 0)
257 : cudaMalloc(p, need);
258 if (e == cudaSuccess) {
259 ++st_.device_mallocs;
260 return true;
261 }
262 if (e == cudaErrorMemoryAllocation) {
263 (void)cudaGetLastError(); // reset the error state
264 return false;
265 }
266 throw std::runtime_error(
267 std::string("GPUMemoryPool: cudaMalloc failed: ") +
268 cudaGetErrorString(e));
269 }
270
271 void *
272 device_malloc(size_t need)
273 {
274 void *p = nullptr;
275 if (try_malloc(&p, need)) return p;
276 const size_t held = st_.bytes_in_use + st_.bytes_cached;
277 clear();
278 ++st_.oom_recoveries;
279 if (try_malloc(&p, need)) return p;
280 char msg[192];
281 std::snprintf(msg, sizeof msg,
282 "GPUMemoryPool: out of device memory allocating %.1f MB "
283 "(the pool held %.1f MB, %.1f MB still in use)",
284 need / 1048576.0, held / 1048576.0, st_.bytes_in_use / 1048576.0);
285 throw std::runtime_error(msg);
286 }
287
288 void
289 device_free(void *p) noexcept
290 {
291 if (mode == Mode::Async)
292 (void)cudaFreeAsync(p, 0);
293 else
294 (void)cudaFree(p);
295 ++st_.device_frees;
296 }
297
298 std::multimap<size_t, void *> free_; // cached blocks by size
299 std::unordered_map<void *, size_t> used_; // live blocks -> size
300 Stats st_;
301 size_t limit_ = 0;
302#if defined(CUSPARSE_WITH)
303 cusparseHandle_t cusparse_ = nullptr;
304#endif
305#if defined(CUDSS_WITH)
306 cudssHandle_t cudss_ = nullptr;
307#endif
308#if defined(CUBLAS_WITH)
309 cublasHandle_t cublas_ = nullptr;
310#endif
311#if defined(CUSOLVER_WITH)
312 cusolverDnHandle_t cusolver_ = nullptr;
313#endif
314};
