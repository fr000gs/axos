3// Dense linear algebra on host tensors (docs/TENSOR_SPEC.md §7, §15.4-15.5).
4// Matrices are tensorET<2, T> and vectors tensorET<1, T>, T = float or
5// double. Errors (singular pivot, shape mismatch) throw std::runtime_error.
6//
7// norm(A, B, p) / norm(A, B, "fro") norms of A - B over all elements
8// conjugateGradient(A, b, x0, maxIter, tol)
9// gaussJordanElimination(A) reduced row-echelon form
10// augmentMatrix(A, b) [A | b]
11// gramSchmidtOrthogonalization(V) orthonormal basis of V's columns
12// luDcmp(A, tol) {L, U}, A = L U (no pivoting)
13// luDcmpPivoted(A, tol) {{L, U}, P}, P A = L U
14// luSolve(L, U, P, B) solves A X = B (B vector or matrix)
15// inverse(A) general inverse (LU + solves)
16// qrDecompositionTile(A) A <- Q (m x n), returns R (n x n)
17// lanczos(A, m, q0) {alpha, beta, Q}
18// expm(A) matrix exponential
19// lobpcg(A, nev, X0, eigvals, X, ...) smallest eigenpairs (symmetric A)
20// symmetricEigen(A, w, V) all eigenpairs (Jacobi, small A)
21// inverse_backs(U, m) inverse of an upper-triangular U
22// revEl / elimStep / matMul(history, ..) tournament elimination with a
23// recorded, replayable history
24//
25// How the speed is obtained (§15): LU is blocked and right-looking (panel on
26// a column-major copy, then a GEMM trailing update that holds > 90% of the
27// flops); triangular solves with several right-hand sides are blocked with
28// GEMM updates, single right-hand sides with GEMV updates; QR is blocked
29// Householder in compact-WY form (GEMM updates); CG preallocates its vectors
30// and fuses A p with p^T A p and the three vector updates with r^T r; the
31// triangular inverse inverts diagonal blocks in parallel and joins them
32// pairwise with GEMMs; history replay is one parallel pass over independent
33// columns (row operations) or rows (column operations).

109// y += a x, serial (for use inside parallel code).
110template <class T>
111AXOS_INLINE void
112axpy_s(size_t n, T a, const T *AXOS_RESTRICT x, T *AXOS_RESTRICT y)
113{
114 using P = simd::Pack<T>;
115 constexpr size_t W = P::W;
116 const auto va = P::set1(a);
117 size_t i = 0;
118 for (; i + W <= n; i += W)
119 P::storeu(y + i, P::fmadd(va, P::loadu(x + i), P::loadu(y + i)));
120 for (; i < n; ++i)
121 y[i] += a * x[i];
122}

124// sum x[i] y[i], serial.
125template <class T>
126AXOS_INLINE T
127dot_s(size_t n, const T *x, const T *y)
128{
129 using P = simd::Pack<T>;
130 constexpr size_t W = P::W;
131 auto c0 = P::zero(), c1 = P::zero();
132 size_t i = 0;
133 for (; i + 2 * W <= n; i += 2 * W) {
134 c0 = P::fmadd(P::loadu(x + i), P::loadu(y + i), c0);
135 c1 = P::fmadd(P::loadu(x + i + W), P::loadu(y + i + W), c1);
136 }
137 T s = P::hsum(P::add(c0, c1));
138 for (; i < n; ++i)
139 s += x[i] * y[i];
140 return s;
141}

145// Solves M X = B in place (B is n x k, row-major, ldb), M lower or upper
146// triangular (n x n, ldm), with unit or general diagonal. Blocked: each
147// diagonal block is solved directly, the rest of B updated with one GEMM (GEMV
148// for k == 1).
149template <class T>
150void
151trsm_left(bool lower, bool unit, size_t n, size_t k, const T *M, size_t ldm,
152 T *B, size_t ldb)
153{
154 if (n == 0 || k == 0) return;
155 constexpr size_t nb = 64;
156 auto solve_block = [&](size_t i0, size_t i1) {
157 if (k == 1) {
158 if (lower) {
159 for (size_t i = i0; i < i1; ++i) {
160 const T *mi = M + i * ldm;
161 T s = B[i * ldb] - dot_s(i - i0, mi + i0, B + i0 * ldb);
162 B[i * ldb] = unit ? s : s / mi[i];
163 }
164 } else {
165 for (size_t i = i1; i-- > i0;) {
166 const T *mi = M + i * ldm;
167 T s = B[i * ldb] -
168 dot_s(i1 - i - 1, mi + i + 1, B + (i + 1) * ldb);
169 B[i * ldb] = unit ? s : s / mi[i];
170 }
171 }
172 return;
173 }
174 // about 64K multiply-adds per chunk: (i1 - i0)^2 / 2 per column
175 const size_t per_col = (i1 - i0) * (i1 - i0) / 2 + 1;
176 detail::parallel_for(
177 k, std::max<size_t>(16, 65536 / per_col),
178 [&](size_t c0, size_t c1) {
179 const size_t w = c1 - c0;
180 if (lower) {
181 for (size_t i = i0; i < i1; ++i) {
182 T *bi = B + i * ldb + c0;
183 const T *mi = M + i * ldm;
184 for (size_t j = i0; j < i; ++j)
185 if (mi[j] != T(0))
186 axpy_s(w, -mi[j], B + j * ldb + c0, bi);
187 if (!unit) {
188 const T inv = T(1) / mi[i];
189 for (size_t c = 0; c < w; ++c)
190 bi[c] *= inv;
191 }
192 }
193 } else {
194 for (size_t i = i1; i-- > i0;) {
195 T *bi = B + i * ldb + c0;
196 const T *mi = M + i * ldm;
197 for (size_t j = i + 1; j < i1; ++j)
198 if (mi[j] != T(0))
199 axpy_s(w, -mi[j], B + j * ldb + c0, bi);
200 if (!unit) {
201 const T inv = T(1) / mi[i];
202 for (size_t c = 0; c < w; ++c)
203 bi[c] *= inv;
204 }
205 }
206 }
207 },
208 16);
209 };
210 auto update = [&](size_t r0, size_t r1, size_t i0, size_t i1) {
211 // B[r0:r1] -= M[r0:r1, i0:i1] B[i0:i1]
212 if (r1 <= r0) return;
213 if (k == 1)
214 kernels::gemv<T>(false, r1 - r0, i1 - i0, T(-1), M + r0 * ldm + i0,
215 ldm, B + i0 * ldb, T(1), B + r0 * ldb);
216 else
217 kernels::gemm<T>(false, false, r1 - r0, k, i1 - i0, T(-1),
218 M + r0 * ldm + i0, ldm, B + i0 * ldb, ldb, T(1), B + r0 * ldb,
219 ldb);
220 };
221 if (k == 1 && ldb != 1) { // gather to a contiguous vector for GEMV
222 std::vector<T> v(n);
223 for (size_t i = 0; i < n; ++i)
224 v[i] = B[i * ldb];
225 trsm_left(lower, unit, n, 1, M, ldm, v.data(), 1);
226 for (size_t i = 0; i < n; ++i)
227 B[i * ldb] = v[i];
228 return;
229 }
230 if (k == 1) {
231 // One right-hand side: left-looking, so the GEMV for each block of rows
232 // reads long contiguous row segments (x[i0:i1] -= M[i0:i1, 0:i0]
233 // x[0:i0] for lower, M[i0:i1, i1:n] x[i1:n] for upper), which streams
234 // at memory bandwidth; the right-looking form reads narrow strips.
235 constexpr size_t rb = 64;
236 const size_t nblk = (n + rb - 1) / rb;
237 for (size_t s = 0; s < nblk; ++s) {
238 const size_t bk = lower ? s : nblk - 1 - s;
239 const size_t i0 = bk * rb, i1 = std::min(n, i0 + rb);
240 if (lower && i0 > 0)
241 kernels::gemv<T>(false, i1 - i0, i0, T(-1), M + i0 * ldm, ldm,
242 B, T(1), B + i0);
243 if (!lower && i1 < n)
244 kernels::gemv<T>(false, i1 - i0, n - i1, T(-1),
245 M + i0 * ldm + i1, ldm, B + i1, T(1), B + i0);
246 solve_block(i0, i1);
247 }
248 return;
249 }
250 if (lower) {
251 for (size_t i0 = 0; i0 < n; i0 += nb) {
252 const size_t i1 = std::min(n, i0 + nb);
253 solve_block(i0, i1);
254 update(i1, n, i0, i1);
255 }
256 } else {
257 const size_t nblk = (n + nb - 1) / nb;
258 for (size_t bk = nblk; bk-- > 0;) {
259 const size_t i0 = bk * nb, i1 = std::min(n, i0 + nb);
260 solve_block(i0, i1);
261 update(0, i0, i0, i1);
262 }
263 }
264}

268// Applies the row interchanges piv[j0..j1) (row j <-> row piv[j], in order)
269// to columns [c0, c1) of the column-major panel P (leading dimension rows),
270// column by column so every access stays inside one contiguous column.
271template <class T>
272void
273panel_swap(T *P, size_t rows, size_t c0, size_t c1, const size_t *piv,
274 size_t j0, size_t j1)
275{
276 for (size_t c = c0; c < c1; ++c) {
277 T *col = P + c * rows;
278 for (size_t j = j0; j < j1; ++j)
279 if (piv[j] != j) std::swap(col[j], col[piv[j]]);
280 }
281}

283// Unblocked LU (partial pivoting when `pivot`) of columns [c0, c0 + w) of a
284// column-major panel P (leading dimension rows), using rows [c0, rows). Row
285// swaps touch only these columns; piv[j] is the panel row swapped with row j.
286// `col0` is the global column of the panel's first column.
287template <class T>
288void
289panel_lu_cols(T *P, size_t rows, size_t c0, size_t w, size_t *piv, bool pivot,
290 double tol, size_t col0, const char *who)
291{
292 for (size_t j = c0; j < c0 + w && j < rows; ++j) {
293 T *col = P + j * rows;
294 size_t p = j;
295 if (pivot) {
296 T best = std::abs(col[j]);
297 for (size_t r = j + 1; r < rows; ++r) {
298 const T v = std::abs(col[r]);
299 if (v > best) {
300 best = v;
301 p = r;
302 }
303 }
304 }
305 piv[j] = p;
306 if (p != j)
307 for (size_t c = c0; c < c0 + w; ++c)
308 std::swap(P[c * rows + j], P[c * rows + p]);
309 const T d = col[j];
310 if (!(std::abs(double(d)) > tol))
311 fail(std::string(who) +
312 ": matrix is singular to working precision "
313 "(pivot " +
314 std::to_string(double(d)) + " in column " +
315 std::to_string(col0 + j) + ")");
316 const T inv = T(1) / d;
317 for (size_t r = j + 1; r < rows; ++r)
318 col[r] *= inv;
319 for (size_t c = j + 1; c < c0 + w; ++c) {
320 T *cc = P + c * rows;
321 const T f = cc[j];
322 if (f != T(0)) axpy_s(rows - j - 1, -f, col + j + 1, cc + j + 1);
323 }
324 }
325}

327// Recursive LU of columns [c0, c0 + w) of the column-major panel (LAPACK
328// dgetrf2): factor the left half, apply its interchanges to the right half,
329// solve for the top of the right half, update the rest with one GEMM, factor
330// it, and apply its interchanges back to the left half. Most of the panel's
331// work is GEMM and every swap runs down a contiguous column.
332template <class T>
333void
334panel_lu_rec(T *P, size_t rows, size_t c0, size_t w, size_t *piv, bool pivot,
335 double tol, size_t col0, const char *who)
336{
337 if (w <= 8 || c0 + w > rows) {
338 panel_lu_cols(P, rows, c0, w, piv, pivot, tol, col0, who);
339 return;
340 }
341 const size_t w1 = w / 2, w2 = w - w1, cm = c0 + w1, c1 = c0 + w;
342 panel_lu_rec(P, rows, c0, w1, piv, pivot, tol, col0, who);
343 if (pivot) panel_swap(P, rows, cm, c1, piv, c0, cm);
344 // A12 = L11^{-1} A12 (column-major, unit lower): rows [c0, cm), cols [cm,
345 // c1)
346 for (size_t c = cm; c < c1; ++c) {
347 T *x = P + c * rows;
348 for (size_t j = c0; j < cm; ++j)
349 if (x[j] != T(0))
350 axpy_s(cm - j - 1, -x[j], P + j * rows + j + 1, x + j + 1);
351 }
352 // A22 -= A21 A12, rows [cm, rows): column-major C = C - A B is the
353 // row-major C^T = C^T - B^T A^T on the same memory.
354 kernels::gemm<T>(false, false, w2, rows - cm, w1, T(-1), P + cm * rows + c0,
355 rows, P + c0 * rows + cm, rows, T(1), P + cm * rows + cm, rows);
356 panel_lu_rec(P, rows, cm, w2, piv, pivot, tol, col0, who);
357 if (pivot) panel_swap(P, rows, c0, cm, piv, cm, c1);
358}

360// Blocked right-looking LU of the n x n row-major matrix A (in place): unit
361// lower L below the diagonal, U on and above it. With perm != nullptr rows are
362// pivoted and perm[i] is the original row ending up in row i. Each panel is
363// transposed into a column-major buffer (tiled SIMD transpose), factored
364// recursively and transposed back; its row interchanges are applied to the rest
365// of the matrix in parallel over column blocks; U12 is a blocked triangular
366// solve and the trailing update one GEMM.
367template <class T>
368void
369lu_inplace(T *A, size_t n, size_t lda, std::vector<size_t> *perm, double tol,
370 const char *who)
371{
372 if (perm) {
373 perm->resize(n);
374 std::iota(perm->begin(), perm->end(), size_t(0));
375 }
376 if (n == 0) return;
377 const size_t nb = n >= 1024 ? 128 : (n >= 256 ? 64 : 32);
378 std::vector<T> panel;
379 std::vector<size_t> piv(nb);
380 for (size_t k0 = 0; k0 < n; k0 += nb) {
381 const size_t b = std::min(nb, n - k0), rows = n - k0;
382 panel.resize(rows * b);
383 T *akk = A + k0 * lda + k0;
384 std::iota(piv.begin(), piv.begin() + b, size_t(0));
385 kernels::transpose<T>(rows, b, akk, lda, panel.data(), rows);
386 {
387 // The panel's GEMMs are small and tall; on one thread they run
388 // faster than with fork/join per call (measured).
389 detail::SerialScope serial;
390 panel_lu_rec(panel.data(), rows, 0, b, piv.data(), perm != nullptr,
391 tol, k0, who);
392 }
393 kernels::transpose<T>(b, rows, panel.data(), rows, akk, lda);
394 if (perm) {
395 for (size_t j = 0; j < b; ++j)
396 if (piv[j] != j)
397 std::swap((*perm)[k0 + j], (*perm)[k0 + piv[j]]);
398 // The same interchanges on the columns outside the panel,
399 // [0, k0) and [k0 + b, n), split into column blocks.
400 detail::parallel_for(
401 n - b, 512,
402 [&](size_t q0, size_t q1) {
403 auto swap_cols = [&](size_t c0, size_t c1) {
404 for (size_t j = 0; j < b; ++j) {
405 if (piv[j] == j) continue;
406 T *ra = A + (k0 + j) * lda,
407 *rb = A + (k0 + piv[j]) * lda;
408 std::swap_ranges(ra + c0, ra + c1, rb + c0);
409 }
410 };
411 if (q0 < k0) swap_cols(q0, std::min(q1, k0));
412 if (q1 > k0) swap_cols(std::max(q0, k0) + b, q1 + b);
413 },
414 64);
415 }
416 if (k0 + b >= n) break;
417 const size_t nr = n - k0 - b;
418 trsm_left(
419 true, true, b, nr, akk, lda, akk + b, lda); // U12 = L11^{-1} A12
420 kernels::gemm<T>(false, false, nr, nr, b, T(-1), akk + b * lda, lda,
421 akk + b, lda, T(1), akk + b * lda + b, lda); // A22 -= A21 U12
422 }
423}

425// Solves A X = B given the in-place LU of A (lu_inplace) and its row
426// permutation; B is n x k (row-major), overwritten with X.
427template <class T>
428void
429lu_solve_inplace(
430 const T *LU, size_t n, const std::vector<size_t> &perm, T *B, size_t k)
431{
432 if (!perm.empty()) {
433 std::vector<T> tmp(n * k);
434 for (size_t i = 0; i < n; ++i)
435 std::copy(B + perm[i] * k, B + perm[i] * k + k, tmp.data() + i * k);
436 std::copy(tmp.begin(), tmp.end(), B);
437 }
438 trsm_left(true, true, n, k, LU, n, B, k);
439 trsm_left(false, false, n, k, LU, n, B, k);
440}

442// The permutations of this thread's recent luDcmpPivoted calls, keyed by the
443// buffer of the returned P. luSolve takes P as a dense n x n matrix; reading
444// all of it would cost more than the triangular solves. A remembered
445// permutation is used only after checking P(i, perm[i]) == 1 for every row,
446// which pins down a permutation matrix completely (O(n) reads).
447template <class T> struct PermCache {
448 struct Entry {
449 const T *key = nullptr;
450 std::vector<size_t> perm;
451 };
452 Entry e[4];
453 size_t next = 0;
454};
455
456template <class T>
457PermCache<T> &
458perm_cache()
459{
460 thread_local PermCache<T> c;
461 return c;
462}
463
464template <class T>
465void
466remember_perm(const T *key, const std::vector<size_t> &perm)
467{
468 PermCache<T> &c = perm_cache<T>();
469 c.e[c.next].key = key;
470 c.e[c.next].perm = perm;
471 c.next = (c.next + 1) % 4;
472}

474// Row permutation encoded by a permutation matrix: perm[i] = j with
475// P(i, j) == 1.
476template <class T, class S>
477std::vector<size_t>
478perm_of(const tensorET<2, T, S> &P)
479{
480 const size_t n = P.size(0), ld = P.stride(0);
481 if (P.size(1) != n) fail("luSolve: P must be square");
482 for (const auto &en : perm_cache<T>().e) {
483 if (en.key != P.data || en.perm.size() != n) continue;
484 bool ok = true;
485 for (size_t i = 0; i < n && ok; ++i)
486 ok = P.data[i * ld + en.perm[i]] == T(1);
487 if (ok) return en.perm;
488 }
489 // Otherwise: the non-zero of every row, in parallel over rows.
490 std::vector<size_t> perm(n);
491 detail::parallel_for(
492 n, 64,
493 [&](size_t r0, size_t r1) {
494 for (size_t i = r0; i < r1; ++i) {
495 const T *row = P.data + i * ld;
496 size_t j = 0;
497 while (j + 4 <= n && row[j] == T(0) && row[j + 1] == T(0) &&
498 row[j + 2] == T(0) && row[j + 3] == T(0))
499 j += 4;
500 while (j < n && row[j] == T(0))
501 ++j;
502 perm[i] = j;
503 }
504 },
505 1);
506 for (size_t i = 0; i < n; ++i)
507 if (perm[i] >= n)
508 fail("luSolve: P is not a permutation matrix (row " +
509 std::to_string(i) + " is zero)");
510 return perm;
511}

513// Splits the in-place LU (n x n, contiguous) into a new unit-lower L and
514// leaves U in LU (strictly lower part zeroed).
515template <class T>
516Mat<T>
517split_lu(Mat<T> &LU)
518{
519 const size_t n = LU.size(0);
520 Mat<T> L = mat<T>(n, n);
521 detail::parallel_for(
522 n, 64,
523 [&](size_t r0, size_t r1) {
524 for (size_t i = r0; i < r1; ++i) {
525 T *l = L.data + i * n, *u = LU.data + i * n;
526 std::copy(u, u + i, l);
527 l[i] = T(1);
528 std::fill(l + i + 1, l + n, T(0));
529 std::fill(u, u + i, T(0));
530 }
531 },
532 1);
533 return L;
534}

536// ---- QR -----------------------------------------------------------------
537
538// Householder reflector (LAPACK dlarfg): given x = [alpha; tail] (length L,
539// contiguous), overwrite alpha with beta and tail with v(1:), return tau so
540// that (I - tau v v^T) x = [beta; 0], v = [1; tail]. Columns with norm below
541// 1e-12 are left as they are (tau = 0).
542template <class T>
543T
544householder(T *x, size_t L)
545{
546 if (L <= 1) return T(0);
547 const T alpha = x[0];
548 T xnorm = std::sqrt(dot_s(L - 1, x + 1, x + 1));
549 const T full = std::hypot(alpha, xnorm);
550 if (xnorm == T(0) || full < T(1e-12)) return T(0);
551 const T beta = alpha >= T(0) ? -full : full;
552 const T tau = (beta - alpha) / beta;
553 const T scal = T(1) / (alpha - beta);
554 for (size_t i = 1; i < L; ++i)
555 x[i] *= scal;
556 x[0] = beta;
557 return tau;
558}

560template <class T> struct QrPanel {
561 size_t k0 = 0, b = 0, rows = 0;
562 std::vector<T>
563 Vt; // b x rows, row-major: reflector i is row i (unit, zeros before)
564 std::vector<T> Tm; // b x b upper triangular (compact WY)
565};

567// Blocked Householder QR of the m x n row-major matrix A (m >= n), in place:
568// R on and above the diagonal; the reflectors are returned per panel.
569template <class T>
570std::vector<QrPanel<T>>
571qr_inplace(T *A, size_t m, size_t n, size_t lda)
572{
573 const size_t nb = n >= 256 ? 48 : 32;
574 std::vector<QrPanel<T>> panels;
575 std::vector<T> pn, W, W2;
576 for (size_t k0 = 0; k0 < n; k0 += nb) {
577 const size_t b = std::min(nb, n - k0), rows = m - k0;
578 pn.resize(rows * b);
579 for (size_t r = 0; r < rows; ++r)
580 for (size_t c = 0; c < b; ++c)
581 pn[c * rows + r] = A[(k0 + r) * lda + k0 + c];
582 std::vector<T> tau(b);
583 for (size_t j = 0; j < b; ++j) {
584 T *x = pn.data() + j * rows + j;
585 const size_t L = rows - j;
586 tau[j] = householder(x, L);
587 if (tau[j] == T(0)) continue;
588 for (size_t c = j + 1; c < b; ++c) {
589 T *y = pn.data() + c * rows + j;
590 const T w = tau[j] * (y[0] + dot_s(L - 1, x + 1, y + 1));
591 y[0] -= w;
592 axpy_s(L - 1, -w, x + 1, y + 1);
593 }
594 }
595 for (size_t r = 0; r < rows; ++r)
596 for (size_t c = 0; c < b; ++c)
597 A[(k0 + r) * lda + k0 + c] = pn[c * rows + r];
598 QrPanel<T> P;
599 P.k0 = k0;
600 P.b = b;
601 P.rows = rows;
602 P.Vt.assign(b * rows, T(0));
603 for (size_t i = 0; i < b; ++i) {
604 T *v = P.Vt.data() + i * rows;
605 v[i] = T(1);
606 for (size_t r = i + 1; r < rows; ++r)
607 v[r] = pn[i * rows + r];
608 }
609 // T (dlarft, forward columnwise): T[0:j, j] = -tau_j T[0:j,0:j]
610 // V[:,0:j]^T v_j
611 P.Tm.assign(b * b, T(0));
612 std::vector<T> z(b);
613 for (size_t j = 0; j < b; ++j) {
614 P.Tm[j * b + j] = tau[j];
615 if (j == 0 || tau[j] == T(0)) continue;
616 const T *vj = P.Vt.data() + j * rows;
617 for (size_t i = 0; i < j; ++i)
618 z[i] = -tau[j] *
619 dot_s(rows - j, P.Vt.data() + i * rows + j, vj + j);
620 for (size_t i = 0; i < j; ++i) {
621 T s(0);
622 for (size_t l = i; l < j; ++l)
623 s += P.Tm[i * b + l] * z[l];
624 P.Tm[i * b + j] = s;
625 }
626 }
627 // Trailing update: C = (I - V T^T V^T) C, C = A[k0:m, k0+b:n]
628 const size_t nt = n - k0 - b;
629 if (nt) {
630 T *C = A + k0 * lda + k0 + b;
631 W.resize(b * nt);
632 W2.resize(b * nt);
633 kernels::gemm<T>(false, false, b, nt, rows, T(1), P.Vt.data(), rows,
634 C, lda, T(0), W.data(), nt);
635 kernels::gemm<T>(true, false, b, nt, b, T(1), P.Tm.data(), b,
636 W.data(), nt, T(0), W2.data(), nt);
637 kernels::gemm<T>(true, false, rows, nt, b, T(-1), P.Vt.data(), rows,
638 W2.data(), nt, T(1), C, lda);
639 }
640 panels.push_back(std::move(P));
641 }
642 return panels;
643}
644
645// Thin Q (m x n) from the panels: Q = H_1 ... H_n [I; 0], accumulated
646// backwards with the block reflectors.
647template <class T>
648Mat<T>
649qr_form_q(const std::vector<QrPanel<T>> &panels, size_t m, size_t n)
650{
651 Mat<T> Q = mat<T>(m, n, T(0));
652 for (size_t i = 0; i < n; ++i)
653 Q.data[i * n + i] = T(1);
654 std::vector<T> W, W2;
655 for (size_t p = panels.size(); p-- > 0;) {
656 const QrPanel<T> &P = panels[p];
657 const size_t nc = n - P.k0;
658 T *C = Q.data + P.k0 * n + P.k0;
659 W.resize(P.b * nc);
660 W2.resize(P.b * nc);
661 kernels::gemm<T>(false, false, P.b, nc, P.rows, T(1), P.Vt.data(),
662 P.rows, C, n, T(0), W.data(), nc);
663 kernels::gemm<T>(false, false, P.b, nc, P.b, T(1), P.Tm.data(), P.b,
664 W.data(), nc, T(0), W2.data(), nc);
665 kernels::gemm<T>(true, false, P.rows, nc, P.b, T(-1), P.Vt.data(),
666 P.rows, W2.data(), nc, T(1), C, n);
667 }
668 return Q;
669}

671// Thin QR of a contiguous m x n matrix: returns {Q, R} with R's diagonal
672// non-negative (the Gram-Schmidt convention).
673template <class T>
674std::pair<Mat<T>, Mat<T>>
675thin_qr(Mat<T> A)
676{
677 const size_t m = A.size(0), n = A.size(1);
678 if (m < n) fail("qrDecompositionTile: needs rows >= columns");
679 auto panels = qr_inplace(A.data, m, n, n);
680 Mat<T> R = mat<T>(n, n, T(0));
681 for (size_t i = 0; i < n; ++i)
682 std::copy(A.data + i * n + i, A.data + i * n + n, R.data + i * n + i);
683 Mat<T> Q = qr_form_q(panels, m, n);
684 for (size_t j = 0; j < n; ++j) {
685 if (R.data[j * n + j] >= T(0)) continue;
686 for (size_t c = j; c < n; ++c)
687 R.data[j * n + c] = -R.data[j * n + c];
688 for (size_t r = 0; r < m; ++r)
689 Q.data[r * n + j] = -Q.data[r * n + j];
690 }
691 return {std::move(Q), std::move(R)};
692}

694// ---- symmetric eigenproblem (cyclic Jacobi) -----------------------------
695
696// All eigenpairs of the symmetric n x n row-major matrix a (destroyed):
697// w ascending, V's columns the eigenvectors. For small matrices.
698template <class T>
699void
700jacobi_eigen(std::vector<T> &a, size_t n, std::vector<T> &w, std::vector<T> &V)
701{
702 V.assign(n * n, T(0));
703 for (size_t i = 0; i < n; ++i)
704 V[i * n + i] = T(1);
705 T fro = 0;
706 for (T v : a)
707 fro += v * v;
708 const T eps = std::numeric_limits<T>::epsilon();
709 for (int sweep = 0; sweep < 64; ++sweep) {
710 T off = 0;
711 for (size_t p = 0; p < n; ++p)
712 for (size_t q = p + 1; q < n; ++q)
713 off += a[p * n + q] * a[p * n + q];
714 if (off <= eps * eps * fro * T(0.25) || off == T(0)) break;
715 for (size_t p = 0; p < n; ++p)
716 for (size_t q = p + 1; q < n; ++q) {
717 const T apq = a[p * n + q];
718 if (std::abs(apq) <= std::numeric_limits<T>::min()) continue;
719 const T theta = (a[q * n + q] - a[p * n + p]) / (T(2) * apq);
720 const T t = (theta >= T(0) ? T(1) : T(-1)) /
721 (std::abs(theta) + std::sqrt(theta * theta + T(1)));
722 const T c = T(1) / std::sqrt(t * t + T(1)), s = t * c;
723 for (size_t k = 0; k < n; ++k) { // columns p, q
724 const T akp = a[k * n + p], akq = a[k * n + q];
725 a[k * n + p] = c * akp - s * akq;
726 a[k * n + q] = s * akp + c * akq;
727 }
728 for (size_t k = 0; k < n; ++k) { // rows p, q
729 const T apk = a[p * n + k], aqk = a[q * n + k];
730 a[p * n + k] = c * apk - s * aqk;
731 a[q * n + k] = s * apk + c * aqk;
732 }
733 for (size_t k = 0; k < n; ++k) {
734 const T vkp = V[k * n + p], vkq = V[k * n + q];
735 V[k * n + p] = c * vkp - s * vkq;
736 V[k * n + q] = s * vkp + c * vkq;
737 }
738 }
739 }
740 std::vector<size_t> idx(n);
741 std::iota(idx.begin(), idx.end(), size_t(0));
742 std::sort(idx.begin(), idx.end(),
743 [&](size_t x, size_t y) { return a[x * n + x] < a[y * n + y]; });
744 w.resize(n);
745 std::vector<T> Vs(n * n);
746 for (size_t j = 0; j < n; ++j) {
747 w[j] = a[idx[j] * n + idx[j]];
748 for (size_t k = 0; k < n; ++k)
749 Vs[k * n + j] = V[k * n + idx[j]];
750 }
751 V.swap(Vs);
752}

754// ---- triangular inverse -------------------------------------------------
755
756// X (n x n, ldx) = inverse of the upper-triangular s x s block of U starting
757// at (i0, i0); only the upper triangle is written.
758template <class T>
759void
760tri_inv_block(const T *U, size_t ldu, size_t i0, size_t s, T *X, size_t ldx)
761{
762 for (size_t j = 0; j < s; ++j) {
763 const T d = U[(i0 + j) * ldu + i0 + j];
764 if (d == T(0))
765 fail("inverse_backs: zero on the diagonal at " +
766 std::to_string(i0 + j));
767 X[(i0 + j) * ldx + i0 + j] = T(1) / d;
768 for (size_t i = j; i-- > 0;) {
769 T sum(0);
770 for (size_t k = i + 1; k <= j; ++k)
771 sum += U[(i0 + i) * ldu + i0 + k] * X[(i0 + k) * ldx + i0 + j];
772 X[(i0 + i) * ldx + i0 + j] = -sum / U[(i0 + i) * ldu + i0 + i];
773 }
774 }
775}

781// p-norm (p >= 1) of A - B over all elements.
782template <int D, class T, class SA, class SB>
783real_type_t<T>
784norm(const tensorET<D, T, SA> &A, const tensorET<D, T, SB> &B, int p)
785{
786 using R = real_type_t<T>;
787 if (p < 1) la_detail::fail("norm: p must be >= 1");
788 for (int d = 0; d < D; ++d)
789 if (A.size(d) != B.size(d)) la_detail::fail("norm: shape mismatch");
790 const auto a = la_detail::dense(A), b = la_detail::dense(B);
791 const T *x = a.data, *y = b.data;
792 const double s = detail::parallel_sum<double>(
793 a.size(), 1 << 14, [=](size_t i0, size_t i1) {
794 double acc = 0;
795 for (size_t i = i0; i < i1; ++i) {
796 const double v = double(std::abs(x[i] - y[i]));
797 acc += p == 1 ? v : (p == 2 ? v * v : std::pow(v, double(p)));
798 }
799 return acc;
800 });
801 return R(p == 1 ? s : (p == 2 ? std::sqrt(s) : std::pow(s, 1.0 / p)));
802}

804// Named norms of A - B: "fro" / "l2" / "2" (Frobenius, i.e. the element-wise
805// 2-norm), "l1" / "1" (sum of absolute values), "inf" / "max" (largest
806// absolute value).
807template <int D, class T, class SA, class SB>
808real_type_t<T>
809norm(const tensorET<D, T, SA> &A, const tensorET<D, T, SB> &B,
810 const std::string &name)
811{
812 using R = real_type_t<T>;
813 if (name == "fro" || name == "l2" || name == "2") return norm(A, B, 2);
814 if (name == "l1" || name == "1") return norm(A, B, 1);
815 if (name == "inf" || name == "max") {
816 for (int d = 0; d < D; ++d)
817 if (A.size(d) != B.size(d)) la_detail::fail("norm: shape mismatch");
818 const auto a = la_detail::dense(A), b = la_detail::dense(B);
819 R m = 0;
820 for (size_t i = 0; i < a.size(); ++i)
821 m = std::max<R>(m, R(std::abs(a.data[i] - b.data[i])));
822 return m;
823 }
824 la_detail::fail("norm: unknown norm \"" + name + "\"");
825}

832// ==== conjugate gradient ====================================================
833
834// Solves A x = b for SPD A from x0. maxIter < 0 means n. Stops when the
835// residual norm drops below tol, or (with a note on stderr) when p^T A p
836// falls below 1e-12 in absolute value.
837template <class T, class SA, class SB, class SX>
838tensorET<1, T>
839conjugateGradient(const tensorET<2, T, SA> &A, const tensorET<1, T, SB> &b,
840 const tensorET<1, T, SX> &x0, int maxIter = -1, double tol = 1e-6)
841{
842 static_assert(la_detail::is_real_v<T>,
843 "conjugateGradient: T must be float or double");
844 la_detail::require_square(A, "conjugateGradient");
845 const size_t n = A.size(0), lda = A.stride(0);
846 if (b.size() != n || x0.size() != n)
847 la_detail::fail("conjugateGradient: dimension mismatch");
848 tensorET<1, T> x = la_detail::dense(x0), r = la_detail::dense(b);
849 tensorET<1, T> p({n}), Ap({n});
850 const T *Ad = A.data;
851 kernels::gemv<T>(false, n, n, T(-1), Ad, lda, x.data, T(1), r.data);
852 T *xd = x.data, *rd = r.data, *pd = p.data, *apd = Ap.data;
853 std::copy(rd, rd + n, pd);
854 double rr = kernels::dot<T>(n, rd, rd);
855 if (std::sqrt(rr) < tol) return x;
856 const long iters = maxIter < 0 ? long(n) : long(maxIter);
857 const size_t rgrain =
858 std::max<size_t>(4, (size_t(1) << 16) / std::max<size_t>(n, 1));
859 for (long it = 0; it < iters; ++it) {
860 // Ap = A p and p^T A p in one pass over A
861 const double pAp = detail::parallel_sum<double>(
862 n, rgrain,
863 [=](size_t i0, size_t i1) {
864 kernels::blas_detail::gemv_rows<T>(
865 i0, i1, n, T(1), Ad, lda, pd, T(0), apd);
866 return double(la_detail::dot_s(i1 - i0, pd + i0, apd + i0));
867 },
868 4);
869 if (std::abs(pAp) < 1e-12) {
870 std::fprintf(stderr,
871 "conjugateGradient: p^T A p = %.3e below 1e-12 after %ld "
872 "iterations; stopping\n",
873 pAp, it);
874 break;
875 }
876 const T alpha = T(rr / pAp);
877 // x += alpha p, r -= alpha A p, and r^T r, in one pass
878 const double rr_new =
879 detail::parallel_sum<double>(n, 1 << 14, [=](size_t i0, size_t i1) {
880 double s = 0;
881 for (size_t i = i0; i < i1; ++i) {
882 xd[i] += alpha * pd[i];
883 const T ri = rd[i] - alpha * apd[i];
884 rd[i] = ri;
885 s += double(ri) * double(ri);
886 }
887 return s;
888 });
889 if (std::sqrt(rr_new) < tol) break;
890 const T beta = T(rr_new / rr);
891 rr = rr_new;
892 detail::parallel_for(n, 1 << 14, [=](size_t i0, size_t i1) {
893 for (size_t i = i0; i < i1; ++i)
894 pd[i] = rd[i] + beta * pd[i];
895 });
896 }
897 return x;
898}

900// ==== Gauss-Jordan ==========================================================
901
902// Reduced row-echelon form of A (partial pivoting). A pivot candidate
903// below max(m, n) * eps * max|A| counts as zero (the column is skipped).
904template <class T, class S>
905tensorET<2, T>
906gaussJordanElimination(const tensorET<2, T, S> &A)
907{
908 static_assert(la_detail::is_real_v<T>,
909 "gaussJordanElimination: T must be float or double");
910 tensorET<2, T> R = la_detail::dense(A);
911 const size_t m = R.size(0), n = R.size(1);
912 if (m == 0 || n == 0) return R;
913 T *a = R.data;
914 T amax = 0;
915 for (size_t i = 0; i < m * n; ++i)
916 amax = std::max(amax, std::abs(a[i]));
917 const T eps = T(std::max(m, n)) * std::numeric_limits<T>::epsilon() * amax;
918 const bool par =
919 double(m) * n * std::min(m, n) > 4e6 && detail::max_threads() > 1;
920 size_t prow = 0;
921 bool skip = false;
922 auto step_serial = [&](size_t col) {
923 size_t p = prow;
924 T best = std::abs(a[prow * n + col]);
925 for (size_t i = prow + 1; i < m; ++i) {
926 const T v = std::abs(a[i * n + col]);
927 if (v > best) {
928 best = v;
929 p = i;
930 }
931 }
932 skip = !(best > eps);
933 if (skip) return;
934 if (p != prow)
935 std::swap_ranges(
936 a + p * n + col, a + p * n + n, a + prow * n + col);
937 const T inv = T(1) / a[prow * n + col];
938 for (size_t c = col; c < n; ++c)
939 a[prow * n + c] *= inv;
940 a[prow * n + col] = T(1);
941 };
942 auto eliminate = [&](size_t i, size_t col) {
943 if (i == prow) return;
944 T *ri = a + i * n;
945 const T f = ri[col];
946 if (f != T(0)) {
947 la_detail::axpy_s(n - col, -f, a + prow * n + col, ri + col);
948 ri[col] = T(0);
949 }
950 };
951#ifdef _OPENMP
952 if (par) {
953#pragma omp parallel
954 {
955 for (size_t col = 0; col < n; ++col) {
956#pragma omp barrier
957 if (prow >= m) break;
958#pragma omp single
959 step_serial(col);
960 if (skip) continue;
961#pragma omp for schedule(static)
962 for (long long i = 0; i < static_cast<long long>(m); ++i)
963 eliminate(size_t(i), col);
964#pragma omp single
965 ++prow;
966 }
967 }
968 return R;
969 }
970#endif
971 (void)par;
972 for (size_t col = 0; col < n && prow < m; ++col) {
973 step_serial(col);
974 if (skip) continue;
975 for (size_t i = 0; i < m; ++i)
976 eliminate(i, col);
977 ++prow;
978 }
979 return R;
980}

1042// ==== LU ====================================================================

1044// {L, U} with A = L U, unit-diagonal L, no pivoting. Throws when a pivot has
1045// absolute value <= tol.
1046template <class T, class S>
1047std::pair<tensorET<2, T>, tensorET<2, T>>
1048luDcmp(const tensorET<2, T, S> &A, double tol = 1e-12)
1049{
1050 static_assert(la_detail::is_real_v<T>, "luDcmp: T must be float or double");

1150// ==== triangular inverse ====================================================
1151
1152// Inverse of the upper-triangular U (only its upper triangle is read). The
1153// diagonal blocks are inverted in parallel, then adjacent inverted blocks are
1154// joined pairwise, X12 = -X11 (U12 X22), with GEMMs (about n^3/3 useful flops,
1155// all in GEMM). `m` is the old block-size hint; any n works.
1156template <class T, class S>
1157tensorET<2, T>
1158inverse_backs(const tensorET<2, T, S> &U, size_t m = 4)
1159{
1160 static_assert(
1161 la_detail::is_real_v<T>, "inverse_backs: T must be float or double");
1162 (void)m;
1163 la_detail::require_square(U, "inverse_backs");
1164 const size_t n = U.size(0), ldu = U.stride(0);
1165 tensorET<2, T> X = la_detail::mat<T>(n, n, T(0));
1166 if (n == 0) return X;
1167 const T *u = U.data;
1168 T *x = X.data;
1169 constexpr size_t nb = 64;
1170 const size_t nblk = (n + nb - 1) / nb;
1171 detail::parallel_for(
1172 nblk, 1,
1173 [&](size_t b0, size_t b1) {
1174 for (size_t bk = b0; bk < b1; ++bk)
1175 la_detail::tri_inv_block(
1176 u, ldu, bk * nb, std::min(nb, n - bk * nb), x, n);
1177 },
1178 1);
1179 std::vector<T> tmp;
1180 for (size_t s = nb; s < n; s *= 2) {
1181 const size_t pairs = (n + 2 * s - 1) / (2 * s);
1182 auto join = [&](size_t p, T *t) {
1183 const size_t i0 = p * 2 * s, i1 = i0 + s;
1184 if (i1 >= n) return;
1185 const size_t w = std::min(s, n - i1); // size of the second block
1186 // t = U12 X22 (s x w), then X12 = -X11 t
1187 kernels::gemm<T>(false, false, s, w, w, T(1), u + i0 * ldu + i1,
1188 ldu, x + i1 * n + i1, n, T(0), t, w);
1189 kernels::gemm<T>(false, false, s, w, s, T(-1), x + i0 * n + i0, n,
1190 t, w, T(0), x + i0 * n + i1, n);
1191 };
1192 const int nt = detail::max_threads();
1193 if (pairs >= size_t(nt) && nt > 1) {
1194 detail::parallel_for(
1195 pairs, 1,
1196 [&](size_t p0, size_t p1) {
1197 std::vector<T> t(s * s);
1198 for (size_t p = p0; p < p1; ++p)
1199 join(p, t.data());
1200 },
1201 1);
1202 } else {
1203 tmp.resize(s * s);
1204 for (size_t p = 0; p < pairs; ++p)
1205 join(p, tmp.data());
1206 }
1207 }
1208 return X;
1209}

1214// matrices): w ascending, V's columns the eigenvectors.
1215template <class T, class S>
1216void
1217symmetricEigen(const tensorET<2, T, S> &A, tensorET<1, T> &w, tensorET<2, T> &V)

1219 static_assert(

1225 for (size_t j = 0; j < n; ++j)

1234// Lanczos with full reorthogonalization (two Gram-Schmidt passes against all
1235// previous vectors): returns {alpha (m), beta (m - 1), Q (m x n)}, the
1236// tridiagonal coefficients and the Lanczos vectors (rows of Q), started at q0
1237// (normalized). On breakdown the next vector is a random vector orthogonal to
1238// the previous ones and its beta is 0.
1239template <class T, class S, class S0>
1240std::tuple<tensorET<1, T>, tensorET<1, T>, tensorET<2, T>>
1241lanczos(const tensorET<2, T, S> &A, size_t m, const tensorET<1, T, S0> &q0)
1242{
1243 static_assert(
1244 la_detail::is_real_v<T>, "lanczos: T must be float or double");
1245 la_detail::require_square(A, "lanczos");
1246 const size_t n = A.size(0), lda = A.stride(0);
1247 if (m == 0 || m > n) la_detail::fail("lanczos: need 1 <= m <= n");
1248 if (q0.size() != n) la_detail::fail("lanczos: q0 has the wrong size");
1249 tensorET<1, T> alpha({m}, T(0)), beta({m - 1}, T(0));
1250 tensorET<2, T> Q = la_detail::mat<T>(m, n, T(0));
1251 std::vector<T> w(n), c(m);
1252 const tensorET<1, T> q = la_detail::dense(q0);
1253 const T qn = std::sqrt(kernels::dot<T>(n, q.data, q.data));
1254 if (!(qn > T(0))) la_detail::fail("lanczos: q0 is zero");
1255 for (size_t i = 0; i < n; ++i)
1256 Q.data[i] = q.data[i] / qn;
1257 T anorm = 0;
1258 std::mt19937_64 rng(12345);
1259 std::normal_distribution<double> nd;
1260 auto reorth = [&](size_t upto) { // w -= Q[0:upto]^T (Q[0:upto] w), twice
1261 for (int pass = 0; pass < 2; ++pass) {
1262 kernels::gemv<T>(
1263 false, upto, n, T(1), Q.data, n, w.data(), T(0), c.data());
1264 kernels::gemv<T>(
1265 true, upto, n, T(-1), Q.data, n, c.data(), T(1), w.data());
1266 }
1267 };
1268 for (size_t j = 0; j < m; ++j) {
1269 const T *qj = Q.data + j * n;
1270 kernels::gemv<T>(false, n, n, T(1), A.data, lda, qj, T(0), w.data());
1271 const T a = kernels::dot<T>(n, qj, w.data());
1272 alpha.data[j] = a;
1273 if (j + 1 == m) break;
1274 kernels::axpy<T>(n, -a, qj, w.data());
1275 if (j > 0)
1276 kernels::axpy<T>(
1277 n, -beta.data[j - 1], Q.data + (j - 1) * n, w.data());
1278 reorth(j + 1);
1279 T bnorm = std::sqrt(kernels::dot<T>(n, w.data(), w.data()));
1280 anorm = std::max({anorm, std::abs(a), bnorm});
1281 if (bnorm <=
1282 T(64) * std::numeric_limits<T>::epsilon() * std::max(anorm, T(1))) {
1283 for (auto &v : w)
1284 v = T(nd(rng)); // breakdown: restart orthogonally
1285 reorth(j + 1);
1286 const T rn = std::sqrt(kernels::dot<T>(n, w.data(), w.data()));
1287 for (auto &v : w)
1288 v /= rn;
1289 beta.data[j] = T(0);
1290 } else {
1291 beta.data[j] = bnorm;
1292 for (auto &v : w)
1293 v /= bnorm;
1294 }
1295 std::copy(w.begin(), w.end(), Q.data + (j + 1) * n);
1296 }
1297 return {std::move(alpha), std::move(beta), std::move(Q)};
1298}

1325// C = sum_k c[k] M_k (+ d I), element-wise over contiguous n x n matrices.

1328lincomb(

1330{

1337 s += t.first * t.second->data[i];

1347// Matrix exponential: scaling and squaring with Pade approximants of degree
1348// 3, 5, 7, 9 or 13 chosen from the 1-norm (Higham 2005). Every product is a
1349// GEMM, the Pade solve a blocked LU solve.
1350template <class T, class S>
1351tensorET<2, T>
1352expm(const tensorET<2, T, S> &A0)
1353{
1354 static_assert(la_detail::is_real_v<T>, "expm: T must be float or double");
1355 using namespace la_detail;
1356 require_square(A0, "expm");
1357 const size_t n = A0.size(0);
1358 Mat<T> A = dense(A0);
1359 if (n == 0) return A;
1360 const double theta[5] = {1.495585217958292e-2, 2.539398330063230e-1,
1361 9.504178996162932e-1, 2.097847961257068e0, 5.371920351148152e0};
1362 const int degs[5] = {3, 5, 7, 9, 13};
1363 const double b3[] = {120, 60, 12, 1};
1364 const double b5[] = {30240, 15120, 3360, 420, 30, 1};
1365 const double b7[] = {17297280, 8648640, 1995840, 277200, 25200, 756, 56, 1};
1366 const double b9[] = {17643225600., 8821612800., 2075673600., 302702400.,
1367 30270240., 2162160., 110880., 3960., 90., 1.};
1368 const double b13[] = {64764752532480000., 32382376266240000.,
1369 7771770303897600., 1187353796428800., 129060195264000., 10559470521600.,
1370 670442572800., 33522128640., 1323241920., 40840800., 960960., 16380.,
1371 182., 1.};
1372 const double a1 = double(norm1(A));
1373 int deg = 13, s = 0;
1374 for (int i = 0; i < 4; ++i)
1375 if (a1 <= theta[i]) {
1376 deg = degs[i];
1377 break;
1378 }
1379 if (deg == 13 && a1 > theta[4]) {
1380 s = std::max(0, int(std::ceil(std::log2(a1 / theta[4]))));
1381 const T scale = T(std::ldexp(1.0, -s));
1382 for (size_t i = 0; i < n * n; ++i)
1383 A.data[i] *= scale;
1384 }
1385 Mat<T> A2 = mat<T>(n, n), U = mat<T>(n, n), V = mat<T>(n, n),
1386 tmp = mat<T>(n, n);
1387 gemm_into(A, A, A2);
1388 if (deg == 13) {
1389 Mat<T> A4 = mat<T>(n, n), A6 = mat<T>(n, n);
1390 gemm_into(A2, A2, A4);
1391 gemm_into(A4, A2, A6);
1392 const double *b = b13;
1393 lincomb<T>(
1394 tmp, {{T(b[13]), &A6}, {T(b[11]), &A4}, {T(b[9]), &A2}}, T(0));
1395 Mat<T> inner = mat<T>(n, n);
1396 gemm_into(A6, tmp, inner);
1397 lincomb<T>(tmp,
1398 {{T(1), &inner}, {T(b[7]), &A6}, {T(b[5]), &A4}, {T(b[3]), &A2}},
1399 T(b[1]));
1400 gemm_into(A, tmp, U);
1401 lincomb<T>(
1402 tmp, {{T(b[12]), &A6}, {T(b[10]), &A4}, {T(b[8]), &A2}}, T(0));
1403 gemm_into(A6, tmp, inner);
1404 lincomb<T>(V,
1405 {{T(1), &inner}, {T(b[6]), &A6}, {T(b[4]), &A4}, {T(b[2]), &A2}},
1406 T(b[0]));
1407 } else {
1408 const double *b = deg == 3 ? b3 : deg == 5 ? b5 : deg == 7 ? b7 : b9;
1409 // powers A^2, A^4, ... up to A^(deg-1)
1410 std::vector<Mat<T>> pw;
1411 pw.push_back(A2);
1412 for (int p = 4; p < deg; p += 2) {
1413 Mat<T> next = mat<T>(n, n);
1414 gemm_into(pw.back(), A2, next);
1415 pw.push_back(std::move(next));
1416 }
1417 // tmp = sum_{odd k} b[k] A^(k-1), V = sum_{even k} b[k] A^k
1418 auto accumulate = [&](Mat<T> &dst, int first) {
1419 T *d = dst.data;
1420 std::fill(d, d + n * n, T(0));
1421 for (size_t i = 0; i < n; ++i)
1422 d[i * n + i] = T(b[first]);
1423 for (int k = first + 2; k <= deg; k += 2) {
1424 const Mat<T> &P = pw[size_t((k - first) / 2 - 1)];
1425 const T c = T(b[k]);
1426 for (size_t i = 0; i < n * n; ++i)
1427 d[i] += c * P.data[i];
1428 }
1429 };
1430 accumulate(tmp, 1);
1431 gemm_into(A, tmp, U);
1432 accumulate(V, 0);
1433 }
1434 // Solve (V - U) X = (V + U)
1435 Mat<T> Q = mat<T>(n, n), X = mat<T>(n, n);
1436 for (size_t i = 0; i < n * n; ++i) {
1437 Q.data[i] = V.data[i] - U.data[i];
1438 X.data[i] = V.data[i] + U.data[i];
1439 }
1440 std::vector<size_t> perm;
1441 lu_inplace(Q.data, n, n, &perm, 0.0, "expm");
1442 lu_solve_inplace(Q.data, n, perm, X.data, n);
1443 for (int k = 0; k < s; ++k) { // undo the scaling by squaring
1444 gemm_into(X, X, tmp);
1445 std::swap(X, tmp);
1446 }
1447 return X;
1448}

1450// ==== LOBPCG =================================================================
1451
1452// The nev smallest eigenpairs of the symmetric matrix A, by LOBPCG from the
1453// initial block X0 (n x k, k >= nev columns). Writes eigvals (ascending) and X
1454// (n x nev, columns = eigenvectors); returns the iterations used. Converged
1455// when every residual norm ||A x - lambda x|| is below tol * max(1, |lambda|).
1456// Each Rayleigh-Ritz step orthonormalizes [X, W, P] (Householder QR, dependent
1457// directions dropped) and solves the small eigenproblem with Jacobi.
1458template <class T, class S, class S0>
1459int
1460lobpcg(const tensorET<2, T, S> &A, int nev, const tensorET<2, T, S0> &X0,
1461 tensorET<1, T> &eigvals, tensorET<2, T> &X, int maxIter = 100,
1462 double tol = 1e-8)
1463{
1464 static_assert(la_detail::is_real_v<T>, "lobpcg: T must be float or double");
1465 using namespace la_detail;
1466 require_square(A, "lobpcg");
1467 const size_t n = A.size(0), k = X0.size(1);
1468 if (nev < 1 || size_t(nev) > k || X0.size(0) != n)
1469 fail("lobpcg: need X0 of size n x k with k >= nev >= 1");
1470 const Mat<T> Ad = dense(A);
1471 auto matmul = [](const Mat<T> &L, const Mat<T> &R, bool tl = false) {
1472 const size_t m = tl ? L.size(1) : L.size(0),
1473 kk = tl ? L.size(0) : L.size(1);
1474 Mat<T> C = mat<T>(m, R.size(1));
1475 kernels::gemm<T>(tl, false, m, R.size(1), kk, T(1), L.data, L.size(1),
1476 R.data, R.size(1), T(0), C.data, R.size(1));
1477 return C;
1478 };
1479 auto finish = [&](const std::vector<T> &lam, const Mat<T> &Xk) {
1480 eigvals = tensorET<1, T>({size_t(nev)});
1481 X = mat<T>(n, size_t(nev));
1482 for (int i = 0; i < nev; ++i)
1483 eigvals.data[i] = lam[size_t(i)];
1484 for (size_t r = 0; r < n; ++r)
1485 std::copy(Xk.data + r * Xk.size(1), Xk.data + r * Xk.size(1) + nev,
1486 X.data + r * size_t(nev));
1487 };
1488 // Small problems: dense Jacobi is exact and cheaper.
1489 if (n <= 3 * k || n <= 32) {
1490 std::vector<T> a(Ad.data, Ad.data + n * n), w, V;
1491 for (size_t i = 0; i < n; ++i)
1492 for (size_t j = i + 1; j < n; ++j)
1493 a[i * n + j] = a[j * n + i] =
1494 (a[i * n + j] + a[j * n + i]) / T(2);
1495 jacobi_eigen(a, n, w, V);
1496 Mat<T> Vm = mat<T>(n, n);
1497 std::copy(V.begin(), V.end(), Vm.data);
1498 finish(w, Vm);
1499 return 0;
1500 }
1501 // Rayleigh-Ritz on the orthonormal basis Q (n x s): the k smallest Ritz
1502 // pairs, coefficients C (s x k).
1503 auto rayleigh_ritz = [&](const Mat<T> &Q, const Mat<T> &AQ,
1504 std::vector<T> &lam, Mat<T> &C) {
1505 const size_t s = Q.size(1);
1506 Mat<T> G = matmul(Q, AQ, true);
1507 std::vector<T> g(s * s), w, V;
1508 for (size_t i = 0; i < s; ++i)
1509 for (size_t j = 0; j < s; ++j)
1510 g[i * s + j] = (G.data[i * s + j] + G.data[j * s + i]) / T(2);
1511 jacobi_eigen(g, s, w, V);
1512 lam.assign(w.begin(), w.begin() + k);
1513 C = mat<T>(s, k);
1514 for (size_t i = 0; i < s; ++i)
1515 std::copy(V.data() + i * s, V.data() + i * s + k, C.data + i * k);
1516 };
1517 // Orthonormal basis of [blocks...] (first block kept first), dropping
1518 // directions whose QR diagonal is below 1e-10 of the largest.
1519 auto orth = [&](std::initializer_list<const Mat<T> *> blocks) {
1520 size_t s = 0;
1521 for (auto *b : blocks)
1522 s += b->size(1);
1523 Mat<T> basis = mat<T>(n, s);
1524 size_t off = 0;
1525 for (auto *b : blocks) {
1526 const size_t c = b->size(1);
1527 for (size_t r = 0; r < n; ++r)
1528 std::copy(b->data + r * c, b->data + r * c + c,
1529 basis.data + r * s + off);
1530 off += c;
1531 }
1532 auto qr = thin_qr(std::move(basis));
1533 T rmax = 0;
1534 for (size_t j = 0; j < s; ++j)
1535 rmax = std::max(rmax, std::abs(qr.second.data[j * s + j]));
1536 std::vector<size_t> keep;
1537 for (size_t j = 0; j < s; ++j)
1538 if (std::abs(qr.second.data[j * s + j]) > T(1e-10) * rmax)
1539 keep.push_back(j);
1540 Mat<T> Q = mat<T>(n, keep.size());
1541 for (size_t r = 0; r < n; ++r)
1542 for (size_t j = 0; j < keep.size(); ++j)
1543 Q.data[r * keep.size() + j] = qr.first.data[r * s + keep[j]];
1544 return Q;
1545 };
1546
1547 const Mat<T> X0d = dense(X0);
1548 Mat<T> Xk = orth({&X0d});
1549 if (Xk.size(1) < k) fail("lobpcg: X0 has linearly dependent columns");
1550 Mat<T> AX = matmul(Ad, Xk);
1551 std::vector<T> lam;
1552 Mat<T> C;
1553 rayleigh_ritz(Xk, AX, lam, C);
1554 Xk = matmul(Xk, C);
1555 AX = matmul(AX, C);
1556 Mat<T> P, AP;
1557 int it = 0;
1558 for (; it < maxIter; ++it) {
1559 // residuals R = AX - X diag(lam)
1560 Mat<T> R = mat<T>(n, k);
1561 std::vector<T> rn(k, T(0));
1562 for (size_t r = 0; r < n; ++r)
1563 for (size_t j = 0; j < k; ++j) {
1564 const T v = AX.data[r * k + j] - lam[j] * Xk.data[r * k + j];
1565 R.data[r * k + j] = v;
1566 rn[j] += v * v;
1567 }
1568 bool done = true;
1569 std::vector<size_t> active;
1570 for (size_t j = 0; j < k; ++j) {
1571 const bool conv =
1572 std::sqrt(rn[j]) <= T(tol) * std::max(T(1), std::abs(lam[j]));
1573 if (!conv) active.push_back(j);
1574 if (j < size_t(nev) && !conv) done = false;
1575 }
1576 if (done) break;
1577 Mat<T> W = mat<T>(n, active.size());
1578 for (size_t r = 0; r < n; ++r)
1579 for (size_t j = 0; j < active.size(); ++j)
1580 W.data[r * active.size() + j] = R.data[r * k + active[j]];
1581 Mat<T> Q = P.size() ? orth({&Xk, &W, &P}) : orth({&Xk, &W});
1582 if (Q.size(1) <= k) break; // no new directions
1583 Mat<T> AQ = matmul(Ad, Q);
1584 rayleigh_ritz(Q, AQ, lam, C);
1585 Mat<T> Xn = matmul(Q, C), AXn = matmul(AQ, C);
1586 // P = the part of the new X outside the old X: Q[:, k:] C[k:, :]
1587 const size_t s = Q.size(1), extra = s - k;
1588 Mat<T> Qe = mat<T>(n, extra), AQe = mat<T>(n, extra),
1589 Ce = mat<T>(extra, k);
1590 for (size_t r = 0; r < n; ++r) {
1591 std::copy(
1592 Q.data + r * s + k, Q.data + r * s + s, Qe.data + r * extra);
1593 std::copy(
1594 AQ.data + r * s + k, AQ.data + r * s + s, AQe.data + r * extra);
1595 }
1596 std::copy(C.data + k * k, C.data + s * k, Ce.data);
1597 P = matmul(Qe, Ce);
1598 AP = matmul(AQe, Ce);
1599 Xk = std::move(Xn);
1600 AX = std::move(AXn);
1601 }
1602 finish(lam, Xk);
1603 return it;

1606// ==== tournament elimination with a recorded history ========================
1607
1608// Row operation "row target -= alpha * row source".
1609template <class T> struct ElimOp {
1610 size_t target_row, source_row;
1611 T alpha;
1612};
1613// Row swap.
1614struct PermOp {
1615 size_t target_row, source_row;
1616};
1617template <class T> using Op = std::variant<ElimOp<T>, PermOp>;
1618template <class T> using History = std::vector<Op<T>>;
1619
1620// One level of pairwise elimination of column `pivot`: rows pivot + 2kd and
1621// pivot + 2kd + d (d = factor) are paired; the row with the larger entry in the
1622// pivot column is swapped to the top and the lower row eliminated against it
1623// (entries below 1e-10 |pivot| are flushed to zero). Returns the operations in
1624// the order applied. `m` is a legacy tile-size hint.
1625template <class T, class S>
1626History<T>
1627elimStep(tensorET<2, T, S> &A, size_t factor, size_t pivot, size_t m = 0)
1628{
1629 static_assert(la_detail::is_real_v<T> && is_host_storage_v<S>,
1630 "elimStep: host float or double matrix");
1631 (void)m;
1632 const size_t rows = A.size(0), cols = A.size(1), lda = A.stride(0);
1633 if (factor == 0 || pivot >= cols) return {};
1634 T *a = A.data;
1635 const size_t npairs =
1636 pivot + factor < rows
1637 ? (rows - pivot - factor + 2 * factor - 1) / (2 * factor)
1638 : 0;
1639 std::vector<ElimOp<T>> el(npairs);
1640 std::vector<char> swapped(npairs, 0), elim(npairs, 0);
1641 detail::parallel_for(
1642 npairs, std::max<size_t>(1, 16384 / std::max<size_t>(cols, 1)),
1643 [&](size_t p0, size_t p1) {
1644 for (size_t p = p0; p < p1; ++p) {
1645 const size_t top = pivot + 2 * factor * p, bot = top + factor;
1646 T *rt = a + top * lda, *rb = a + bot * lda;
1647 if (std::abs(rb[pivot]) > std::abs(rt[pivot])) {
1648 std::swap_ranges(rt, rt + cols, rb);
1649 swapped[p] = 1;
1650 }
1651 const T pv = rt[pivot];
1652 if (pv == T(0) || rb[pivot] == T(0)) continue;
1653 const T alpha = rb[pivot] / pv;
1654 la_detail::axpy_s(
1655 cols - pivot - 1, -alpha, rt + pivot + 1, rb + pivot + 1);
1656 rb[pivot] = T(0);
1657 const T thr = T(1e-10) * std::abs(pv);
1658 for (size_t c = pivot + 1; c < cols; ++c)
1659 if (std::abs(rb[c]) < thr) rb[c] = T(0);
1660 el[p] = ElimOp<T>{bot, top, alpha};
1661 elim[p] = 1;
1662 }
1663 },
1664 1);
1665 History<T> h;
1666 for (size_t p = 0; p < npairs; ++p) {
1667 const size_t top = pivot + 2 * factor * p;
1668 if (swapped[p]) h.push_back(PermOp{top, top + factor});
1669 if (elim[p]) h.push_back(el[p]);
1670 }
1671 return h;
1672}

1674// Reduces A to upper-triangular form in place by pairwise (tournament)
1675// elimination, column by column at distances 1, 2, 4, ..., and returns the
1676// history H of row operations, so that H A_original = U (H = E_k ... E_1).
1677template <class T, class S>
1678History<T>
1679revEl(tensorET<2, T, S> &A, size_t m = 0)
1680{
1681 History<T> h;
1682 const size_t rows = A.size(0), cols = A.size(1);
1683 for (size_t pivot = 0; pivot < std::min(rows, cols); ++pivot)
1684 for (size_t d = 1; pivot + d < rows; d *= 2) {
1685 History<T> s = elimStep(A, d, pivot, m);
1686 h.insert(h.end(), s.begin(), s.end());
1687 }
1688 return h;
1689}

1691// H A: replays the history as row operations on a copy of A, oldest
1692// first. One parallel pass: columns are independent.
1693template <class T, class S>
1694tensorET<2, T>
1695matMul(const History<T> &H, const tensorET<2, T, S> &A, size_t m = 0)
1696{
1697 (void)m;
1698 tensorET<2, T> R = la_detail::dense(A);
1699 const size_t cols = R.size(1);
1700 T *a = R.data;
1701 detail::parallel_for(
1702 cols, 256,
1703 [&](size_t c0, size_t c1) {
1704 const size_t w = c1 - c0;
1705 for (const Op<T> &op : H) {
1706 if (const auto *e = std::get_if<ElimOp<T>>(&op))
1707 la_detail::axpy_s(w, -e->alpha,
1708 a + e->source_row * cols + c0,
1709 a + e->target_row * cols + c0);
1710 else {
1711 const PermOp &p = std::get<PermOp>(op);
1712 std::swap_ranges(a + p.target_row * cols + c0,
1713 a + p.target_row * cols + c1,
1714 a + p.source_row * cols + c0);
1715 }
1716 }
1717 },
1718 16);
1719 return R;
1720}

1722// A H: replays the history as column operations on a copy of A, most recent
1723// first (an elimination "row t -= a row s" becomes "column s -= a column t", a
1724// swap swaps the columns). Rows are independent: blocks of rows are transposed
1725// into a small column-major buffer so every operation is one contiguous vector
1726// update. With revEl and inverse_backs:
1727// H = revEl(Acopy), inverse = matMul(inverse_backs(Acopy), H).
1728template <class T, class S>
1729tensorET<2, T>
1730matMul(const tensorET<2, T, S> &A, const History<T> &H, size_t m = 0)
1731{
1732 (void)m;
1733 tensorET<2, T> R = la_detail::dense(A);
1734 const size_t rows = R.size(0), cols = R.size(1);
1735 constexpr size_t RB = 8;
1736 T *a = R.data;
1737 detail::parallel_for((rows + RB - 1) / RB, 1,
1738 [&](size_t b0, size_t b1) {
1739 std::vector<T> buf(cols * RB);
1740 for (size_t bk = b0; bk < b1; ++bk) {
1741 const size_t r0 = bk * RB, nr = std::min(RB, rows - r0);
1742 for (size_t c = 0; c < cols; ++c)
1743 for (size_t r = 0; r < RB; ++r)
1744 buf[c * RB + r] =
1745 r < nr ? a[(r0 + r) * cols + c] : T(0);
1746 for (size_t k = H.size(); k-- > 0;) {
1747 const Op<T> &op = H[k];
1748 if (const auto *e = std::get_if<ElimOp<T>>(&op)) {
1749 T *s = buf.data() + e->source_row * RB;
1750 const T *t = buf.data() + e->target_row * RB;
1751 const T al = e->alpha;
1752 for (size_t r = 0; r < RB; ++r)
1753 s[r] -= al * t[r];
1754 } else {
1755 const PermOp &p = std::get<PermOp>(op);
1756 std::swap_ranges(buf.data() + p.target_row * RB,
1757 buf.data() + p.target_row * RB + RB,
1758 buf.data() + p.source_row * RB);
1759 }
1760 }
1761 for (size_t c = 0; c < cols; ++c)
1762 for (size_t r = 0; r < nr; ++r)
1763 a[(r0 + r) * cols + c] = buf[c * RB + r];
1764 }
1765 },
1766 1);
1767 return R;
1768}
