13 // y = alpha * A * x + beta * y. x has cols() entries, y has rows().
14 // When beta == 0, y is not read (so it may hold NaN / garbage).
15 template <typename M>
16 static void
17 spmv(const M &A, const typename M::value_type *x, typename M::value_type *y,
18 typename M::value_type alpha, typename M::value_type beta)
19 {
20 using T = typename M::value_type;
21 const auto *rp = A.row_ptr();
22 const auto *ci = A.col_ind();
23 const T *v = A.values();
24 const long m = static_cast<long>(A.rows());
25#pragma omp parallel for schedule(static) if (A.nnz() > 20000)
26 for (long i = 0; i < m; ++i) {
27 T s = T(0);
28 for (auto k = rp[i]; k < rp[i + 1]; ++k)
29 s += v[k] * x[ci[k]];
30 y[i] = (beta == T(0)) ? alpha * s : alpha * s + beta * y[i];
31 }
32 }
33
34 // Y = alpha * A * X + beta * Y with row-major dense X (cols x k) and
35 // Y (rows x k).
36 template <typename M>
37 static void
38 spmm(const M &A, const typename M::value_type *X, size_t k,
39 typename M::value_type *Y, typename M::value_type alpha,
40 typename M::value_type beta)
41 {
42 using T = typename M::value_type;
43 const auto *rp = A.row_ptr();
44 const auto *ci = A.col_ind();
45 const T *v = A.values();
46 const long m = static_cast<long>(A.rows());
47#pragma omp parallel for schedule(static) if (A.nnz() > 20000)
48 for (long i = 0; i < m; ++i) {
49 T *yi = Y + static_cast<size_t>(i) * k;
50 for (size_t c = 0; c < k; ++c)
51 yi[c] = (beta == T(0)) ? T(0) : beta * yi[c];
52 for (auto p = rp[i]; p < rp[i + 1]; ++p) {
53 const T a = alpha * v[p];
54 const T *xr = X + static_cast<size_t>(ci[p]) * k;
55 for (size_t c = 0; c < k; ++c)
56 yi[c] += a * xr[c];
57 }
58 }
59 }

61 // A_ij *= r[i] * c[j]; either factor may be null.
62 template <typename M>
63 static void
64 scale(M &A, const real_of_t<typename M::value_type> *r,
65 const real_of_t<typename M::value_type> *c)
66 {
67 const auto *rp = A.row_ptr();
68 const auto *ci = A.col_ind();
69 auto *v = A.values_mut();
70 const long m = static_cast<long>(A.rows());
71#pragma omp parallel for schedule(static) if (A.nnz() > 20000)
72 for (long i = 0; i < m; ++i) {
73 const auto ri = r ? r[i] : real_of_t<typename M::value_type>(1);
74 for (auto k = rp[i]; k < rp[i + 1]; ++k)
75 v[k] *= c ? ri * c[ci[k]] : ri;
76 }
77 }

79 // Row-wise norm of A into out[rows()].
80 template <typename M>
81 static void
82 row_norms(const M &A, real_of_t<typename M::value_type> *out, Norm p)
83 {
84 using R = real_of_t<typename M::value_type>;
85 const auto *rp = A.row_ptr();
86 const auto *v = A.values();
87 const long m = static_cast<long>(A.rows());
88#pragma omp parallel for schedule(static) if (A.nnz() > 20000)
89 for (long i = 0; i < m; ++i) {
90 R acc = 0;
91 for (auto k = rp[i]; k < rp[i + 1]; ++k) {
92 R a = std::abs(v[k]);
93 if (p == Norm::L1)
94 acc += a;
95 else if (p == Norm::L2)
96 acc += a * a;
97 else
98 acc = std::max(acc, a);
99 }
100 out[i] = (p == Norm::L2) ? std::sqrt(acc) : acc;
101 }
102 }

104 template <typename M>
105 static M
106 transpose(const M &A)
107 {
108 using T = typename M::value_type;
109 using Idx = typename M::index_type;
110 const size_t m = A.rows(), n = A.cols(), nz = A.nnz();
111 M B(n, m, nz);
112 std::vector<Idx> cnt(n + 1, 0);
113 const Idx *rp = A.row_ptr();
114 const Idx *ci = A.col_ind();
115 const T *v = A.values();
116 for (size_t k = 0; k < nz; ++k)
117 cnt[ci[k] + 1]++;
118 for (size_t j = 0; j < n; ++j)
119 cnt[j + 1] += cnt[j];
120 std::memcpy(B.row_ptr_mut(), cnt.data(), (n + 1) * sizeof(Idx));
121 Idx *bci = B.col_ind_mut();
122 T *bv = B.values_mut();
123 std::vector<Idx> next(cnt.begin(), cnt.end() - 1);
124 for (size_t i = 0; i < m; ++i) {
125 for (Idx k = rp[i]; k < rp[i + 1]; ++k) {
126 Idx dst = next[ci[k]]++;
127 bci[dst] = static_cast<Idx>(i);
128 bv[dst] = v[k];
129 }
130 }
131 return B;
132 }

134 // C = A * B (Gustavson, sorted output).
135 template <typename M>
136 static M
137 spgemm(const M &A, const M &B)
138 {
139 using T = typename M::value_type;
140 using Idx = typename M::index_type;
141 const size_t m = A.rows(), n = B.cols();
142 const Idx *arp = A.row_ptr(), *aci = A.col_ind();
143 const Idx *brp = B.row_ptr(), *bci = B.col_ind();
144 const T *av = A.values(), *bv = B.values();
145
146 std::vector<Idx> rp(m + 1, 0);
147 // Symbolic pass: number of distinct columns per output row.
148#pragma omp parallel
149 {
150 std::vector<Idx> mark(n, -1);
151#pragma omp for schedule(dynamic, 64)
152 for (long i = 0; i < static_cast<long>(m); ++i) {
153 Idx cnt = 0;
154 for (Idx p = arp[i]; p < arp[i + 1]; ++p) {
155 const Idx r = aci[p];
156 for (Idx q = brp[r]; q < brp[r + 1]; ++q) {
157 if (mark[bci[q]] != static_cast<Idx>(i)) {
158 mark[bci[q]] = static_cast<Idx>(i);
159 ++cnt;
160 }
161 }
162 }
163 rp[i + 1] = cnt;
164 }
165 }
166 for (size_t i = 0; i < m; ++i)
167 rp[i + 1] += rp[i];
168
169 M C(m, n, static_cast<size_t>(rp[m]));
170 std::memcpy(C.row_ptr_mut(), rp.data(), (m + 1) * sizeof(Idx));
171 Idx *cci = C.col_ind_mut();
172 T *cv = C.values_mut();
173
174#pragma omp parallel
175 {
176 std::vector<Idx> mark(n, -1);
177 std::vector<T> acc(n, T(0));
178 std::vector<Idx> cols;
179#pragma omp for schedule(dynamic, 64)
180 for (long i = 0; i < static_cast<long>(m); ++i) {
181 cols.clear();
182 for (Idx p = arp[i]; p < arp[i + 1]; ++p) {
183 const Idx r = aci[p];
184 const T a = av[p];
185 for (Idx q = brp[r]; q < brp[r + 1]; ++q) {
186 const Idx j = bci[q];
187 if (mark[j] != static_cast<Idx>(i)) {
188 mark[j] = static_cast<Idx>(i);
189 acc[j] = a * bv[q];
190 cols.push_back(j);
191 } else {
192 acc[j] += a * bv[q];
193 }
194 }
195 }
196 std::sort(cols.begin(), cols.end());
197 Idx w = rp[i];
198 for (Idx j : cols) {
199 cci[w] = j;
200 cv[w] = acc[j];
201 ++w;
202 }
203 }
204 }
205 return C;
206 }
