3// solve_milp(problem, options): the AXOS MILP solver.
4//
5// 1. presolve: the LP presolve (presolve/presolve.h: empty, fixed and
6// singleton rows / columns, duplicate rows), integrality carried to the
7// reduced columns through its column map, integer bounds rounded
8// 2. branch and cut on the reduced problem (bnb.h)
9// 3. postsolve of the best solution and a feasibility check on the ORIGINAL
10// problem (check_milp): a solution failing it is not reported
11//
12// Objective values and bounds are in minimization form (io/mps.h negates the
13// costs of a maximization problem and sets maximize): negate them for output
14// when orig.maximize is set.

26// MIP coefficient tightening (M. Savelsbergh, ORSA J. Computing 6, 1994) on
27// one-sided rows a x <= b: when a binary x_k leaves the row redundant at one of
28// its values (the largest activity then stays below b by d > 0), its
29// coefficient moves toward zero by d (and b by d when a_k > 0). The integer
30// points are unchanged; the LP relaxation gets tighter. Returns the number of
31// changes.
32inline int
33tighten_coefficients(LpProblem &p)
34{
35 const size_t m = p.rows(), n = p.cols();
36 std::vector<int32_t> rp(p.A.row_ptr(), p.A.row_ptr() + m + 1);
37 std::vector<int32_t> ci(p.A.col_ind(), p.A.col_ind() + p.A.nnz());
38 std::vector<double> va(p.A.values(), p.A.values() + p.A.nnz());
39 auto fin = [](double v) { return std::abs(v) < 1e20; };
40 int changed = 0;
41 for (size_t i = 0; i < m; ++i) {
42 const bool lf = fin(p.row_lb[i]), uf = fin(p.row_ub[i]);
43 if (lf == uf) continue; // ranged, equality or free
44 const double sgn = uf ? 1.0 : -1.0; // sgn a x <= b
45 double b = uf ? p.row_ub[i] : -p.row_lb[i];
46 double maxact = 0;
47 bool finite = true;
48 for (int32_t k = rp[i]; k < rp[i + 1] && finite; ++k) {
49 const double a = sgn * va[k];
50 const int j = ci[k];
51 const double bd = a > 0 ? p.col_ub[j] : p.col_lb[j];
52 if (!fin(bd))
53 finite = false;
54 else
55 maxact += a * bd;
56 }
57 if (!finite) continue;
58 for (int32_t k = rp[i]; k < rp[i + 1]; ++k) {
59 const int j = ci[k];
60 if (!is_int(p, j) || p.col_lb[j] != 0 || p.col_ub[j] != 1) continue;
61 const double a = sgn * va[k], tol = 1e-9 * (1 + std::abs(b));
62 if (a > 0) {
63 const double d = b - (maxact - a); // slack at x_k = 0
64 if (d > tol && d < a - tol) {
65 va[k] = sgn * (a - d);
66 b -= d;
67 maxact -= d;
68 ++changed;
69 }
70 } else if (a < 0) {
71 const double d = b - (maxact + a); // slack at x_k = 1
72 if (d > tol && d < -a - tol) {
73 va[k] = sgn * (a + d);
74 ++changed;
75 }
76 }
77 }
78 if (uf)
79 p.row_ub[i] = b;
80 else
81 p.row_lb[i] = -b;
82 }
83 if (changed) p.A = HostMatrix(m, n, rp, ci, va);
84 return changed;
85}

87inline MilpSolution
88solve_milp(const LpProblem &orig, const MilpOptions &opt)
89{
90 const auto t0 = std::chrono::steady_clock::now();
91 auto elapsed = [&] {
92 return std::chrono::duration<double>(
93 std::chrono::steady_clock::now() - t0)
94 .count();
95 };
96 MilpSolution out;
97 const std::string bad = orig.validate();
98 if (!bad.empty()) throw std::invalid_argument("solve_milp: " + bad);
99
100 LpProblem work;
101 std::unique_ptr<Presolve> pre;
102 if (opt.presolve) {
103 pre.reset(new Presolve(orig));
104 if (pre->status() == Status::Infeasible) {
105 out.status = Status::Infeasible;
106 out.seconds = elapsed();
107 return out;
108 }
109 work = pre->reduced();
110 work.is_integer.assign(work.cols(), 0);
111 if (!orig.is_integer.empty())
112 for (size_t k = 0; k < work.cols(); ++k)
113 work.is_integer[k] = orig.is_integer[pre->col_map()[k]];
114 } else {
115 work = orig;
116 if (work.is_integer.empty()) work.is_integer.assign(work.cols(), 0);
117 }
118 for (size_t j = 0; j < work.cols(); ++j) {
119 if (!work.is_integer[j]) continue;
120 work.col_lb[j] = std::ceil(work.col_lb[j] - 1e-9);
121 work.col_ub[j] = std::floor(work.col_ub[j] + 1e-9);
122 if (work.col_lb[j] > work.col_ub[j]) {
123 out.status = Status::Infeasible;
124 out.seconds = elapsed();
125 return out;
126 }
127 }
128
129 const int tightened = opt.presolve ? tighten_coefficients(work) : 0;
130 if (opt.verbose && tightened)
131 std::printf("[milp] coefficient tightening changed %d coefficients\n",
132 tightened);
133
134 MilpSolution s;
135 if (work.cols() == 0 || work.rows() == 0) {
136 // nothing left to search: every column at the bound its cost prefers
137 std::vector<double> x(work.cols());
138 bool unbounded = false;
139 for (size_t j = 0; j < work.cols(); ++j) {
140 const double c = work.c[j];
141 double v = c > 0 ? work.col_lb[j] : (c < 0 ? work.col_ub[j] : 0.0);
142 if (c == 0)
143 v = std::min(std::max(0.0, work.col_lb[j]), work.col_ub[j]);
144 if (!std::isfinite(v)) unbounded = true;
145 x[j] = v;
146 }
147 s.status = unbounded ? Status::Unbounded : Status::Optimal;
148 if (!unbounded) {
149 s.x = x;
150 s.objective = check_milp(work, x).objective;
151 s.bound = s.objective;
152 }
153 } else {
154 milp::BranchAndBound bnb(work, opt, t0);
155 s = bnb.solve();
156 }
157
158 out = s;
159 out.x.clear();
160 out.objective = kInf;
161 if (s.has_solution()) {
162 std::vector<double> x;
163 if (pre) {
164 LpSolution r;
165 r.x = s.x;
166 r.y.assign(work.rows(), 0.0);
167 r.z.assign(work.cols(), 0.0);
168 x = pre->postsolve(r).x;
169 } else {
170 x = s.x;
171 }
172 if (!orig.is_integer.empty())
173 for (size_t j = 0; j < x.size(); ++j)
174 if (orig.is_integer[j]) x[j] = std::round(x[j]);
175 const MilpCheck c = check_milp(orig, x);
176 if (c.ok(10 * opt.feas_tol, opt.int_tol)) {
177 out.x = x;
178 out.objective = c.objective;
179 } else if (opt.verbose) {
180 std::printf("[milp] postsolved solution fails the check: bounds "
181 "%.2e rows %.2e integrality %.2e\n",
182 c.bound_viol, c.row_viol, c.int_viol);
183 }
184 if (out.x.empty() && out.status == Status::Optimal)
185 out.status = Status::NumericalError;
186 }
187 out.seconds = elapsed();
188 return out;
189}
