3// Root probing on binary columns (CPU; the GPU version, which probes many
4// binaries at once, is gpu::probe_binaries in milp_cuda.h).
5//
6// Each binary x_j is propagated at 0 and at 1 (propagate.h). A probe that
7// empties the domain fixes x_j the other way (keeping that probe's
8// implications); otherwise, for every other column, the weaker of the two
9// probes' bounds holds whichever value x_j takes ("double probing") and
10// replaces the global bound when tighter. Both probes empty: infeasible. The
11// work per probe is that of its propagation: only the columns the probe changed
12// (its trail) are compared.

24struct ProbingStats {
25 int probes = 0, fixed = 0, tightened = 0;
26 bool infeasible = false;
27 double seconds = 0;
28};

30inline ProbingStats
31probe_binaries_cpu(Propagator &prop, const LpProblem &p,
32 std::vector<double> &lb, std::vector<double> &ub, double seconds)
33{
34 const auto t0 = std::chrono::steady_clock::now();
35 auto elapsed = [&] {
36 return std::chrono::duration<double>(
37 std::chrono::steady_clock::now() - t0)
38 .count();
39 };
40 ProbingStats st;
41 const size_t n = p.cols();
42 std::vector<double> wl = lb,
43 wu =
44 ub; // working domain, equal to lb/ub between probes
45 std::vector<int> mark(n, -1), c0;
46 std::vector<double> pl(n),
47 pu(n); // probe 0 bounds of the columns it changed
48 std::vector<int> changed(1);
49 std::vector<std::pair<int, std::pair<double, double>>> upd;
50 Trail tr;
51 for (size_t jj = 0; jj < n && elapsed() < seconds; ++jj) {
52 const int j = static_cast<int>(jj);
53 if (!is_int(p, jj) || lb[jj] != 0 || ub[jj] != 1) continue;
54 changed[0] = j;
55 // probe x_j = 0
56 tr.undo(0, wl, wu);
57 tr.record(j, wl[j], wu[j]);
58 wu[j] = 0;
59 const bool f0 = prop.propagate(wl, wu, &changed, &tr, 25.0);
60 c0.clear();
61 for (size_t k = 0; k < tr.col.size(); ++k) {
62 const int t = tr.col[k];
63 if (mark[t] != j) {
64 mark[t] = j;
65 c0.push_back(t);
66 }
67 }
68 for (int t : c0) {
69 pl[t] = wl[t];
70 pu[t] = wu[t];
71 }
72 tr.undo(0, wl, wu);
73 // probe x_j = 1
74 tr.record(j, wl[j], wu[j]);
75 wl[j] = 1;
76 const bool f1 = prop.propagate(wl, wu, &changed, &tr, 25.0);
77 st.probes += 2;
78 if (!f0 && !f1) {
79 tr.undo(0, wl, wu);
80 st.infeasible = true;
81 break;
82 }
83 if (!f0) { // x_j = 1: keep probe 1
84 for (int t : tr.col) {
85 lb[t] = wl[t];
86 ub[t] = wu[t];
87 }
88 tr.col.clear();
89 tr.old_lb.clear();
90 tr.old_ub.clear();
91 ++st.fixed;
92 continue;
93 }
94 if (!f1) { // x_j = 0: restore probe 0
95 tr.undo(0, wl, wu);
96 for (int t : c0) {
97 lb[t] = wl[t] = pl[t];
98 ub[t] = wu[t] = pu[t];
99 }
100 ++st.fixed;
101 continue;
102 }
103 // both feasible: the weaker bound of the two probes, where both moved
104 // it
105 upd.clear();
106 for (int t : tr.col) {
107 if (t == j || mark[t] != j) continue;
108 const double l2 = std::min(pl[t], wl[t]),
109 u2 = std::max(pu[t], wu[t]);
110 if (l2 > lb[t] + 1e-9 || u2 < ub[t] - 1e-9)
111 upd.push_back({t, {l2, u2}});
112 }
113 tr.undo(0, wl, wu);
114 for (auto &e : upd) {
115 const int t = e.first;
116 if (mark[t] == -2 - j)
117 continue; // the trail may list a column twice
118 mark[t] = -2 - j;
119 if (e.second.first > lb[t] + 1e-9) {
120 lb[t] = wl[t] = e.second.first;
121 ++st.tightened;
122 }
123 if (e.second.second < ub[t] - 1e-9) {
124 ub[t] = wu[t] = e.second.second;
125 ++st.tightened;
126 }
127 }
128 }
129 st.seconds = elapsed();
130 return st;
131}
