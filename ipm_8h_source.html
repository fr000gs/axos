3// Primal-dual interior-point method (Mehrotra predictor-corrector) for LP,
4// with a regularized quasi-definite augmented KKT system solved by the sparse
5// LDL^T solver (cuDSS on the GPU, up-looking LDL^T with AMD on the CPU).
6//
7// min c^T x s.t. A x - w = 0, lb <= x <= ub, l <= w <= u
8//
9// with z = (x, w), lambda for the equalities and bound duals sl, su. Each
10// iteration factors
11// [ -(Theta_x^{-1} + rho I) A^T ]
12// K = [ A Theta_w + delta I ]
13// (Theta = (sl/tl + su/tu)^{-1}; rows with equal bounds have Theta_w = 0),
14// whose pattern is fixed, so only the diagonal changes between iterations and
15// the symbolic analysis is done once. Both the predictor and corrector solves
16// reuse one factorization, and each solve is refined against the
17// unregularized K. Free columns need no special handling (rho keeps K
18// quasi-definite), which is why the augmented rather than the normal-equation
19// system is used.
20//
21// Limitations: no infeasibility / unboundedness certificates (a failed run
22// reports NumericalError or IterationLimit; use PDLP or LpMethod::Auto for
23// those), no crossover.

39template <template <typename> class Store = Cpu::HostStorage> class Ipm {
40 using VS = Store<double>;
41 using Vec = tensorET<1, double, VS>;
42 using Mat = Sparse::Csr<double, int32_t, Store>;
43 using Par = Parallel<typename VS::backend_type>;
44 using Ker = Sparse::Kernels<typename VS::backend_type>;
45 using Ldl = Sparse::SparseLdlt<double, int32_t, Store>;
46
47 public:
48 // Solves p (no presolve). Requires at least one row and one column, no
49 // free rows and no fixed columns (solve_lp() takes care of that).
50 LpSolution
51 solve(const LpProblem &p, const SolverOptions &opt)
52 {
53 using namespace ipm;
54 const auto t_start = std::chrono::steady_clock::now();
55 auto elapsed = [&] {
56 return std::chrono::duration<double>(
57 std::chrono::steady_clock::now() - t_start)
58 .count();
59 };
60 const size_t m = p.rows(), n = p.cols(), N = n + m, N2 = n + m;
61 if (m == 0 || n == 0)
62 throw std::invalid_argument("Ipm::solve needs rows and columns");
63 std::string bad = p.validate();
64 if (!bad.empty()) throw std::invalid_argument("Ipm::solve: " + bad);
65
66 // ---- scaling ----------------------------------------------------
67 Scaling sc;
68 LpProblem q;
69 if (opt.scaling) {
70 sc = compute_scaling(
71 p.A, opt.ruiz_iterations, opt.pock_chambolle_alpha);
72 q = apply_scaling(p, sc);
73 } else {
74 sc.row.assign(m, 1.0);
75 sc.col.assign(n, 1.0);
76 q = p;
77 }
78
79 // ---- KKT pattern (host), positions of the diagonals -------------
80 HostMatrix At = q.A.transpose();
81 std::vector<int32_t> krp(N2 + 1, 0), kci, dxp(n), dwp(m);
82 std::vector<double> kv;
83 kci.reserve(2 * q.A.nnz() + N2);
84 kv.reserve(2 * q.A.nnz() + N2);
85 for (size_t j = 0; j < n; ++j) {
86 dxp[j] = static_cast<int32_t>(kci.size());
87 kci.push_back(static_cast<int32_t>(j));
88 kv.push_back(0.0);
89 for (int k = At.row_ptr()[j]; k < At.row_ptr()[j + 1]; ++k) {
90 kci.push_back(static_cast<int32_t>(n) + At.col_ind()[k]);
91 kv.push_back(At.values()[k]);
92 }
93 krp[j + 1] = static_cast<int32_t>(kci.size());
94 }
95 for (size_t i = 0; i < m; ++i) {
96 for (int k = q.A.row_ptr()[i]; k < q.A.row_ptr()[i + 1]; ++k) {
97 kci.push_back(q.A.col_ind()[k]);
98 kv.push_back(q.A.values()[k]);
99 }
100 dwp[i] = static_cast<int32_t>(kci.size());
101 kci.push_back(static_cast<int32_t>(n + i));
102 kv.push_back(0.0);
103 krp[n + i + 1] = static_cast<int32_t>(kci.size());
104 }
105 Mat K(Sparse::Csr<double, int32_t, Cpu::HostStorage>(
106 N2, N2, krp, kci, kv));
107 Mat A(q.A);
108 const Mat &AT = A.transposed();
109 Store<int32_t> dxpos(n), dwpos(m);
110 Sparse::detail::host_to_store(dxpos, dxp.data(), n);
111 Sparse::detail::host_to_store(dwpos, dwp.data(), m);
112
113 // ---- host-side starting point --------------------------------------
114 std::vector<double> hlb(N), hub(N), hz(N), htl(N, 1.0), htu(N, 1.0),
115 hsl(N, 0.0), hsu(N, 0.0);
116 for (size_t j = 0; j < n; ++j) {
117 hlb[j] = q.col_lb[j];
118 hub[j] = q.col_ub[j];
119 }
120 for (size_t i = 0; i < m; ++i) {
121 hlb[n + i] = q.row_lb[i];
122 hub[n + i] = q.row_ub[i];
123 }
124 double cinf = 0;
125 for (double v : q.c)
126 cinf = std::max(cinf, std::abs(v));
127 // Starting scale: slacks to finite bounds and bound duals start at
128 // about M (large starts keep the early steps from being blocked).
129 const double M = 1e2;
130 const double s0 = std::max(M, cinf);
131 auto interior = [&](double lo, double hi, double target) {
132 if (lo == hi) return lo;
133 const bool lf = std::isfinite(lo), uf = std::isfinite(hi);
134 if (lf && uf) {
135 const double d = std::min(M, 0.5 * (hi - lo));
136 return std::min(std::max(target, lo + d), hi - d);
137 }
138 if (lf) return std::max(target, lo + M);
139 if (uf) return std::min(target, hi - M);
140 return target;
141 };
142 for (size_t j = 0; j < n; ++j)
143 hz[j] = interior(hlb[j], hub[j], 0.0);
144 {
145 std::vector<double> ax(m, 0.0);
146 for (size_t i = 0; i < m; ++i)
147 for (int k = q.A.row_ptr()[i]; k < q.A.row_ptr()[i + 1]; ++k)
148 ax[i] += q.A.values()[k] * hz[q.A.col_ind()[k]];
149 for (size_t i = 0; i < m; ++i)
150 hz[n + i] = interior(hlb[n + i], hub[n + i], ax[i]);
151 }
152 for (size_t j = 0; j < N; ++j) {
153 if (bl(hlb[j], hub[j])) {
154 htl[j] = hz[j] - hlb[j];
155 hsl[j] = s0;
156 }
157 if (bu(hlb[j], hub[j])) {
158 htu[j] = hub[j] - hz[j];
159 hsu[j] = s0;
160 }
161 }
162
163 auto upload = [](const std::vector<double> &h) {
164 tensorET<1, double> t({h.size()}, 0.0);
165 std::memcpy(t.data, h.data(), h.size() * sizeof(double));
166 return Vec(t);
167 };
168 Vec z = upload(hz), tl = upload(htl), tu = upload(htu),
169 sl = upload(hsl), su = upload(hsu), lbz = upload(hlb),
170 ubz = upload(hub);
171 Vec c = upload(q.c);
172 Vec lam({m}, 0.0), dlam({m}, 0.0), dlam_a({m}, 0.0), rp({m}, 0.0),
173 ax({m}, 0.0), thw({m}, 0.0), tmpm({m}, 0.0);
174 Vec aty({n}, 0.0), tmpn({n}, 0.0);
175 Vec rd({N}, 0.0), hzv({N}, 0.0), rcl({N}, 0.0), rcu({N}, 0.0),
176 g({N}, 0.0), dz({N}, 0.0), dsl({N}, 0.0), dsu({N}, 0.0),
177 dz_a({N}, 0.0), dsl_a({N}, 0.0), dsu_a({N}, 0.0);
178 Vec rhs({N2}, 0.0), sol({N2}, 0.0), res({N2}, 0.0), dsol({N2}, 0.0);
179 Par par;
180 Ldl ldl(Sparse::Symmetry::Symmetric,
181 opt.ipm_ordering == 1 ? Sparse::Ordering::NestedDissection
182 : Sparse::Ordering::MinDegree);
183 bool analyzed = false;
184 std::unique_ptr<NormalKkt> nk; // normal-equations alternative (CPU)
185 bool use_normal = false;
186
187 // ---- norms for the relative termination tests --------------------
188 double nb = 0, nc = 0;
189 for (size_t i = 0; i < m; ++i) {
190 if (std::isfinite(p.row_lb[i])) nb += p.row_lb[i] * p.row_lb[i];
191 if (std::isfinite(p.row_ub[i])) nb += p.row_ub[i] * p.row_ub[i];
192 }
193 for (double v : p.c)
194 nc += v * v;
195 nb = std::sqrt(nb);
196 nc = std::sqrt(nc);
197 double qb = 0, qc = 0;
198 for (size_t i = 0; i < m; ++i) {
199 if (std::isfinite(q.row_lb[i])) qb += q.row_lb[i] * q.row_lb[i];
200 if (std::isfinite(q.row_ub[i])) qb += q.row_ub[i] * q.row_ub[i];
201 }
202 for (double v : q.c)
203 qc += v * v;
204 qb = std::sqrt(qb);
205 qc = std::sqrt(qc);
206
207 double rho = 1e-8, delta = 1e-8; // static primal / dual regularization
208 const int ref_steps = 3; // refinement steps per KKT solve
209 double tol_eff = std::min({opt.eps_primal, opt.eps_dual, opt.eps_gap});
210 int tightened = 0, stall = 0;
211
212 auto finish = [&](Status st, long iters) {
213 tensorET<1, double> hx(z), hy(lam);
214 std::vector<double> xv(hx.data, hx.data + n),
215 yv(hy.data, hy.data + m);
216 unscale_solution(xv, yv, sc);
217 LpSolution s = evaluate_solution(p, xv, yv);
218 s.status = st;
219 s.iterations = iters;
220 s.seconds = elapsed();
221 return s;
222 };
223
224 // Solve K sol = r with the current factorization, refining against the
225 // unregularized system.
226 double last_ref_res = 0;
227 auto solve_kkt = [&](Vec &r, Vec &out) {
228 if (use_normal)
229 nk->solve(r.data, out.data);
230 else
231 ldl.solve(r, out);
232 for (int step = 0; step < ref_steps; ++step) {
233 Ker::spmv(A, out.data, tmpm.data, 1.0, 0.0);
234 Ker::spmv(AT, out.data + n, tmpn.data, 1.0, 0.0);
235 par.for_each(
236 N2, RefineRes{res.data, r.data, out.data, tmpn.data,
237 tmpm.data, hzv.data, thw.data, n});
238 par.zero();
239 par.reduce(0, N2, SqNorm{res.data});
240 par.reduce(1, N2, SqNorm{r.data});
241 const double *s = par.fetch();
242 last_ref_res = std::sqrt(s[0]) / (1 + std::sqrt(s[1]));
243 if (std::sqrt(s[0]) <= 1e-13 * (1 + std::sqrt(s[1]))) return;
244 if (use_normal)
245 nk->solve(res.data, dsol.data);
246 else
247 ldl.solve(res, dsol);
248 par.for_each(N2, Axpy{out.data, dsol.data, 1.0});
249 }
250 };
251
252 for (long it = 0; it <= opt.ipm_max_iterations; ++it) {
253 // ---- residuals, objectives, complementarity ---------------
254 Ker::spmv(A, z.data, ax.data, 1.0, 0.0);
255 Ker::spmv(AT, lam.data, aty.data, 1.0, 0.0);
256 par.for_each(m, ResP{rp.data, z.data + n, ax.data});
257 par.for_each(N, ResD{rd.data, c.data, aty.data, lam.data, sl.data,
258 su.data, lbz.data, ubz.data, n});
259 par.zero();
260 par.reduce(0, N,
261 Stats{rd.data, z.data, c.data, lam.data, tl.data, tu.data,
262 sl.data, su.data, lbz.data, ubz.data, n});
263 par.reduce(5, m, SqNorm{rp.data});
264 const double *s = par.fetch();
265 const double rd2 = s[0], cx = s[1], dobj = s[2] + p.offset,
266 comp = s[3], ncomp = s[4], rp2 = s[5];
267 const double pobj = cx + p.offset;
268 const double mu = ncomp > 0 ? comp / ncomp : 0.0;
269 const double rel_p = std::sqrt(rp2) / (1 + qb);
270 const double rel_d = std::sqrt(rd2) / (1 + qc);
271 const double rel_g =
272 std::abs(pobj - dobj) / (1 + std::abs(pobj) + std::abs(dobj));
273 if (opt.verbose)
274 std::printf(
275 "[ipm] it %-3ld pobj % .8e dobj % .8e rp %.1e rd %.1e "
276 "gap %.1e mu %.1e\n",
277 it, pobj, dobj, rel_p, rel_d, rel_g, mu);
278 if (!std::isfinite(pobj) || !std::isfinite(dobj) ||
279 !std::isfinite(mu))
280 return finish(Status::NumericalError, it);
281
282 if (rel_p <= tol_eff && rel_d <= tol_eff && rel_g <= tol_eff) {
283 LpSolution cand = finish(Status::Optimal, it);
284 // accept only if the ORIGINAL problem meets the tolerances
285 const double op = cand.primal_residual / (1 + nb);
286 const double od = cand.dual_residual / (1 + nc);
287 const double og =
288 cand.gap / (1 + std::abs(cand.primal_objective) +
289 std::abs(cand.dual_objective));
290 const bool trusted =
291 cand.error_bound <=
292 10 * opt.eps_gap * (1 + std::abs(cand.primal_objective));
293 if (op <= opt.eps_primal && od <= opt.eps_dual &&
294 og <= opt.eps_gap && trusted)
295 return cand;
296 if (tightened >= 3) {
297 // still not trustworthy: do not claim optimality
298 cand.status = Status::NumericalError;
299 return cand;
300 }
301 tol_eff *= 0.05;
302 ++tightened;
303 continue;
304 }
305 if (it >= opt.ipm_max_iterations)
306 return finish(Status::IterationLimit, it);
307 if (elapsed() > opt.time_limit)
308 return finish(Status::TimeLimit, it);
309
310 if (mu > 0)
311 par.for_each(N, Recenter{sl.data, su.data, tl.data, tu.data,
312 lbz.data, ubz.data, mu, 1e10});
313
314 // ---- factor the KKT matrix ---------------------------------
315 par.for_each(N, BuildTheta{hzv.data, tl.data, tu.data, sl.data,
316 su.data, lbz.data, ubz.data});
317 bool factored = false;
318 for (int attempt = 0; attempt < 5 && !factored; ++attempt) {
319 par.for_each(
320 N, ScatterKkt{K.values_mut(), thw.data, dxpos.data(),
321 dwpos.data(), hzv.data, rho, delta, n});
322 if (!analyzed) {
323 const double ta = elapsed();
324 analyzed = true;
325 double kkt_flops = 0;
326 bool aug_analyzed = false;
327 if constexpr (std::is_same_v<Store<double>,
328 Cpu::HostStorage<double>>) {
329 // the normal equations are analyzed first: when cheap
330 // they are used without the (much larger) augmented
331 // analysis
332 if (opt.ipm_normal >= 0) {
333 nk.reset(new NormalKkt(
334 q, opt.ipm_ordering == 1
335 ? Sparse::Ordering::NestedDissection
336 : Sparse::Ordering::MinDegree));
337 if (opt.verbose)
338 std::printf("[ipm] normal pattern built at "
339 "%.0f ms, eligible %d\n",
340 (elapsed() - ta) * 1000,
341 (int)nk->eligible());
342 bool nk_ok = nk->eligible();
343 if (nk_ok) {
344 try {
345 nk->analyze();
346 } catch (const std::bad_alloc &) {
347 nk_ok = false;
348 }
349 }
350 if (nk_ok) {
351 if (opt.verbose)
352 std::printf(
353 "[ipm] normal analyzed at %.0f ms\n",
354 (elapsed() - ta) * 1000);
355 kkt_flops = nk->factor_flops();
356 if (opt.ipm_normal == 1 || kkt_flops < 2e9 ||
357 (opt.ipm_max_flops > 0 &&
358 kkt_flops > 50 * opt.ipm_max_flops)) {
359 // cheap, or hopeless (the augmented system
360 // has about the same fill): no need to
361 // analyze it as well
362 use_normal = true;
363 } else {
364 ldl.analyze(K);
365 aug_analyzed = true;
366 if (opt.verbose)
367 std::printf(
368 "[ipm] normal equations: %.3e "
369 "flops vs augmented %.3e\n",
370 kkt_flops, ldl.factor_flops());
371 if (kkt_flops < 0.5 * ldl.factor_flops())
372 use_normal = true;
373 else
374 kkt_flops = ldl.factor_flops();
375 }
376 }
377 }
378 if (!use_normal && !aug_analyzed) {
379 try {
380 ldl.analyze(K);
381 kkt_flops = ldl.factor_flops();
382 } catch (const std::bad_alloc
383 &) { // factors do not fit in memory
384 LpSolution ns = finish(Status::NotSolved, 0);
385 ns.factor_flops = 1e300;
386 return ns;
387 }
388 }
389 } else {
390 ldl.analyze(K);
391 }
392 if constexpr (std::is_same_v<Store<double>,
393 Cpu::HostStorage<double>>) {
394 if (opt.verbose)
395 std::printf("[ipm] est. factor flops %.3e (%s)\n",
396 kkt_flops,
397 use_normal ? "normal equations" : "augmented");
398 if (opt.ipm_max_flops > 0 &&
399 kkt_flops > opt.ipm_max_flops) {
400 if (opt.verbose)
401 std::printf("[ipm] factorization needs %.2e "
402 "flops (limit %.2e): giving up\n",
403 kkt_flops, opt.ipm_max_flops);
404 LpSolution ns = finish(Status::NotSolved, 0);
405 ns.factor_flops = kkt_flops;
406 return ns;
407 }
408 }
409 if (opt.verbose)
410 std::printf(
411 "[ipm] KKT size %zu, nnz(L) %zu, analyze %.0f ms\n",
412 N2,
413 use_normal ? nk->factor_nnz() : ldl.factor_nnz(),
414 (elapsed() - ta) * 1000);
415 }
416 const double tf = elapsed();
417 if (use_normal) {
418 if constexpr (std::is_same_v<Store<double>,
419 Cpu::HostStorage<double>>)
420 factored =
421 nk->factorize(hzv.data, thw.data, rho, delta);
422 } else {
423 factored = ldl.factorize(K);
424 }
425 if (opt.verbose)
426 std::printf("[ipm] factorize %.0f ms\n",
427 (elapsed() - tf) * 1000);
428 if (opt.verbose && std::getenv("AXOS_MF_PROFILE")) {
429 if (use_normal)
430 nk->print_profile();
431 else
432 ldl.print_profile();
433 }
434 if (!factored) {
435 rho *= 100;
436 delta *= 100;
437 }
438 }
439 if (!factored) return finish(Status::NumericalError, it);
440
441 // ---- predictor -----------------------------------------------
442 par.for_each(N, CompAff{rcl.data, rcu.data, tl.data, tu.data,
443 sl.data, su.data, lbz.data, ubz.data});
444 par.for_each(N, BuildG{g.data, rd.data, rcl.data, rcu.data, tl.data,
445 tu.data, lbz.data, ubz.data});
446 par.for_each(N2, PackRhs{rhs.data, g.data, rp.data, thw.data, n});
447 solve_kkt(rhs, sol);
448 par.for_each(
449 N, Unpack{dz_a.data, dsl_a.data, dsu_a.data, dlam_a.data,
450 sol.data, g.data, thw.data, rcl.data, rcu.data, tl.data,
451 tu.data, sl.data, su.data, lbz.data, ubz.data, n});
452
453 par.zero();
454 par.set_slot(0, 1e300);
455 par.set_slot(1, 1e300);
456 par.reduce_min(
457 0, N, StepP{tl.data, tu.data, dz_a.data, lbz.data, ubz.data});
458 par.reduce_min(1, N,
459 StepD{sl.data, su.data, dsl_a.data, dsu_a.data, lbz.data,
460 ubz.data});
461 const double *sa = par.fetch();
462 const double ap_aff = std::min(1.0, sa[0]),
463 ad_aff = std::min(1.0, sa[1]);
464 par.zero();
465 par.reduce(0, N,
466 MuAff{tl.data, tu.data, sl.data, su.data, dz_a.data, dsl_a.data,
467 dsu_a.data, lbz.data, ubz.data, ap_aff, ad_aff});
468 const double mu_aff = ncomp > 0 ? par.fetch()[0] / ncomp : 0.0;
469 double sigma = mu > 0 ? std::pow(mu_aff / mu, 3.0) : 0.0;
470 sigma = std::min(1.0, std::max(sigma, 0.0));
471
472 // ---- corrector -----------------------------------------------
473 par.for_each(N, CompCorr{rcl.data, rcu.data, tl.data, tu.data,
474 sl.data, su.data, lbz.data, ubz.data, dz_a.data,
475 dsl_a.data, dsu_a.data, sigma * mu});
476 par.for_each(N, BuildG{g.data, rd.data, rcl.data, rcu.data, tl.data,
477 tu.data, lbz.data, ubz.data});
478 par.for_each(N2, PackRhs{rhs.data, g.data, rp.data, thw.data, n});
479 solve_kkt(rhs, sol);
480 par.for_each(
481 N, Unpack{dz.data, dsl.data, dsu.data, dlam.data, sol.data,
482 g.data, thw.data, rcl.data, rcu.data, tl.data, tu.data,
483 sl.data, su.data, lbz.data, ubz.data, n});
484
485 par.zero();
486 par.set_slot(0, 1e300);
487 par.set_slot(1, 1e300);
488 par.reduce_min(
489 0, N, StepP{tl.data, tu.data, dz.data, lbz.data, ubz.data});
490 par.reduce_min(1, N,
491 StepD{
492 sl.data, su.data, dsl.data, dsu.data, lbz.data, ubz.data});
493 const double *sb = par.fetch();
494 const double eta = 0.995;
495 const double ap = std::min(1.0, eta * sb[0]),
496 ad = std::min(1.0, eta * sb[1]);
497 if (opt.verbose)
498 std::printf("[ipm] ap %.2e ad %.2e sigma %.2e "
499 "refine-res %.1e rho %.0e\n",
500 ap, ad, sigma, last_ref_res, rho);
501 if (!(ap > 1e-10) && !(ad > 1e-10)) {
502 if (++stall >= 3) return finish(Status::NumericalError, it);
503 } else {
504 stall = 0;
505 }
506 par.for_each(N,
507 UpdatePrimalDual{z.data, tl.data, tu.data, sl.data, su.data,
508 dz.data, dsl.data, dsu.data, lbz.data, ubz.data, ap, ad});
509 par.for_each(m, Axpy{lam.data, dlam.data, ad});
510 }
511 return finish(Status::IterationLimit, opt.ipm_max_iterations);
512 }
