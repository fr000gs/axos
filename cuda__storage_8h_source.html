3// Cuda::CudaStorage<T>: a flat array of T in device memory, owned or
4// borrowed, and the Cuda::Backend tag with its transfer functions
5// (docs/TENSOR_SPEC.md §3-4). Same interface and copy semantics as
6// Cpu::HostStorage (see storage/host_storage.h):
7//
8// - owning memory comes from GPUMemoryPool::get() and goes back to it; there
9// is no cudaMalloc/cudaFree per object in steady state
10// - (n, x) fills on the device: one cudaMemset when every byte of x is the
11// same (0, 0.0, -1), otherwise a small seed block doubled by
12// device-to-device copies (no kernel, so plain g++ can compile this)
13// - deep copies are device-to-device; constructors and assignments are
14// ordered on the default stream, so later default-stream work sees the
15// data in place
16//
17// Without AXOS_ENABLE_CUDA (and outside nvcc) the type still exists with the
18// same interface, so host-only builds can instantiate templates that mention
19// it, but it can never allocate: doing so throws std::runtime_error.

67inline void
68copy2d(void *dst, size_t dpitch, const void *src, size_t spitch, size_t width,
69 size_t height, cudaMemcpyKind kind)
70{
71 if (width == 0 || height == 0) return;
72 if (height == 1 || (dpitch == width && spitch == width))
73 copy(dst, src, width * height, kind); // contiguous: no pitch limits
74 else
75 check(cudaMemcpy2D(dst, dpitch, src, spitch, width, height, kind),
76 "cudaMemcpy2D");
77}

79template <class T>
80void
81fill(T *p, size_t n, const T &x)
82{
83 if (n == 0) return;
84 unsigned char byte;
85 if (AXOS::detail::uniform_bytes(x, byte)) {
86 check(cudaMemset(p, byte, n * sizeof(T)), "cudaMemset");
87 return;
88 }
89 const size_t seed = std::min<size_t>(n, 4096);
90 std::vector<T> h(seed, x);
91 copy(p, h.data(), seed * sizeof(T), cudaMemcpyHostToDevice);
92 for (size_t done = seed; done < n;) {
93 const size_t c = std::min(done, n - done);
94 copy(p + done, p, c * sizeof(T), cudaMemcpyDeviceToDevice);
95 done += c;
96 }
97}
98#endif

100// Visits the 2-D planes (last two dimensions) of t in row-major logical order:
101// f(offset, flat, height, width, pitch), with the plane starting at
102// t.data + offset, holding logical elements [flat, flat + height * width), rows
103// `pitch` elements apart. A contiguous tensor is a single plane.
104template <class Tensor, class F>
105void
106for_each_plane(const Tensor &t, F &&f)
107{
108 const size_t n = t.size();
109 if (n == 0) return;
110 constexpr int D = Tensor::rank;
111 if constexpr (D == 1) {
112 f(size_t(0), size_t(0), size_t(1), n, n);
113 } else {
114 if (AXOS::detail::tensor_is_contiguous(t)) {
115 f(size_t(0), size_t(0), size_t(1), n, n);
116 return;
117 }
118 const size_t h = t.size(D - 2), w = t.size(D - 1);
119 const size_t pitch = t.stride(D - 2), plane = h * w;
120 std::array<size_t, D> idx{};
121 for (size_t flat = 0; flat < n; flat += plane) {
122 size_t off = 0;
123 for (int d = 0; d < D - 2; ++d)
124 off += idx[d] * t.stride(d);
125 f(off, flat, h, w, pitch);
126 for (int d = D - 3; d >= 0; --d) {
127 if (++idx[d] < t.size(d)) break;
128 idx[d] = 0;
129 }
130 }
131 }
132}

136// Backend tag for device memory. All transfers are blocking and move exactly
137// t.size() elements (views copied in row-major logical order). read_element is
138// a one-element device-to-host copy (about 10 us): do not use it in loops.
139struct Backend {
140 static constexpr bool is_host = false;
141
142 template <class Tensor>
143 static auto
144 read_element(const Tensor &t, size_t i)
145 {
146 using T =
147 std::remove_cv_t<std::remove_reference_t<decltype(t.data[0])>>;
148 T v{};
149#if defined(AXOS_CUDA_STORAGE)
150 cuda_detail::copy(&v, t.data + i, sizeof(T), cudaMemcpyDeviceToHost);
151#else
152 (void)t, (void)i;
153 cuda_detail::no_cuda();
154#endif
155 return v;
156 }
157
158 // dst and src have the same shape.
159 template <class Tensor>
160 static void
161 copy_data(Tensor &dst, const Tensor &src)
162 {
163#if defined(AXOS_CUDA_STORAGE)
164 using T = std::remove_reference_t<decltype(*dst.data)>;
165 if (AXOS::detail::tensor_is_contiguous(dst) &&
166 AXOS::detail::tensor_is_contiguous(src)) {
167 cuda_detail::copy(dst.data, src.data, src.size() * sizeof(T),
168 cudaMemcpyDeviceToDevice);
169 return;
170 }
171 cuda_detail::for_each_plane(src,
172 [&](size_t off, size_t flat, size_t h, size_t w, size_t pitch) {
173 size_t doff = flat, dpitch = w;
174 if (!AXOS::detail::tensor_is_contiguous(dst)) {
175 doff = AXOS::detail::logical_offset(dst, flat);
176 dpitch = dst.stride(Tensor::rank - 2);
177 }
178 cuda_detail::copy2d(dst.data + doff, dpitch * sizeof(T),
179 src.data + off, pitch * sizeof(T), w * sizeof(T), h,
180 cudaMemcpyDeviceToDevice);
181 });
182#else
183 (void)dst, (void)src;
184 cuda_detail::no_cuda();
185#endif
186 }

188 template <class Tensor, class U>
189 static void
190 copy_to_host(const Tensor &t, U *host_dst)
191 {
192#if defined(AXOS_CUDA_STORAGE)
193 using T =
194 std::remove_cv_t<std::remove_reference_t<decltype(t.data[0])>>;
195 if constexpr (!std::is_same_v<T, U>) {
196 std::vector<T> tmp(t.size());
197 copy_to_host(t, tmp.data());
198 AXOS::detail::convert_n(host_dst, tmp.data(), tmp.size());
199 } else {
200 cuda_detail::for_each_plane(t,
201 [&](size_t off, size_t flat, size_t h, size_t w, size_t pitch) {
202 cuda_detail::copy2d(host_dst + flat, w * sizeof(T),
203 t.data + off, pitch * sizeof(T), w * sizeof(T), h,
204 cudaMemcpyDeviceToHost);
205 });
206 }
207#else
208 (void)t, (void)host_dst;
209 cuda_detail::no_cuda();
210#endif
211 }
212
213 template <class Tensor, class U>
214 static void
215 copy_from_host(Tensor &t, const U *host_src)
216 {
217#if defined(AXOS_CUDA_STORAGE)
218 using T =
219 std::remove_cv_t<std::remove_reference_t<decltype(t.data[0])>>;
220 if constexpr (!std::is_same_v<T, U>) {
221 std::vector<T> tmp(t.size());
222 AXOS::detail::convert_n(tmp.data(), host_src, tmp.size());
223 copy_from_host(t, static_cast<const T *>(tmp.data()));
224 } else {
225 cuda_detail::for_each_plane(t,
226 [&](size_t off, size_t flat, size_t h, size_t w, size_t pitch) {
227 cuda_detail::copy2d(t.data + off, pitch * sizeof(T),
228 host_src + flat, w * sizeof(T), w * sizeof(T), h,
229 cudaMemcpyHostToDevice);
230 });
231 }
232#else
233 (void)t, (void)host_src;
234 cuda_detail::no_cuda();
235#endif
236 }

238 // Sets every element of t (also of a view) to x.
239 template <class Tensor, class U>
240 static void
241 fill(Tensor &t, const U &x)
242 {
243#if defined(AXOS_CUDA_STORAGE)
244 using T = std::remove_reference_t<decltype(*t.data)>;
245 const T v = static_cast<T>(x);
246 if (AXOS::detail::tensor_is_contiguous(t)) {
247 cuda_detail::fill(t.data, t.size(), v);
248 return;
249 }
250 std::vector<T> plane;
251 cuda_detail::for_each_plane(t, [&](size_t off, size_t, size_t h,
252 size_t w, size_t pitch) {
253 plane.assign(h * w, v);
254 cuda_detail::copy2d(t.data + off, pitch * sizeof(T), plane.data(),
255 w * sizeof(T), w * sizeof(T), h, cudaMemcpyHostToDevice);
256 });
257#else
258 (void)t, (void)x;
259 cuda_detail::no_cuda();
260#endif
261 }

264template <typename T> class CudaStorage {

287 CudaStorage(T *raw, size_t n, bool own = false) noexcept
288 : ptr_(raw), n_(raw ? n : 0), own_(own && raw != nullptr)
289 {

291
292 CudaStorage(const CudaStorage &o)
293 {
294 if (o.own_) {
295 ptr_ = allocate_raw(o.n_);
296 n_ = o.n_;
297 own_ = n_ > 0;
298 copy_device(ptr_, o.ptr_, n_);
299 } else {
300 ptr_ = o.ptr_;
301 n_ = o.n_;
302 }
303 }
304
305 CudaStorage &
306 operator=(const CudaStorage &o)
307 {
308 if (this == &o) return *this;
309 if (o.own_ && own_ && n_ == o.n_) { // same-size deep copy: reuse
310 copy_device(ptr_, o.ptr_, n_);
311 return *this;
312 }
313 CudaStorage tmp(o);
314 swap(tmp);
315 return *this;
316 }
317
318 CudaStorage(CudaStorage &&o) noexcept
319 : ptr_(std::exchange(o.ptr_, nullptr)), n_(std::exchange(o.n_, 0)),
320 own_(std::exchange(o.own_, false))
321 {
322 }
323
324 CudaStorage &
325 operator=(CudaStorage &&o) noexcept
326 {
327 if (this != &o) {
328 release();
329 ptr_ = std::exchange(o.ptr_, nullptr);
330 n_ = std::exchange(o.n_, 0);
331 own_ = std::exchange(o.own_, false);
332 }
333 return *this;
334 }

369 // The allocator behind owning storages (the GPU pool).
370 static T *
371 allocate_raw(size_t n)
372 {
373 if (n == 0) return nullptr;
374#if defined(AXOS_CUDA_STORAGE)
375 if (n > std::numeric_limits<size_t>::max() / sizeof(T))
376 throw std::bad_alloc();
377 return static_cast<T *>(GPUMemoryPool::get().allocate(n * sizeof(T)));
378#else
379 cuda_detail::no_cuda();
380#endif
381 }
382
383 static void
384 free_raw(T *p, size_t n) noexcept
385 {
386#if defined(AXOS_CUDA_STORAGE)
387 if (p) GPUMemoryPool::get().deallocate(p, n * sizeof(T));
388#else
389 (void)p, (void)n;
390#endif
391 }
392
393 private:
394 static void
395 copy_device(T *dst, const T *src, size_t n)
396 {
397#if defined(AXOS_CUDA_STORAGE)
398 cuda_detail::copy(dst, src, n * sizeof(T), cudaMemcpyDeviceToDevice);
399#else
400 (void)dst, (void)src, (void)n;
401 if (n) cuda_detail::no_cuda();
402#endif
403 }
404
405 void
406 release() noexcept
407 {
408 if (own_) free_raw(ptr_, n_);
409 ptr_ = nullptr;
410 n_ = 0;
411 own_ = false;
412 }
413
414 T *ptr_ = nullptr;
415 size_t n_ = 0;
416 bool own_ = false;
417};
