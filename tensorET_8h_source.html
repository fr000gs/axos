3// tensorET<DIM, T, Storage>: a dense, row-major, rank-DIM array whose
4// elements live in the memory of Storage (docs/TENSOR_SPEC.md §5).
5//
6// tensorET<1, double> v({n}, 0.0); // host vector, zero-filled
7// tensorET<2, float> A({m, k}); // uninitialized (no page
8// touched) tensorET<1, double, Cuda::CudaStorage<double>> d(v); // one
9// upload tensorET<1, double> h(d); // one
10// download
11//
12// Layout and ownership:
13// - `data` (a public field) points at element 0 in the storage's memory
14// space; it always equals storage_.data(). Owning tensors hold their own
15// buffer; views (slice, reshape, the raw-pointer constructor, Csr's
16// values_view) hold a borrowing storage and are valid while the memory
17// they came from lives.
18// - Copies follow the storage: an owning tensor deep-copies, a view stays a
19// view of the same memory. Moves steal the buffer and leave the source
20// empty (size() == 0, data == nullptr).
21// - Copying between storages (host <-> device) allocates in the target's
22// storage and transfers once, blocking.
23// - The innermost stride is always 1; views keep their parent's strides.
24//
25// Element access on the host: operator[] / value(i) (flat buffer index, no
26// bounds check) and operator()(i0, ..., i_{DIM-1}). On the device only the
27// const by-value forms exist, each a device-to-host transfer; writable
28// element access does not compile (per-element round trips in loops are a
29// performance bug).
30//
31// Host tensors are expressions (tensor/expr.h): `C = A + B * 2.0` is one
32// fused pass; see tensorMath.h for the operators and functions.

412 // ---- views (valid while the source memory lives) --------------------

414 // The box [start, start + length); strides are this tensor's.
415 tensorET
416 slice(const shape_type &start, const shape_type &length) const
417 {
418 tensorET v;
419 size_t off = 0;
420 for (int d = 0; d < DIM; ++d)
421 off += start[d] * strides_[d];
422 v.dims_ = length;
423 v.strides_ = strides_;
424 v.size_ = product_(length);
425 size_t span = 0;
426 if (v.size_) {
427 span = 1;
428 for (int d = 0; d < DIM; ++d)
429 span += (length[d] - 1) * strides_[d];
430 }
431 v.storage_ = Storage(data ? data + off : nullptr, span, false);
432 v.data = v.storage_.data();
433 return v;
434 }

436 // Rank-N view of the same (contiguous) buffer.
437 template <int N>
438 tensorET<N, T, Storage>
439 reshape(const std::array<size_t, N> &dims) const
440 {
441 size_t n = 1;
442 for (size_t e : dims)
443 n *= e;
444 if (n != size_)
445 detail::tensor_error("reshape: element count " + std::to_string(n) +
446 " != " + std::to_string(size_));
447 if (!is_contiguous())
448 detail::tensor_error("reshape: the tensor is not contiguous");
449 size_t d[N];
450 for (int k = 0; k < N; ++k)
451 d[k] = dims[k];
452 return tensorET<N, T, Storage>(data, d, false);
453 }
454
455 tensorET<1, T, Storage>
456 flatten() const
457 { return reshape<1>(std::array<size_t, 1>{size_}); }

459 // ---- domain ---------------------------------------------------------
460
461 Domain<DIM>
462 getIndex() const
463 {
464 Domain<DIM> r;
465 for (int d = 0; d < DIM; ++d) {
466 r.ext_[d] = dims_[d];
467 r.lo_[d] = custom_domain_ ? dlo_[d] : 0;
468 r.hi_[d] =
469 custom_domain_ ? dhi_[d] : static_cast<long long>(dims_[d]) - 1;
470 }
471 return r;
472 }

474 // ---- 2-D tiling helpers (host) --------------------------------------
475 // Tile (i, j) of size m covers rows [i*m, i*m+m) and columns
476 // [j*m, j*m+m), clipped to the matrix.
477
478 tensorET
479 blockView(size_t i, size_t j, size_t m) const
480 {
481 static_assert(DIM == 2, "blockView needs a matrix");
482 return slice({i * m, j * m}, {m, m});
483 }
484
485 // Writes tile (i, j), or with `transpose` the transpose of tile (j, i),
486 // into the caller's m*m row-major buffer, zero outside the matrix, and
487 // returns a non-owning m x m view of buf.
488 tensorET<2, T, Storage>
489 blockCopy(
490 size_t i, size_t j, size_t m, T *buf, bool transpose = false) const
491 {
492 static_assert(DIM == 2 && on_host, "blockCopy needs a host matrix");
493 std::fill(buf, buf + m * m, T(0));
494 const size_t r0 = (transpose ? j : i) * m, c0 = (transpose ? i : j) * m;
495 const size_t nr = r0 < dims_[0] ? std::min(m, dims_[0] - r0) : 0;
496 const size_t nc = c0 < dims_[1] ? std::min(m, dims_[1] - c0) : 0;
497 for (size_t r = 0; r < nr; ++r) {
498 const T *src = data + (r0 + r) * strides_[0] + c0;
499 if (!transpose)
500 std::copy(src, src + nc, buf + r * m);
501 else
502 for (size_t c = 0; c < nc; ++c)
503 buf[c * m + r] = src[c];
504 }
505 size_t d[2] = {m, m};
506 return tensorET<2, T, Storage>(buf, d, false);
507 }

513 static_assert(DIM == 2 && on_host, "blockCopy needs a host matrix");
514 tensorET t(shape_type{m, m});

530 // ---- random fill ----------------------------------------------------
531 // i.i.d. samples; without a seed the values are non-deterministic, with a
532 // seed they depend only on the seed (not on the thread count). Complex
533 // types draw real and imaginary parts independently from the real parts of
534 // the bounds / mean.
535
536 static tensorET
537 uniform(const shape_type &dims, T lo = T(0), T hi = T(1),
538 std::optional<std::uint64_t> seed = std::nullopt)
539 {
540 return random_(
541 dims, seed, [lo, hi](std::mt19937_64 &g, T *p, size_t n) {
542 if constexpr (std::is_integral_v<T>) {
543 std::uniform_int_distribution<T> d(lo, hi);
544 for (size_t i = 0; i < n; ++i)
545 p[i] = d(g);
546 } else if constexpr (is_complex_v<T>) {
547 using R = real_type_t<T>;
548 std::uniform_real_distribution<R> d(
549 std::real(lo), std::real(hi));
550 for (size_t i = 0; i < n; ++i) {
551 const R re = d(g);
552 p[i] = T(re, d(g));
553 }
554 } else {
555 std::uniform_real_distribution<double> d(
556 static_cast<double>(lo), static_cast<double>(hi));
557 for (size_t i = 0; i < n; ++i)
558 p[i] = static_cast<T>(d(g));
559 }
560 });
561 }
562
563 static tensorET
564 gaussian(const shape_type &dims, T mean = T(0), T stddev = T(1),
565 std::optional<std::uint64_t> seed = std::nullopt)
566 {
567 static_assert(!std::is_integral_v<T>,
568 "gaussian needs a floating-point or complex element type");
569 return random_(
570 dims, seed, [mean, stddev](std::mt19937_64 &g, T *p, size_t n) {
571 if constexpr (is_complex_v<T>) {
572 using R = real_type_t<T>;
573 std::normal_distribution<R> d(
574 std::real(mean), std::real(stddev));
575 for (size_t i = 0; i < n; ++i) {
576 const R re = d(g);
577 p[i] = T(re, d(g));
578 }
579 } else {
580 std::normal_distribution<double> d(
581 static_cast<double>(mean), static_cast<double>(stddev));
582 for (size_t i = 0; i < n; ++i)
583 p[i] = static_cast<T>(d(g));
584 }
585 });
586 }

686 void

696 AXOS_INLINE size_t
697 offset_(I... idx) const noexcept
698 {
699 const size_t ix[DIM] = {static_cast<size_t>(idx)...};
700 size_t off = ix[DIM - 1]; // innermost stride is 1
701 for (int d = 0; d < DIM - 1; ++d)
702 off += ix[d] * strides_[d];

704 }

706 // *this is owning, contiguous and has o's shape.
707 template <class S2>
708 void
709 transfer_from_(const tensorET<DIM, T, S2> &o)
710 {
711 using SB = typename S2::backend_type;
712 if (size_ == 0) return;
713 if constexpr (on_host) {
714 SB::copy_to_host(o, data);
715 } else if constexpr (is_host_storage_v<S2>) {
716 if (o.is_contiguous()) {
717 backend_type::copy_from_host(*this, o.data);
718 } else {
719 const tensorET<DIM, T, S2> h = o.clone();
720 backend_type::copy_from_host(*this, h.data);
721 }
722 } else { // between two device storages: through the host
723 const tensorET<DIM, T> h(o);
724 backend_type::copy_from_host(*this, h.data);
725 }
726 }

728 template <class S2>
729 void
730 write_back_(
731 size_t i, size_t j, size_t m, const tensorET<2, T, S2> &tile, bool add)
732 {
733 static_assert(DIM == 2 && on_host && is_host_storage_v<S2>,
734 "blockWriteBack needs host matrices");
735 const size_t r0 = i * m, c0 = j * m;
736 const size_t nr = r0 < dims_[0] ? std::min(m, dims_[0] - r0) : 0;
737 const size_t nc = c0 < dims_[1] ? std::min(m, dims_[1] - c0) : 0;
738 nr_nc_check_(tile, m);
739 for (size_t r = 0; r < nr; ++r) {
740 const T *src = tile.data + r * tile.stride(0);
741 T *dst = data + (r0 + r) * strides_[0] + c0;
742 if (add)
743 for (size_t c = 0; c < nc; ++c)
744 dst[c] += src[c];
745 else
746 std::copy(src, src + nc, dst);
747 }
748 }

754 if (tile.size(0) < m || tile.size(1) < m)
755 detail::tensor_error("block write-back: tile smaller than m x m");

758 template <class F>
759 static tensorET
760 random_(const shape_type &dims, std::optional<std::uint64_t> seed, F fill)
761 {
762 tensorET<DIM, T> h(dims);
763 std::uint64_t base;
764 if (seed) {
765 base = *seed;
766 } else {
767 std::random_device rd;
768 base = (std::uint64_t(rd()) << 32) ^ rd();
769 }
770 constexpr size_t kBlock = size_t(1) << 16; // values per engine
771 const size_t n = h.size(), nb = (n + kBlock - 1) / kBlock;
772 T *p = h.data;
773 detail::parallel_for(
774 nb, 1,
775 [&](size_t b0, size_t b1) {
776 for (size_t b = b0; b < b1; ++b) {
777 std::mt19937_64 g(
778 detail::mix_seed(base ^ detail::mix_seed(b)));
779 const size_t s = b * kBlock;
780 fill(g, p + s, std::min(n, s + kBlock) - s);
781 }
782 },
783 1);
784 if constexpr (on_host)
785 return h;
786 else
787 return tensorET(h);
788 }
