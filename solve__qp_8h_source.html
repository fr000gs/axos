3// Entry point of the QP solvers: solve_qp(problem, options) runs the chosen
4// first-order method (QpOptions::method) on the CPU or, when the build has
5// CUDA (-DAXOS_ENABLE_CUDA) and opt.use_gpu is set (or opt.auto_device finds
6// the problem large enough), on the GPU. With opt.comm (several MPI ranks,
7// -DAXOS_ENABLE_MPI) HPR-QP runs distributed, one GPU or CPU per rank.

26inline bool
27qp_gpu_available()
28{
29#if defined(AXOS_ENABLE_CUDA)
30 int count = 0;
31 return cudaGetDeviceCount(&count) == cudaSuccess && count > 0;
32#else
33 return false;
34#endif
35}

37template <class B>
38QpSolution
39solve_qp_on(const QpProblem &p, const QpOptions &opt)
40{
41 if (opt.method == QpMethod::Pdhcg) return qp::Pdhcg<B>().solve(p, opt);
42 return qp::HprQp<B>().solve(p, opt);
43}

45// The device solve_qp uses for p under opt.
46inline bool
47qp_use_gpu(const QpProblem &p, const QpOptions &opt)
48{
49 if (!opt.auto_device) return opt.use_gpu;
50 const size_t work = p.lp.A.nnz() + p.Q.nnz() + p.cols() + p.rows();
51 return work >= opt.gpu_min_work && qp_gpu_available();
52}

54inline QpSolution
55solve_qp(const QpProblem &p, const QpOptions &opt)
56{
57 if (opt.comm && opt.comm->distributed()) {
58 // over MPI ranks: HPR-QP (Auto included), on every rank's GPU or CPU
59 if (opt.method == QpMethod::Pdhcg || opt.method == QpMethod::Ipm)
60 throw std::invalid_argument(
61 "solve_qp: only HPR-QP runs over several MPI ranks");
62 QpOptions o = opt;
63 o.method = QpMethod::HprQp;
64 if (qp_use_gpu(p, o)) {
65#if defined(AXOS_ENABLE_CUDA)
66 return solve_qp_on<qp::CudaBackend>(p, o);
67#else
68 throw std::runtime_error(
69 "solve_qp: this build has no CUDA (-DAXOS_ENABLE_CUDA)");
70#endif
71 }
72 return solve_qp_on<qp::CpuBackend>(p, o);
73 }
74 if (opt.method == QpMethod::Ipm)
75 return qp::QpIpm().solve(p, opt); // CPU only
76 if (opt.method == QpMethod::Auto) {
77 QpOptions o = opt;
78 o.method = QpMethod::Ipm;
79 o.ipm_max_flops = std::min(opt.ipm_max_flops, opt.ipm_auto_flops);
80 o.time_limit =
81 0.25 * opt.time_limit; // the rest is the first-order method's
82 o.ipm_project = true;
83 const QpSolution s = qp::QpIpm().solve(p, o);
84 if (s.status == Status::Optimal || s.status == Status::Infeasible)
85 return s;
86 o = opt;
87 o.method = QpMethod::HprQp;
88 o.auto_device = true;
89 o.gpu_min_work = std::min(opt.gpu_min_work, opt.gpu_min_work_hard);
90 o.time_limit = opt.time_limit - s.seconds;
91 QpSolution t = solve_qp(p, o);
92 t.seconds +=
93 s.seconds; // the declined or failed interior-point attempt counts
94 t.setup_seconds += s.seconds;
95 t.method =
96 "auto: hpr-qp after ipm (" + std::string(to_string(s.status)) + ")";
97 return t;
98 }
99 if (qp_use_gpu(p, opt)) {
100#if defined(AXOS_ENABLE_CUDA)
101 return solve_qp_on<qp::CudaBackend>(p, opt);
102#else
103 throw std::runtime_error(
104 "solve_qp: this build has no CUDA (-DAXOS_ENABLE_CUDA)");
105#endif
106 }
107 return solve_qp_on<qp::CpuBackend>(p, opt);
108}
