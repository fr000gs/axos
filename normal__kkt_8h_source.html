3// Normal-equations solver for the interior-point KKT system (CPU).
4//
5// Instead of the augmented system
6// [ -D A^T ] [dx] [r1]
7// [ A E ] [dy] = [r2], D = Theta_x^{-1} + rho, E = Theta_w +
8// delta,
9// eliminate dx: S dy = r2 + A D^{-1} r1 with S = A D^{-1} A^T + E (symmetric
10// positive definite), then dx = D^{-1}(A^T dy - r1). Wins when A has far more
11// columns than rows and no dense columns (transportation / assignment LPs, set
12// covering): S is only m x m, factored with the multifrontal Cholesky. Not used
13// with free columns (D -> rho makes S badly conditioned) or when a column
14// touches many rows (S would be dense).

27class NormalKkt {
28 using Ldl = Sparse::SparseLdlt<double, int32_t, Cpu::HostStorage>;
29 using Mat = Sparse::Csr<double, int32_t, Cpu::HostStorage>;
30 using Vec = tensorET<1, double, Cpu::HostStorage<double>>;

32 public:
33 // Eligibility: no free columns, columns not too dense, S not too large.
34 // On success the pattern of S is built and eligible() is true.
35 NormalKkt(const LpProblem &q, Sparse::Ordering ord)
36 : ldl_(Sparse::Symmetry::SPD, ord)
37 {
38 m_ = q.rows();
39 n_ = q.cols();
40 for (size_t j = 0; j < n_; ++j)
41 if (!std::isfinite(q.col_lb[j]) && !std::isfinite(q.col_ub[j]))
42 return;
43 A_ = &q.A;
44 AT_ = q.A.transpose();
45 const int *cp = AT_.row_ptr();
46 double work = 0;
47 for (size_t j = 0; j < n_; ++j) {
48 const double c = cp[j + 1] - cp[j];
49 if (c > 60) return; // dense column
50 work += c * c;
51 }
52 if (work > 30.0 * (q.A.nnz() + m_)) return; // S would fill in too much
53 par_ = work > 2e5;
54 // pattern of S row by row (sorted, with the diagonal)
55 std::vector<int32_t> rp(m_ + 1, 0), ci;
56 std::vector<int32_t> mark(m_, -1), tmp;
57 const int *ar = A_->row_ptr(), *ac = A_->col_ind();
58 const int *tr = AT_.row_ptr(), *tc = AT_.col_ind();
59 for (size_t i = 0; i < m_; ++i) {
60 tmp.clear();
61 mark[i] = static_cast<int32_t>(i);
62 tmp.push_back(static_cast<int32_t>(i));
63 for (int k = ar[i]; k < ar[i + 1]; ++k) {
64 const int j = ac[k];
65 for (int t = tr[j]; t < tr[j + 1]; ++t) {
66 const int r = tc[t];
67 if (mark[r] != static_cast<int32_t>(i)) {
68 mark[r] = static_cast<int32_t>(i);
69 tmp.push_back(r);
70 }
71 }
72 }
73 std::sort(tmp.begin(), tmp.end());
74 ci.insert(ci.end(), tmp.begin(), tmp.end());
75 rp[i + 1] = static_cast<int32_t>(ci.size());
76 if (ci.size() > 60u * m_ + 4000000u) return; // too large
77 }
78 srp_ = rp;
79 sci_ = ci;
80 diag_.resize(m_);
81 for (size_t i = 0; i < m_; ++i)
82 for (int32_t p = rp[i]; p < rp[i + 1]; ++p)
83 if (static_cast<size_t>(ci[p]) == i) diag_[i] = p;
84 S_ = Mat(m_, m_, rp, ci, std::vector<double>(ci.size(), 0.0));
85 dinv_.assign(n_, 0.0);
86 y_ = Vec(m_);
87 rhs_ = Vec(m_);
88 ok_ = true;
89 }

111 // hz: Theta_x^{-1} for the n columns; thw: Theta_w for the m rows.
112 bool
113 factorize(const double *hz, const double *thw, double rho, double delta)
114 {
115 for (size_t j = 0; j < n_; ++j)
116 dinv_[j] = 1.0 / (hz[j] + rho);
117 const int *ar = A_->row_ptr(), *ac = A_->col_ind();
118 const double *av = A_->values();
119 const int *tr = AT_.row_ptr(), *tc = AT_.col_ind();
120 const double *tv = AT_.values();
121 double *sv = S_.values_mut();
122#pragma omp parallel if (par_)
123 {
124 std::vector<double> acc(m_, 0.0);
125#pragma omp for schedule(dynamic, 64)
126 for (long i = 0; i < static_cast<long>(m_); ++i) {
127 for (int k = ar[i]; k < ar[i + 1]; ++k) {
128 const int j = ac[k];
129 const double f = av[k] * dinv_[j];
130 for (int t = tr[j]; t < tr[j + 1]; ++t)
131 acc[tc[t]] += f * tv[t];
132 }
133 for (int32_t p = srp_[i]; p < srp_[i + 1]; ++p) {
134 sv[p] = acc[sci_[p]];
135 acc[sci_[p]] = 0.0;
136 }
137 sv[diag_[i]] += thw[i] + delta;
138 }
139 }
140 return ldl_.factorize(S_);
141 }

143 // Solves the (regularized) augmented system: rhs = [r1 (n); r2 (m)].
144 void
145 solve(const double *r, double *out)
146 {
147 const double *r1 = r, *r2 = r + n_;
148 // y := r2 + A D^{-1} r1
149 std::vector<double> t(n_);
150 for (size_t j = 0; j < n_; ++j)
151 t[j] = dinv_[j] * r1[j];
152 for (size_t i = 0; i < m_; ++i) {
153 double s = r2[i];
154 for (int k = A_->row_ptr()[i]; k < A_->row_ptr()[i + 1]; ++k)
155 s += A_->values()[k] * t[A_->col_ind()[k]];
156 rhs_.data[i] = s;
157 }
158 ldl_.solve(rhs_, y_);
159 double *dx = out, *dy = out + n_;
160 for (size_t i = 0; i < m_; ++i)
161 dy[i] = y_.data[i];
162 // dx = D^{-1} (A^T dy - r1)
163 for (size_t j = 0; j < n_; ++j) {
164 double s = 0;
165 for (int k = AT_.row_ptr()[j]; k < AT_.row_ptr()[j + 1]; ++k)
166 s += AT_.values()[k] * dy[AT_.col_ind()[k]];
167 dx[j] = dinv_[j] * (s - r1[j]);
168 }
169 }

178 Vec y_, rhs_;
179 size_t m_ = 0, n_ = 0;
180 bool ok_ = false, analyzed_ = false, par_ = false;
181};
