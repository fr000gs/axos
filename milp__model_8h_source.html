3// Mixed-integer linear programs for the MILP solver (solve_milp.h):

63struct MilpSolution {
64 Status status =
65 Status::NotSolved; // Optimal (gap closed), Infeasible, TimeLimit, ...
66 std::vector<double> x; // best solution (original columns), empty if none
67 double objective =
68 std::numeric_limits<double>::infinity(); // its objective (with offset)
69 double bound =
70 -std::numeric_limits<double>::infinity(); // proven lower bound
71 long nodes = 0;
72 long lp_iterations = 0;
73 double seconds = 0;
74 double first_solution_seconds = -1; // time of the first incumbent
75 double root_bound =

78 std::string incumbent_source; // heuristic that found the final incumbent
79 std::string
80 notes; // root statistics ("probing_gpu=probes/fixed/tightened;...")
81
82 bool
83 has_solution() const
84 { return !x.empty(); }
85 double
86 gap() const
87 {
88 if (!has_solution()) return std::numeric_limits<double>::infinity();
89 const double d = objective - bound;
90 return d <= 0
91 ? 0.0
92 : d / std::max({std::abs(objective), std::abs(bound), 1e-9});
93 }
94};

96// Feasibility of x for p: the largest relative violation of a bound, a row
97// (relative to 1 + |rhs|) and integrality. Returns the objective in obj.
98struct MilpCheck {
99 double bound_viol = 0, row_viol = 0, int_viol = 0, objective = 0;
100 bool
101 ok(double feas_tol, double int_tol) const
102 {
103 return bound_viol <= feas_tol && row_viol <= feas_tol &&
104 int_viol <= int_tol;
105 }
106};

108inline MilpCheck
109check_milp(const LpProblem &p, const std::vector<double> &x)
110{
111 MilpCheck c;
112 const size_t n = p.cols(), m = p.rows();
113 if (x.size() != n) {
114 c.bound_viol = c.row_viol = c.int_viol =
115 std::numeric_limits<double>::infinity();
116 return c;
117 }
118 double obj = p.offset;
119 for (size_t j = 0; j < n; ++j) {
120 obj += p.c[j] * x[j];
121 const double v = std::max(p.col_lb[j] - x[j], x[j] - p.col_ub[j]);
122 if (v > 0)
123 c.bound_viol = std::max(c.bound_viol, v / (1 + std::abs(x[j])));
124 if (!p.is_integer.empty() && p.is_integer[j])
125 c.int_viol =
126 std::max(c.int_viol, std::abs(x[j] - std::round(x[j])));
127 }
128 const auto *rp = p.A.row_ptr();
129 const auto *ci = p.A.col_ind();
130 const double *va = p.A.values();
131 for (size_t i = 0; i < m; ++i) {
132 double ax = 0;
133 for (auto k = rp[i]; k < rp[i + 1]; ++k)
134 ax += va[k] * x[ci[k]];
135 if (ax < p.row_lb[i])
136 c.row_viol = std::max(
137 c.row_viol, (p.row_lb[i] - ax) / (1 + std::abs(p.row_lb[i])));
138 if (ax > p.row_ub[i])
139 c.row_viol = std::max(
140 c.row_viol, (ax - p.row_ub[i]) / (1 + std::abs(p.row_ub[i])));
141 }
142 c.objective = obj;
143 return c;
144}

146inline bool
147is_int(const LpProblem &p, size_t j)
148{ return !p.is_integer.empty() && p.is_integer[j]; }
