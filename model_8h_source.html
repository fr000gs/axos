3// Optimization problem model shared by all AXOS solvers.
4//
5// minimize c^T x + offset
6// subject to row_lb <= A x <= row_ub
7// col_lb <= x <= col_ub
8//
9// Any bound may be +-infinity (Solver::kInf). Equalities have row_lb == row_ub.
10// A maximization problem is stored negated (c -> -c, offset -> -offset) with
11// `maximize` set, so results can be reported in the original sense.
12//
13// Duals follow the convention used throughout the solvers:
14// z = c - A^T y (reduced costs); y_i >= 0 only at a lower row bound, <= 0
15// only at an upper one; z_j >= 0 only at a lower column bound, <= 0 only at
16// an upper one.
17// dual objective = offset + sum_i (y_i > 0 ? row_lb_i : row_ub_i) * y_i
18// + sum_j (z_j > 0 ? col_lb_j : col_ub_j) * z_j

43inline const char *
44to_string(Status s)

55 case Status::IterationLimit:

57 case Status::TimeLimit:

67struct LpProblem {
68 std::string name;
69 HostMatrix A; // rows x cols
70 std::vector<double> c, col_lb, col_ub, row_lb, row_ub;
71 double offset = 0;
72 bool maximize = false;
73 std::vector<uint8_t> is_integer; // empty for a pure LP
74 std::vector<std::string> row_names, col_names;
75
76 size_t
77 rows() const
78 { return A.rows(); }
79 size_t
80 cols() const
81 { return A.cols(); }
82 bool
83 has_integers() const
84 {
85 for (auto v : is_integer)
86 if (v) return true;
87 return false;
88 }
89
90 // Empty string when the problem is well formed.
91 std::string
92 validate() const
93 {
94 std::ostringstream e;
95 const size_t m = rows(), n = cols();
96 if (c.size() != n || col_lb.size() != n || col_ub.size() != n)
97 e << "column vector sizes must equal cols(); ";
98 if (row_lb.size() != m || row_ub.size() != m)
99 e << "row vector sizes must equal rows(); ";
100 if (!is_integer.empty() && is_integer.size() != n)
101 e << "is_integer must be empty or have cols() entries; ";
102 if (!e.str().empty()) return e.str();
103 std::string v = A.validate();
104 if (!v.empty()) return "matrix: " + v;
105 for (size_t j = 0; j < n; ++j) {
106 if (std::isnan(c[j]) || std::isnan(col_lb[j]) ||
107 std::isnan(col_ub[j]))
108 return "NaN in column data at " + std::to_string(j);
109 if (col_lb[j] > col_ub[j])
110 return "column " + std::to_string(j) + " has lb > ub";
111 }
112 for (size_t i = 0; i < m; ++i) {
113 if (std::isnan(row_lb[i]) || std::isnan(row_ub[i]))
114 return "NaN in row bounds at " + std::to_string(i);
115 if (row_lb[i] > row_ub[i])
116 return "row " + std::to_string(i) + " has lb > ub";
117 }
118 return "";
119 }
120
121 // Objective value c^T x + offset for a given x (in the stored, minimized
122 // sense).
123 double
124 objective(const std::vector<double> &x) const
125 {
126 double s = offset;
127 for (size_t j = 0; j < cols(); ++j)
128 s += c[j] * x[j];
129 return s;
130 }
131};

133// Solution in the problem's stored (minimization) form.
134struct LpSolution {
135 Status status = Status::NotSolved;
136 std::vector<double> x; // primal, cols()
137 std::vector<double> y; // row duals, rows()
138 std::vector<double> z; // reduced costs, cols()
139 double primal_objective = 0;
140 double dual_objective = 0;
141 double primal_residual = 0; // ||violation of Ax in [row_lb,row_ub]||_2
142 double dual_residual = 0; // ||violation of reduced-cost sign||_2
143 double gap = 0; // |primal - dual objective|
144 // Bound on the objective error of (x, y): the gap plus each residual times
145 // the size of the variables it multiplies,
146 // gap + ||dual residual||_2 ||x||_2 + ||primal residual||_2 ||y||_2.
147 // A small relative gap with a large error_bound means the point is not
148 // trustworthy (badly scaled problem).
149 double error_bound = 0;
150 long iterations = 0;
151 double seconds = 0;
152 double factor_flops =
153 0; // Ipm (CPU): estimated flops of one KKT factorization, when known
154};

156// Algorithm of solve_lp(). Pdlp: cheap per iteration, moderate accuracy,
157// detects infeasibility (best on the GPU and huge problems); Ipm: accurate in
158// few iterations, no infeasibility certificates; Simplex: dual simplex (exact
159// vertex solutions, certificates, warm starts); Auto: Ipm, then Simplex, then
160// Pdlp, until one reports a result.
161enum class LpMethod { Pdlp, Ipm, Simplex, Auto };
162
163// Termination tolerances follow the PDLP convention:
164// ||primal residual||_2 <= eps_primal * (1 + ||finite row bounds||_2)
165// ||dual residual||_2 <= eps_dual * (1 + ||c||_2)
166// |primal obj - dual obj| <= eps_gap * (1 + |primal obj| + |dual obj|)
167struct SolverOptions {
168 LpMethod method = LpMethod::Pdlp;
169 int ipm_max_iterations = 200;
170 int ipm_ordering = 0; // 0 default (AMD), 1 nested dissection (GPU only)
171 double eps_primal = 1e-6;
172 double eps_dual = 1e-6;
173 double eps_gap = 1e-6;
174 long max_iterations = 200000;
175 double time_limit = 1e100; // seconds
176 bool verbose = false;
177 bool pdlp_polish =
178 false; // Pdlp: reach tight tolerances by polishing a 1e-4 solution
179 // (primal / dual feasibility solves)
180 bool pdlp_halpern = true; // Pdlp: reflected Halpern PDHG with fixed steps
181 // instead of adaptive-step PDHG with averaging
182 int ipm_normal = 0; // Ipm (CPU): 0 choose between normal equations and the
183 // augmented system, 1 force normal, -1 never
184 double ipm_max_flops =
185 0; // Ipm (CPU): give up (NotSolved) if one factorization exceeds this
186 // many flops; 0 = no limit
187 bool crossover = false; // Ipm: finish with a simplex crossover to a basic
188 // (vertex) solution
189 bool presolve = true;
190 bool scaling = true;
191 int ruiz_iterations = 10;
192 double pock_chambolle_alpha = 1.0; // <0 disables
193 int check_frequency = 64; // iterations between termination checks
194
195 void
196 set_tolerance(double eps)
197 { eps_primal = eps_dual = eps_gap = eps; }
198};

200// Residuals of a candidate (x, y) for `p`, computed on the host in double
201// precision. z is recomputed as c - A^T y. Used by tests and by the solvers'
202// final verification against the original problem.
203inline LpSolution
204evaluate_solution(const LpProblem &p, const std::vector<double> &x,
205 const std::vector<double> &y)
206{
207 const size_t m = p.rows(), n = p.cols();
208 LpSolution s;
209 s.x = x;
210 s.y = y;
211 s.z.assign(n, 0.0);
212 std::vector<double> ax(m, 0.0);
213 const auto *rp = p.A.row_ptr();
214 const auto *ci = p.A.col_ind();
215 const double *v = p.A.values();
216 for (size_t i = 0; i < m; ++i)
217 for (int k = rp[i]; k < rp[i + 1]; ++k) {
218 ax[i] += v[k] * x[ci[k]];
219 s.z[ci[k]] -= v[k] * y[i];
220 }
221 for (size_t j = 0; j < n; ++j)
222 s.z[j] += p.c[j];
223
224 double pr = 0, dr = 0, dual = p.offset;
225 for (size_t i = 0; i < m; ++i) {
226 double viol = std::max({p.row_lb[i] - ax[i], ax[i] - p.row_ub[i], 0.0});
227 pr += viol * viol;
228 double yi = y[i];
229 // A dual on an infinite bound is a dual infeasibility.
230 if (yi > 0 && std::isinf(p.row_lb[i]))
231 dr += yi * yi;
232 else if (yi < 0 && std::isinf(p.row_ub[i]))
233 dr += yi * yi;
234 else if (yi > 0)
235 dual += yi * p.row_lb[i];
236 else if (yi < 0)
237 dual += yi * p.row_ub[i];
238 }
239 for (size_t j = 0; j < n; ++j) {
240 double viol = std::max({p.col_lb[j] - x[j], x[j] - p.col_ub[j], 0.0});
241 pr += viol * viol;
242 double zj = s.z[j];
243 if (zj > 0 && std::isinf(p.col_lb[j]))
244 dr += zj * zj;
245 else if (zj < 0 && std::isinf(p.col_ub[j]))
246 dr += zj * zj;
247 else if (zj > 0)
248 dual += zj * p.col_lb[j];
249 else if (zj < 0)
250 dual += zj * p.col_ub[j];
251 }
252 s.primal_objective = p.objective(x);
253 s.dual_objective = dual;
254 s.primal_residual = std::sqrt(pr);
255 s.dual_residual = std::sqrt(dr);
256 s.gap = std::abs(s.primal_objective - s.dual_objective);
257 double xn = 0, yn = 0;
258 for (double v : x)
259 xn += v * v;
260 for (double v : y)
261 yn += v * v;
262 s.error_bound = s.gap + s.dual_residual * std::sqrt(xn) +
263 s.primal_residual * std::sqrt(yn);
264 return s;
265}
