3// CPU backend of the first-order QP solvers: the per-element ops of
4// qp_ops.h run in OpenMP loops (tensor/parallel.h), sparse products use the
5// CSR kernels of sparse_cpu.h. Reductions add per-chunk partial results in
6// chunk order, so results do not depend on thread timing.
7//
8// Backend interface (shared with qp_backend_cuda.h):
9// Vec vec(n, v), upload(host), download(vec, host), ptr(vec)
10// Mat upload(HostMatrix), spmv(Mat, x, y) y = A x
11// spmv_epi(name, Mat, x, args, f) f(i, (A x)_i, args) for every row i
12// map(name, n, args, f) f(i, args) for i < n
13// reduce(name, n, args, f) array of Args::K results (sum or max)
14// reduce_to(name, n, args, f, slot), copy_results(n), sync(), results()
15// the same, deferred: results()[slot + k]
16// results_dev() the slots where device ops can read them
17// run_block(key, fn) fn() (the GPU backend replays it as a graph),
18// forget(key) drops that graph; use_graphs(false): no graphs
19// epi(name, n, s, args, f) f(i, s[i], args): an epilogue on given row sums
20// through_host(v, n, fn) fn(h) on a host copy h of v[0..n), copied back

91 template <class Args, class F>
92 void
93 spmv_epi(
94 const char *, const Mat &M, const double *x, const Args &a, F f) const
95 {
96 const auto *rp = M.A.row_ptr();
97 const auto *ci = M.A.col_ind();
98 const double *v = M.A.values();
99 detail::parallel_for(M.rows(), kGrain / 8, [&](size_t b, size_t e) {
100 for (size_t i = b; i < e; ++i) {
101 double s = 0.0;
102 for (auto k = rp[i]; k < rp[i + 1]; ++k)
103 s += v[k] * x[ci[k]];
104 f(static_cast<axos_qp::idx>(i), s, a);
105 }
106 });
107 }

129 template <class Args, class F>
130 std::array<double, 8>
131 reduce(const char *, size_t n, const Args &a, F f) const
132 {
133 constexpr int K = Args::K;
134 std::array<double, 8> out{};
135 const int nt = detail::max_threads();
136 const size_t chunks = detail::chunk_count(n, kGrain, nt);
137 std::vector<std::array<double, K>> part(chunks);
138 detail::parallel_for(
139 chunks, 1,
140 [&](size_t c0, size_t c1) {
141 for (size_t c = c0; c < c1; ++c) {
142 double acc[Args::K] =
143 {}; // (MSVC: no constexpr locals in lambdas)
144 const size_t b = detail::chunk_begin(n, chunks, c, 1);
145 const size_t e = detail::chunk_begin(n, chunks, c + 1, 1);
146 for (size_t i = b; i < e; ++i)
147 f(static_cast<axos_qp::idx>(i), a, acc);
148 for (int k = 0; k < Args::K; ++k)
149 part[c][k] = acc[k];
150 }
151 },
152 1);
153 for (size_t c = 0; c < chunks; ++c)
154 for (int k = 0; k < K; ++k)
155 out[k] = ((Args::kMax >> k) & 1u) ? std::max(out[k], part[c][k])
156 : out[k] + part[c][k];
157 return out;
158 }
