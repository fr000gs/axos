3// LP-free primal heuristics for MILP.
4//
5// FeasibilityJump (B. Luteberget, G. Sartor, Math. Prog. Comp. 2023): the point
6// x stays within the bounds; a row's violation is its distance to [l, u],
7// weighted by w_i. One variable moves at a time, to its "jump value": the value
8// (integer for integer columns) minimizing the weighted violation of the rows
9// of its column, the others fixed. That one-variable function is convex and
10// piecewise linear, so its minimum is at a breakpoint (a row becoming
11// satisfied) or a bound. Moves come from a violated row; when no variable of it
12// improves, the row's weight grows (the Lagrangian part), eventually making
13// some move pay off. Every step needs one column and its rows, no LP.
14//
15// FixAndPropagate: integer columns are fixed one by one in a given order to a
16// preferred value (rounded LP value or a guess), each fix followed by domain
17// propagation (propagate.h); a fix that empties a domain is retried with the
18// other rounding once, then the heuristic gives up. The continuous columns are
19// left to an LP over the fixed integers (the caller's).

36 explicit FeasibilityJump(const LpProblem &p) : p_(p), At_(p.A.transpose())
37 {
38 isint_.assign(p.cols(), 0);
39 for (size_t j = 0; j < p.cols(); ++j)
40 isint_[j] = is_int(p, j) ? 1 : 0;

43 // Looks for a point satisfying all rows within [lb, ub], starting from x0
44 // when given (else the bound nearest zero), for at most time_limit
45 // seconds. On success x holds the point.
46 bool
47 run(const std::vector<double> &lb, const std::vector<double> &ub,
48 const std::vector<double> *x0, double time_limit,
49 std::vector<double> &x, uint64_t seed = 1)
50 {
51 const auto t0 = std::chrono::steady_clock::now();
52 auto elapsed = [&] {
53 return std::chrono::duration<double>(
54 std::chrono::steady_clock::now() - t0)
55 .count();
56 };
57 const size_t n = p_.cols(), m = p_.rows();
58 rng_.seed(seed);
59 x.assign(n, 0.0);
60 for (size_t j = 0; j < n; ++j) {
61 double v = x0 ? (*x0)[j] : 0.0;
62 v = std::min(std::max(v, lb[j]), ub[j]);
63 if (!std::isfinite(v) || std::abs(v) > 1e15)
64 v = std::abs(lb[j]) < 1e15
65 ? lb[j]
66 : (std::abs(ub[j]) < 1e15 ? ub[j] : 0.0);
67 if (isint_[j])
68 v = std::min(std::max(std::round(v), std::ceil(lb[j] - 1e-9)),
69 std::floor(ub[j] + 1e-9));
70 x[j] = v;
71 }
72 act_.assign(m, 0.0);
73 w_.assign(m, 1.0);
74 const auto *rp = p_.A.row_ptr();
75 const auto *ci = p_.A.col_ind();
76 const double *va = p_.A.values();
77 for (size_t i = 0; i < m; ++i)
78 for (auto k = rp[i]; k < rp[i + 1]; ++k)
79 act_[i] += va[k] * x[ci[k]];
80 // violated rows as an indexed set
81 pos_.assign(m, -1);
82 viol_.clear();
83 for (size_t i = 0; i < m; ++i)
84 update_violated(static_cast<int>(i));
85 long steps = 0;
86 std::vector<int> cand;
87 while (!viol_.empty()) {
88 if ((++steps & 255) == 0 && elapsed() > time_limit) return false;
89 const int r = viol_[rng_() % viol_.size()];
90 // candidate columns of row r (a sample of long rows)
91 cand.clear();
92 const auto b = rp[r], e = rp[r + 1];
93 const long len = static_cast<long>(e - b);
94 if (len <= 48) {
95 for (auto k = b; k < e; ++k)
96 cand.push_back(ci[k]);
97 } else {
98 for (int s = 0; s < 48; ++s)
99 cand.push_back(ci[b + static_cast<long>(rng_() % len)]);
100 }
101 int best_j = -1;
102 double best_v = 0, best_score = 1e-9;
103 for (int j : cand) {
104 if (lb[j] == ub[j]) continue;
105 double v, score;
106 jump(j, lb, ub, x, v, score);
107 if (score > best_score ||
108 (score == best_score && best_j >= 0 && (rng_() & 1))) {
109 best_score = score;
110 best_j = j;
111 best_v = v;
112 }
113 }
114 if (best_j < 0) {
115 w_[r] += 1.0; // local minimum for this row: weigh it up
116 continue;
117 }
118 const double delta = best_v - x[best_j];
119 x[best_j] = best_v;
120 for (auto k = At_.row_ptr()[best_j]; k < At_.row_ptr()[best_j + 1];
121 ++k) {
122 const int i = At_.col_ind()[k];
123 act_[i] += At_.values()[k] * delta;
124 update_violated(i);
125 }
126 }
127 // exact activities (drift) and a final check
128 for (size_t i = 0; i < m; ++i) {
129 double a = 0;
130 for (auto k = rp[i]; k < rp[i + 1]; ++k)
131 a += va[k] * x[ci[k]];
132 if (a < p_.row_lb[i] - 1e-6 * (1 + std::abs(p_.row_lb[i])) ||
133 a > p_.row_ub[i] + 1e-6 * (1 + std::abs(p_.row_ub[i])))
134 return false;
135 }
136 return true;
137 }
138
139 private:
140 const LpProblem &p_;
141 HostMatrix At_;
142 std::vector<uint8_t> isint_;
143 std::vector<double> act_, w_;
144 std::vector<int> pos_, viol_;
145 std::mt19937_64 rng_;
146 std::vector<double> cands_;
147
148 double
149 row_viol(int i, double a) const
150 {
151 const double l = p_.row_lb[i], u = p_.row_ub[i];
152 if (a > u) return a - u;
153 if (a < l) return l - a;
154 return 0.0;
155 }
156 void
157 update_violated(int i)
158 {
159 const double a = act_[i];
160 const bool v = a > p_.row_ub[i] + 1e-6 * (1 + std::abs(p_.row_ub[i])) ||
161 a < p_.row_lb[i] - 1e-6 * (1 + std::abs(p_.row_lb[i]));
162 if (v && pos_[i] < 0) {
163 pos_[i] = static_cast<int>(viol_.size());
164 viol_.push_back(i);
165 } else if (!v && pos_[i] >= 0) {
166 const int last = viol_.back();
167 viol_[pos_[i]] = last;
168 pos_[last] = pos_[i];
169 viol_.pop_back();
170 pos_[i] = -1;
171 }
172 }
173
174 // Weighted violation of the rows of column j with x_j = v.
175 double
176 column_cost(int j, double v, double xj) const
177 {
178 double f = 0;
179 for (auto k = At_.row_ptr()[j]; k < At_.row_ptr()[j + 1]; ++k) {
180 const int i = At_.col_ind()[k];
181 const double a = At_.values()[k];
182 f += w_[i] * row_viol(i, act_[i] + a * (v - xj));
183 }
184 return f;
185 }
186
187 // Best value of column j (jump value) and its score (decrease of the
188 // weighted violation); the minimum of a convex piecewise-linear
189 // function lies at a breakpoint or a bound.
190 void
191 jump(int j, const std::vector<double> &lb, const std::vector<double> &ub,
192 const std::vector<double> &x, double &best_v, double &score)
193 {
194 const double xj = x[j];
195 cands_.clear();
196 auto add = [&](double v) {
197 if (!std::isfinite(v)) return;
198 v = std::min(std::max(v, lb[j]), ub[j]);
199 if (std::abs(v) > 1e15) return;
200 if (isint_[j]) {
201 cands_.push_back(std::floor(v + 1e-9));
202 cands_.push_back(std::ceil(v - 1e-9));
203 } else {
204 cands_.push_back(v);
205 }
206 };
207 for (auto k = At_.row_ptr()[j]; k < At_.row_ptr()[j + 1]; ++k) {
208 const int i = At_.col_ind()[k];
209 const double a = At_.values()[k];
210 if (a == 0) continue;
211 const double rest = act_[i] - a * xj;
212 if (std::abs(p_.row_ub[i]) < 1e20) add((p_.row_ub[i] - rest) / a);
213 if (std::abs(p_.row_lb[i]) < 1e20) add((p_.row_lb[i] - rest) / a);
214 }
215 add(lb[j]);
216 add(ub[j]);
217 const double f0 = column_cost(j, xj, xj);
218 double best_f = f0;
219 best_v = xj;
220 for (double v : cands_) {
221 if (v == xj) continue;
222 if (isint_[j] && (v < lb[j] - 1e-9 || v > ub[j] + 1e-9)) continue;
223 const double f = column_cost(j, v, xj);
224 if (f < best_f - 1e-12) {
225 best_f = f;
226 best_v = v;
227 }
228 }
229 score = f0 - best_f;
230 }
231};

233// Fixes the integer columns in `order` to `want` (rounded into the current
234// domain), propagating after each fix; a fix that empties a domain is undone
235// (through the trail) and retried with the other rounding once, then the
236// heuristic gives up. lb/ub hold the final domains: the integers fixed, the
237// continuous columns propagated.
238inline bool
239fix_and_propagate(Propagator &prop, const LpProblem &p,
240 const std::vector<int> &order, const std::vector<double> &want,
241 std::vector<double> &lb, std::vector<double> &ub, double deadline_s,
242 const std::chrono::steady_clock::time_point &t0)
243{
244 std::vector<int> changed(1);
245 Trail trail;
246 long count = 0;
247 for (int j : order) {
248 if (!is_int(p, j) || lb[j] == ub[j]) continue;
249 if ((++count & 63) == 0 &&
250 std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
251 .count() > deadline_s)
252 return false;
253 const double v = std::min(std::max(std::round(want[j]), lb[j]), ub[j]);
254 const double alt =
255 want[j] >= v ? std::min(v + 1, ub[j]) : std::max(v - 1, lb[j]);
256 bool ok = false;
257 for (int attempt = 0; attempt < 2 && !ok; ++attempt) {
258 const double val = attempt == 0 ? v : alt;
259 if (attempt == 1 && val == v) break;
260 const size_t mark = trail.mark();
261 trail.record(j, lb[j], ub[j]);
262 lb[j] = ub[j] = val;
263 changed[0] = j;
264 ok = prop.propagate(lb, ub, &changed, &trail, 2.0);
265 if (!ok) trail.undo(mark, lb, ub);
266 }
267 if (!ok) return false;
268 trail.col.clear(); // keep the fixes: the trail only serves the retry
269 trail.old_lb.clear();
270 trail.old_ub.clear();
271 }
272 return true;
273}
