3// QPS reader: MPS plus a quadratic objective section (QUADOBJ, QMATRIX or
4// QSECTION), as used by the Maros-Meszaros test set. Returns a QpProblem
5// with Q stored as a full symmetric CSR matrix (duplicates summed). A
6// maximization problem is negated (c, offset and Q), matching the LP reader.

19inline QpProblem
20read_qps(std::istream &in, bool fixed_format = false)
21{
22 std::vector<QuadEntry> quad;
23 QpProblem p;
24 p.lp = read_mps(in, &quad, fixed_format);
25 const size_t n = p.lp.cols();
26 Sparse::CooBuilder<double> b(n, n);
27 b.reserve(quad.size());
28 const double sgn = p.lp.maximize ? -1.0 : 1.0;
29 for (const auto &e : quad)
30 if (e.v != 0)
31 b.add(
32 static_cast<size_t>(e.i), static_cast<size_t>(e.j), sgn * e.v);
33 p.Q = b.build(true);
34 return p;
35}

37inline QpProblem
38read_qps_file(const std::string &path)
39{
40 std::vector<char> buf(1 << 20); // large files: a bigger stream buffer
41 std::ifstream f;
42 f.rdbuf()->pubsetbuf(buf.data(), static_cast<std::streamsize>(buf.size()));
43 f.open(path, std::ios::binary);
44 if (!f) throw std::runtime_error("cannot open QPS file: " + path);
45 QpProblem p;
46 try {
47 p = read_qps(f);
48 } catch (const std::runtime_error &) {
49 // names with spaces: retry reading by column positions
50 f.clear();
51 f.seekg(0);
52 p = read_qps(f, true);
53 }
54 if (p.lp.name.empty()) {
55 const size_t s = path.find_last_of("/\\");
56 p.lp.name = path.substr(s == std::string::npos ? 0 : s + 1);
57 }
58 return p;
59}
