3// Small dense kernels for the multifrontal LDL^T (column-major, lower
4// triangle). No BLAS dependency: the rank-k update is a register-blocked
5// 8x4 micro-kernel written so the compiler vectorizes it (-O3 -march=native).
6//
7// syrk_d C -= L diag(d) L^T (lower part of C)
8// partial_ldlt factor the leading ns columns of an fs x fs front and
9// form the Schur complement in its trailing block
10// trsv kernels used by the triangular solves

55typedef double vd __attribute__((vector_size(VW * 8), aligned(8)));

57// MSVC has no vector extensions: the same 4-wide vector on AVX2 intrinsics

60struct vd {
61 __m256d x;
62 vd() : x(_mm256_setzero_pd()) {}
63 vd(__m256d y) : x(y) {}
64 double
65 operator[](int i) const
66 {

68 _mm256_store_pd(t, x);
69 return t[i];
70 }

72 operator+=(vd o)
73 {
74 x = _mm256_add_pd(x, o.x);

118// pack rows [i0, i0+mr) x cols p in [0,kc) of L (col-major, ld) into MR-wide
119// panel layout
120inline void
121pack_a(
122 const double *L, int ld, int i0, int mr, int p0, int kc, int m, double *dst)
123{
124 for (int ir = 0; ir < mr; ir += MR) {
125 const int rows = std::min(MR, m - (i0 + ir));
126 double *d = dst + static_cast<size_t>(ir) * kc;
127 for (int p = 0; p < kc; ++p) {
128 const double *src = L + static_cast<size_t>(p0 + p) * ld + i0 + ir;
129 int r = 0;
130 for (; r < rows; ++r)
131 d[p * MR + r] = src[r];
132 for (; r < MR; ++r)
133 d[p * MR + r] = 0.0;
134 }
135 }
136}
137// pack scaled columns: B[c][p] = L(j0+c, p0+p) * d[p0+p]; NR-wide panels
138inline void
139pack_b(const double *L, int ld, const double *d, int j0, int nc, int p0, int kc,
140 int m, double *dst)
141{
142 for (int jr = 0; jr < nc; jr += NR) {
143 const int cols = std::min(NR, m - (j0 + jr));
144 double *o = dst + static_cast<size_t>(jr) * kc;
145 for (int p = 0; p < kc; ++p) {
146 const double dp = d[p0 + p];
147 const double *src = L + static_cast<size_t>(p0 + p) * ld + j0 + jr;
148 int c = 0;
149 for (; c < cols; ++c)
150 o[p * NR + c] = src[c] * dp;
151 for (; c < NR; ++c)
152 o[p * NR + c] = 0.0;
153 }
154 }
155}

157inline void
158micro(int kc, const double *Ap, const double *Bp, double *C, int ldc, int i0,
159 int j0, int m)
160{
161 vd acc[NR][MRV];
162 for (int c = 0; c < NR; ++c)
163 for (int r = 0; r < MRV; ++r)
164 acc[c][r] = vd{};
165 for (int p = 0; p < kc; ++p) {
166 vd a[MRV];
167 for (int r = 0; r < MRV; ++r)
168 a[r] = *reinterpret_cast<const vd *>(Ap + r * VW);
169 for (int c = 0; c < NR; ++c) {
170 const double b = Bp[c];
171 for (int r = 0; r < MRV; ++r)
172 acc[c][r] += a[r] * b;
173 }
174 Ap += MR;
175 Bp += NR;
176 }
177 const int mr = std::min(MR, m - i0), nr = std::min(NR, m - j0);
178 if (mr == MR && i0 >= j0 + NR - 1) { // entire tile in the lower triangle
179 for (int c = 0; c < nr; ++c) {
180 double *cc = C + static_cast<size_t>(j0 + c) * ldc + i0;
181 for (int r = 0; r < MRV; ++r) {
182 vd v;
183 std::memcpy(&v, cc + r * VW, sizeof v);
184 v -= acc[c][r];
185 std::memcpy(cc + r * VW, &v, sizeof v);
186 }
187 }
188 } else {
189 for (int c = 0; c < nr; ++c) {
190 double *cc = C + static_cast<size_t>(j0 + c) * ldc + i0;
191 for (int r = 0; r < mr; ++r)
192 if (i0 + r >= j0 + c) cc[r] -= acc[c][r / VW][r % VW];
193 }
194 }
195}

197// C(lower) -= L diag(d) L^T ; L is m x k, leading dimension ldl.
198// Column blocks of NCB columns are independent tasks (private packing buffers).
199inline void
200syrk_d_packed(int m, int k, const double *L, int ldl, const double *d,
201 double *C, int ldc, bool allow_parallel)
202{
203 if (m <= 0 || k <= 0) return;
204 const double flops = static_cast<double>(m) * m * k;
205 bool par = allow_parallel && flops > 2e6;
206#ifdef _OPENMP
207 if (par && omp_in_parallel()) par = false;
208#endif
209 int ncb = NCB;
210#ifdef _OPENMP
211 if (par) { // finer tasks for load balance when there are few column blocks
212 const int T = omp_get_max_threads();
213 ncb = std::max(NR * 2, std::min(NCB, ((m / (4 * T)) / NR) * NR));
214 }
215#endif
216 const int njc = (m + ncb - 1) / ncb;
217 auto task = [&](int jb) {
218 static thread_local Aligned At, Bt;
219 Bt.reserve(static_cast<size_t>(NCB) * KC);
220 At.reserve(static_cast<size_t>(MC) * KC);
221 const int jc = jb * ncb;
222 const int nc = std::min(ncb, m - jc);
223 const int ic_begin = (jc / MC) * MC;
224 for (int pc = 0; pc < k; pc += KC) {
225 const int kc = std::min(KC, k - pc);
226 pack_b(L, ldl, d, jc, nc, pc, kc, m, Bt.p);
227 for (int ic = ic_begin; ic < m; ic += MC) {
228 const int mc = std::min(MC, m - ic);
229 pack_a(L, ldl, ic, mc, pc, kc, m, At.p);
230 for (int jr = 0; jr < nc; jr += NR) {
231 const double *Bp = Bt.p + static_cast<size_t>(jr) * kc;
232 for (int ir = 0; ir < mc; ir += MR) {
233 if (ic + ir + MR - 1 < jc + jr)
234 continue; // above the diagonal
235 micro(kc, At.p + static_cast<size_t>(ir) * kc, Bp, C,
236 ldc, ic + ir, jc + jr, m);
237 }
238 }
239 }
240 }
241 };
242#ifdef _OPENMP
243 if (par && njc > 1) {
244#pragma omp parallel for schedule(dynamic, 1)
245 for (int t = 0; t < njc; ++t)
246 task(t); // early blocks have the most work and run first
247 return;
248 }
249#endif
250 for (int jb = 0; jb < njc; ++jb)
251 task(jb);
252}
253} // namespace packed

255// C(i, j) -= sum_p L(i, p) * d(p) * L(j, p) for i >= j, all m x m, where L is
256// m x k (leading dimension ldl) and C has leading dimension ldc.
257// `work` must hold at least m * k elements (L scaled by d).
258template <typename T>
259void
260syrk_d(int m, int k, const T *L, int ldl, const T *d, T *C, int ldc, T *work,
261 bool allow_parallel)
262{
263 if (m <= 0 || k <= 0) return;
264 if constexpr (std::is_same_v<T, double>) {
265 packed::syrk_d_packed(m, k, L, ldl, d, C, ldc, allow_parallel);
266 (void)work;
267 return;
268 }
269 // W = L * diag(d), tight (leading dimension m)
270 for (int p = 0; p < k; ++p) {
271 const T dp = d[p];
272 const T *lp = L + static_cast<size_t>(p) * ldl;
273 T *wp = work + static_cast<size_t>(p) * m;
274 AXOS_OMP_SIMD
275 for (int i = 0; i < m; ++i)
276 wp[i] = lp[i] * dp;
277 }
278 const int nblocks = (m + kNR - 1) / kNR;
279 auto column_block = [&](int jb) {
280 const int j0 = jb * kNR;
281 const int nr = std::min(kNR, m - j0);
282 // rows i >= j0 only (lower triangle), in tiles of kMR
283 for (int p0 = 0; p0 < k; p0 += kKC) {
284 const int kc = std::min(kKC, k - p0);
285 for (int i0 = j0; i0 < m; i0 += kMR) {
286 const int mr = std::min(kMR, m - i0);
287 T acc[kNR][kMR] = {};
288 if (mr == kMR && nr == kNR) {
289 for (int p = 0; p < kc; ++p) {
290 const T *lp =
291 L + static_cast<size_t>(p0 + p) * ldl + i0;
292 const T *wp =
293 work + static_cast<size_t>(p0 + p) * m + j0;
294 for (int c = 0; c < kNR; ++c) {
295 const T w = wp[c];
296 AXOS_OMP_SIMD
297 for (int r = 0; r < kMR; ++r)
298 acc[c][r] += lp[r] * w;
299 }
300 }
301 } else {
302 for (int p = 0; p < kc; ++p) {
303 const T *lp =
304 L + static_cast<size_t>(p0 + p) * ldl + i0;
305 const T *wp =
306 work + static_cast<size_t>(p0 + p) * m + j0;
307 for (int c = 0; c < nr; ++c)
308 for (int r = 0; r < mr; ++r)
309 acc[c][r] += lp[r] * wp[c];
310 }
311 }
312 for (int c = 0; c < nr; ++c) {
313 T *cc = C + static_cast<size_t>(j0 + c) * ldc + i0;
314 for (int r = 0; r < mr; ++r)
315 if (i0 + r >= j0 + c) cc[r] -= acc[c][r];
316 }
317 }
318 }
319 };
320 const double flops = static_cast<double>(m) * m * k;
321 bool par = allow_parallel && flops > 2e6;
322#ifdef _OPENMP
323 if (par && !omp_in_parallel()) {
324#pragma omp parallel for schedule(dynamic, 1)
325 for (int jb = 0; jb < nblocks; ++jb)
326 column_block(jb);
327 return;
328 }
329#endif
330 (void)par;
331 for (int jb = 0; jb < nblocks; ++jb)
332 column_block(jb);
333}

335// Factors the leading ns columns of the fs x fs column-major lower front F
336// (leading dimension ldf): on return F(0:ns, 0:ns) holds the strictly lower
337// part of the unit L11, F(ns:fs, 0:ns) holds L21, d[0:ns) the pivots, and
338// F(ns:fs, ns:fs) (lower) the Schur complement F22 - L21 D L21^T. sign[j] (may
339// be null) is the expected pivot sign of column j (+1/-1/0); with eps > 0 a
340// pivot whose sign*value is below eps is replaced by sign*eps. `check(j,
341// value)` must return false for an unacceptable pivot; factorization then stops
342// and the failing column index goes to *fail (else -1). Returns the regularized
343// count.
344template <typename T, typename CheckFn>
345size_t
346partial_ldlt(int fs, int ns, T *F, int ldf, T *d, const signed char *sign,
347 T eps, CheckFn &&check, int *fail, T *work, bool allow_parallel)
348{
349 *fail = -1;
350 size_t nreg = 0;
351 // wider panels for big fronts: fewer passes over the trailing matrix
352 const int pw = fs >= 3000 ? 128 : 64;
353 for (int j0 = 0; j0 < ns; j0 += pw) {
354 const int jb = std::min(pw, ns - j0);
355 // ---- diagonal block: columns j0 .. j0+jb, rows j0 .. j0+jb --------
356 const int jend = j0 + jb;
357 for (int c = j0; c < jend; ++c) {
358 T *col = F + static_cast<size_t>(c) * ldf;
359 T piv = col[c];
360 if (sign && sign[c] != 0) {
361 const T sg = static_cast<T>(sign[c]);
362 if (!(sg * piv >= eps)) {
363 piv = sg * eps;
364 ++nreg;
365 }
366 }
367 if (!check(c, piv)) {
368 *fail = c;
369 return nreg;
370 }
371 d[c] = piv;
372 for (int cj = c + 1; cj < jend; ++cj) {
373 T *colj = F + static_cast<size_t>(cj) * ldf;
374 const T lj = col[cj] / piv; // L(cj, c)
375 for (int i = cj; i < jend; ++i)
376 colj[i] -= col[i] * lj;
377 }
378 for (int i = c + 1; i < jend; ++i)
379 col[i] /= piv;
380 }
381 // ---- rows below the block: L21 = A21 (L11 D)^-T, independent row
382 // chunks
383 if (jend < fs) {
384 auto chunk = [&](int ra, int rb) {
385 for (int c = j0; c < jend; ++c) {
386 T *col = F + static_cast<size_t>(c) * ldf;
387 for (int p = j0; p < c; ++p) {
388 const T *cp = F + static_cast<size_t>(p) * ldf;
389 const T t = d[p] * cp[c]; // d_p L(c, p)
390 AXOS_OMP_SIMD
391 for (int i = ra; i < rb; ++i)
392 col[i] -= t * cp[i];
393 }
394 const T inv = T(1) / d[c];
395 AXOS_OMP_SIMD
396 for (int i = ra; i < rb; ++i)
397 col[i] *= inv;
398 }
399 };
400 constexpr int kChunk = 256;
401 const int nrows = fs - jend;
402 bool parc = allow_parallel && nrows >= 2 * kChunk;
403#ifdef _OPENMP
404 if (parc && !omp_in_parallel()) {
405 const int nch = (nrows + kChunk - 1) / kChunk;
406#pragma omp parallel for schedule(static)
407 for (int ch = 0; ch < nch; ++ch)
408 chunk(jend + ch * kChunk,
409 std::min(fs, jend + (ch + 1) * kChunk));
410 } else
411#endif
412 {
413 (void)parc;
414 for (int ra = jend; ra < fs; ra += kChunk)
415 chunk(ra, std::min(fs, ra + kChunk));
416 }
417 }
418 // ---- trailing update: columns j0+jb .. fs -------------------------
419 const int m = fs - (j0 + jb);
420 if (m > 0) {
421 syrk_d(m, jb, F + static_cast<size_t>(j0) * ldf + (j0 + jb), ldf,
422 d + j0, F + static_cast<size_t>(j0 + jb) * ldf + (j0 + jb), ldf,
423 work, allow_parallel);
424 }
425 }
426 return nreg;
427}
