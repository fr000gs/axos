25// y = alpha * A * x + beta * y
26template <typename T, typename Idx, template <typename> class S, typename VS>
27void
28spmv(const Csr<T, Idx, S> &A, const tensorET<1, T, VS> &x,
29 tensorET<1, T, VS> &y, T alpha = T(1), T beta = T(0))
30{
31 using M = Csr<T, Idx, S>;
32 static_assert(
33 std::is_same_v<typename M::backend_type, typename VS::backend_type>,
34 "sparse matrix and dense tensor must use the same backend");
35 if (x.size() != A.cols() || y.size() != A.rows())
36 throw std::invalid_argument("spmv: dimension mismatch");
37 if (A.rows() == 0) return;
38 Kernels<typename M::backend_type>::spmv(A, x.data, y.data, alpha, beta);
39}
40
41// y = alpha * A^T * x + beta * y (uses the cached explicit transpose).
42template <typename T, typename Idx, template <typename> class S, typename VS>
43void
44spmv_t(const Csr<T, Idx, S> &A, const tensorET<1, T, VS> &x,
45 tensorET<1, T, VS> &y, T alpha = T(1), T beta = T(0))
46{
47 if (x.size() != A.rows() || y.size() != A.cols())
48 throw std::invalid_argument("spmv_t: dimension mismatch");
49 if (A.cols() == 0) return;
50 spmv(A.transposed(), x, y, alpha, beta);
51}

53// Returns A * x.
54template <typename T, typename Idx, template <typename> class S, typename VS>
55tensorET<1, T, VS>
56spmv(const Csr<T, Idx, S> &A, const tensorET<1, T, VS> &x)
57{
58 tensorET<1, T, VS> y({A.rows()}, T(0));
59 spmv(A, x, y);
60 return y;
61}

63// Y = alpha * A * X + beta * Y for row-major dense X (cols x k), Y (rows x k).
64template <typename T, typename Idx, template <typename> class S, typename VS>
65void
66spmm(const Csr<T, Idx, S> &A, const tensorET<2, T, VS> &X,
67 tensorET<2, T, VS> &Y, T alpha = T(1), T beta = T(0))
68{
69 using M = Csr<T, Idx, S>;
70 static_assert(
71 std::is_same_v<typename M::backend_type, typename VS::backend_type>,
72 "sparse matrix and dense tensor must use the same backend");
73 if (X.size(0) != A.cols() || Y.size(0) != A.rows() ||
74 X.size(1) != Y.size(1))
75 throw std::invalid_argument("spmm: dimension mismatch");
76 if (A.rows() == 0 || X.size(1) == 0) return;
77 Kernels<typename M::backend_type>::spmm(
78 A, X.data, X.size(1), Y.data, alpha, beta);
79}

81// A_ij *= r[i] * c[j] (r or c may be null). r has rows() entries and c has
82// cols(); both are real-valued and live on the matrix's device.
83template <typename T, typename Idx, template <typename> class S>
84void
85scale_rows_cols(Csr<T, Idx, S> &A, const real_of_t<T> *r, const real_of_t<T> *c)
86{
87 using M = Csr<T, Idx, S>;
88 Kernels<typename M::backend_type>::scale(A, r, c);
89}

91// Row-wise norms: out has rows() entries.
92template <typename T, typename Idx, template <typename> class S>
93void
94row_norms(const Csr<T, Idx, S> &A, real_of_t<T> *out, Norm p)
95{
96 using M = Csr<T, Idx, S>;
97 if (A.rows() == 0) return;
98 Kernels<typename M::backend_type>::row_norms(A, out, p);
99}
100
101// Column-wise norms: out has cols() entries (through the cached transpose).
102template <typename T, typename Idx, template <typename> class S>
103void
104col_norms(const Csr<T, Idx, S> &A, real_of_t<T> *out, Norm p)
105{ row_norms(A.transposed(), out, p); }

107// C = A * B
108template <typename T, typename Idx, template <typename> class S>
109Csr<T, Idx, S>
110spgemm(const Csr<T, Idx, S> &A, const Csr<T, Idx, S> &B)
111{
112 using M = Csr<T, Idx, S>;
113 if (A.cols() != B.rows())
114 throw std::invalid_argument("spgemm: dimension mismatch");
115 return Kernels<typename M::backend_type>::spgemm(A, B);
116}

118// A * diag(d) * A^T, the normal-equations matrix of interior-point methods.
119// d has cols() entries, on the matrix's device.
120template <typename T, typename Idx, template <typename> class S>
121Csr<T, Idx, S>
122AdAt(const Csr<T, Idx, S> &A, const real_of_t<T> *d)
123{
124 Csr<T, Idx, S> Ad(A);
125 scale_rows_cols(Ad, nullptr, d);
126 return spgemm(Ad, A.transposed());
127}
