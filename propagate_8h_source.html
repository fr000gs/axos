3// Domain propagation (activity-based bound tightening): for a row
4// l <= sum_j a_j x_j <= u, the min and max activities over the current bounds
5// give, for every x_j in the row,
6// a_j > 0: x_j <= (u - minact_{-j}) / a_j, x_j >= (l - maxact_{-j}) /
7// a_j a_j < 0: the same with the inequalities reversed,
8// where minact_{-j} is the minimum activity of the other terms. Integer bounds
9// are rounded. A row whose minimum activity exceeds u (or maximum is below l)
10// proves the domain empty.
11//
12// Rows come from a queue: initially all rows (or the rows of the columns whose
13// bounds changed, the node / probing case), then the rows of every column a row
14// tightened, until nothing changes or the work budget is spent. Activities are
15// recomputed per visit (O(row length)); infinite contributions are counted so a
16// row with one infinite term can still tighten that term's variable.

27// Bound changes in order, with the values they replaced: undo() restores
28// the domains to any earlier mark (fix-and-propagate, probing, diving).
29struct Trail {
30 std::vector<int> col;
31 std::vector<double> old_lb, old_ub;
32 size_t
33 mark() const
34 { return col.size(); }
35 void
36 record(int j, double l, double u)
37 {
38 col.push_back(j);
39 old_lb.push_back(l);
40 old_ub.push_back(u);
41 }
42 void
43 undo(size_t to, std::vector<double> &lb, std::vector<double> &ub)
44 {
45 while (col.size() > to) {
46 lb[col.back()] = old_lb.back();
47 ub[col.back()] = old_ub.back();
48 col.pop_back();
49 old_lb.pop_back();
50 old_ub.pop_back();
51 }
52 }
53};

55class Propagator {
56 public:
57 explicit Propagator(const LpProblem &p) : p_(p), At_(p.A.transpose())
58 {
59 const size_t n = p.cols();
60 isint_.assign(n, 0);
61 for (size_t j = 0; j < n; ++j)
62 isint_[j] = is_int(p, j) ? 1 : 0;
63 inq_.assign(p.rows(), 0);
64 }
65
66 // Tightens lb/ub (column bounds, size n) in place. changed: the columns
67 // whose bounds changed since the last fixpoint (nullptr: start from all
68 // rows). trail: receives every bound change with the value it replaced.
69 // Returns false when the domain is proven empty (the bounds may then be
70 // partly tightened: undo through the trail).
71 bool
72 propagate(std::vector<double> &lb, std::vector<double> &ub,
73 const std::vector<int> *changed = nullptr, Trail *trail = nullptr,
74 double work_factor = 10.0)
75 {
76 const size_t m = p_.rows();
77 queue_.clear();
78 if (changed) {
79 for (int j : *changed)
80 push_rows_of(j);
81 } else {
82 for (size_t i = 0; i < m; ++i)
83 push_row(static_cast<int>(i));
84 }
85 const double budget =
86 work_factor * static_cast<double>(p_.A.nnz() + m) + 1000;
87 double work = 0;
88 size_t head = 0;
89 bool feasible = true;
90 while (head < queue_.size()) {
91 const int i = queue_[head++];
92 inq_[i] = 0;
93 const auto b = p_.A.row_ptr()[i], e = p_.A.row_ptr()[i + 1];
94 work += static_cast<double>(e - b);
95 if (!row(i, lb, ub, trail)) {
96 feasible = false;
97 break;
98 }
99 if (work > budget) break;
100 }
101 for (size_t k = head; k < queue_.size(); ++k)
102 inq_[queue_[k]] = 0;
103 queue_.clear();
104 return feasible;
105 }

107 const HostMatrix &
108 transposed() const
109 { return At_; }

111 private:
112 const LpProblem &p_;
113 HostMatrix At_;
114 std::vector<uint8_t> isint_, inq_;
115 std::vector<int> queue_;

117 static bool
118 fin(double v)
119 { return std::abs(v) < 1e20; }
120
121 void
122 push_row(int i)
123 {
124 if (!inq_[i]) {
125 inq_[i] = 1;
126 queue_.push_back(i);
127 }
128 }
129 void
130 push_rows_of(int j)
131 {
132 for (auto k = At_.row_ptr()[j]; k < At_.row_ptr()[j + 1]; ++k)
133 push_row(At_.col_ind()[k]);
134 }

136 // Tightens the bounds of the columns of row i; false if infeasible.
137 bool
138 row(int i, std::vector<double> &lb, std::vector<double> &ub, Trail *trail)
139 {
140 const auto *ci = p_.A.col_ind();
141 const double *va = p_.A.values();
142 const auto b = p_.A.row_ptr()[i], e = p_.A.row_ptr()[i + 1];
143 const double l = p_.row_lb[i], u = p_.row_ub[i];
144 double minact = 0, maxact = 0;
145 int ninf_min = 0, ninf_max = 0;
146 for (auto k = b; k < e; ++k) {
147 const double a = va[k];
148 const int j = ci[k];
149 if (a > 0) {
150 if (fin(lb[j]))
151 minact += a * lb[j];
152 else
153 ++ninf_min;
154 if (fin(ub[j]))
155 maxact += a * ub[j];
156 else
157 ++ninf_max;
158 } else if (a < 0) {
159 if (fin(ub[j]))
160 minact += a * ub[j];
161 else
162 ++ninf_min;
163 if (fin(lb[j]))
164 maxact += a * lb[j];
165 else
166 ++ninf_max;
167 }
168 }
169 const double tol_u = 1e-6 * (1 + std::abs(u)),
170 tol_l = 1e-6 * (1 + std::abs(l));
171 if (fin(u) && ninf_min == 0 && minact > u + tol_u) return false;
172 if (fin(l) && ninf_max == 0 && maxact < l - tol_l) return false;
173 const bool use_u = fin(u) && ninf_min <= 1,
174 use_l = fin(l) && ninf_max <= 1;
175 if (!use_u && !use_l) return true;
176 for (auto k = b; k < e; ++k) {
177 const double a = va[k];
178 const int j = ci[k];
179 if (std::abs(a) < 1e-9) continue;
180 double nlb = lb[j], nub = ub[j];
181 // residual activities of the other terms
182 if (use_u) {
183 double rest;
184 bool ok = true;
185 const double own = a > 0 ? lb[j] : ub[j];
186 if (ninf_min == 0)
187 rest = minact - a * own;
188 else if (!fin(own))
189 rest = minact; // j is the one infinite term
190 else
191 ok = false;
192 if (ok) {
193 const double v = (u - rest) / a;
194 if (a > 0)
195 nub = std::min(nub, v);
196 else
197 nlb = std::max(nlb, v);
198 }
199 }
200 if (use_l) {
201 double rest;
202 bool ok = true;
203 const double own = a > 0 ? ub[j] : lb[j];
204 if (ninf_max == 0)
205 rest = maxact - a * own;
206 else if (!fin(own))
207 rest = maxact;
208 else
209 ok = false;
210 if (ok) {
211 const double v = (l - rest) / a;
212 if (a > 0)
213 nlb = std::max(nlb, v);
214 else
215 nub = std::min(nub, v);
216 }
217 }
218 if (isint_[j]) {
219 nlb = std::ceil(nlb - 1e-6);
220 nub = std::floor(nub + 1e-6);
221 }
222 const double range =
223 fin(lb[j]) && fin(ub[j]) ? ub[j] - lb[j] : 1e300;
224 // continuous bounds move only by a meaningful amount (no creeping)
225 const double step =
226 isint_[j]
227 ? 0.5
228 : 1e-3 * std::max(1.0, std::min(range,
229 std::abs(nlb) + std::abs(nub)));
230 const bool up_lb = nlb > lb[j] + step && std::abs(nlb) < 1e15;
231 const bool up_ub = nub < ub[j] - step && std::abs(nub) < 1e15;
232 if (!up_lb && !up_ub) continue;
233 if (trail) trail->record(j, lb[j], ub[j]);
234 if (up_lb) lb[j] = nlb;
235 if (up_ub) ub[j] = nub;
236 if (lb[j] > ub[j] + 1e-6 * (1 + std::abs(ub[j]))) return false;
237 if (lb[j] > ub[j]) lb[j] = ub[j]; // within tolerance
238 push_rows_of(j);
239 }
240 return true;
241 }
