3// PDHCG: restarted primal-dual hybrid gradient for convex QP with an inexact
4// primal subproblem (Y. Huang et al., arXiv:2405.16160), in the Halpern form
5// of PDHCG-II.
6//
7// saddle min_{x in C} max_y 1/2 x^T Q x + c^T x - y^T A x + p_K(y)
8// x+ ~ argmin_{x in C} 1/2 x^T Q x + (c - A^T y)^T x + |x - x|^2 / (2 tau)
9// y+ = prox of p_K at y + sigma(... - A(2 x+ - x)) (PDLP dual step)
10// Halpern z <- (t+1)/(t+2) (2 T(z) - z) + 1/(t+2) z0
11//
12// Unlike HPR-QP (and rAPDHG/PDQP), Q is not linearized, so the step sizes
13// depend on |A| only: tau = eta / omega, sigma = eta omega, eta < 1 / |A|. The
14// subproblem is solved by conjugate gradients when x has no finite bounds and
15// by projected Barzilai-Borwein otherwise (the paper's switch), warm-started at
16// x, up to a tolerance tied to the current KKT error; without Q it is the exact
17// projection. Q x, A x and A^T y are carried along, so a termination check
18// needs no extra products.
19//
20// Restarts (every opt.check_every iterations) compare the relative KKT error at
21// T(z) with the one at the last restart (0.2 sufficient, 0.8 necessary,
22// artificial after 0.2 of all iterations); the primal weight omega is
23// re-balanced from the movement since the last restart. Preconditioning and
24// termination as in HPR-QP (qp_scaling.h, qp_model.h).

47 QpSolution
48 solve(const QpProblem &p, const QpOptions &opt)
49 {
50 using namespace axos_qp;
51 const auto t_start = std::chrono::steady_clock::now();
52 auto elapsed = [&] {
53 return std::chrono::duration<double>(
54 std::chrono::steady_clock::now() - t_start)
55 .count();
56 };
57 const std::string bad = p.validate();
58 if (!bad.empty()) throw std::invalid_argument("Pdhcg: " + bad);
59 const size_t n = p.cols(), m = p.rows();
60
61 QpScaling sc;
62 QpProblem q;
63 if (opt.scaling) {
64 sc = compute_qp_scaling(
65 p, opt.ruiz_iterations, opt.pock_chambolle_alpha);
66 q = apply_qp_scaling(p, sc);
67 } else {
68 sc.col.assign(n, 1.0);
69 sc.row.assign(m, 1.0);
70 q = p;
71 }
72 B be;
73 const bool hasQ = q.Q.nnz() > 0;
74 bool bounded = false;
75 for (size_t j = 0; j < n && !bounded; ++j)
76 bounded =
77 std::isfinite(q.lp.col_lb[j]) || std::isfinite(q.lp.col_ub[j]);
78 const bool use_cg = hasQ && !bounded;
79 auto A = be.upload(q.lp.A);
80 auto AT = be.upload(q.lp.A.transpose());
81 auto Qm = be.upload(q.Q);
82 std::vector<double> cinv(n), rinv(m);
83 for (size_t j = 0; j < n; ++j)
84 cinv[j] = 1.0 / sc.col[j];
85 for (size_t i = 0; i < m; ++i)
86 rinv[i] = 1.0 / sc.row[i];
87 Vec c = be.upload(q.lp.c), lb = be.upload(q.lp.col_lb),
88 ub = be.upload(q.lp.col_ub);
89 Vec l = be.upload(q.lp.row_lb), u = be.upload(q.lp.row_ub);
90 Vec dcinv = be.upload(cinv), drinv = be.upload(rinv);
91
92 // state z = (x, y) with Q x, A x, A^T y; anchors; T(z) outputs
93 Vec x = be.vec(n), qx = be.vec(n), aty = be.vec(n), ax = be.vec(m),
94 y = be.vec(m);
95 Vec x0 = be.vec(n), qx0 = be.vec(n), aty0 = be.vec(n), ax0 = be.vec(m),
96 y0 = be.vec(m);
97 Vec xn = be.vec(n), qxn = be.vec(n), atyn = be.vec(n), axn = be.vec(m),
98 yn = be.vec(m);
99 Vec g = be.vec(n), s = be.vec(n), qs = be.vec(n), hp = be.vec(n);
100 Vec r = use_cg ? be.vec(n) : be.vec(0),
101 pdir = use_cg ? be.vec(n) : be.vec(0);
102 Vec xr = be.vec(n),
103 yr = be.vec(m); // last restart point (primal weight)
104 auto P = [](Vec &v) { return B::ptr(v); };
105
106 const double lamA =
107 m ? power_method(be, A, &AT, n, m, opt.power_iterations) : 0.0;
108 const double lamQ =
109 hasQ ? power_method(be, Qm, nullptr, n, n, opt.power_iterations)
110 : 0.0;
111 const double eta = lamA > 0 ? 0.998 / std::sqrt(1.01 * lamA) : 1.0;
112 double omega = 1.0;
113 {
114 double nb = 0, nc = 0;
115 for (size_t i = 0; i < m; ++i) {
116 if (std::isfinite(q.lp.row_lb[i]))
117 nb += q.lp.row_lb[i] * q.lp.row_lb[i];
118 if (std::isfinite(q.lp.row_ub[i]))
119 nb += q.lp.row_ub[i] * q.lp.row_ub[i];
120 }
121 for (double v : q.lp.c)
122 nc += v * v;
123 if (nb > 1e-20 && nc > 1e-20) omega = std::sqrt(nc) / std::sqrt(nb);
124 }
125 double bmax = 0, cmax = 0;
126 for (size_t i = 0; i < m; ++i) {
127 if (std::isfinite(p.lp.row_lb[i]))
128 bmax = std::max(bmax, std::abs(p.lp.row_lb[i]));
129 if (std::isfinite(p.lp.row_ub[i]))
130 bmax = std::max(bmax, std::abs(p.lp.row_ub[i]));
131 }
132 for (double v : p.lp.c)
133 cmax = std::max(cmax, std::abs(v));
134 // x = P_C(0)
135 {
136 std::vector<double> h(n);
137 for (size_t j = 0; j < n; ++j)
138 h[j] = std::min(std::max(0.0, q.lp.col_lb[j]), q.lp.col_ub[j]);
139 x = be.upload(h);
140 if (hasQ) be.spmv(Qm, P(x), P(qx));
141 if (m) be.spmv(A, P(x), P(ax));
142 }
143 auto copy = [&](Vec &dst, Vec &src, size_t len) {
144 AXOS_QP_MAP(be, copy, len, (Copy{P(dst), P(src)}));
145 };
146 auto anchor = [&]() {
147 copy(x0, x, n);
148 copy(qx0, qx, n);
149 copy(aty0, aty, n);
150 copy(ax0, ax, m);
151 copy(y0, y, m);
152 copy(xr, x, n);
153 copy(yr, y, m);
154 };
155 anchor();
156 const double setup_s = elapsed();
157
158 struct Kkt {
159 double eta_p = 0, eta_d = 0, eta_gap = 0, pobj = 0, dobj = 0;
160 double
161 err() const
162 { return std::max({eta_p, eta_d, eta_gap}); }
163 };
164 auto eval = [&](Vec &xv, Vec &qxv, Vec &axv, Vec &yv, Vec &atyv) {
165 Kkt k;
166 double pviol = 0, axmax = 0, drow = 0;
167 if (m) {
168 const auto rr = AXOS_QP_RED(be, kkt_rows, m,
169 (KktRows{P(axv), P(yv), P(l), P(u), P(drinv)}));
170 pviol = rr[0];
171 axmax = rr[1];
172 drow = rr[2];
173 }
174 const auto cc = AXOS_QP_RED(be, pd_kkt_cols, n,
175 (PdKktCols{
176 P(xv), P(qxv), P(atyv), P(c), P(lb), P(ub), P(dcinv)}));
177 k.pobj = 0.5 * cc[3] + cc[4] + q.lp.offset;
178 k.dobj = -0.5 * cc[3] + drow + cc[5] + q.lp.offset;
179 k.eta_p = pviol / (1 + std::max(bmax, axmax));
180 k.eta_d = cc[0] / (1 + std::max({cmax, cc[1], cc[2]}));
181 k.eta_gap = std::abs(k.pobj - k.dobj) /
182 (1 + std::max(std::abs(k.pobj), std::abs(k.dobj)));
183 return k;
184 };
185 long inner_total = 0;
186 auto finish = [&](Status st, long iters, Vec &xv, Vec &yv,
187 const Kkt &k) {
188 std::vector<double> hx, hy;
189 be.download(xv, hx);
190 be.download(yv, hy);
191 for (size_t j = 0; j < n; ++j)
192 hx[j] *= sc.col[j];
193 for (size_t i = 0; i < m; ++i)
194 hy[i] *= sc.row[i];
195 QpSolution sol = evaluate_qp(p, hx, hy);
196 sol.status = st;
197 sol.iterations = iters;
198 sol.seconds = elapsed();
199 sol.setup_seconds = setup_s;
200 sol.method = "pdhcg";
201 sol.device = be.name();
202 if (opt.verbose)
203 std::printf("[pdhcg] %s after %ld iterations (%ld inner), %.3f "
204 "s (setup %.3f s): "
205 "obj %.10e eta_p %.2e eta_d %.2e gap %.2e "
206 "(internal %.2e)\n",
207 to_string(st), iters, inner_total, sol.seconds, setup_s,
208 sol.primal_objective, sol.rel_primal, sol.rel_dual,
209 sol.rel_gap, k.err());
210 return sol;
211 };
212
213 if (opt.verbose)
214 std::printf("[pdhcg] %s: n %zu m %zu nnz(A) %zu nnz(Q) %zu, |A| "
215 "%.3e lambda_Q %.3e "
216 "omega0 %.3e inner %s, setup %.3f s\n",
217 be.name().c_str(), n, m, q.lp.A.nnz(), q.Q.nnz(),
218 std::sqrt(lamA), lamQ, omega,
219 use_cg ? "CG" : (hasQ ? "projected BB" : "projection"),
220 setup_s);
221
222 long it = 0, t = 0;
223 double kkt_restart = -1, kkt_prev = 1e300, kkt_last = 1.0;
224 Kkt kkt;
225 while (true) {
226 const double tau = eta / omega, sigma = eta * omega,
227 inv_tau = 1.0 / tau;
228 // ---- primal subproblem, warm-started at x
229 // -------------------------------
230 AXOS_QP_MAP(be, pd_start, n,
231 (PdStart{P(x), P(qx), P(c), P(aty), P(xn), P(qxn), P(g),
232 use_cg ? P(r) : nullptr, use_cg ? P(pdir) : nullptr}));
233 if (!hasQ) {
234 // exact: x+ = P_C(x - tau (c - A^T y))
235 AXOS_QP_MAP(be, pd_pg_step, n,
236 (PdPgStep{P(xn), P(s), P(g), P(lb), P(ub), tau}));
237 ++inner_total;
238 } else if (!use_cg) {
239 // projected Barzilai-Borwein on the strongly convex subproblem
240 const double L = lamQ * 1.01 + inv_tau, mu = inv_tau;
241 double alpha = 1.0 / L;
242 const double tol_in = 0.05 * kkt_last;
243 for (int k = 0; k < max_inner; ++k) {
244 AXOS_QP_MAP(be, pd_pg_step, n,
245 (PdPgStep{P(xn), P(s), P(g), P(lb), P(ub), alpha}));
246 be.spmv(Qm, P(s), P(qs));
247 const auto rr = AXOS_QP_RED(be, pd_pg_update, n,
248 (PdPgUpdate{
249 P(g), P(qxn), P(s), P(qs), P(xn), inv_tau}));
250 ++inner_total;
251 if (rr[2] <= tol_in * (1.0 + rr[3]) || rr[0] <= 0) break;
252 alpha = rr[1] > 0 ? rr[0] / rr[1] : 1.0 / L;
253 alpha = std::min(std::max(alpha, 1.0 / L), 1.0 / mu);
254 }
255 } else {
256 // conjugate gradients on (Q + I/tau) x = x/tau - c + A^T y
257 double rho = AXOS_QP_RED(be, dot2, n, (Dot2{P(r), P(r)}))[1];
258 const double tol_in = 0.05 * kkt_last;
259 for (int k = 0; k < max_inner && rho > 0; ++k) {
260 be.spmv(Qm, P(pdir), P(qs));
261 const double php = AXOS_QP_RED(be, pd_cg_curv, n,
262 (PdCgCurv{P(pdir), P(qs), P(hp), inv_tau}))[0];
263 if (!(php > 0)) break;
264 const double alpha = rho / php;
265 const auto rr = AXOS_QP_RED(be, pd_cg_step, n,
266 (PdCgStep{P(xn), P(qxn), P(r), P(pdir), P(qs), P(hp),
267 alpha}));
268 ++inner_total;
269 if (rr[1] <= tol_in * (1.0 + rr[2])) break;
270 const double beta = rr[0] / rho;
271 rho = rr[0];
272 AXOS_QP_MAP(
273 be, pd_cg_dir, n, (PdCgDir{P(pdir), P(r), beta}));
274 }
275 }
276 // ---- dual step
277 // ---------------------------------------------------------------
278 if (m) {
279 be.spmv(A, P(xn), P(axn));
280 AXOS_QP_MAP(be, pd_dual, m,
281 (PdDual{P(y), P(ax), P(axn), P(l), P(u), P(yn), sigma}));
282 be.spmv(AT, P(yn), P(atyn));
283 }
284 const bool check = it % opt.check_every == 0;
285 if (check) {
286 kkt = eval(xn, qxn, axn, yn, atyn);
287 kkt_last = std::max(kkt.err(), 1e-12);
288 if (opt.verbose >= 2)
289 std::printf("[pdhcg] it %-8ld t %-7ld pobj % .10e dobj % "
290 ".10e eta_p %.2e "
291 "eta_d %.2e gap %.2e omega %.3e inner %ld\n",
292 it, t, kkt.pobj, kkt.dobj, kkt.eta_p, kkt.eta_d,
293 kkt.eta_gap, omega, inner_total);
294 if (!std::isfinite(kkt.err()))
295 return finish(Status::NumericalError, it, xn, yn, kkt);
296 if (kkt.err() <= opt.tol)
297 return finish(Status::Optimal, it + 1, xn, yn, kkt);
298 if (it + 1 >= opt.max_iterations)
299 return finish(Status::IterationLimit, it + 1, xn, yn, kkt);
300 if (elapsed() > opt.time_limit)
301 return finish(Status::TimeLimit, it + 1, xn, yn, kkt);
302 if (kkt_restart < 0) kkt_restart = kkt.err();
303 const double e = kkt.err();
304 const bool suff = e <= 0.2 * kkt_restart;
305 const bool nec = e <= 0.8 * kkt_restart && e > kkt_prev;
306 const bool art = t > 0 && static_cast<double>(t) >=
307 0.2 * static_cast<double>(it);
308 kkt_prev = e;
309 if (t > 0 && (suff || nec || art)) {
310 // restart at T(z); re-balance the primal weight
311 const double dx = std::sqrt(
312 AXOS_QP_RED(be, diff_sq, n, (DiffSq{P(xn), P(xr)}))[0]);
313 const double dy = m ? std::sqrt(AXOS_QP_RED(be, diff_sq, m,
314 (DiffSq{P(yn), P(yr)}))[0])
315 : 0.0;
316 if (dx > 1e-10 && dy > 1e-10)
317 omega = std::exp(
318 0.5 * std::log(dy / dx) + 0.5 * std::log(omega));
319 copy(x, xn, n);
320 copy(y, yn, m);
321 if (hasQ)
322 be.spmv(Qm, P(x), P(qx)); // exact products, no drift
323 if (m) {
324 be.spmv(A, P(x), P(ax));
325 be.spmv(AT, P(y), P(aty));
326 }
327 anchor();
328 kkt_restart = e;
329 kkt_prev = 1e300;
330 t = 0;
331 ++it;
332 continue;
333 }
334 }
335 // ---- reflected Halpern step
336 // --------------------------------------------------
337 const double tt = static_cast<double>(t);
338 const double w1 = (tt + 1.0) / (tt + 2.0), w0c = 1.0 / (tt + 2.0);
339 AXOS_QP_MAP(be, pd_halpern_n, n,
340 (PdHalpernN{P(x), P(qx), P(aty), P(xn), P(qxn), P(atyn), P(x0),
341 P(qx0), P(aty0), w1, w0c}));
342 AXOS_QP_MAP(
343 be, halpern, m, (Halpern{P(ax), P(axn), P(ax0), w1, w0c}));
344 AXOS_QP_MAP(be, halpern, m, (Halpern{P(y), P(yn), P(y0), w1, w0c}));
345 ++it;
346 ++t;
347 }
348 }
