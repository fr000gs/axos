3// CUDA execution of the PDLP kernels (Parallel<Cuda::Backend>). Reductions
4// are two-stage (per-block partials, then one finishing block), so they need
5// no atomics; the host reads all slots back with one copy in fetch().

18template <typename F>
19__global__ void
20for_each_kernel(size_t n, F f)
21{
22 const size_t stride = static_cast<size_t>(gridDim.x) * blockDim.x;
23 for (size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
24 i < n; i += stride)
25 f(i);
26}

28// Stage 1: each block writes its partial sums to part[block * K + k].
29template <int K, typename F>
30__global__ void
31reduce_kernel(size_t n, F f, double *part)
32{
33 __shared__ double sh[K][32];
34 double acc[K];
35#pragma unroll
36 for (int k = 0; k < K; ++k)
37 acc[k] = 0.0;
38 const size_t stride = static_cast<size_t>(gridDim.x) * blockDim.x;
39 for (size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
40 i < n; i += stride)
41 f(i, acc);
42 const int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
43 const int nwarps = (blockDim.x + 31) >> 5;
44#pragma unroll
45 for (int k = 0; k < K; ++k) {
46 double v = acc[k];
47 for (int off = 16; off > 0; off >>= 1)
48 v += __shfl_down_sync(0xffffffffu, v, off);
49 if (lane == 0) sh[k][warp] = v;
50 }
51 __syncthreads();
52 if (warp == 0) {
53#pragma unroll
54 for (int k = 0; k < K; ++k) {
55 double v = lane < nwarps ? sh[k][lane] : 0.0;
56 for (int off = 16; off > 0; off >>= 1)
57 v += __shfl_down_sync(0xffffffffu, v, off);
58 if (lane == 0) part[static_cast<size_t>(blockIdx.x) * K + k] = v;
59 }
60 }
61}

63// Stage 2 (one block): out[k] += sum over blocks of part[b * K + k].
64template <int K>
65__global__ void
66reduce_finish_kernel(const double *part, int nblocks, double *out)
67{
68 __shared__ double sh[K][32];
69 const int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
70 const int nwarps = (blockDim.x + 31) >> 5;
71#pragma unroll
72 for (int k = 0; k < K; ++k) {
73 double v = 0.0;
74 for (int b = threadIdx.x; b < nblocks; b += blockDim.x)
75 v += part[static_cast<size_t>(b) * K + k];
76 for (int off = 16; off > 0; off >>= 1)
77 v += __shfl_down_sync(0xffffffffu, v, off);
78 if (lane == 0) sh[k][warp] = v;
79 }
80 __syncthreads();
81 if (warp == 0) {
82#pragma unroll
83 for (int k = 0; k < K; ++k) {
84 double v = lane < nwarps ? sh[k][lane] : 0.0;
85 for (int off = 16; off > 0; off >>= 1)
86 v += __shfl_down_sync(0xffffffffu, v, off);
87 if (lane == 0) out[k] += v;
88 }
89 }
90}

92// Minimum reduction: stage 1 writes one minimum per block, stage 2 folds
93// them into out[0] (which the caller preloads, see set_slot).
94template <typename F>
95__global__ void
96reduce_min_kernel(size_t n, F f, double *part)
97{
98 __shared__ double sh[32];
99 double m = INFINITY;
100 const size_t stride = static_cast<size_t>(gridDim.x) * blockDim.x;
101 for (size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
102 i < n; i += stride) {
103 const double v = f(i);
104 m = v < m ? v : m;
105 }
106 for (int off = 16; off > 0; off >>= 1) {
107 const double o = __shfl_down_sync(0xffffffffu, m, off);
108 m = o < m ? o : m;
109 }
110 const int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
111 if (lane == 0) sh[warp] = m;
112 __syncthreads();
113 if (warp == 0) {
114 double v = lane < ((blockDim.x + 31) >> 5) ? sh[lane] : INFINITY;
115 for (int off = 16; off > 0; off >>= 1) {
116 const double o = __shfl_down_sync(0xffffffffu, v, off);
117 v = o < v ? o : v;
118 }
119 if (lane == 0) part[blockIdx.x] = v;
120 }
121}
122
123__global__ void
124reduce_min_finish_kernel(const double *part, int nblocks, double *out)
125{
126 __shared__ double sh[32];
127 double m = INFINITY;
128 for (int b = threadIdx.x; b < nblocks; b += blockDim.x)
129 m = part[b] < m ? part[b] : m;
130 for (int off = 16; off > 0; off >>= 1) {
131 const double o = __shfl_down_sync(0xffffffffu, m, off);
132 m = o < m ? o : m;
133 }
134 const int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
135 if (lane == 0) sh[warp] = m;
136 __syncthreads();
137 if (warp == 0) {
138 double v = lane < ((blockDim.x + 31) >> 5) ? sh[lane] : INFINITY;
139 for (int off = 16; off > 0; off >>= 1) {
140 const double o = __shfl_down_sync(0xffffffffu, v, off);
141 v = o < v ? o : v;
142 }
143 if (lane == 0) out[0] = v < out[0] ? v : out[0];
144 }
145}

149template <> struct Parallel<Cuda::Backend> {
150 double *dev = nullptr; // accumulated slots
151 double *part = nullptr; // per-block partial sums (kMaxBlocks * kMaxK)
152 double host[pdlp::kSlots] = {};
153 static constexpr int kBlock = 256;
154 static constexpr int kMaxBlocks = 1024;
155 static constexpr int kMaxK = 8;
156
157 Parallel()
158 {
159 check(cudaMalloc(&dev, pdlp::kSlots * sizeof(double)), "cudaMalloc");
160 check(cudaMalloc(&part, kMaxBlocks * kMaxK * sizeof(double)),
161 "cudaMalloc");
162 zero();
163 }
164 Parallel(const Parallel &) = delete;
165 Parallel &operator=(const Parallel &) = delete;
166 ~Parallel()
167 {
168 if (dev) cudaFree(dev);
169 if (part) cudaFree(part);
170 }
171
172 static void
173 check(cudaError_t e, const char *what)
174 {
175 if (e != cudaSuccess)
176 throw std::runtime_error(std::string("CUDA error in ") + what +
177 ": " + cudaGetErrorString(e));
178 }
179
180 static int
181 grid_for(size_t n)
182 {
183 size_t g = (n + kBlock - 1) / kBlock;
184 return static_cast<int>(g > kMaxBlocks ? kMaxBlocks : (g == 0 ? 1 : g));
185 }
186
187 template <typename F>
188 void
189 for_each(size_t n, F f)
190 {
191 if (n == 0) return;
192 pdlp::for_each_kernel<<<grid_for(n), kBlock>>>(n, f);
193 }
194
195 void
196 zero()
197 { check(cudaMemset(dev, 0, pdlp::kSlots * sizeof(double)), "zero slots"); }
198
199 template <typename F>
200 void
201 reduce(int slot, size_t n, F f)
202 {
203 if (n == 0) return;
204 static_assert(F::K <= kMaxK, "too many reduction values");
205 const int blocks = grid_for(n);
206 pdlp::reduce_kernel<F::K, F><<<blocks, kBlock>>>(n, f, part);
207 pdlp::reduce_finish_kernel<F::K>
208 <<<1, kBlock>>>(part, blocks, dev + slot);
209 }
210
211 void
212 set_slot(int slot, double v)
213 {
214 check(
215 cudaMemcpy(dev + slot, &v, sizeof(double), cudaMemcpyHostToDevice),
216 "set_slot");
217 }
218
219 // slots[slot] = min(slots[slot], min over i of f(i)).
220 template <typename F>
221 void
222 reduce_min(int slot, size_t n, F f)
223 {
224 if (n == 0) return;
225 const int blocks = grid_for(n);
226 pdlp::reduce_min_kernel<<<blocks, kBlock>>>(n, f, part);
227 pdlp::reduce_min_finish_kernel<<<1, kBlock>>>(part, blocks, dev + slot);
228 }
229
230 const double *
231 fetch()
232 {
233 check(cudaGetLastError(), "kernel launch");
234 check(cudaMemcpy(host, dev, sizeof(host), cudaMemcpyDeviceToHost),
235 "fetch");
236 return host;
237 }
238
239 void
240 copy(double *dst, const double *src, size_t n)
241 {
242 if (n)
243 check(cudaMemcpy(
244 dst, src, n * sizeof(double), cudaMemcpyDeviceToDevice),
245 "copy");
246 }
247};
