3// HPR-QP over several MPI processes, one GPU (or one CPU) per rank.
4//
5// The constraint rows are split into contiguous blocks, balanced on nonzeros.
6// Rank r holds rows [r0, r1) of A, the n x (r1 - r0) matrix A^T of those rows,
7// and the m-sized vectors of those rows (y, ybar, y0, the row bounds). Every
8// rank keeps a full copy of the n-sized vectors and of Q. Then:
9// * A v needs no communication, because v (n-sized) is replicated.
10// * A^T y is a sum over the row blocks: each rank computes its block's
11// partial product, and one MPI_Allreduce (SUM, n doubles) per iteration
12// adds them up before the fused epilogue runs on the sum.
13// * Q products are repeated on every rank, so the work divided is the work in
14// A: LPs and QPs whose cost is in the constraints.
15// Reductions over rows (merit and KKT terms at a check) are combined over the
16// ranks: a few doubles every 10 to 100 iterations. Host-side decisions (stop,
17// restart, new penalty) use rank 0's values, broadcast, so the ranks never take
18// different paths.
19//
20// GPU vectors are exchanged through pinned host memory (the backend's
21// through_host). With a CUDA-aware MPI (Open MPI or MVAPICH2 built with CUDA,
22// Cray MPICH), define AXOS_MPI_CUDA_AWARE to pass device pointers to MPI
23// directly. MS-MPI on Windows is not CUDA-aware.
24//
25// Without AXOS_ENABLE_MPI a QpComm is a single process, every operation is a
26// no-op, and nothing here needs an MPI library.

77 // Rows [first, second) of rank r: contiguous blocks with about equal
78 // nnz + rows (row_ptr[i] + i is nondecreasing in i)
79 std::pair<size_t, size_t>
80 rows_of(const HostMatrix &A, int r) const
81 {
82 const size_t m = A.rows();
83 auto start = [&](int k) -> size_t {
84 if (k <= 0) return 0;
85 if (k >= size_) return m;
86 const auto *rp = A.row_ptr();
87 const uint64_t total = uint64_t(A.nnz()) + m;
88 const uint64_t target = total * uint64_t(k) / uint64_t(size_);
89 size_t lo = 0, hi = m; // first i with rp[i] + i >= target
90 while (lo < hi) {
91 const size_t mid = (lo + hi) / 2;
92 if (uint64_t(rp[mid]) + mid < target)
93 lo = mid + 1;
94 else
95 hi = mid;
96 }
97 return lo;
98 };
99 return {start(r), start(r + 1)};
100 }

124 // The concatenation of every rank's rows (rows_of(A, r) of a vector
125 // with one entry per row of A), in rank order, on every rank.
126 std::vector<double>
127 gather_rows(const HostMatrix &A, const std::vector<double> &mine) const
128 {
129 if (!distributed()) return mine;
130 std::vector<double> all(A.rows());
131#if defined(AXOS_ENABLE_MPI)
132 std::vector<int> counts(size_), displs(size_);
133 for (int r = 0; r < size_; ++r) {
134 const auto b = rows_of(A, r);
135 counts[r] = int(b.second - b.first);
136 displs[r] = int(b.first);
137 }
138 if (int(mine.size()) != counts[rank_])
139 throw std::logic_error("QpComm::gather_rows: wrong local length");
140 const auto t0 = std::chrono::steady_clock::now();
141 MPI_Allgatherv(mine.data(), counts[rank_], MPI_DOUBLE, all.data(),
142 counts.data(), displs.data(), MPI_DOUBLE, comm_);
143 account(t0, A.rows());
144#endif
145 return all;
146 }

161 void
162 reduce(double *v, size_t n, bool add) const
163 {
164#if defined(AXOS_ENABLE_MPI)
165 if (!distributed() || n == 0) return;
166 const auto t0 = std::chrono::steady_clock::now();
167 MPI_Allreduce(MPI_IN_PLACE, v, int(n), MPI_DOUBLE,
168 add ? MPI_SUM : MPI_MAX, comm_);
169 account(t0, n);
170#else
171 (void)v, (void)n, (void)add;
172#endif
173 }

175 void
176 account(std::chrono::steady_clock::time_point t0, size_t n) const
177 {
178 seconds_ +=
179 std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
180 .count();
181 ++calls_;
182 doubles_ += double(n);
183 }

187#endif
188 int rank_ = 0, size_ = 1, local_rank_ = 0;
189 mutable double seconds_ = 0, doubles_ = 0;
190 mutable long calls_ = 0;

193// Rows [r0, r1) of A as a matrix of their own (same columns).
194inline HostMatrix
195row_block(const HostMatrix &A, size_t r0, size_t r1)
196{
197 const auto *rp = A.row_ptr();
198 const auto *ci = A.col_ind();
199 const double *v = A.values();
200 std::vector<int32_t> brp(r1 - r0 + 1);
201 for (size_t i = r0; i <= r1; ++i)
202 brp[i - r0] = rp[i] - rp[r0];
203 std::vector<int32_t> bci(ci + rp[r0], ci + rp[r1]);
204 std::vector<double> bv(v + rp[r0], v + rp[r1]);
205 return HostMatrix(r1 - r0, A.cols(), brp, bci, bv);
206}

210// v (n doubles of the backend) summed over the ranks, in place, ordered
211// with the work queued on the backend
212template <class B>
213void
214allreduce_sum(B &be, double *v, size_t n, const QpComm &c)
215{
216 if (!c.distributed() || n == 0) return;
217#if defined(AXOS_MPI_CUDA_AWARE)
218 if constexpr (B::is_gpu) {
219 be.sync();
220 c.sum(v, n);
221 return;
222 }
223#endif
224 be.through_host(v, n, [&](double *h) { c.sum(h, n); });
225}
