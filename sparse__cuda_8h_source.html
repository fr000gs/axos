37template <typename T>
38constexpr cudaDataType
39value_type()
40{
41 if constexpr (std::is_same_v<T, float>)
42 return CUDA_R_32F;
43 else if constexpr (std::is_same_v<T, double>)
44 return CUDA_R_64F;
45 else if constexpr (std::is_same_v<T, std::complex<float>>)
46 return CUDA_C_32F;
47 else
48 return CUDA_C_64F;
49}

56// Scratch memory taken from the GPU pool.
57struct PoolBuf {
58 void *p = nullptr;
59 size_t bytes = 0;
60 PoolBuf() = default;
61 PoolBuf(const PoolBuf &) = delete;
62 PoolBuf &operator=(const PoolBuf &) = delete;
63 ~PoolBuf() { release(); }
64 void
65 reserve(size_t n)
66 {
67 if (n <= bytes) return;
68 release();
69 p = GPUMemoryPool::get().allocate(n);
70 bytes = n;
71 }
72 void
73 release()
74 {
75 if (p) GPUMemoryPool::get().deallocate(p, bytes);
76 p = nullptr;
77 bytes = 0;
78 }
79};

81// Per-matrix cuSPARSE state: the sparse-matrix descriptor and the SpMV
82// workspace (with preprocessing done once), reused across calls.
83struct CusparseCache : Sparse::detail::DeviceCache {
84 cusparseSpMatDescr_t mat = nullptr;
85 PoolBuf spmv_buf;
86 bool spmv_ready = false;
87 PoolBuf spmm_buf;
88 ~CusparseCache() override
89 {
90 if (mat) cusparseDestroySpMat(mat);
91 }
92};

94template <typename M>
95inline CusparseCache &
96cache_of(const M &A)
97{
98 using T = typename M::value_type;
99 using Idx = typename M::index_type;
100 auto *c = dynamic_cast<CusparseCache *>(A.device_cache());
101 if (!c) {
102 auto sp = std::make_shared<CusparseCache>();
103 A.set_device_cache(sp);
104 c = sp.get();
105 }
106 if (!c->mat) {
107 check(
108 cusparseCreateCsr(&c->mat, A.rows(), A.cols(), A.nnz(),
109 const_cast<Idx *>(A.row_ptr()), const_cast<Idx *>(A.col_ind()),
110 const_cast<T *>(A.values()), index_type<Idx>(),
111 index_type<Idx>(), CUSPARSE_INDEX_BASE_ZERO, value_type<T>()),
112 "cusparseCreateCsr");
113 }
114 return *c;
115}

126 template <typename M>
127 static void
128 spmv(const M &A, const typename M::value_type *x, typename M::value_type *y,
129 typename M::value_type alpha, typename M::value_type beta)
130 {
131 using T = typename M::value_type;
132 using namespace cuda_detail;
133 if (A.nnz() == 0) {
134 // y = beta * y (rare edge case, done on the host)
135 std::vector<T> h(A.rows(), T(0));
136 if (beta != T(0)) {
137 check(cudaMemcpy(h.data(), y, A.rows() * sizeof(T),
138 cudaMemcpyDeviceToHost),
139 "spmv copy");
140 for (auto &v : h)
141 v *= beta;
142 }
143 check(cudaMemcpy(y, h.data(), A.rows() * sizeof(T),
144 cudaMemcpyHostToDevice),
145 "spmv copy");
146 return;
147 }
148 CusparseCache &c = cache_of(A);
149 cusparseDnVecDescr_t vx, vy;
150 check(cusparseCreateDnVec(
151 &vx, A.cols(), const_cast<T *>(x), value_type<T>()),
152 "cusparseCreateDnVec");
153 check(cusparseCreateDnVec(&vy, A.rows(), y, value_type<T>()),
154 "cusparseCreateDnVec");
155 if (!c.spmv_ready) {
156 size_t sz = 0;
157 check(
158 cusparseSpMV_bufferSize(handle(),
159 CUSPARSE_OPERATION_NON_TRANSPOSE, &alpha, c.mat, vx, &beta,
160 vy, value_type<T>(), CUSPARSE_SPMV_ALG_DEFAULT, &sz),
161 "cusparseSpMV_bufferSize");
162 c.spmv_buf.reserve(sz);
163#if CUSPARSE_VERSION >= 12400
164 check(cusparseSpMV_preprocess(handle(),
165 CUSPARSE_OPERATION_NON_TRANSPOSE, &alpha, c.mat, vx,
166 &beta, vy, value_type<T>(), CUSPARSE_SPMV_ALG_DEFAULT,
167 c.spmv_buf.p),
168 "cusparseSpMV_preprocess");
169#endif
170 c.spmv_ready = true;
171 }
172 check(cusparseSpMV(handle(), CUSPARSE_OPERATION_NON_TRANSPOSE, &alpha,
173 c.mat, vx, &beta, vy, value_type<T>(),
174 CUSPARSE_SPMV_ALG_DEFAULT, c.spmv_buf.p),
175 "cusparseSpMV");
176 cusparseDestroyDnVec(vx);
177 cusparseDestroyDnVec(vy);
178 }

180 template <typename M>
181 static void
182 spmm(const M &A, const typename M::value_type *X, size_t k,
183 typename M::value_type *Y, typename M::value_type alpha,
184 typename M::value_type beta)
185 {
186 using T = typename M::value_type;
187 using namespace cuda_detail;
188 if (A.nnz() == 0) {
189 std::vector<T> h(A.rows() * k, T(0));
190 if (beta != T(0)) {
191 check(cudaMemcpy(h.data(), Y, h.size() * sizeof(T),
192 cudaMemcpyDeviceToHost),
193 "spmm copy");
194 for (auto &v : h)
195 v *= beta;
196 }
197 check(cudaMemcpy(Y, h.data(), h.size() * sizeof(T),
198 cudaMemcpyHostToDevice),
199 "spmm copy");
200 return;
201 }
202 CusparseCache &c = cache_of(A);
203 cusparseDnMatDescr_t mx, my;
204 check(cusparseCreateDnMat(&mx, A.cols(), k, k, const_cast<T *>(X),
205 value_type<T>(), CUSPARSE_ORDER_ROW),
206 "cusparseCreateDnMat");
207 check(cusparseCreateDnMat(
208 &my, A.rows(), k, k, Y, value_type<T>(), CUSPARSE_ORDER_ROW),
209 "cusparseCreateDnMat");
210 size_t sz = 0;
211 check(
212 cusparseSpMM_bufferSize(handle(), CUSPARSE_OPERATION_NON_TRANSPOSE,
213 CUSPARSE_OPERATION_NON_TRANSPOSE, &alpha, c.mat, mx, &beta, my,
214 value_type<T>(), CUSPARSE_SPMM_ALG_DEFAULT, &sz),
215 "cusparseSpMM_bufferSize");
216 c.spmm_buf.reserve(sz);
217 check(cusparseSpMM(handle(), CUSPARSE_OPERATION_NON_TRANSPOSE,
218 CUSPARSE_OPERATION_NON_TRANSPOSE, &alpha, c.mat, mx, &beta,
219 my, value_type<T>(), CUSPARSE_SPMM_ALG_DEFAULT, c.spmm_buf.p),
220 "cusparseSpMM");
221 cusparseDestroyDnMat(mx);
222 cusparseDestroyDnMat(my);
223 }

225 // A_ij *= r[i] * c[j]; r, c are device pointers (either may be null).
226 template <typename M>
227 static void
228 scale(M &A, const real_of_t<typename M::value_type> *r,
229 const real_of_t<typename M::value_type> *c)
230 {
231 using T = typename M::value_type;
232 using R = real_of_t<T>;
233 using namespace cuda_detail;
234 if (A.nnz() == 0) return;
235 if constexpr (is_complex_value<T>::value) {
236 // No complex kernel: scale on the host.
237 Csr<T, typename M::index_type, Cpu::HostStorage> h(A);
238 std::vector<R> hr(A.rows()), hc(A.cols());
239 if (r)
240 check(cudaMemcpy(hr.data(), r, hr.size() * sizeof(R),
241 cudaMemcpyDeviceToHost),
242 "scale copy");
243 if (c)
244 check(cudaMemcpy(hc.data(), c, hc.size() * sizeof(R),
245 cudaMemcpyDeviceToHost),
246 "scale copy");
247 HostK::scale(h, r ? hr.data() : nullptr, c ? hc.data() : nullptr);
248 check(cudaMemcpy(A.values_mut(), h.values(), A.nnz() * sizeof(T),
249 cudaMemcpyHostToDevice),
250 "scale copy");
251 } else {
252 constexpr int W = Sparse::cuda_kernels::SPARSE_WARPS_PER_BLOCK;
253 Sparse::cuda_kernels::scaleKernel<T, typename M::index_type>
254 <<<static_cast<int>((A.rows() + W - 1) / W), W * 32>>>(
255 A.row_ptr(), A.col_ind(), A.values_mut(), r, c,
256 static_cast<long>(A.rows()));
257 check(cudaGetLastError(), "scaleKernel");
258 }
259 }

261 template <typename M>
262 static void
263 row_norms(const M &A, real_of_t<typename M::value_type> *out, Norm p)
264 {
265 using T = typename M::value_type;
266 using R = real_of_t<T>;
267 using namespace cuda_detail;
268 if (A.nnz() == 0) {
269 check(cudaMemset(out, 0, A.rows() * sizeof(R)), "row_norms");
270 return;
271 }
272 if constexpr (is_complex_value<T>::value) {
273 Csr<T, typename M::index_type, Cpu::HostStorage> h(A);
274 std::vector<R> hn(A.rows());
275 HostK::row_norms(h, hn.data(), p);
276 check(cudaMemcpy(out, hn.data(), hn.size() * sizeof(R),
277 cudaMemcpyHostToDevice),
278 "row_norms copy");
279 } else {
280 constexpr int W = Sparse::cuda_kernels::SPARSE_WARPS_PER_BLOCK;
281 Sparse::cuda_kernels::rowNormKernel<T, typename M::index_type>
282 <<<static_cast<int>((A.rows() + W - 1) / W), W * 32>>>(
283 A.row_ptr(), A.values(), out, static_cast<long>(A.rows()),
284 p == Norm::L1 ? 0 : (p == Norm::L2 ? 1 : 2));
285 check(cudaGetLastError(), "rowNormKernel");
286 }

289 template <typename M>
290 static M
291 transpose(const M &A)
292 {
293 using T = typename M::value_type;
294 using Idx = typename M::index_type;
295 using namespace cuda_detail;
296 if constexpr (sizeof(Idx) != 4) {
297 // cusparseCsr2cscEx2 is 32-bit only: go through the host.
298 Csr<T, Idx, Cpu::HostStorage> h(A);
299 return M(HostK::transpose(h));
300 } else {
301 const size_t m = A.rows(), n = A.cols(), nz = A.nnz();
302 M B(n, m, nz);
303 if (nz == 0) return B;
304 size_t sz = 0;
305 check(cusparseCsr2cscEx2_bufferSize(handle(), m, n, nz, A.values(),
306 A.row_ptr(), A.col_ind(), B.values_mut(), B.row_ptr_mut(),
307 B.col_ind_mut(), value_type<T>(), CUSPARSE_ACTION_NUMERIC,
308 CUSPARSE_INDEX_BASE_ZERO, CUSPARSE_CSR2CSC_ALG1, &sz),
309 "cusparseCsr2cscEx2_bufferSize");
310 PoolBuf buf;
311 buf.reserve(sz);
312 check(cusparseCsr2cscEx2(handle(), m, n, nz, A.values(),
313 A.row_ptr(), A.col_ind(), B.values_mut(), B.row_ptr_mut(),
314 B.col_ind_mut(), value_type<T>(), CUSPARSE_ACTION_NUMERIC,
315 CUSPARSE_INDEX_BASE_ZERO, CUSPARSE_CSR2CSC_ALG1, buf.p),
316 "cusparseCsr2cscEx2");
317 return B;
318 }

321 template <typename M>
322 static M
323 spgemm(const M &A, const M &B)
324 {
325 using T = typename M::value_type;
326 using Idx = typename M::index_type;
327 using namespace cuda_detail;
328 const size_t m = A.rows(), n = B.cols();
329 if (A.nnz() == 0 || B.nnz() == 0) return M(m, n, 0);
330 if constexpr (sizeof(Idx) != 4) {
331 Csr<T, Idx, Cpu::HostStorage> ha(A), hb(B);
332 return M(HostK::spgemm(ha, hb));
333 } else {
334 CusparseCache &ca = cache_of(A);
335 CusparseCache &cb = cache_of(B);
336 cusparseSpMatDescr_t matC;
337 check(cusparseCreateCsr(&matC, m, n, 0, nullptr, nullptr, nullptr,
338 CUSPARSE_INDEX_32I, CUSPARSE_INDEX_32I,
339 CUSPARSE_INDEX_BASE_ZERO, value_type<T>()),
340 "cusparseCreateCsr(C)");
341 cusparseSpGEMMDescr_t d;
342 check(cusparseSpGEMM_createDescr(&d), "cusparseSpGEMM_createDescr");
343 const T alpha = T(1), beta = T(0);
344 const auto op = CUSPARSE_OPERATION_NON_TRANSPOSE;
345 const auto vt = value_type<T>();
346 const auto alg = CUSPARSE_SPGEMM_DEFAULT;
347 size_t s1 = 0, s2 = 0;
348 check(cusparseSpGEMM_workEstimation(handle(), op, op, &alpha,
349 ca.mat, cb.mat, &beta, matC, vt, alg, d, &s1, nullptr),
350 "SpGEMM workEstimation(size)");
351 PoolBuf b1, b2;
352 b1.reserve(s1);
353 check(cusparseSpGEMM_workEstimation(handle(), op, op, &alpha,
354 ca.mat, cb.mat, &beta, matC, vt, alg, d, &s1, b1.p),
355 "SpGEMM workEstimation");
356 check(cusparseSpGEMM_compute(handle(), op, op, &alpha, ca.mat,
357 cb.mat, &beta, matC, vt, alg, d, &s2, nullptr),
358 "SpGEMM compute(size)");
359 b2.reserve(s2);
360 check(cusparseSpGEMM_compute(handle(), op, op, &alpha, ca.mat,
361 cb.mat, &beta, matC, vt, alg, d, &s2, b2.p),
362 "SpGEMM compute");
363 int64_t cr, cc, cn;
364 check(cusparseSpMatGetSize(matC, &cr, &cc, &cn),
365 "cusparseSpMatGetSize");
366 M C(m, n, static_cast<size_t>(cn));
367 if (cn > 0) {
368 check(cusparseCsrSetPointers(matC, C.row_ptr_mut(),
369 C.col_ind_mut(), C.values_mut()),
370 "cusparseCsrSetPointers");
371 check(cusparseSpGEMM_copy(handle(), op, op, &alpha, ca.mat,
372 cb.mat, &beta, matC, vt, alg, d),
373 "cusparseSpGEMM_copy");
374 }
375 cusparseSpGEMM_destroyDescr(d);
376 cusparseDestroySpMat(matC);
377 check(cudaDeviceSynchronize(), "SpGEMM sync");
378 return C;
379 }
