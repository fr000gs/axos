3// Element-wise math and matrix products on host tensors
4// (docs/TENSOR_SPEC.md §6).
5//
6// Element-wise operations are lazy (tensor/expr.h): the operators and
7// functions below build small expression objects, and assigning one to a
8// tensor evaluates the whole chain in a single pass with no temporaries:
9//
10// C = exp(sin(A) + cos(A)) * 2.0 + B;
11//
12// x + y, x - y, x * y, x / y element-wise (* is not a matrix product)
13// x op s, s op x with a scalar s, for + - * /
14// -x
15// sqrt exp Log sin cos tan sinh cosh tanh fabs
16//
17// Operands must have the same shape (checked when the node is built).
18// Element i of the result depends only on element i of the operands, so
19// `A = A + B` is safe. The evaluation runs on all threads above a size
20// threshold that shrinks with the cost of the chain; +, -, *, /, sqrt and
21// fabs use explicit SIMD, the transcendental functions call the scalar
22// std:: versions (so results are bit-identical to the scalar expression).
23//
24// Matrix products (all accept views, whose row stride is used directly):
25// matMul(A, B) lazy A (m x k) * B (k x n); value(i) is entry
26// (i / n, i % n). Assigned directly it runs the fast
27// GEMM into the target; inside a larger expression the
28// product is computed once into a cache first. Do not
29// assign it to one of its operands' views.
30// matMulTile(A, B) eager product, new m x n tensor (GEMV for a vector)
31// matMulBlocked(A, B, C, m, alpha, bias, beta, transA, transB, accumulate)
32// C = alpha op(A) op(B) + beta bias, or C += ... with
33// accumulate; m is a legacy tile-size hint (the GEMM
34// picks its own blocking)
35// matMulND(A, B) batched over the leading dimensions (rank >= 3)
36// gemv, matVec y = alpha op(A) x + beta y
37// transpose(A) new n x m tensor
38// extractSlice / insertSlice, getValue / setValue

56// Rank of a node: that of the first child that declares one.
57template <class L, class R = void, class = void> struct rank_from {};
58template <class L, class R>
59struct rank_from<L, R, std::enable_if_t<has_rank<L>::value>> {
60 static constexpr int rank = L::rank;
61};
62template <class L, class R>
63struct rank_from<L, R,
64 std::enable_if_t<!has_rank<L>::value && !std::is_void_v<R> &&
65 has_rank<R>::value>> {
66 static constexpr int rank = R::rank;
67};

69// Per-element cost without the >= 1 clamp (tensors cost 0).
70template <class E>
71constexpr int
72raw_cost()
73{
74 if constexpr (has_cost<E>::value)
75 return E::cost;
76 else
77 return 1;
78}

80template <class E>
81constexpr bool
82packet_ok_of()
83{
84 if constexpr (has_packet_ok<E>::value)
85 return E::packet_ok;
86 else
87 return false;
88}

90template <class L, class R>
91void
92check_same_shape(const L &l, const R &r)
93{
94 if constexpr (has_rank<L>::value && has_rank<R>::value) {
95 static_assert(
96 L::rank == R::rank, "element-wise operation: rank mismatch");
97 for (int d = 0; d < L::rank; ++d)
98 if (l.size(d) != r.size(d))
99 throw std::runtime_error(
100 "element-wise operation: shape mismatch in dimension " +
101 std::to_string(d) + " (" + std::to_string(l.size(d)) +
102 " vs " + std::to_string(r.size(d)) + ")");
103 }
104}

164// Scalar math that also works for half types (through float) and
165// integers (through double).
166template <class T, class F>
167T
168via(T x, F f)
169{
170 if constexpr (is_half_v<T>)
171 return T(f(static_cast<float>(x)));
172 else if constexpr (std::is_integral_v<T>)
173 return static_cast<T>(f(static_cast<double>(x)));
174 else
175 return f(x);
176}

248template <class T, class L, class R, class Op>
249class BinaryExpr : public Expr<T, BinaryExpr<T, L, R, Op>>,
250 public detail::rank_from<L, R> {
251 public:
252 static constexpr int cost =
253 detail::raw_cost<L>() + detail::raw_cost<R>() + Op::cost;
254 static constexpr bool packet_ok = Op::has_packet &&
255 detail::packet_ok_of<L>() &&
256 detail::packet_ok_of<R>();
257
258 BinaryExpr(const L &l, const R &r) : l_(l), r_(r)
259 { detail::check_same_shape(l, r); }
260
261 T
262 value(size_t i) const
263 { return Op::apply(T(l_.value(i)), T(r_.value(i))); }
264 size_t
265 size(int d) const
266 { return l_.size(d); }
267 bool
268 contiguous() const
269 { return detail::expr_contiguous(l_) && detail::expr_contiguous(r_); }
270 T
271 lvalue(size_t i) const
272 {
273 return Op::apply(
274 T(detail::expr_lvalue(l_, i)), T(detail::expr_lvalue(r_, i)));
275 }
276 void
277 prepare() const
278 {
279 detail::expr_prepare(l_);
280 detail::expr_prepare(r_);
281 }
282 template <class P>
283 AXOS_INLINE typename P::reg
284 packet(size_t i) const
285 {
286 return Op::template packet<P>(
287 l_.template packet<P>(i), r_.template packet<P>(i));
288 }
289
290 private:
291 detail::nested_t<L> l_;
292 detail::nested_t<R> r_;
293};

295// Left ? (s op x) : (x op s)
296template <class T, class E, class Op, bool Left>
297class ScalarExpr : public Expr<T, ScalarExpr<T, E, Op, Left>>,
298 public detail::rank_from<E> {
299 public:
300 static constexpr int cost = detail::raw_cost<E>() + Op::cost;
301 static constexpr bool packet_ok =
302 Op::has_packet && detail::packet_ok_of<E>();
303
304 ScalarExpr(const E &e, T s) : e_(e), s_(s) {}
305
306 T
307 value(size_t i) const
308 { return apply(T(e_.value(i))); }
309 size_t
310 size(int d) const
311 { return e_.size(d); }
312 bool
313 contiguous() const
314 { return detail::expr_contiguous(e_); }
315 T
316 lvalue(size_t i) const
317 { return apply(T(detail::expr_lvalue(e_, i))); }
318 void
319 prepare() const
320 { detail::expr_prepare(e_); }
321 template <class P>
322 AXOS_INLINE typename P::reg
323 packet(size_t i) const
324 {
325 const auto s = P::set1(s_);
326 const auto x = e_.template packet<P>(i);
327 if constexpr (Left)
328 return Op::template packet<P>(s, x);
329 else
330 return Op::template packet<P>(x, s);
331 }
332
333 private:
334 T
335 apply(const T &x) const
336 {
337 if constexpr (Left)
338 return Op::apply(s_, x);
339 else
340 return Op::apply(x, s_);
341 }
342 detail::nested_t<E> e_;
343 T s_;
344};

426// [first, last) byte range covered by a tensor (views included).
427template <class Tn>
428std::pair<const char *, const char *>
429byte_span(const Tn &t)
430{
431 using T = typename Tn::value_type;
432 if (t.size() == 0 || !t.data) return {nullptr, nullptr};
433 size_t last = 0;
434 for (int d = 0; d < Tn::rank; ++d)
435 last += (t.size(d) - 1) * t.stride(d);
436 const char *b = reinterpret_cast<const char *>(t.data);
437 return {b, b + (last + 1) * sizeof(T)};
438}
439
440template <class A, class B>
441bool
442overlaps(const A &a, const B &b)
443{
444 const auto x = byte_span(a), y = byte_span(b);
445 return x.first && y.first && x.first < y.second && y.first < x.second;
446}
447
448template <class T>
449bool
450overlaps_ptr(const T *p, size_t n, std::pair<const char *, const char *> s)
451{
452 if (!p || !s.first || n == 0) return false;
453 const char *b = reinterpret_cast<const char *>(p);
454 return b < s.second && s.first < b + n * sizeof(T);
455}

459// Lazy matrix product. See the header comment.
460template <class T, class SA, class SB>
461class MatMulExpr : public Expr<T, MatMulExpr<T, SA, SB>> {
462 public:
463 using MA = tensorET<2, T, SA>;
464 using MB = tensorET<2, T, SB>;
465 static constexpr int rank = 2;
466 static constexpr int cost = 1;
467 static constexpr bool packet_ok = true; // reads the cache (after prepare)
468
469 MatMulExpr(const MA &A, const MB &B) : A_(A), B_(B)
470 {
471 if (A.size(1) != B.size(0))
472 throw std::runtime_error("matMul: inner dimensions differ (" +
473 std::to_string(A.size(1)) + " vs " +
474 std::to_string(B.size(0)) + ")");
475 }
476
477 size_t
478 size(int d) const
479 { return d == 0 ? A_.size(0) : B_.size(1); }
480
481 T
482 value(size_t i) const
483 {
484 if (cache_.data) return cache_.data[i];
485 const size_t n = B_.size(1), r = i / n, c = i % n;
486 T s(0);
487 for (size_t p = 0; p < A_.size(1); ++p)
488 s += A_(r, p) * B_(p, c);
489 return s;
490 }
491
492 // Computes the product once into an internal cache.
493 void
494 prepare() const
495 {
496 if (cache_.data || A_.size(0) * B_.size(1) == 0) return;
497 cache_ = tensorET<2, T>(std::array<size_t, 2>{A_.size(0), B_.size(1)});
498 run(cache_.data);
499 }
500
501 template <class P>
502 AXOS_INLINE typename P::reg
503 packet(size_t i) const
504 { return P::loadu(cache_.data + i); }
505
506 // Direct evaluation into a contiguous m x n destination, unless it
507 // overlaps an operand.
508 bool
509 assign_into(T *dst) const
510 {
511 const size_t n = A_.size(0) * B_.size(1);
512 if (detail::overlaps_ptr(dst, n, detail::byte_span(A_)) ||
513 detail::overlaps_ptr(dst, n, detail::byte_span(B_)))
514 return false;
515 run(dst);
516 return true;
517 }
518
519 private:
520 void
521 run(T *dst) const
522 {
523 kernels::gemm<T>(false, false, A_.size(0), B_.size(1), A_.size(1), T(1),
524 A_.data, A_.stride(0), B_.data, B_.stride(0), T(0), dst,
525 B_.size(1));
526 }
527 const MA &A_;
528 const MB &B_;
529 mutable tensorET<2, T> cache_;
530};
531
532template <class T, class SA, class SB>
533MatMulExpr<T, SA, SB>
534matMul(const tensorET<2, T, SA> &A, const tensorET<2, T, SB> &B)
535{
536 static_assert(is_host_storage_v<SA> && is_host_storage_v<SB>,
537 "matMul works on host tensors");
538 return MatMulExpr<T, SA, SB>(A, B);
539}

541// y = alpha op(A) x + beta y.
542template <class T, class SA, class SX, class SY>
543void
544gemv(const tensorET<2, T, SA> &A, const tensorET<1, T, SX> &x,
545 tensorET<1, T, SY> &y, detail::scalar_arg_t<T> alpha = T(1),
546 detail::scalar_arg_t<T> beta = T(0), bool trans = false)
547{
548 const size_t m = A.size(0), n = A.size(1);
549 if (x.size() != (trans ? m : n) || y.size() != (trans ? n : m))
550 throw std::runtime_error("gemv: dimension mismatch");
551 kernels::gemv<T>(
552 trans, m, n, alpha, A.data, A.stride(0), x.data, beta, y.data);
553}

555template <class T, class SA, class SX>
556tensorET<1, T>
557matVec(const tensorET<2, T, SA> &A, const tensorET<1, T, SX> &x,
558 bool trans = false)
559{
560 tensorET<1, T> y(trans ? A.size(1) : A.size(0));
561 gemv(A, x, y, T(1), T(0), trans);

565// Eager product: m x n matrix (GEMM), or a vector for a vector argument.
566template <class T, class SA, class SB>
567tensorET<2, T>
568matMulTile(const tensorET<2, T, SA> &A, const tensorET<2, T, SB> &B)
569{
570 const size_t m = A.size(0), k = A.size(1), n = B.size(1);
571 if (B.size(0) != k)
572 throw std::runtime_error("matMulTile: inner dimensions differ");
573 tensorET<2, T> C(std::array<size_t, 2>{m, n});
574 if (n == 1 && B.is_contiguous())
575 kernels::gemv<T>(
576 false, m, k, T(1), A.data, A.stride(0), B.data, T(0), C.data);
577 else
578 kernels::gemm<T>(false, false, m, n, k, T(1), A.data, A.stride(0),
579 B.data, B.stride(0), T(0), C.data, n);
580 return C;
581}

588// C = alpha op(A) op(B) + beta bias (accumulate: C += the same).
589// C is allocated when empty; otherwise its shape must match. `m` is a
590// legacy tile-size hint and is not used.
591template <class T, class SA, class SB, class SC>
592void
593matMulBlocked(const tensorET<2, T, SA> &A, const tensorET<2, T, SB> &B,
594 tensorET<2, T, SC> &C, size_t m, detail::scalar_arg_t<T> alpha = T(1),
595 const detail::scalar_arg_t<tensorET<2, T>> *bias = nullptr,
596 detail::scalar_arg_t<T> beta = T(1), bool transA = false,
597 bool transB = false, bool accumulate = false)
598{
599 static_assert(std::is_same_v<SC, Cpu::HostStorage<T>>,
600 "matMulBlocked writes a host matrix");
601 (void)m;
602 const size_t M = transA ? A.size(1) : A.size(0);
603 const size_t K = transA ? A.size(0) : A.size(1);
604 const size_t N = transB ? B.size(0) : B.size(1);
605 if ((transB ? B.size(1) : B.size(0)) != K)
606 throw std::runtime_error("matMulBlocked: inner dimensions differ");
607 if (C.size() == 0 && C.data == nullptr) {
608 C = tensorET<2, T, SC>(std::array<size_t, 2>{M, N});
609 if (accumulate) C.fill(T(0));
610 }
611 if (C.size(0) != M || C.size(1) != N)
612 throw std::runtime_error("matMulBlocked: C has the wrong shape");
613 if (bias && (bias->size(0) != M || bias->size(1) != N))
614 throw std::runtime_error("matMulBlocked: bias has the wrong shape");
615 if (detail::overlaps(C, A) ||
616 detail::overlaps(C, B)) { // C aliases an operand
617 tensorET<2, T> tmp = C.clone();
618 matMulBlocked(
619 A, B, tmp, m, alpha, bias, beta, transA, transB, accumulate);
620 Cpu::Backend::copy_data(C, tmp);
621 return;
622 }
623 T cbeta = accumulate ? T(1) : T(0);
624 if (bias) {
625 const size_t ldc = C.stride(0), ldb = bias->stride(0);
626 detail::parallel_for(
627 M, std::max<size_t>(1, 32768 / std::max<size_t>(N, 1)),
628 [&](size_t r0, size_t r1) {
629 for (size_t i = r0; i < r1; ++i) {
630 T *c = C.data + i * ldc;
631 const T *b = bias->data + i * ldb;
632 for (size_t j = 0; j < N; ++j)
633 c[j] = accumulate ? c[j] + beta * b[j] : beta * b[j];
634 }
635 },
636 1);
637 cbeta = T(1);
638 }
639 kernels::gemm<T>(transA, transB, M, N, K, alpha, A.data, A.stride(0),
640 B.data, B.stride(0), cbeta, C.data, C.stride(0));
641}

643// Batched product over the leading dimensions (rank >= 3): the last two
644// dimensions multiply, the leading ones must match.
645template <int R, class T, class SA, class SB>
646tensorET<R, T>
647matMulND(const tensorET<R, T, SA> &A, const tensorET<R, T, SB> &B)
648{
649 static_assert(
650 R >= 3, "matMulND needs rank >= 3 (use matMulTile for matrices)");
651 std::array<size_t, R> shape{};
652 size_t batch = 1;
653 for (int d = 0; d < R - 2; ++d) {
654 if (A.size(d) != B.size(d))
655 throw std::runtime_error("matMulND: leading dimensions differ");
656 shape[d] = A.size(d);
657 batch *= A.size(d);
658 }
659 const size_t m = A.size(R - 2), k = A.size(R - 1), n = B.size(R - 1);
660 if (B.size(R - 2) != k)
661 throw std::runtime_error("matMulND: inner dimensions differ");
662 if (!A.is_contiguous() || !B.is_contiguous())
663 throw std::runtime_error("matMulND: operands must be contiguous");
664 shape[R - 2] = m;
665 shape[R - 1] = n;
666 tensorET<R, T> C(shape);
667 auto one = [&](size_t b) {
668 kernels::gemm<T>(false, false, m, n, k, T(1), A.data + b * m * k, k,
669 B.data + b * k * n, n, T(0), C.data + b * m * n, n);
670 };
671 if (double(m) * n * k < kernels::blas_detail::kParallelFma && batch > 1)
672 detail::parallel_for(
673 batch, 1,
674 [&](size_t b0, size_t b1) {
675 for (size_t b = b0; b < b1; ++b)
676 one(b);
677 },
678 1);
679 else
680 for (size_t b = 0; b < batch; ++b)
681 one(b);
682 return C;
683}

685// The b-th matrix (last two dimensions) of a rank >= 3 tensor, with the
686// leading dimensions flattened in row-major order.
687template <int R, class T, class S>
688tensorET<2, T>
689extractSlice(const tensorET<R, T, S> &t, size_t b)
690{
691 static_assert(R >= 3, "extractSlice needs rank >= 3");
692 const size_t m = t.size(R - 2), n = t.size(R - 1);
693 const tensorET<R, T, S> c = t.is_contiguous() ? t : t.clone();
694 if (b >= c.size() / std::max<size_t>(1, m * n))
695 throw std::runtime_error("extractSlice: index out of range");
696 tensorET<2, T> M(std::array<size_t, 2>{m, n});
697 std::copy(c.data + b * m * n, c.data + (b + 1) * m * n, M.data);
698 return M;
699}

701template <int R, class T, class S, class SM>
702void
703insertSlice(tensorET<R, T, S> &t, size_t b, const tensorET<2, T, SM> &M)
704{
705 static_assert(R >= 3, "insertSlice needs rank >= 3");
706 const size_t m = t.size(R - 2), n = t.size(R - 1);
707 if (M.size(0) != m || M.size(1) != n)
708 throw std::runtime_error("insertSlice: matrix shape does not match");
709 if (!t.is_contiguous())
710 throw std::runtime_error("insertSlice: target must be contiguous");
711 if (b >= t.size() / std::max<size_t>(1, m * n))
712 throw std::runtime_error("insertSlice: index out of range");
713 for (size_t i = 0; i < m; ++i)
714 std::copy(M.data + i * M.stride(0), M.data + i * M.stride(0) + n,
715 t.data + b * m * n + i * n);
716}
