3// Elementwise / reduction kernels for the PDLP solver, written once as
4// host+device functors and executed by Parallel<Backend>:
5//
6// for_each(n, f) f(i)
7// reduce<K>(slot, n, f) f(i, double acc[K]) accumulates K sums
8// fetch() returns the accumulated slots on the host
9//
10// Parallel<Cpu::Backend> is OpenMP; Parallel<Cuda::Backend> lives in
11// pdlp_cuda.h.

33AXOS_HD inline double
34dmin(double a, double b)
35{ return a < b ? a : b; }
36AXOS_HD inline double
37dmax(double a, double b)
38{ return a > b ? a : b; }
39AXOS_HD inline bool
40fin(double v)
41{ return v > -1e300 && v < 1e300; }
42AXOS_HD inline double
43clampd(double v, double lo, double hi)
44{ return v < lo ? lo : (v > hi ? hi : v); }

46// x+ = proj_[lb,ub](x - tau (c - A^T y))
47struct XUpdate {
48 const double *c, *aty, *lb, *ub, *x;
49 double *xn;
50 double tau;
51 AXOS_HD void
52 operator()(size_t i) const
53 { xn[i] = clampd(x[i] - tau * (c[i] - aty[i]), lb[i], ub[i]); }
54};
55
56// y+ = max(0, y + s (l - t)) + min(0, y + s (u - t)), t = 2 A x+ - A x
57struct YUpdate {
58 const double *l, *u, *y, *ax, *axn;
59 double *yn;
60 double sigma;
61 AXOS_HD void
62 operator()(size_t i) const
63 {
64 const double t = 2.0 * axn[i] - ax[i];
65 const double a = y[i] + sigma * (l[i] - t);
66 const double b = y[i] + sigma * (u[i] - t);
67 yn[i] = dmax(a, 0.0) + dmin(b, 0.0);
68 }
69};
70
71// Fused x update and ||dx||^2 (one pass over the columns).
72struct XUpdateStep {
73 const double *c, *aty, *lb, *ub, *x;
74 double *xn;
75 double tau;
76 static constexpr int K = 1;
77 AXOS_HD void
78 operator()(size_t i, double *acc) const
79 {
80 const double v = clampd(x[i] - tau * (c[i] - aty[i]), lb[i], ub[i]);
81 xn[i] = v;
82 const double d = v - x[i];
83 acc[0] += d * d;
84 }
85};
86
87// Fused y update, ||dy||^2 and dy^T (A dx) (one pass over the rows).
88struct YUpdateStep {
89 const double *l, *u, *y, *ax, *axn;
90 double *yn;
91 double sigma;
92 static constexpr int K = 2;
93 AXOS_HD void
94 operator()(size_t i, double *acc) const
95 {
96 const double t = 2.0 * axn[i] - ax[i];
97 const double a = y[i] + sigma * (l[i] - t);
98 const double b = y[i] + sigma * (u[i] - t);
99 const double v = dmax(a, 0.0) + dmin(b, 0.0);
100 yn[i] = v;
101 const double dy = v - y[i];
102 acc[0] += dy * dy;
103 acc[1] += dy * (axn[i] - ax[i]);
104 }
105};

107// Reflected Halpern step, in place on the iterate z:
108// z = w1 (2 T(z) - z) + w0 anchor,
109// where `image` holds T(z) and `anchor` is the last restart point.
110struct HalpernMix {
111 double *z;
112 const double *image, *anchor;
113 double w1, w0;
114 AXOS_HD void
115 operator()(size_t i) const
116 { z[i] = w1 * (2.0 * image[i] - z[i]) + w0 * anchor[i]; }
117};

168// KKT pieces over rows: primal residual (scaled and original scale) and the
169// row part of the dual objective.
170struct KktM {
171 const double *ax, *y, *l, *u, *rinv;
172 static constexpr int K = 3;
173 AXOS_HD void
174 operator()(size_t i, double *acc) const
175 {
176 const double d = ax[i] - clampd(ax[i], l[i], u[i]);
177 const double dorig = d * rinv[i];
178 acc[0] += d * d;
179 acc[1] += dorig * dorig;
180 const double yi = y[i];
181 acc[2] += yi > 0 ? l[i] * yi : (yi < 0 ? u[i] * yi : 0.0);
182 }
183};
184
185// KKT pieces over columns: dual residual (scaled / original), column part of
186// the dual objective, and c^T x.
187struct KktN {
188 const double *x, *aty, *c, *lb, *ub, *cinv;
189 static constexpr int K = 4;
190 AXOS_HD void
191 operator()(size_t i, double *acc) const
192 {
193 const double z = c[i] - aty[i];
194 const double zlo = fin(ub[i]) ? -INFINITY : 0.0;
195 const double zhi = fin(lb[i]) ? INFINITY : 0.0;
196 const double lam = clampd(z, zlo, zhi);
197 const double dr = z - lam;
198 const double drorig = dr * cinv[i];
199 acc[0] += dr * dr;
200 acc[1] += drorig * drorig;
201 acc[2] += lam > 0 ? lb[i] * lam : (lam < 0 ? ub[i] * lam : 0.0);
202 acc[3] += c[i] * x[i];
203 }
204};

206// Candidate primal ray dx = x - xr (columns): |dx|^2, c^T dx and the
207// violation of the recession cone of the variable bounds.
208struct PrimalRayN {
209 const double *x, *xr, *c, *lb, *ub;
210 static constexpr int K = 3;
211 AXOS_HD void
212 operator()(size_t i, double *acc) const
213 {
214 const double dx = x[i] - xr[i];
215 acc[0] += dx * dx;
216 acc[1] += c[i] * dx;
217 const bool lf = fin(lb[i]), uf = fin(ub[i]);
218 const double v = (lf && !uf) ? dmin(dx, 0.0)
219 : (!lf && uf) ? dmax(dx, 0.0)
220 : (lf && uf) ? dx
221 : 0.0;
222 acc[2] += v * v;
223 }
224};
225
226// ... and of the row activity: violation of the recession cone of [l, u].
227struct PrimalRayM {
228 const double *ax, *axr, *l, *u;
229 static constexpr int K = 1;
230 AXOS_HD void
231 operator()(size_t i, double *acc) const
232 {
233 const double d = ax[i] - axr[i];
234 const bool lf = fin(l[i]), uf = fin(u[i]);
235 const double v = (lf && !uf) ? dmin(d, 0.0)
236 : (!lf && uf) ? dmax(d, 0.0)
237 : (lf && uf) ? d
238 : 0.0;
239 acc[0] += v * v;
240 }
241};

243// Candidate dual ray dy = y - yr (rows): |dy|^2, violation of the sign cone
244// and the row part of the homogeneous dual objective.
245struct DualRayM {
246 const double *y, *yr, *l, *u;
247 static constexpr int K = 3;
248 AXOS_HD void
249 operator()(size_t i, double *acc) const
250 {
251 const double dy = y[i] - yr[i];
252 acc[0] += dy * dy;
253 const bool lf = fin(l[i]), uf = fin(u[i]);
254 const double v = (lf && !uf) ? dmin(dy, 0.0)
255 : (!lf && uf) ? dmax(dy, 0.0)
256 : (!lf && !uf) ? dy
257 : 0.0;
258 acc[1] += v * v;
259 const double p = (lf && !uf) ? dmax(dy, 0.0)
260 : (!lf && uf) ? dmin(dy, 0.0)
261 : (lf && uf) ? dy
262 : 0.0;
263 acc[2] += p > 0 ? l[i] * p : (p < 0 ? u[i] * p : 0.0);
264 }
265};

267// ... and over columns: reduced-cost violation for c = 0, lambda = -A^T dy.
268struct DualRayN {
269 const double *aty, *atyr, *lb, *ub;
270 static constexpr int K = 2;
271 AXOS_HD void
272 operator()(size_t i, double *acc) const
273 {
274 const double lin = -(aty[i] - atyr[i]);
275 const double zlo = fin(ub[i]) ? -INFINITY : 0.0;
276 const double zhi = fin(lb[i]) ? INFINITY : 0.0;
277 const double lam = clampd(lin, zlo, zhi);
278 const double dr = lin - lam;
279 acc[0] += dr * dr;
280 acc[1] += lam > 0 ? lb[i] * lam : (lam < 0 ? ub[i] * lam : 0.0);
281 }
282};

342template <typename Backend> struct Parallel;
343
344template <> struct Parallel<Cpu::Backend> {
345 static constexpr size_t kParallelMin = pdlp::kParallelMin;
346 double slots[pdlp::kSlots] = {};
347
348 template <typename F>
349 void
350 for_each(size_t n, F f)
351 {
352#pragma omp parallel for schedule(static) if (n > kParallelMin)
353 for (long i = 0; i < static_cast<long>(n); ++i)
354 f(static_cast<size_t>(i));
355 }
356
357 void
358 zero()
359 { std::memset(slots, 0, sizeof(slots)); }
360
361 // Adds the K sums of f over [0, n) into slots[slot .. slot+K).
362 template <typename F>
363 void
364 reduce(int slot, size_t n, F f)
365 {
366 constexpr int K = F::K;
367#pragma omp parallel if (n > kParallelMin)
368 {
369 double loc[K] = {};
370#pragma omp for schedule(static) nowait
371 for (long i = 0; i < static_cast<long>(n); ++i)
372 f(static_cast<size_t>(i), loc);
373#pragma omp critical(axos_pdlp_reduce)
374 for (int k = 0; k < K; ++k)
375 slots[slot + k] += loc[k];
376 }
377 }
378
379 void
380 set_slot(int slot, double v)
381 { slots[slot] = v; }
382
383 // slots[slot] = min(slots[slot], min over i of f(i)).
384 template <typename F>
385 void
386 reduce_min(int slot, size_t n, F f)
387 {
388 double m = slots[slot];
389 // per-thread minima (MSVC's OpenMP has no min reduction)
390#pragma omp parallel if (n > kParallelMin)
391 {
392 double loc = m;
393#pragma omp for schedule(static) nowait
394 for (long i = 0; i < static_cast<long>(n); ++i) {
395 const double v = f(static_cast<size_t>(i));
396 if (v < loc) loc = v;
397 }
398#pragma omp critical(axos_pdlp_reduce_min)
399 if (loc < m) m = loc;
400 }
401 slots[slot] = m;
402 }
403
404 const double *
405 fetch()
406 { return slots; }
407
408 void
409 copy(double *dst, const double *src, size_t n)
410 { std::memcpy(dst, src, n * sizeof(double)); }
411};
