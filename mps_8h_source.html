3// MPS reader and writer (free-format tokenization, which also reads normal
4// fixed-format files as long as names contain no spaces).
5//
6// Supported: NAME, OBJSENSE / OBJSENSE MAX|MIN, OBJNAME, ROWS (N/L/G/E, extra
7// N rows are dropped), COLUMNS (with MARKER INTORG/INTEND), RHS (a value on
8// the objective row is minus the objective constant), RANGES, BOUNDS
9// (UP LO FX FR MI PL BV LI UI), ENDATA. Values with magnitude >= 1e20 are
10// treated as infinite. SC (semi-continuous) bounds are rejected.
11// Quadratic objective sections (QPS: QUADOBJ with one triangle, QMATRIX with
12// both, QSECTION for the objective row) are returned through `quad` as full
13// symmetric entries of Q (objective 1/2 x^T Q x); without `quad` they are an
14// error. See solver/io/qps.h.
15//
16// A maximization problem is returned negated with `maximize = true`, matching
17// the LpProblem convention.

39inline double
40clamp_inf(double v)
41{
42 if (v >= kMpsInf) return kInf;
43 if (v <= -kMpsInf) return -kInf;
44 return v;

47inline std::vector<std::string>
48tokens(const std::string &line)
49{
50 std::vector<std::string> t;
51 std::istringstream is(line);
52 std::string w;
53 while (is >> w)
54 t.push_back(w);
55 return t;
56}

58// Fixed-format MPS data line: the non-empty fields at columns 2-3, 5-12,
59// 15-22, 25-36, 40-47 and 50-61, trimmed (names may contain spaces).
60inline std::vector<std::string>
61fixed_tokens(const std::string &line)
62{
63 static const size_t pos[6][2] = {
64 {1, 3}, {4, 12}, {14, 22}, {24, 36}, {39, 47}, {49, 61}};
65 std::vector<std::string> t;
66 for (const auto &f : pos) {
67 if (f[0] >= line.size()) break;
68 std::string s = line.substr(f[0], f[1] - f[0]);
69 const size_t a = s.find_first_not_of(" \t"),
70 b = s.find_last_not_of(" \t");
71 if (a != std::string::npos) t.push_back(s.substr(a, b - a + 1));
72 }
73 return t;
74}

76inline double
77to_double(const std::string &s, size_t line)
78{
79 char *end = nullptr;
80 double v = std::strtod(s.c_str(), &end);
81 if (end == s.c_str() || *end != '\0')
82 throw std::runtime_error(
83 "MPS line " + std::to_string(line) + ": bad number '" + s + "'");
84 return v;
85}

88upper(std::string s)

102// fixed_format: split data lines by column position (for files whose names
103// contain spaces); header lines are always split on whitespace.
104inline LpProblem
105read_mps(std::istream &in, std::vector<QuadEntry> *quad = nullptr,
106 bool fixed_format = false)
107{
108 using namespace mps_detail;
109 LpProblem p;
110 enum class Sec {
111 None,
112 Name,
113 ObjSense,
114 ObjName,
115 Rows,
116 Columns,
117 Rhs,
118 Ranges,
119 Bounds,
120 Quad,
121 End
122 };
123 bool quad_full = false; // QMATRIX lists both triangles, QUADOBJ one
124 Sec sec = Sec::None;
125
126 std::string obj_row, obj_name_wanted;
127 std::vector<char> row_type; // per constraint row
128 std::vector<double> rhs, range; // per constraint row
129 std::vector<std::string> row_names;
130 std::unordered_map<std::string, int> row_index; // constraint rows only
131 std::unordered_set<std::string> dropped_rows; // extra N rows
132 std::unordered_map<std::string, int> col_index;
133 std::vector<double> c, lb, ub;
134 std::vector<uint8_t> integer;
135 std::vector<std::string> col_names;
136 std::vector<int> ci_i, ci_j; // triplets
137 std::vector<double> ci_v;
138 bool maximize = false, in_int = false, have_rhs_obj = false;
139 double obj_rhs = 0;
140
141 auto col_of = [&](const std::string &name) {
142 auto it = col_index.find(name);
143 if (it != col_index.end()) return it->second;
144 int id = static_cast<int>(col_names.size());
145 col_index[name] = id;
146 col_names.push_back(name);
147 c.push_back(0);
148 lb.push_back(0);
149 ub.push_back(kInf);
150 integer.push_back(in_int ? 1 : 0);
151 return id;
152 };
153
154 std::string line;
155 size_t ln = 0;
156 while (std::getline(in, line)) {
157 ++ln;
158 if (!line.empty() && line.back() == '\r') line.pop_back();
159 if (line.empty() || line[0] == '*') continue;
160 const bool data_line =
161 std::isspace(static_cast<unsigned char>(line[0]));
162 auto t =
163 (fixed_format && data_line) ? fixed_tokens(line) : tokens(line);
164 if (t.empty()) continue;
165
166 if (!std::isspace(static_cast<unsigned char>(line[0]))) { // header
167 std::string h = upper(t[0]);
168 if (h == "NAME") {
169 sec = Sec::Name;
170 p.name = t.size() > 1 ? t[1] : "";
171 } else if (h == "OBJSENSE") {
172 sec = Sec::ObjSense;
173 if (t.size() > 1) maximize = upper(t[1]).rfind("MAX", 0) == 0;
174 } else if (h == "OBJNAME") {
175 sec = Sec::ObjName;
176 } else if (h == "ROWS")
177 sec = Sec::Rows;
178 else if (h == "COLUMNS")
179 sec = Sec::Columns;
180 else if (h == "RHS")
181 sec = Sec::Rhs;
182 else if (h == "RANGES")
183 sec = Sec::Ranges;
184 else if (h == "BOUNDS")
185 sec = Sec::Bounds;
186 else if (h == "QUADOBJ" || h == "QMATRIX" || h == "QSECTION") {
187 if (!quad)
188 throw std::runtime_error(
189 "MPS line " + std::to_string(ln) +
190 ": quadratic objective; read it with read_qps");
191 sec = Sec::Quad;
192 quad_full = (h == "QMATRIX");
193 if (h == "QSECTION" && t.size() > 1 && t[1] != obj_row)
194 throw std::runtime_error(
195 "MPS line " + std::to_string(ln) +
196 ": quadratic constraints are not supported");
197 } else if (h == "ENDATA") {
198 sec = Sec::End;
199 break;
200 } else
201 throw std::runtime_error("MPS line " + std::to_string(ln) +
202 ": unknown section '" + t[0] + "'");
203 continue;
204 }
205
206 switch (sec) {
207 case Sec::ObjSense:
208 maximize = upper(t[0]).rfind("MAX", 0) == 0;
209 break;
210 case Sec::ObjName:
211 obj_name_wanted = t[0];
212 break;
213 case Sec::Rows: {
214 if (t.size() < 2)
215 throw std::runtime_error(
216 "MPS line " + std::to_string(ln) + ": bad ROWS entry");
217 char ty = static_cast<char>(std::toupper(t[0][0]));
218 if (ty == 'N') {
219 if (obj_row.empty() &&
220 (obj_name_wanted.empty() || obj_name_wanted == t[1]))
221 obj_row = t[1];
222 else
223 dropped_rows.insert(t[1]); // extra N rows are ignored
224 } else if (ty == 'L' || ty == 'G' || ty == 'E') {
225 row_index[t[1]] = static_cast<int>(row_type.size());
226 row_type.push_back(ty);
227 rhs.push_back(0);
228 range.push_back(std::nan(""));
229 row_names.push_back(t[1]);
230 } else {
231 throw std::runtime_error("MPS line " + std::to_string(ln) +
232 ": bad row type '" + t[0] + "'");
233 }
234 break;
235 }
236 case Sec::Columns: {
237 if (t.size() >= 3 && t[1] == "'MARKER'") {
238 in_int = (t[2] == "'INTORG'");
239 break;
240 }
241 if (t.size() < 3 || t.size() % 2 == 0)
242 throw std::runtime_error(
243 "MPS line " + std::to_string(ln) + ": bad COLUMNS entry");
244 int j = col_of(t[0]);
245 for (size_t k = 1; k + 1 < t.size(); k += 2) {
246 double v = to_double(t[k + 1], ln);
247 if (t[k] == obj_row) {
248 c[j] += v;
249 } else {
250 auto it = row_index.find(t[k]);
251 if (it == row_index.end()) {
252 if (dropped_rows.count(t[k])) continue;
253 throw std::runtime_error(
254 "MPS line " + std::to_string(ln) +
255 ": unknown row '" + t[k] + "'");
256 }
257 ci_i.push_back(it->second);
258 ci_j.push_back(j);
259 ci_v.push_back(v);
260 }
261 }
262 break;
263 }
264 case Sec::Rhs:
265 case Sec::Ranges: {
266 size_t start = (t.size() % 2 == 1) ? 1 : 0; // optional set name
267 for (size_t k = start; k + 1 < t.size(); k += 2) {
268 double v = to_double(t[k + 1], ln);
269 if (sec == Sec::Rhs && t[k] == obj_row) {
270 obj_rhs = v;
271 have_rhs_obj = true;
272 continue;
273 }
274 auto it = row_index.find(t[k]);
275 if (it == row_index.end()) {
276 if (dropped_rows.count(t[k])) continue;
277 throw std::runtime_error("MPS line " + std::to_string(ln) +
278 ": unknown row '" + t[k] + "'");
279 }
280 if (sec == Sec::Rhs)
281 rhs[it->second] = v;
282 else
283 range[it->second] = v;
284 }
285 break;
286 }
287 case Sec::Bounds: {
288 std::string ty = upper(t[0]);
289 bool valued =
290 !(ty == "FR" || ty == "MI" || ty == "PL" || ty == "BV");
291 size_t need = valued ? 3 : 2; // without a bound-set name
292 if (t.size() < need)
293 throw std::runtime_error(
294 "MPS line " + std::to_string(ln) + ": bad BOUNDS entry");
295 size_t name_at = (t.size() == need) ? 1 : 2;
296 int j = col_of(t[name_at]);
297 double v = valued ? clamp_inf(to_double(t[name_at + 1], ln)) : 0;
298 if (ty == "UP") {
299 ub[j] = v;
300 if (v < 0 && lb[j] == 0) lb[j] = -kInf;
301 } else if (ty == "LO")
302 lb[j] = v;
303 else if (ty == "FX")
304 lb[j] = ub[j] = v;
305 else if (ty == "FR") {
306 lb[j] = -kInf;
307 ub[j] = kInf;
308 } else if (ty == "MI")
309 lb[j] = -kInf;
310 else if (ty == "PL")
311 ub[j] = kInf;
312 else if (ty == "BV") {
313 lb[j] = 0;
314 ub[j] = 1;
315 integer[j] = 1;
316 } else if (ty == "LI") {
317 lb[j] = v;
318 integer[j] = 1;
319 } else if (ty == "UI") {
320 ub[j] = v;
321 integer[j] = 1;
322 } else
323 throw std::runtime_error("MPS line " + std::to_string(ln) +
324 ": unsupported bound type '" + t[0] +
325 "'");
326 break;
327 }
328 case Sec::Quad: {
329 if (t.size() < 3)
330 throw std::runtime_error(
331 "MPS line " + std::to_string(ln) + ": bad quadratic entry");
332 const int a = col_of(t[0]), b = col_of(t[1]);
333 const double v = to_double(t[2], ln);
334 quad->push_back({a, b, v});
335 if (!quad_full && a != b) quad->push_back({b, a, v});
336 break;
337 }
338 default:
339 throw std::runtime_error(
340 "MPS line " + std::to_string(ln) + ": data outside a section");
341 }
342 }
343 if (obj_row.empty() && row_type.empty() && col_names.empty())
344 throw std::runtime_error("MPS: empty or malformed file");
345
346 const size_t m = row_type.size(), n = col_names.size();
347 p.row_lb.assign(m, -kInf);
348 p.row_ub.assign(m, kInf);
349 for (size_t i = 0; i < m; ++i) {
350 const double r = clamp_inf(rhs[i]);
351 const bool has_range = !std::isnan(range[i]);
352 const double R = has_range ? std::abs(range[i]) : 0;
353 switch (row_type[i]) {
354 case 'L':
355 p.row_ub[i] = r;
356 if (has_range) p.row_lb[i] = r - R;
357 break;
358 case 'G':
359 p.row_lb[i] = r;
360 if (has_range) p.row_ub[i] = r + R;
361 break;
362 default: // E
363 if (!has_range) {
364 p.row_lb[i] = p.row_ub[i] = r;
365 } else if (range[i] >= 0) {
366 p.row_lb[i] = r;
367 p.row_ub[i] = r + R;
368 } else {
369 p.row_lb[i] = r - R;
370 p.row_ub[i] = r;
371 }
372 }
373 }
374 Sparse::CooBuilder<double> b(m, n);
375 b.reserve(ci_v.size());
376 for (size_t k = 0; k < ci_v.size(); ++k)
377 b.add(ci_i[k], ci_j[k], ci_v[k]);
378 p.A = b.build();
379 p.c = c;
380 p.col_lb = lb;
381 p.col_ub = ub;
382 p.offset = have_rhs_obj ? -obj_rhs : 0.0;
383 p.row_names = row_names;
384 p.col_names = col_names;
385 bool any_int = false;
386 for (auto v : integer)
387 any_int |= (v != 0);
388 if (any_int) p.is_integer = integer;
389 if (maximize) {
390 p.maximize = true;
391 for (auto &v : p.c)
392 v = -v;
393 p.offset = -p.offset;
394 }
395 return p;
396}

406// Writes free-format MPS that read_mps() (and other readers) accept.
407inline void
408write_mps(std::ostream &out, const LpProblem &p)
409{
410 const size_t m = p.rows(), n = p.cols();
411 auto rname = [&](size_t i) {
412 return i < p.row_names.size() ? p.row_names[i]
413 : "R" + std::to_string(i + 1);
414 };
415 auto cname = [&](size_t j) {
416 return j < p.col_names.size() ? p.col_names[j]
417 : "C" + std::to_string(j + 1);
418 };
419 out.precision(17);
420 const double sgn = p.maximize ? -1.0 : 1.0;
421 out << "NAME " << (p.name.empty() ? "AXOS" : p.name) << "\n";
422 if (p.maximize) out << "OBJSENSE\n MAX\n";
423 out << "ROWS\n N OBJ\n";
424 // A free row (both bounds infinite) constrains nothing; it is written as
425 // an extra N row, which readers drop.
426 for (size_t i = 0; i < m; ++i) {
427 const bool lo = std::isfinite(p.row_lb[i]),
428 hi = std::isfinite(p.row_ub[i]);
429 char ty = (lo && hi && p.row_lb[i] == p.row_ub[i]) ? 'E'
430 : hi ? 'L'
431 : lo ? 'G'
432 : 'N';
433 out << ' ' << ty << ' ' << rname(i) << "\n";
434 }
435 // Column-wise entries via the transpose.
436 HostMatrix At = p.A.transpose();
437 out << "COLUMNS\n";
438 bool in_int = false;
439 int marker = 0;
440 for (size_t j = 0; j < n; ++j) {
441 const bool is_int = !p.is_integer.empty() && p.is_integer[j];
442 if (is_int != in_int) {
443 out << " MARKER" << marker++ << " 'MARKER' '"
444 << (is_int ? "INTORG" : "INTEND") << "'\n";
445 in_int = is_int;
446 }
447 if (p.c[j] != 0)
448 out << " " << cname(j) << " OBJ " << sgn * p.c[j] << "\n";
449 for (int k = At.row_ptr()[j]; k < At.row_ptr()[j + 1]; ++k)
450 out << " " << cname(j) << ' ' << rname(At.col_ind()[k]) << ' '
451 << At.values()[k] << "\n";
452 if (p.c[j] == 0 && At.row_ptr()[j] == At.row_ptr()[j + 1])
453 out << " " << cname(j) << " OBJ 0\n"; // keep empty columns
454 }
455 if (in_int) out << " MARKER" << marker++ << " 'MARKER' 'INTEND'\n";
456 out << "RHS\n";
457 if (p.offset != 0) out << " RHS OBJ " << -sgn * p.offset << "\n";
458 std::vector<int> ranged;
459 for (size_t i = 0; i < m; ++i) {
460 const bool lo = std::isfinite(p.row_lb[i]),
461 hi = std::isfinite(p.row_ub[i]);
462 double r = (lo && hi && p.row_lb[i] == p.row_ub[i]) ? p.row_lb[i]
463 : hi ? p.row_ub[i]
464 : lo ? p.row_lb[i]
465 : 0.0;
466 if (r != 0) out << " RHS " << rname(i) << ' ' << r << "\n";
467 if (lo && hi && p.row_lb[i] != p.row_ub[i])
468 ranged.push_back(static_cast<int>(i));
469 }
470 if (!ranged.empty()) {
471 out << "RANGES\n";
472 for (int i : ranged)
473 out << " RNG " << rname(i) << ' ' << (p.row_ub[i] - p.row_lb[i])
474 << "\n";
475 }
476 out << "BOUNDS\n";
477 for (size_t j = 0; j < n; ++j) {
478 const double l = p.col_lb[j], u = p.col_ub[j];
479 const bool is_int = !p.is_integer.empty() && p.is_integer[j];
480 if (l == u) {
481 out << " FX BND " << cname(j) << ' ' << l << "\n";
482 continue;
483 }
484 if (std::isinf(l) && std::isinf(u)) {
485 out << " FR BND " << cname(j) << "\n";
486 continue;
487 }
488 if (std::isinf(l))
489 out << " MI BND " << cname(j) << "\n";
490 else if (l != 0)
491 out << " LO BND " << cname(j) << ' ' << l << "\n";
492 if (std::isfinite(u))
493 out << " UP BND " << cname(j) << ' ' << u << "\n";
494 else if (is_int)
495 out << " PL BND " << cname(j) << "\n";
496 }
497 out << "ENDATA\n";
498}
