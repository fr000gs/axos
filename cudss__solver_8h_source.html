3// SparseLdlt on the GPU, built on NVIDIA cuDSS. Same interface as the CPU
4// solver in sparse_ldl.h. Requires -DCUDSS_WITH and -lcudss.
5//
6// analyze() runs cuDSS's reordering + symbolic phases, factorize() the numeric
7// phase (the first call factorizes, later calls refactorize with the stored
8// analysis), and solve() the triangular/diagonal solves. The values are read
9// from the Csr each time, so a new Csr with the same pattern can be passed to
10// factorize().

24inline void
25check(cudssStatus_t s, const char *what)
26{
27 if (s != CUDSS_STATUS_SUCCESS)
28 throw std::runtime_error(std::string("cuDSS error in ") + what +
29 " (status " +
30 std::to_string(static_cast<int>(s)) + ")");
31}

33template <typename Idx>
34constexpr cudssDataType_t
35index_dt()
36{ return sizeof(Idx) == 4 ? CUDSS_R_32I : CUDSS_R_64I; }
37
38template <typename T>
39constexpr cudssDataType_t
40value_dt()
41{ return std::is_same_v<T, float> ? CUDSS_R_32F : CUDSS_R_64F; }

45template <typename T, typename Idx>
46class SparseLdlt<T, Idx, Cuda::CudaStorage> {
47 static_assert(
48 std::is_floating_point_v<T>, "SparseLdlt supports float and double");
49
50 public:
51 using matrix_type = Csr<T, Idx, Cuda::CudaStorage>;
52 using vector_type = tensorET<1, T, Cuda::CudaStorage<T>>;
53
54 explicit SparseLdlt(
55 Symmetry kind = Symmetry::SPD, Ordering ord = Ordering::MinDegree)
56 : kind_(kind)
57 {
58 using namespace cudss_detail;
59 check(cudssConfigCreate(&config_), "cudssConfigCreate");
60 if (ord == Ordering::MinDegree || ord == Ordering::NestedDissection) {
61 cudssReorderingAlg_t alg =
62 ord == Ordering::MinDegree
63 ? CUDSS_REORDERING_ALG_AMD
64 : CUDSS_REORDERING_ALG_NESTED_DISSECTION;
65 check(cudssConfigSet(
66 config_, CUDSS_CONFIG_REORDERING_ALG, &alg, sizeof(alg)),
67 "cudssConfigSet(reordering)");
68 }
69 check(cudssDataCreate(handle(), &data_), "cudssDataCreate");
70 }
71
72 SparseLdlt(const SparseLdlt &) = delete;
73 SparseLdlt &operator=(const SparseLdlt &) = delete;
74
75 ~SparseLdlt()
76 {
77 destroy_matrices();
78 if (data_) cudssDataDestroy(handle(), data_);
79 if (config_) cudssConfigDestroy(config_);
80 }

82 void
83 analyze(const matrix_type &A)
84 {
85 using namespace cudss_detail;
86 if (A.rows() != A.cols())
87 throw std::invalid_argument("SparseLdlt: matrix must be square");
88 destroy_matrices();
89 n_ = A.rows();
90 nnz_ = A.nnz();
91 // Placeholder right-hand side / solution for the descriptors; the
92 // real vectors are bound in solve() with cudssMatrixSetValues.
93 scratch_ = Cuda::CudaStorage<T>(2 * n_);
94 T *bp = scratch_.data(), *xp = scratch_.data() + n_;
95 check(cudssMatrixCreateDn(
96 &b_, n_, 1, n_, bp, value_dt<T>(), CUDSS_LAYOUT_COL_MAJOR),
97 "cudssMatrixCreateDn(b)");
98 check(cudssMatrixCreateDn(
99 &x_, n_, 1, n_, xp, value_dt<T>(), CUDSS_LAYOUT_COL_MAJOR),
100 "cudssMatrixCreateDn(x)");
101 check(cudssMatrixCreateCsr(&a_, n_, n_, nnz_, A.row_ptr(), nullptr,
102 A.col_ind(), const_cast<T *>(A.values()), index_dt<Idx>(),
103 index_dt<Idx>(), value_dt<T>(),
104 kind_ == Symmetry::SPD ? CUDSS_MTYPE_SPD
105 : CUDSS_MTYPE_SYMMETRIC,
106 CUDSS_MVIEW_FULL, CUDSS_BASE_ZERO),
107 "cudssMatrixCreateCsr");
108 check(cudssExecute(
109 handle(), CUDSS_PHASE_ANALYSIS, config_, data_, a_, x_, b_),
110 "cudssExecute(analysis)");
111 analyzed_ = true;
112 factored_ = false;
113 numeric_done_ = false;
114 }
115
116 // Returns false when cuDSS reports a failed factorization.
117 bool
118 factorize(const matrix_type &A)
119 {
120 using namespace cudss_detail;
121 if (!analyzed_) analyze(A);
122 if (A.rows() != n_ || A.nnz() != nnz_)
123 throw std::invalid_argument(
124 "SparseLdlt::factorize: pattern differs from analyze()");
125 check(cudssMatrixSetCsrPointers(a_, A.row_ptr(), nullptr, A.col_ind(),
126 const_cast<T *>(A.values())),
127 "cudssMatrixSetCsrPointers");
128 check(cudssExecute(handle(),
129 numeric_done_ ? CUDSS_PHASE_REFACTORIZATION
130 : CUDSS_PHASE_FACTORIZATION,
131 config_, data_, a_, x_, b_),
132 "cudssExecute(factorization)");
133 cudaDeviceSynchronize();
134 int info = 0;
135 size_t written = 0;
136 check(cudssDataGet(handle(), data_, CUDSS_DATA_INFO, &info,
137 sizeof(info), &written),
138 "cudssDataGet(info)");
139 numeric_done_ = true;
140 factored_ = (info == 0);
141 failed_ = factored_ ? -1 : info;
142 return factored_;
143 }
144
145 void
146 solve(const vector_type &b, vector_type &x) const
147 {
148 using namespace cudss_detail;
149 if (!factored_)
150 throw std::runtime_error("SparseLdlt::solve: not factorized");
151 if (b.size() != n_ || x.size() != n_)
152 throw std::invalid_argument("SparseLdlt::solve: size mismatch");
153 check(cudssMatrixSetValues(b_, const_cast<T *>(b.data)),
154 "cudssMatrixSetValues(b)");
155 check(cudssMatrixSetValues(x_, x.data), "cudssMatrixSetValues(x)");
156 check(cudssExecute(
157 handle(), CUDSS_PHASE_SOLVE, config_, data_, a_, x_, b_),
158 "cudssExecute(solve)");
159 cudaDeviceSynchronize();
160 }

162 size_t
163 factor_nnz() const
164 {
165 int64_t v = 0;
166 size_t written = 0;
167 if (analyzed_)
168 cudssDataGet(
169 handle(), data_, CUDSS_DATA_LU_NNZ, &v, sizeof(v), &written);
170 return static_cast<size_t>(v);
171 }

192 // Positive / negative eigenvalue counts as reported by cuDSS for
193 // indefinite matrices (zero is n - pos - neg).
194 void
195 inertia(size_t &pos, size_t &neg, size_t &zero) const
196 {
197 Idx v[2] = {
198 0, 0}; // cuDSS: {positive, negative}, same width as the index
199 size_t written = 0;
200 cudss_detail::check(cudssDataGet(handle(), data_, CUDSS_DATA_INERTIA, v,
201 sizeof(v), &written),
202 "cudssDataGet(inertia)");
203 pos = static_cast<size_t>(v[0]);
204 neg = static_cast<size_t>(v[1]);
205 zero = n_ - pos - neg;
206 }

208 private:
209 static cudssHandle_t
210 handle()
211 { return GPUMemoryPool::get().get_cudss(); }

213 void
214 destroy_matrices()
215 {
216 if (a_) cudssMatrixDestroy(a_);
217 if (b_) cudssMatrixDestroy(b_);
218 if (x_) cudssMatrixDestroy(x_);
219 a_ = b_ = x_ = nullptr;
220 }
221
222 Symmetry kind_;
223 size_t n_ = 0, nnz_ = 0;
224 bool analyzed_ = false, factored_ = false, numeric_done_ = false;
225 long failed_ = -1;
226 cudssConfig_t config_ = nullptr;
227 cudssData_t data_ = nullptr;
228 cudssMatrix_t a_ = nullptr, b_ = nullptr, x_ = nullptr;
229 Cuda::CudaStorage<T> scratch_;
230};
