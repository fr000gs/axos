3// Threading helpers for the dense layer: OpenMP when the translation unit is
4// compiled with it, serial otherwise. Every helper opens at most one parallel
5// region per call, runs small work inline (no fork/join), and never nests:
6// inside an existing parallel region everything runs on the calling thread.
7//
8// parallel_for(n, grain, f) f(begin, end) over chunks of [0, n)
9// parallel_sum<R>(n, grain, f) sum of f(begin, end) over the chunks, added
10// in chunk order (deterministic for a fixed
11// thread count)
12// copy_bytes / fill_elems memcpy / fill, parallel above a few MiB

34// While alive, the dense kernels called from this thread run on it alone
35// (for small, latency-bound phases where fork/join costs more than it saves).
36struct SerialScope {
37 SerialScope() { ++serial_depth(); }
38 ~SerialScope() { --serial_depth(); }
39 SerialScope(const SerialScope &) = delete;
40 SerialScope &operator=(const SerialScope &) = delete;
41};

43// Threads a new parallel region would get from this point (1 inside one).
44inline int
45max_threads()
46{
47#ifdef _OPENMP
48 if (serial_depth() > 0) return 1;
49 return omp_in_parallel() ? 1 : omp_get_max_threads();
50#else
51 return 1;
52#endif
53}

55// Number of chunks for n items with at least `grain` items per chunk: up to
56// 4 per thread, so the dynamic schedule can balance hybrid (P/E) cores.
57inline size_t
58chunk_count(size_t n, size_t grain, int threads)
59{
60 if (threads <= 1 || grain == 0 || n < 2 * grain) return 1;
61 return std::max<size_t>(1, std::min(n / grain, size_t(4) * threads));
62}
63
64// Start of chunk k of c over [0, n), rounded down to a multiple of `align`.
65inline size_t
66chunk_begin(size_t n, size_t c, size_t k, size_t align)
67{
68 if (k == 0) return 0;
69 if (k >= c) return n;
70 const size_t b = (n / c) * k + std::min(k, n % c);
71 return align > 1 ? b / align * align : b;
72}

74template <class F>
75void
76parallel_for(size_t n, size_t grain, F &&f, size_t align = 16)
77{
78 if (n == 0) return;
79 const int nt = max_threads();
80 const size_t c = chunk_count(n, grain, nt);
81 if (c <= 1) {
82 f(size_t(0), n);
83 return;
84 }
85#ifdef _OPENMP
86 const int use = static_cast<int>(std::min<size_t>(nt, c));
87#pragma omp parallel for schedule(dynamic, 1) num_threads(use)
88 for (long long k = 0; k < static_cast<long long>(c); ++k) {
89 const size_t b = chunk_begin(n, c, size_t(k), align);
90 const size_t e = chunk_begin(n, c, size_t(k) + 1, align);
91 if (b < e) f(b, e);
92 }
93#else
94 f(size_t(0), n);
95#endif
96}

98template <class R, class F>
99R
100parallel_sum(size_t n, size_t grain, F &&f, size_t align = 16)
101{
102 if (n == 0) return R(0);
103 const int nt = max_threads();
104 const size_t c = chunk_count(n, grain, nt);
105 if (c <= 1) return f(size_t(0), n);
106#ifdef _OPENMP
107 std::vector<R> part(c, R(0));
108 const int use = static_cast<int>(std::min<size_t>(nt, c));
109#pragma omp parallel for schedule(dynamic, 1) num_threads(use)
110 for (long long k = 0; k < static_cast<long long>(c); ++k) {
111 const size_t b = chunk_begin(n, c, size_t(k), align);
112 const size_t e = chunk_begin(n, c, size_t(k) + 1, align);
113 if (b < e) part[size_t(k)] = f(b, e);
114 }
115 R s(0);
116 for (const R &v : part)
117 s += v;
118 return s;
119#else
120 return f(size_t(0), n);
121#endif
122}

124// Bytes per chunk for parallel copies and fills; below 4 chunks' worth a
125// single memcpy/memset is as fast as a parallel one.
126inline constexpr size_t kCopyGrain = size_t(1) << 20;
127
128inline void
129copy_bytes(void *dst, const void *src, size_t bytes)
130{
131 if (bytes == 0) return;
132 if (bytes < 4 * kCopyGrain) {
133 std::memcpy(dst, src, bytes);
134 return;
135 }
136 char *d = static_cast<char *>(dst);
137 const char *s = static_cast<const char *>(src);
138 parallel_for(
139 bytes, kCopyGrain,
140 [=](size_t b, size_t e) { std::memcpy(d + b, s + b, e - b); }, 4096);
141}

143// True when every byte of x is the same, so a fill is a memset of that byte
144// (covers 0, 0.0 and -1).
145template <class T>
146bool
147uniform_bytes(const T &x, unsigned char &byte)
148{
149 unsigned char b[sizeof(T)];
150 std::memcpy(b, &x, sizeof(T));
151 for (size_t i = 1; i < sizeof(T); ++i)
152 if (b[i] != b[0]) return false;
153 byte = b[0];
154 return true;
155}

157template <class T>
158void
159fill_elems(T *p, size_t n, const T &x)
160{
161 if (n == 0) return;
162 unsigned char byte;
163 const bool bytes = uniform_bytes(x, byte);
164 const size_t grain = std::max<size_t>(1, kCopyGrain / sizeof(T));
165 if (n < 4 * grain) {
166 if (bytes)
167 std::memset(static_cast<void *>(p), byte, n * sizeof(T));
168 else
169 std::fill_n(p, n, x);
170 return;
171 }
172 parallel_for(
173 n, grain,
174 [=](size_t b, size_t e) {
175 if (bytes)
176 std::memset(
177 static_cast<void *>(p + b), byte, (e - b) * sizeof(T));
178 else
179 std::fill(p + b, p + e, x);
180 },
181 std::max<size_t>(1, 4096 / sizeof(T)));
182}
