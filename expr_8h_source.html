3// Expression-template core (docs/TENSOR_SPEC.md §6, §15.1).
4//
5// Expr<T, E> is the CRTP base of every element-wise expression. The library's
6// nodes (tensorMath.h) and host tensors derive from it; user expressions may
7// too. The requirements on an expression type E are:
8//
9// T value(size_t i) const element with row-major flat index i
10// size_t size(int d) const extent of dimension d
11//
12// and, optionally (detected at compile time):
13//
14// static constexpr int rank checked against the target's rank
15// bool contiguous() const false when value(i) is not the logical
16// element i (strided views); lvalue(i) is
17// then used instead
18// T lvalue(size_t i) const element with logical index i
19// void prepare() const called once before evaluation
20// bool assign_into(T *dst) const evaluate straight into a contiguous
21// destination; return false to decline
22// static constexpr int cost relative cost per element (default 1),
23// sets the threshold for threading
24// static constexpr bool packet_ok with `template <class P> typename P::reg
25// packet(size_t i) const`: SIMD evaluation
26// of W = P::W elements starting at i
27//
28// Nodes hold tensors by reference and other nodes by value, so building an
29// expression allocates nothing, and assigning it is one pass over the data
30// (split across threads above a size threshold) with no temporaries.

158// Outputs at least this large are written with non-temporal stores: they
159// would not stay in cache anyway, and streaming skips the read-for-ownership
160// of every destination line (a quarter of the traffic of C = A + B).
161inline constexpr size_t kStreamBytes = size_t(8) << 20;

163template <class T, class E>
164AXOS_INLINE void
165eval_range(T *d, const E &e, size_t b, size_t en, bool stream = false)
166{
167 if constexpr (expr_packet_ok<E, T>()) {
168 using P = simd::Pack<T>;
169 constexpr size_t W = P::W;
170 size_t i = b;
171 if (stream) {
172 constexpr size_t A = W * sizeof(T); // stream stores need alignment
173 for (; i < en && reinterpret_cast<std::uintptr_t>(d + i) % A; ++i)
174 d[i] = e.value(i);
175 for (; i + W <= en; i += W)
176 P::stream(d + i, e.template packet<P>(i));
177 simd::stream_fence();
178 for (; i < en; ++i)
179 d[i] = e.value(i);
180 return;
181 }
182 for (; i + 2 * W <= en; i += 2 * W) {
183 const auto v0 = e.template packet<P>(i);
184 const auto v1 = e.template packet<P>(i + W);
185 P::storeu(d + i, v0);
186 P::storeu(d + i + W, v1);
187 }
188 for (; i + W <= en; i += W)
189 P::storeu(d + i, e.template packet<P>(i));
190 for (; i < en; ++i)
191 d[i] = e.value(i);
192 } else {
193 for (size_t i = b; i < en; ++i)
194 d[i] = e.value(i);
195 }
196}

198template <class Dst, class E>
199void
200check_expr_shape(const Dst &dst, const E &e)
201{
202 for (int d = 0; d < Dst::rank; ++d)
203 if (e.size(d) != dst.size(d))
204 throw std::runtime_error("tensorET = expression: shape mismatch "
205 "in dimension " +
206 std::to_string(d) + " (" +
207 std::to_string(dst.size(d)) + " vs " +
208 std::to_string(e.size(d)) + ")");
209}

211// dst = e, element-wise, into dst's existing buffer.
212template <class Dst, class E>
213void
214assign_expr(Dst &dst, const E &e)
215{
216 using T = typename Dst::value_type;
217 static_assert(Dst::on_host, "expressions are evaluated on the host only");
218 if constexpr (has_rank<E>::value)
219 static_assert(
220 E::rank == Dst::rank, "tensorET = expression: rank mismatch");
221 check_expr_shape(dst, e);
222 const size_t n = dst.size();
223 if (n == 0) return;
224 const bool dc = dst.is_contiguous();
225 if constexpr (has_assign_into<E, T>::value)
226 if (dc && e.assign_into(dst.data)) return;
227 expr_prepare(e);
228 T *d = dst.data;
229 if (dc && expr_contiguous(e)) {
230 const bool stream = n * sizeof(T) >= kStreamBytes;
231 parallel_for(n, expr_grain<E>(), [d, &e, stream](size_t b, size_t en) {
232 eval_range(d, e, b, en, stream);
233 });
234 } else {
235 parallel_for(n, expr_grain<E>(), [&](size_t b, size_t en) {
236 for (size_t i = b; i < en; ++i)
237 d[dc ? i : logical_offset(dst, i)] = expr_lvalue(e, i);
238 });
239 }
240}
