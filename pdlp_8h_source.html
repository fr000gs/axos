3// PDLP: restarted primal-dual hybrid gradient for LP (Applegate et al.,
4// "Practical Large-Scale Linear Programming using Primal-Dual Hybrid
5// Gradient"), with the KKT-based restart rule used by cuPDLP.
6//
7// min c^T x s.t. l <= A x <= u, lb <= x <= ub
8// saddle form L(x, y) = c^T x - y^T A x + sum_i phi_i(y_i),
9// phi_i(y) = l_i y (y > 0), u_i y (y < 0)
10//
11// One iteration is an x projection, an SpMV with A, a closed-form y update
12// y+ = max(0, y + s (l - t)) + min(0, y + s (u - t)), t = 2 A x+ - A x
13// and an SpMV with A^T. Step sizes adapt to the local Lipschitz estimate,
14// iterates are averaged, and the solver restarts from the better of the
15// current and averaged iterate when the KKT error drops enough. The primal
16// weight is rebalanced at every restart. Termination uses the tolerances in
17// SolverOptions on the ORIGINAL (unscaled) problem. Infeasibility and
18// unboundedness are detected from normalized iterate differences.
19//
20// Runs on the CPU (Store = Cpu::HostStorage) or the GPU (Cuda::CudaStorage,
21// include pdlp_cuda.h first through solver.h).

36template <template <typename> class Store = Cpu::HostStorage> class Pdlp {
37 using VS = Store<double>;
38 using Vec = tensorET<1, double, VS>;
39 using Mat = Sparse::Csr<double, int32_t, Store>;
40 using Par = Parallel<typename VS::backend_type>;
41
42 struct Kkt {
43 double pr_scaled = 0, pr_orig = 0, dr_scaled = 0, dr_orig = 0;
44 double pobj = 0, dobj = 0, gap = 0, err = 0, bound = 0;
45 };
46
47 public:
48 // Optional starting point (original scale) and termination mode for the
49 // polishing sub-solves: 0 full KKT, 1 primal residual only, 2 dual residual
50 // only.
51 struct Start {
52 const std::vector<double> *x = nullptr, *y = nullptr;
53 int mode = 0;
54 };
55
56 // Solves p (no presolve). Requires at least one row and one column. With
57 // opt.pdlp_polish and a tight tolerance, PDLP runs to a looser tolerance
58 // first, then polishes: a feasibility solve for x (objective dropped) and
59 // one for y (right-hand sides and bounds zeroed) from that point, which
60 // converge much faster than continuing on the full problem.
61 LpSolution
62 solve(const LpProblem &p, const SolverOptions &opt)
63 {
64 const double tight =
65 std::min({opt.eps_primal, opt.eps_dual, opt.eps_gap});
66 if (!opt.pdlp_polish || tight >= 5e-6)
67 return solve_core(p, opt, nullptr);
68 const auto t0 = std::chrono::steady_clock::now();
69 auto since = [&] {
70 return std::chrono::duration<double>(
71 std::chrono::steady_clock::now() - t0)
72 .count();
73 };
74 SolverOptions loose = opt;
75 loose.set_tolerance(std::max(tight, 1e-4));
76 LpSolution main = solve_core(p, loose, nullptr);
77 if (main.status != Status::Optimal) return main;
78 // primal feasibility from x (objective dropped)
79 LpProblem pp = p;
80 std::fill(pp.c.begin(), pp.c.end(), 0.0);
81 pp.offset = 0;
82 SolverOptions o1 = opt;
83 o1.time_limit = std::max(1.0, opt.time_limit - since());
84 std::vector<double> y0(p.rows(), 0.0);
85 Start s1{&main.x, &y0, 1};
86 LpSolution a = solve_core(pp, o1, &s1);
87 // dual feasibility from y (b and finite bounds zeroed, objective kept)
88 LpProblem pd = p;
89 for (double &v : pd.row_lb)
90 if (std::isfinite(v)) v = 0;
91 for (double &v : pd.row_ub)
92 if (std::isfinite(v)) v = 0;
93 for (double &v : pd.col_lb)
94 if (std::isfinite(v)) v = 0;
95 for (double &v : pd.col_ub)
96 if (std::isfinite(v)) v = 0;
97 pd.offset = 0;
98 SolverOptions o2 = opt;
99 o2.time_limit = std::max(1.0, opt.time_limit - since());
100 std::vector<double> x0(p.cols(), 0.0);
101 Start s2{&x0, &main.y, 2};
102 LpSolution b = solve_core(pd, o2, &s2);
103 long iters = main.iterations + a.iterations + b.iterations;
104 if (a.status == Status::Optimal || a.status == Status::IterationLimit ||
105 a.status == Status::TimeLimit) {
106 // x from a, y from b; keep the better-gap combination
107 LpSolution c = evaluate_solution(p, a.x, b.y);
108 const double nbn = norm_b(p), ncn = norm_c(p);
109 const bool ok =
110 c.primal_residual <= opt.eps_primal * (1 + nbn) &&
111 c.dual_residual <= opt.eps_dual * (1 + ncn) &&
112 c.gap <= opt.eps_gap * (1 + std::abs(c.primal_objective) +
113 std::abs(c.dual_objective)) &&
114 c.error_bound <=
115 10 * opt.eps_gap * (1 + std::abs(c.primal_objective));
116 if (ok) {
117 c.status = Status::Optimal;
118 c.iterations = iters;
119 c.seconds = since();
120 return c;
121 }
122 }
123 // polishing did not reach the tolerance: continue from the loose
124 // solution
125 SolverOptions o3 = opt;
126 o3.time_limit = std::max(1.0, opt.time_limit - since());
127 Start s3{&main.x, &main.y, 0};
128 LpSolution r = solve_core(p, o3, &s3);
129 r.iterations += iters;
130 r.seconds = since();
131 return r;
132 }
133
134 private:
135 static double
136 norm_b(const LpProblem &p)
137 {
138 double nb = 0;
139 for (size_t i = 0; i < p.rows(); ++i) {
140 if (std::isfinite(p.row_lb[i])) nb += p.row_lb[i] * p.row_lb[i];
141 if (std::isfinite(p.row_ub[i])) nb += p.row_ub[i] * p.row_ub[i];
142 }
143 return std::sqrt(nb);
144 }
145 static double
146 norm_c(const LpProblem &p)
147 {
148 double nc = 0;
149 for (double v : p.c)
150 nc += v * v;
151 return std::sqrt(nc);
152 }
153
154 LpSolution
155 solve_core(const LpProblem &p, const SolverOptions &opt, const Start *start)
156 {
157 using namespace pdlp;
158 const auto t_start = std::chrono::steady_clock::now();
159 auto elapsed = [&] {
160 return std::chrono::duration<double>(
161 std::chrono::steady_clock::now() - t_start)
162 .count();
163 };
164 const size_t m = p.rows(), n = p.cols();
165 if (m == 0 || n == 0)
166 throw std::invalid_argument(
167 "Pdlp::solve needs rows and columns; use solve_lp()");
168 std::string bad = p.validate();
169 if (!bad.empty()) throw std::invalid_argument("Pdlp::solve: " + bad);
170
171 // ---- scaling and device data ------------------------------------
172 Scaling sc;
173 LpProblem q;
174 if (opt.scaling) {
175 sc = compute_scaling(
176 p.A, opt.ruiz_iterations, opt.pock_chambolle_alpha);
177 q = apply_scaling(p, sc);
178 } else {
179 sc.row.assign(m, 1.0);
180 sc.col.assign(n, 1.0);
181 q = p;
182 }
183 Mat A(q.A);
184 const Mat &AT = A.transposed();
185
186 auto upload = [](const std::vector<double> &h) {
187 tensorET<1, double> t({h.size()}, 0.0);
188 std::memcpy(t.data, h.data(), h.size() * sizeof(double));
189 return Vec(t);
190 };
191 std::vector<double> cinv(n), rinv(m);
192 for (size_t j = 0; j < n; ++j)
193 cinv[j] = 1.0 / sc.col[j];
194 for (size_t i = 0; i < m; ++i)
195 rinv[i] = 1.0 / sc.row[i];
196 Vec c = upload(q.c), lb = upload(q.col_lb), ub = upload(q.col_ub);
197 Vec l = upload(q.row_lb), u = upload(q.row_ub);
198 Vec dcinv = upload(cinv), drinv = upload(rinv);
199 Vec dcol = upload(sc.col), drow = upload(sc.row);
200
201 Vec X0({n}, 0.0), X1({n}, 0.0), Xr({n}, 0.0), Sx({n}, 0.0);
202 Vec Aty0({n}, 0.0), Aty1({n}, 0.0), Atyr({n}, 0.0), Saty({n}, 0.0);
203 Vec Y0({m}, 0.0), Y1({m}, 0.0), Yr({m}, 0.0), Sy({m}, 0.0);
204 Vec Ax0({m}, 0.0), Ax1({m}, 0.0), Axr({m}, 0.0), Sax({m}, 0.0);
205 Vec *x = &X0, *xn = &X1, *y = &Y0, *yn = &Y1;
206 Vec *ax = &Ax0, *axn = &Ax1, *aty = &Aty0, *atyn = &Aty1;
207
208 Par par;
209
210 // ---- problem norms for tolerances and the initial primal weight ---
211 double nc = 0, nb = 0;
212 for (size_t j = 0; j < n; ++j)
213 nc += p.c[j] * p.c[j];
214 for (size_t i = 0; i < m; ++i) {
215 if (std::isfinite(p.row_lb[i])) nb += p.row_lb[i] * p.row_lb[i];
216 if (std::isfinite(p.row_ub[i])) nb += p.row_ub[i] * p.row_ub[i];
217 }
218 nc = std::sqrt(nc);
219 nb = std::sqrt(nb);
220 double qc = 0, qb = 0, amax = 0; // scaled-space norms
221 for (size_t j = 0; j < n; ++j)
222 qc += q.c[j] * q.c[j];
223 for (size_t i = 0; i < m; ++i) {
224 if (std::isfinite(q.row_lb[i])) qb += q.row_lb[i] * q.row_lb[i];
225 if (std::isfinite(q.row_ub[i])) qb += q.row_ub[i] * q.row_ub[i];
226 }
227 for (size_t k = 0; k < q.A.nnz(); ++k)
228 amax = std::max(amax, std::abs(q.A.values()[k]));
229 qc = std::sqrt(qc);
230 qb = std::sqrt(qb);
231 double omega = (qc > 1e-10 && qb > 1e-10) ? qc / qb : 1.0;
232 double eta = amax > 0 ? 1.0 / amax : 1.0;
233
234 // Halpern PDHG uses fixed steps: eta = 0.998 / ||A||_2 (power
235 // iteration)
236 if (opt.pdlp_halpern) {
237 Vec &v = X1, &w = Ax1, &t = Aty1;
238 par.for_each(n, pdlp::Fill{v.data, 1.0});
239 double sig = 1.0;
240 for (int k = 0; k < 60; ++k) {
241 Sparse::spmv(A, v, w);
242 Sparse::spmv(AT, w, t);
243 par.zero();
244 par.reduce(0, n, SqNormN{t.data});
245 par.reduce(1, n, SqNormN{v.data});
246 const double *s = par.fetch();
247 if (s[1] <= 0 || s[0] <= 0) break;
248 sig = std::sqrt(std::sqrt(
249 s[0] / s[1])); // ||A^T A v|| / ||v|| ~ sigma_max^2
250 const double inv = 1.0 / std::sqrt(s[0]);
251 par.for_each(n, pdlp::ScaleCopy{v.data, t.data, inv});
252 }
253 eta = 0.998 / (1.05 * sig);
254 par.for_each(n, pdlp::Fill{X1.data, 0.0});
255 par.for_each(m, pdlp::Fill{Ax1.data, 0.0});
256 par.for_each(n, pdlp::Fill{Aty1.data, 0.0});
257 }
258
259 // ---- initial point: x = proj(0), y = 0, or the given start
260 // -----------
261 {
262 tensorET<1, double> h({n}, 0.0);
263 for (size_t j = 0; j < n; ++j) {
264 const double v =
265 (start && start->x) ? (*start->x)[j] / sc.col[j] : 0.0;
266 h.data[j] = std::min(std::max(v, q.col_lb[j]), q.col_ub[j]);
267 }
268 Vec init(h);
269 par.copy(x->data, init.data, n);
270 if (start && start->y) {
271 tensorET<1, double> hy({m}, 0.0);
272 for (size_t i = 0; i < m; ++i)
273 hy.data[i] = (*start->y)[i] / sc.row[i];
274 Vec inity(hy);
275 par.copy(y->data, inity.data, m);
276 }
277 }
278 Sparse::spmv(A, *x, *ax);
279 if (start && start->y) Sparse::spmv(AT, *y, *aty);
280 snapshot(par, n, m, *x, *y, *ax, *aty, Xr, Yr, Axr, Atyr);
281 double se = 0;
282
283 auto eval = [&](const Vec &xv, const Vec &yv, const Vec &axv,
284 const Vec &atyv) {
285 par.zero();
286 par.reduce(
287 0, m, KktM{axv.data, yv.data, l.data, u.data, drinv.data});
288 par.reduce(3, n,
289 KktN{xv.data, atyv.data, c.data, lb.data, ub.data, dcinv.data});
290 const double *s = par.fetch();
291 Kkt k;
292 k.pr_scaled = std::sqrt(s[0]);
293 k.pr_orig = std::sqrt(s[1]);
294 k.dr_scaled = std::sqrt(s[3]);
295 k.dr_orig = std::sqrt(s[4]);
296 k.pobj = s[6] + p.offset;
297 k.dobj = s[2] + s[5] + p.offset;
298 k.gap = std::abs(k.pobj - k.dobj);
299 // objective-error bound in the original scale (see LpSolution)
300 par.zero();
301 par.reduce(0, n, WSqNorm{xv.data, dcol.data});
302 par.reduce(1, m, WSqNorm{yv.data, drow.data});
303 const double *w = par.fetch();
304 k.bound = k.gap + k.dr_orig * std::sqrt(w[0]) +
305 k.pr_orig * std::sqrt(w[1]);
306 k.err = std::sqrt(omega * omega * k.pr_scaled * k.pr_scaled +
307 k.dr_scaled * k.dr_scaled / (omega * omega) +
308 k.gap * k.gap);
309 return k;
310 };
311 const int mode = start ? start->mode : 0;
312 auto converged = [&](const Kkt &k) {
313 if (mode == 1) return k.pr_orig <= opt.eps_primal * (1 + nb);
314 if (mode == 2) return k.dr_orig <= opt.eps_dual * (1 + nc);
315 return k.pr_orig <= opt.eps_primal * (1 + nb) &&
316 k.dr_orig <= opt.eps_dual * (1 + nc) &&
317 k.gap <= opt.eps_gap *
318 (1 + std::abs(k.pobj) + std::abs(k.dobj)) &&
319 k.bound <= 10 * opt.eps_gap * (1 + std::abs(k.pobj));
320 };
321
322 auto finish = [&](Status st, const Vec &fx, const Vec &fy, long iters) {
323 tensorET<1, double> hx(fx), hy(fy);
324 std::vector<double> xv(hx.data, hx.data + n),
325 yv(hy.data, hy.data + m);
326 unscale_solution(xv, yv, sc);
327 LpSolution s = evaluate_solution(p, xv, yv);
328 s.status = st;
329 s.iterations = iters;
330 s.seconds = elapsed();
331 return s;
332 };
333
334 Kkt cur = eval(*x, *y, *ax, *aty);
335 double last_restart_err = cur.err;
336 double prev_cand_err = kInf;
337 long it = 0, since_restart = 0;
338 const double eps_inf = 1e-8;
339 // Minimum objective improvement of a certificate ray (unit norm),
340 // relative to the problem's scale and the requested accuracy.
341 const double cert_margin_dual = 10 * opt.eps_primal * (1 + qb);
342 const double cert_margin_primal = 10 * opt.eps_dual * (1 + qc);
343
344 while (true) {
345 // Halpern PDHG: the output point is T(z) (feasible for the bounds),
346 // not z
347 Vec *cx = x, *cy = y, *cax = ax, *caty = aty;
348 if (opt.pdlp_halpern && it > 0) {
349 cx = xn;
350 cy = yn;
351 cax = axn;
352 caty = atyn;
353 }
354 if (it % opt.check_frequency == 0) {
355 cur = eval(*cx, *cy, *cax, *caty);
356 if (opt.verbose)
357 std::printf(
358 "[pdlp] it %-8ld pobj % .8e dobj % .8e "
359 "pres %.2e dres %.2e gap %.2e omega %.2e eta %.2e\n",
360 it, cur.pobj, cur.dobj, cur.pr_orig, cur.dr_orig,
361 cur.gap, omega, eta);
362 if (!std::isfinite(cur.err))
363 return finish(Status::NumericalError, *cx, *cy, it);
364
365 Kkt avg;
366 const bool have_avg = se > 0;
367 if (have_avg) {
368 const double inv = 1.0 / se;
369 par.for_each(n, ScaleCopy{xn->data, Sx.data, inv});
370 par.for_each(n, ScaleCopy{atyn->data, Saty.data, inv});
371 par.for_each(m, ScaleCopy{yn->data, Sy.data, inv});
372 par.for_each(m, ScaleCopy{axn->data, Sax.data, inv});
373 avg = eval(*xn, *yn, *axn, *atyn);
374 }
375 if (have_avg && converged(avg) &&
376 (!converged(cur) || avg.err < cur.err))
377 return finish(Status::Optimal, *xn, *yn, it);
378 if (converged(cur))
379 return finish(Status::Optimal, *cx, *cy, it);
380
381 // infeasibility / unboundedness certificates
382 if (since_restart > 0) {
383 par.zero();
384 par.reduce(
385 0, m, DualRayM{cy->data, Yr.data, l.data, u.data});
386 par.reduce(3, n,
387 DualRayN{caty->data, Atyr.data, lb.data, ub.data});
388 par.reduce(5, n,
389 PrimalRayN{
390 cx->data, Xr.data, c.data, lb.data, ub.data});
391 par.reduce(
392 8, m, PrimalRayM{cax->data, Axr.data, l.data, u.data});
393 const double *s = par.fetch();
394 const double dy2 = s[0], dx2 = s[5];
395 if (dy2 > 1e-20) {
396 const double ny = std::sqrt(dy2);
397 const double rho = (s[2] + s[4]) / ny;
398 const double viol = std::sqrt(s[1] + s[3]) / ny;
399 if (opt.verbose)
400 std::printf("[pdlp] dual ray: |dy| %.3e rho %.3e "
401 "viol %.3e margin %.3e\n",
402 ny, rho, viol, cert_margin_dual);
403 // The ray must improve the homogeneous dual objective
404 // by a real margin: redundant constraints give null
405 // directions whose objective is only round-off.
406 if (rho > cert_margin_dual && viol <= eps_inf * rho)
407 return finish(Status::Infeasible, *cx, *cy, it);
408 }
409 if (dx2 > 1e-20) {
410 const double nx = std::sqrt(dx2);
411 const double slope = s[6] / nx;
412 const double viol = std::sqrt(s[7] + s[8]) / nx;
413 if (-slope > cert_margin_primal &&
414 viol <= eps_inf * (-slope))
415 return finish(Status::Unbounded, *cx, *cy, it);
416 }
417 }
418
419 if (it >= opt.max_iterations)
420 return finish(Status::IterationLimit, *cx, *cy, it);
421 if (elapsed() > opt.time_limit)
422 return finish(Status::TimeLimit, *cx, *cy, it);
423
424 // restart?
425 const bool use_avg = have_avg && avg.err < cur.err;
426 const double cand_err = use_avg ? avg.err : cur.err;
427 const bool suff = cand_err <= 0.2 * last_restart_err;
428 const bool nec = cand_err <= 0.8 * last_restart_err &&
429 cand_err > prev_cand_err;
430 const bool art =
431 since_restart >= 0.36 * static_cast<double>(it);
432 if (it > 0 && (suff || nec || art)) {
433 if (opt.pdlp_halpern) { // restart from T(z)
434 par.copy(x->data, cx->data, n);
435 par.copy(y->data, cy->data, m);
436 par.copy(ax->data, cax->data, m);
437 par.copy(aty->data, caty->data, n);
438 } else if (use_avg) {
439 std::swap(x, xn);
440 std::swap(y, yn);
441 std::swap(ax, axn);
442 std::swap(aty, atyn);
443 }
444 // primal weight: balance the movement since last restart
445 par.zero();
446 par.reduce(0, n, DiffSq{x->data, Xr.data});
447 par.reduce(1, m, DiffSq{y->data, Yr.data});
448 const double *s = par.fetch();
449 const double dx = std::sqrt(s[0]), dy = std::sqrt(s[1]);
450 if (dx > 1e-10 && dy > 1e-10)
451 omega = std::exp(
452 0.5 * std::log(dy / dx) + 0.5 * std::log(omega));
453 snapshot(par, n, m, *x, *y, *ax, *aty, Xr, Yr, Axr, Atyr);
454 par.for_each(n, pdlp::Fill{Sx.data, 0.0});
455 par.for_each(n, pdlp::Fill{Saty.data, 0.0});
456 par.for_each(m, pdlp::Fill{Sy.data, 0.0});
457 par.for_each(m, pdlp::Fill{Sax.data, 0.0});
458 se = 0;
459 since_restart = 0;
460 last_restart_err = eval(*x, *y, *ax, *aty).err;
461 prev_cand_err = kInf;
462 } else {
463 prev_cand_err = cand_err;
464 }
465 }
466
467 if (opt.pdlp_halpern) {
468 // ---- reflected Halpern PDHG step (fixed step size) ---------
469 const double tau = eta / omega, sigma = eta * omega;
470 par.zero();
471 par.reduce(0, n,
472 XUpdateStep{c.data, aty->data, lb.data, ub.data, x->data,
473 xn->data, tau});
474 Sparse::spmv(A, *xn, *axn);
475 par.reduce(1, m,
476 YUpdateStep{l.data, u.data, y->data, ax->data, axn->data,
477 yn->data, sigma});
478 Sparse::spmv(AT, *yn, *atyn);
479 const double k1 = static_cast<double>(since_restart);
480 const double w1 = (k1 + 1.0) / (k1 + 2.0),
481 w0 = 1.0 / (k1 + 2.0);
482 // z <- w1 (2 T(z) - z) + w0 z0, in place; T(z) stays in
483 // xn/yn/axn/atyn
484 par.for_each(
485 n, pdlp::HalpernMix{x->data, xn->data, Xr.data, w1, w0});
486 par.for_each(
487 m, pdlp::HalpernMix{y->data, yn->data, Yr.data, w1, w0});
488 par.for_each(
489 m, pdlp::HalpernMix{ax->data, axn->data, Axr.data, w1, w0});
490 par.for_each(n,
491 pdlp::HalpernMix{aty->data, atyn->data, Atyr.data, w1, w0});
492 ++it;
493 ++since_restart;
494 continue;
495 }
496
497 // ---- one PDHG iteration with adaptive step size ---------------
498 double eta_used = 0;
499 bool accepted = false;
500 for (int tries = 0; tries < 60 && !accepted; ++tries) {
501 const double tau = eta / omega, sigma = eta * omega;
502 par.zero();
503 par.reduce(0, n,
504 XUpdateStep{c.data, aty->data, lb.data, ub.data, x->data,
505 xn->data, tau});
506 Sparse::spmv(A, *xn, *axn);
507 par.reduce(1, m,
508 YUpdateStep{l.data, u.data, y->data, ax->data, axn->data,
509 yn->data, sigma});
510 Sparse::spmv(AT, *yn, *atyn);
511 const double *s = par.fetch();
512 const double dz = omega * s[0] + s[1] / omega;
513 const double denom = 2.0 * std::abs(s[2]);
514 if (!std::isfinite(dz) || !std::isfinite(denom))
515 return finish(Status::NumericalError, *x, *y, it);
516 const double eta_bar = denom > 0 ? dz / denom : kInf;
517 const double kk = static_cast<double>(it + 2);
518 const double eta_prime =
519 std::min((1.0 - std::pow(kk, -0.3)) * eta_bar,
520 (1.0 + std::pow(kk, -0.6)) * eta);
521 if (eta <= eta_bar) {
522 accepted = true;
523 eta_used = eta;
524 }
525 eta = eta_prime;
526 if (!(eta > 0) || !std::isfinite(eta))
527 return finish(Status::NumericalError, *x, *y, it);
528 }
529 if (!accepted) return finish(Status::NumericalError, *x, *y, it);
530
531 se += eta_used;
532 par.for_each(
533 n, AvgN{eta_used, xn->data, atyn->data, Sx.data, Saty.data});
534 par.for_each(
535 m, AvgM{eta_used, yn->data, axn->data, Sy.data, Sax.data});
536 std::swap(x, xn);
537 std::swap(y, yn);
538 std::swap(ax, axn);
539 std::swap(aty, atyn);
540 ++it;
541 ++since_restart;
542 }
543 }
