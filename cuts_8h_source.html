3// Gomory mixed-integer (GMI) cuts from the optimal simplex tableau (MILP root).
4//
5// A tableau row of a basic integer column x_k with fractional value reads
6// x_k + sum_{j nonbasic} a_j v_j = 0
7// (v_j a column, or the activity w_i = (A x)_i of row i, see
8// DualSimplex::tableau_row). Writing each nonbasic through its distance to the
9// bound it sits at, s_j = v_j - L_j (at lower) or U_j - v_j (at upper), s_j >=
10// 0, gives x_k + sum a'_j s_j = b with b = the current x_k and a'_j = +-a_j.
11// With f0 = frac(b) the GMI inequality is
12// sum g_j s_j >= 1,
13// g_j = f_j / f0 or (1 - f_j) / (1 - f0) (s_j integer, f_j = frac(a'_j))
14// g_j = a'_j / f0 or -a'_j / (1 - f0) (s_j continuous, by sign)
15// s_j is integer when v_j is an integer column with integral bounds; row
16// activities count as continuous. Substituting back and expanding the row
17// activities gives a cut sum pi_t x_t >= pi_0 valid for every integer point
18// within the bounds it was derived from (the global root bounds).
19//
20// A cut is kept only if numerically safe: no free nonbasic in the row,
21// coefficients within a 1e6 range (tiny ones moved to the right-hand side
22// through the column bounds, keeping validity), and a violation by the LP point
23// of at least 1e-4 relative to its norm (efficacy).

44// GMI cut from basis position r of the last solve of `lp` by `spx`. lb/ub:
45// the column bounds the cut must be valid for. x: the LP point (columns).
46inline bool
47gmi_cut(const DualSimplex &spx, const LpProblem &lp, int r,
48 const std::vector<double> &lb, const std::vector<double> &ub,
49 const std::vector<double> &x, Cut &cut, std::vector<int> &tidx,
50 std::vector<double> &tcoef, std::vector<double> &dense,
51 std::vector<int> &touched)
52{
53 const int n = static_cast<int>(lp.cols());
54 const int k = spx.basic_at(r);
55 if (k >= n || !is_int(lp, k)) return false;
56 const double xk = spx.value_of(k);
57 const double f0 = xk - std::floor(xk);
58 if (f0 < 0.01 || f0 > 0.99) return false;
59 spx.tableau_row(r, tidx, tcoef);
60 // cut in the nonbasic variables: sum g_j s_j >= 1
61 // pi accumulates the coefficients over the columns, rhs the constant
62 double rhs = 1.0;
63 auto add = [&](int t, double v) {
64 if (dense[t] == 0.0) touched.push_back(t);
65 dense[t] += v;
66 if (dense[t] == 0.0) dense[t] = 1e-300; // keep it marked as touched
67 };
68 touched.clear();
69 const auto *rp = lp.A.row_ptr();
70 const auto *ci = lp.A.col_ind();
71 const double *va = lp.A.values();
72 bool ok = true;
73 for (size_t t = 0; t < tidx.size() && ok; ++t) {
74 const int j = tidx[t];
75 const double a = tcoef[t];
76 double L, U;
77 if (j < n) {
78 L = lb[j];
79 U = ub[j];
80 } else {
81 L = lp.row_lb[j - n];
82 U = lp.row_ub[j - n];
83 }
84 if (L == U) continue; // fixed: s_j = 0
85 const VarStatus st = spx.status_of(j);
86 if (st != VarStatus::AtLower && st != VarStatus::AtUpper) {
87 ok = false; // a free nonbasic: no valid GMI from this row
88 break;
89 }
90 const bool at_lower = st == VarStatus::AtLower;
91 const double bound = at_lower ? L : U;
92 if (std::abs(bound) > 1e9) {
93 ok = false;
94 break;
95 }
96 const double ap = at_lower ? a : -a;
97 const bool integer_s =
98 j < n && is_int(lp, j) && bound == std::round(bound);
99 double g;
100 if (integer_s) {
101 const double fj = ap - std::floor(ap);
102 g = fj <= f0 ? fj / f0 : (1 - fj) / (1 - f0);
103 } else {
104 g = ap >= 0 ? ap / f0 : -ap / (1 - f0);
105 }
106 if (g == 0) continue;
107 // g s_j = g (v_j - L) at lower, g (U - v_j) at upper
108 const double sign = at_lower ? 1.0 : -1.0;
109 rhs += sign * g * bound;
110 if (j < n) {
111 add(j, sign * g);
112 } else {
113 const int i = j - n;
114 for (auto q = rp[i]; q < rp[i + 1]; ++q)
115 add(ci[q], sign * g * va[q]);
116 }
117 }
118 if (!ok) {
119 for (int t : touched)
120 dense[t] = 0.0;
121 return false;
122 }
123 // collect, drop tiny coefficients through the bounds, check the range
124 double mx = 0;
125 for (int t : touched)
126 mx = std::max(mx, std::abs(dense[t]));
127 cut.idx.clear();
128 cut.coef.clear();
129 if (mx == 0) {
130 for (int t : touched)
131 dense[t] = 0.0;
132 return false;
133 }
134 double mn = 1e300;
135 for (int t : touched) {
136 const double v = dense[t];
137 dense[t] = 0.0;
138 if (std::abs(v) < 1e-9 * mx) {
139 // sum pi x >= rhs -> drop pi_t x_t: rhs -= max over the bounds of
140 // pi_t x_t
141 const double worst = v > 0 ? v * ub[t] : v * lb[t];
142 if (!std::isfinite(worst) || std::abs(worst) > 1e12) {
143 ok = false;
144 continue;
145 }
146 rhs -= worst;
147 continue;
148 }
149 cut.idx.push_back(t);
150 cut.coef.push_back(v);
151 mn = std::min(mn, std::abs(v));
152 }
153 if (!ok || cut.idx.empty() || mx / mn > 1e6) return false;
154 cut.rhs = rhs;
155 // violation and efficacy at the LP point
156 double act = 0, norm2 = 0;
157 for (size_t t = 0; t < cut.idx.size(); ++t) {
158 act += cut.coef[t] * x[cut.idx[t]];
159 norm2 += cut.coef[t] * cut.coef[t];
160 }
161 const double viol = rhs - act;
162 cut.efficacy = viol / std::sqrt(norm2);
163 if (viol <= 1e-6 * (1 + std::abs(rhs)) || cut.efficacy < 1e-4) return false;
164 // scale to a unit largest coefficient
165 const double s = 1.0 / mx;
166 for (double &c : cut.coef)
167 c *= s;
168 cut.rhs *= s;
169 return true;
170}

172// Appends cuts as rows (rhs <= row <= +inf) to lp.
173inline void
174append_cuts(LpProblem &lp, const std::vector<Cut> &cuts)
175{
176 if (cuts.empty()) return;
177 const size_t m = lp.rows(), n = lp.cols();
178 std::vector<int32_t> rp(lp.A.row_ptr(), lp.A.row_ptr() + m + 1);
179 std::vector<int32_t> ci(lp.A.col_ind(), lp.A.col_ind() + lp.A.nnz());
180 std::vector<double> va(lp.A.values(), lp.A.values() + lp.A.nnz());
181 for (const Cut &c : cuts) {
182 std::vector<std::pair<int, double>> e;
183 for (size_t t = 0; t < c.idx.size(); ++t)
184 e.emplace_back(c.idx[t], c.coef[t]);
185 std::sort(e.begin(), e.end());
186 for (auto &pr : e) {
187 ci.push_back(pr.first);
188 va.push_back(pr.second);
189 }
190 rp.push_back(static_cast<int32_t>(ci.size()));
191 lp.row_lb.push_back(c.rhs);
192 lp.row_ub.push_back(kInf);
193 }
194 lp.A = HostMatrix(m + cuts.size(), n, rp, ci, va);
195}

197// Keeps the rows i with keep[i] set (cut removal).
198inline void
199remove_rows(LpProblem &lp, const std::vector<char> &keep)
200{
201 const size_t m = lp.rows(), n = lp.cols();
202 const int32_t *rp0 = lp.A.row_ptr(), *ci0 = lp.A.col_ind();
203 const double *va0 = lp.A.values();
204 std::vector<int32_t> rp(1, 0), ci;
205 std::vector<double> va, rl, ru;
206 for (size_t i = 0; i < m; ++i) {
207 if (!keep[i]) continue;
208 for (int32_t t = rp0[i]; t < rp0[i + 1]; ++t) {
209 ci.push_back(ci0[t]);
210 va.push_back(va0[t]);
211 }
212 rp.push_back(static_cast<int32_t>(ci.size()));
213 rl.push_back(lp.row_lb[i]);
214 ru.push_back(lp.row_ub[i]);
215 }
216 lp.A = HostMatrix(rl.size(), n, rp, ci, va);
217 lp.row_lb.swap(rl);
218 lp.row_ub.swap(ru);
219 if (lp.row_names.size() == m) {
220 std::vector<std::string> nm;
221 for (size_t i = 0; i < m; ++i)
222 if (keep[i]) nm.push_back(lp.row_names[i]);
223 lp.row_names.swap(nm);
224 }
225}
