3// HPR-QP: dual Halpern Peaceman-Rachford for convex QP, Algorithm 4 of
4// K. Chen, D. Sun, Y. Yuan, G. Zhang, X. Zhao, arXiv:2507.02470.
5//
6// primal min 1/2 <x,Qx> + <c,x> + delta_C(x) s.t. Ax in K
7// dual min 1/2 <w,Qw> + delta*_K(-y) + delta*_C(-z)
8// s.t. -Q w + A^T y + z = c, w in range(Q)
9//
10// Iteration (state u = (y, w, x); only Q w and A^T y are needed, so w is never
11// projected onto range(Q) - the paper's "shadow sequence"):
12// z/x xbar = P_C(x + s(A^T y - Q w - c)), xhat = 2 xbar - x
13// w whalf = (s lq w + xhat) / (1 + s lq) (sGS forward)
14// y ybar = (P_K(r) - r) / (s la), r = A(xhat + s(Qw - Q whalf)) - s la
15// y w wbar = whalf + s/(1 + s lq) A^T(ybar - y) (sGS
16// backward) Halpern u <- (t+1)/(t+2) (2 T(u) - u) + 1/(t+2) u0
17// with la >= |A A^T|, lq >= |Q| (power method). For diagonal Q the w step is
18// exact (no semi-proximal term), lq replaced per element by q_i, as in
19// HPR-QP.jl. When over 80% of the columns are free (opt.hpr_free_variant) the
20// HPR-QP.jl free sweep w, x, y is used instead (no second half w step): three
21// products per iteration instead of four, up to 1.5x faster.
22//
23// An iteration is four kernels: two products with Q, one with A, one with A^T,
24// each fusing the vector updates into the epilogue of the product that feeds
25// them (qp_ops.h). The scalars are read from device memory, so the iterations
26// between two checks are queued as blocks of 64, 32, ..., 1 that the GPU
27// backend captures once each as a CUDA graph and replays with one launch; a
28// check iteration (reductions and KKT) is one more graph plus one sync.
29//
30// Checks every 10 iterations up to 1000, then every opt.check_every (100).
31// Restarts use the merit R = |T(u) - u|_M (metric of the method, x-(w, y)
32// coupling terms): the first check restarts; later, restart when R <= 0.2 R0
33// (R0 = merit at the epoch's first iteration), or R <= 0.8 R0 and R grew since
34// the last check, or the epoch is longer than 1/2 (1/5 once R fell below a
35// tenth of the first check's merit) of all iterations; the new anchor is T(u).
36// The penalty s minimizes the merit a s + b / s + s^2 g(s) of the epoch's
37// displacement (golden section on log s), blended with the best s so far
38// (weight exp(-0.05 R / R_best)), and near convergence (residuals below 1e-9)
39// scaled by the dual / primal infeasibility ratio. These rules follow
40// HPR-QP.jl. Preconditioning: Ruiz (10 passes) and Pock-Chambolle (alpha = 1)
41// on [Q A^T; A 0] (qp_scaling.h).
42//
43// Termination: max(eta_p, eta_d, eta_gap) <= opt.tol at T(u), on the original
44// (unscaled) problem with the bound duals of evaluate_qp() (qp_model.h), which
45// re-evaluates the solution on the host.
46//
47// MPI (opt.comm with more than one rank, qp_dist.h): each rank solves its
48// block of constraint rows, with the n-sized state and Q replicated; the A^T
49// products are summed over the ranks (one MPI_Allreduce of n doubles per
50// iteration), the row reductions of a check combined, decisions follow rank 0.
51// No CUDA graphs then (host steps between iterations). Every rank returns the
52// same solution (y gathered from the blocks).

65// map / reduce / sparse-product epilogue of an op of qp_ops.h on either
66// backend (GPU: kernel "k_<op>")
67#define AXOS_QP_MAP(be, f, n, args) \
68 (be).map("k_" #f, (n), (args), \
69 [](axos_qp::idx i_, const auto &a_) { axos_qp::f(i_, a_); })
70#define AXOS_QP_RED(be, f, n, args) \
71 (be).reduce("k_" #f, (n), (args), \
72 [](axos_qp::idx i_, const auto &a_, double *acc_) { \
73 axos_qp::f(i_, a_, acc_); \
74 })
75#define AXOS_QP_RED_TO(be, f, n, args, slot) \
76 (be).reduce_to( \
77 "k_" #f, (n), (args), \
78 [](axos_qp::idx i_, const auto &a_, double *acc_) { \
79 axos_qp::f(i_, a_, acc_); \
80 }, \
81 (slot))
82#define AXOS_QP_SPMV(be, f, M, x, args) \
83 (be).spmv_epi("k_" #f, (M), (x), (args), \
84 [](axos_qp::idx i_, double s_, const auto &a_) { \
85 axos_qp::f(i_, s_, a_); \
86 })
87// the epilogue on row sums s formed elsewhere (GPU: kernel "k_<op>_vec")
88#define AXOS_QP_EPI(be, f, n, s, args) \
89 (be).epi("k_" #f, (n), (s), (args), \
90 [](axos_qp::idx i_, double s_, const auto &a_) { \
91 axos_qp::f(i_, s_, a_); \
92 })
93// A product with M = A^T (n x local rows) and its epilogue. With the rows
94// split over MPI ranks (qp_dist.h) the local partial product goes to part
95// (n doubles), is summed over the ranks, and the epilogue runs on the sum.
96#define AXOS_QP_SPMV_AT(be, cm, f, M, x, part, args) \
97 do { \
98 if ((cm).distributed()) { \
99 (be).spmv((M), (x), (part)); \
100 ::AXOS::Solver::qp::allreduce_sum((be), (part), (M).rows(), (cm)); \
101 AXOS_QP_EPI(be, f, (M).rows(), (part), args); \
102 } else { \
103 AXOS_QP_SPMV(be, f, M, x, args); \
104 } \
105 } while (0)

111// Largest eigenvalue of M^T M (MT given) or of symmetric M by the power method
112// on the backend, from a fixed generic start vector. The vector is normalized
113// on the device, so blocks of 10 iterations run as one graph and convergence
114// (relative change 1e-6 over a block) is tested between blocks. Without
115// convergence within max_iter the estimate is raised by 5%. With cm, M holds
116// this rank's rows of a matrix split over MPI ranks: the products with MT are
117// summed over the ranks and rank 0's estimate counts.
118template <class B>
119double
120power_method(B &be, const typename B::Mat &M, const typename B::Mat *MT,
121 size_t n, size_t m, int max_iter, const QpComm *cm = nullptr)
122{
123 using namespace axos_qp;
124 if (n == 0) return 0.0;
125 std::vector<double> h(n);
126 for (size_t j = 0; j < n; ++j)
127 h[j] = 1.0 + 0.5 * std::sin(1.7 * double(j) + 0.3);
128 double nrm = 0;
129 for (double v : h)
130 nrm += v * v;
131 for (double &v : h)
132 v /= std::sqrt(nrm);
133 auto v = be.upload(h);
134 auto t = be.vec(std::max<size_t>(m, 1));
135 auto s = be.vec(n);
136 const long key = MT ? -10 : -11;
137 const int kBlock = 10;
138 auto step = [&] {
139 if (MT) {
140 be.spmv(M, B::ptr(v), B::ptr(t));
141 be.spmv(*MT, B::ptr(t), B::ptr(s));
142 if (cm) allreduce_sum(be, B::ptr(s), n, *cm);
143 } else {
144 be.spmv(M, B::ptr(v), B::ptr(s));
145 }
146 // slot 0: <s, v> (Rayleigh quotient, |v| = 1), slot 1: |s|^2
147 AXOS_QP_RED_TO(be, dot2, n, (Dot2{B::ptr(s), B::ptr(v)}), 0);
148 AXOS_QP_MAP(be, scale_norm, n,
149 (ScaleNorm{B::ptr(v), B::ptr(s), be.results_dev() + 1}));
150 };
151 double lam = 0, prev = 0;
152 bool converged = false;
153 for (int k = 0; k < max_iter && !converged; k += kBlock) {
154 be.run_block(key, [&] {
155 for (int j = 0; j < kBlock; ++j)
156 step();
157 be.copy_results(2);
158 });
159 be.sync();
160 double r[2] = {be.results()[0], be.results()[1]};
161 if (cm) cm->bcast(r, 2);
162 if (!(r[1] > 0) || !std::isfinite(r[0]))
163 break; // M v = 0: keep the last estimate
164 lam = r[0];
165 converged = k > 0 && std::abs(lam - prev) <= 1e-6 * std::abs(lam);
166 prev = lam;
167 }
168 be.forget(key);
169 return converged ? lam : 1.05 * lam;
170}

172template <class B> class HprQp {
173 using Vec = typename B::Vec;
174
175 public:
176 QpSolution
177 solve(const QpProblem &p, const QpOptions &opt)
178 {
179 using namespace axos_qp;
180 const auto t_start = std::chrono::steady_clock::now();
181 auto elapsed = [&] {
182 return std::chrono::duration<double>(
183 std::chrono::steady_clock::now() - t_start)
184 .count();
185 };
186 const std::string bad = p.validate();
187 if (!bad.empty()) throw std::invalid_argument("HprQp: " + bad);
188 const size_t n = p.cols(), m = p.rows();
189 const QpComm one;
190 const QpComm &cm = opt.comm ? *opt.comm : one;
191 const bool dist = cm.distributed();
192 const bool talk = cm.rank() == 0; // the rank that prints
193
194 // ---- preconditioning and device data
195 // ----------------------------------
196 QpScaling sc;
197 QpProblem q;
198 if (opt.scaling) {
199 sc = compute_qp_scaling(
200 p, opt.ruiz_iterations, opt.pock_chambolle_alpha);
201 q = apply_qp_scaling(p, sc);
202 } else {
203 sc.col.assign(n, 1.0);
204 sc.row.assign(m, 1.0);
205 q = p;
206 }
207 // this rank's block of constraint rows: [r0, r0 + ml) (all of them
208 // on one process)
209 const auto rows = cm.rows_of(q.lp.A, cm.rank());
210 const size_t r0 = rows.first, ml = rows.second - rows.first;
211 HostMatrix Ablock;
212 if (dist) Ablock = row_block(q.lp.A, r0, r0 + ml);
213 const HostMatrix &Al = dist ? Ablock : q.lp.A;
214 auto local = [&](const std::vector<double> &v) {
215 return std::vector<double>(v.begin() + r0, v.begin() + r0 + ml);
216 };
217 B be;
218 if (dist) be.use_graphs(false); // MPI exchanges inside the iterations
219 const bool hasQ = q.Q.nnz() > 0;
220 // diagonal Q: exact w step (no semi-proximal term, hpr_factors)
221 std::vector<double> hq;
222 if (hasQ && q.Q.rows() == n) {
223 bool diag = true;
224 const auto *qrp = q.Q.row_ptr();
225 const auto *qci = q.Q.col_ind();
226 for (size_t i = 0; i < n && diag; ++i)
227 for (auto k = qrp[i]; k < qrp[i + 1]; ++k)
228 if (static_cast<size_t>(qci[k]) != i) diag = false;
229 if (diag) {
230 hq.assign(n, 0.0);
231 for (size_t i = 0; i < n; ++i)
232 for (auto k = qrp[i]; k < qrp[i + 1]; ++k)
233 hq[i] += q.Q.values()[k];
234 }
235 }
236 const bool qdiag_mode = !hq.empty();
237 // sweep order for (mostly) free columns (see HprFreeXW in qp_ops.h)
238 size_t nfree = 0;
239 for (size_t j = 0; j < n; ++j)
240 if (!std::isfinite(q.lp.col_lb[j]) &&
241 !std::isfinite(q.lp.col_ub[j]))
242 ++nfree;
243 const bool free_mode =
244 opt.hpr_free_variant == 2 ||
245 (opt.hpr_free_variant == 1 && nfree > 0.8 * static_cast<double>(n));
246 auto A = be.upload(Al); // ml x n
247 auto AT = be.upload(Al.transpose()); // n x ml (n empty rows if ml == 0)
248 // n x n even without Q: the fused steps are epilogues of Q products
249 auto Qm = be.upload(q.Q.rows() == n ? q.Q : HostMatrix(n, n, 0));
250 std::vector<double> cinv(n), rinv(ml);
251 for (size_t j = 0; j < n; ++j)
252 cinv[j] = 1.0 / sc.col[j];
253 for (size_t i = 0; i < ml; ++i)
254 rinv[i] = 1.0 / sc.row[r0 + i];
255 Vec c = be.upload(q.lp.c), lb = be.upload(q.lp.col_lb),
256 ub = be.upload(q.lp.col_ub);
257 Vec l = be.upload(local(q.lp.row_lb)),
258 u = be.upload(local(q.lp.row_ub));
259 Vec dcinv = be.upload(cinv), drinv = be.upload(rinv);
260
261 Vec x = be.vec(n), w = be.vec(n), qw = be.vec(n), aty = be.vec(n),
262 y = be.vec(ml);
263 Vec x0 = be.vec(n), w0 = be.vec(n), qw0 = be.vec(n), aty0 = be.vec(n),
264 y0 = be.vec(ml);
265 Vec xbar = be.vec(n), wbar = be.vec(n), qwbar = be.vec(n);
266 Vec atybar = be.vec(n), ybar = be.vec(ml);
267 Vec xhat = be.vec(n), whalf = be.vec(n), qwhalf = be.vec(n);
268 Vec vv = be.vec(n), d = be.vec(n), qd = be.vec(n);
269 Vec tn = be.vec(n), tn2 = be.vec(n),
270 tm = be.vec(ml); // scratch (KKT, restarts)
271 Vec sn = be.vec(dist ? n : 0); // partial A^T products (MPI)
272 Vec scal = be.vec(kScalars); // block scalars (HprScalar)
273 Vec qdv = be.upload(hq); // diag(Q) (diagonal Q only)
274 Vec qbv = be.vec(hq.size()); // 1 / (1 + sigma diag(Q))
275 const double *QD = qdiag_mode ? B::ptr(qbv) : nullptr;
276 auto P = [](Vec &v) { return B::ptr(v); };
277
278 // ---- spectral estimates and initial penalty
279 // ---------------------------
280 const QpComm *pcm = dist ? &cm : nullptr;
281 const double lamA = m ? 1.01 * power_method(be, A, &AT, n, ml,
282 opt.power_iterations, pcm)
283 : 0.0;
284 double lamQ = 0.0;
285 if (qdiag_mode)
286 for (double v : hq)
287 lamQ = std::max(lamQ, v);
288 else if (hasQ)
289 lamQ = 1.01 * power_method(
290 be, Qm, nullptr, n, n, opt.power_iterations, pcm);
291 double sigma = 1.0;
292 {
293 double nb = 0, nc = 0;
294 for (size_t i = 0; i < m; ++i) {
295 double b = 0;
296 if (std::isfinite(q.lp.row_lb[i])) b = std::abs(q.lp.row_lb[i]);
297 if (std::isfinite(q.lp.row_ub[i]))
298 b = std::max(b, std::abs(q.lp.row_ub[i]));
299 nb += b * b;
300 }
301 for (double v : q.lp.c)
302 nc += v * v;
303 nb = std::sqrt(nb);
304 nc = std::sqrt(nc);
305 if (nb > 1e-16 && nb < 1e16 && nc > 1e-16 && nc < 1e16)
306 sigma = nb / nc;
307 }
308 // original-scale norms for the relative residuals
309 double bmax = 0, cmax = 0;
310 for (size_t i = 0; i < m; ++i) {
311 if (std::isfinite(p.lp.row_lb[i]))
312 bmax = std::max(bmax, std::abs(p.lp.row_lb[i]));
313 if (std::isfinite(p.lp.row_ub[i]))
314 bmax = std::max(bmax, std::abs(p.lp.row_ub[i]));
315 }
316 for (double v : p.lp.c)
317 cmax = std::max(cmax, std::abs(v));
318 const double setup_s = elapsed();
319
320 // ---- the iteration, as queued work
321 // -------------------------------------
322 const double *S = P(scal);
323 // scalars of the iterations t0, t0 + 1, ... that follow
324 auto set_scalars = [&](long t0) {
325 const double sl = sigma * lamQ;
326 AXOS_QP_MAP(be, hpr_scalars, kMaxBlock,
327 (HprScalars{P(scal), sigma, sl / (1.0 + sl), 1.0 / (1.0 + sl),
328 sigma * lamA, sigma / (1.0 + sl),
329 static_cast<double>(t0)}));
330 };
331 auto set_qb =
332 [&] { // per-element factors of a diagonal Q, after sigma changed
333 if (qdiag_mode)
334 AXOS_QP_MAP(be, hpr_qb, n, (HprQb{P(qbv), P(qdv), sigma}));
335 };
336 set_qb();
337 auto primal = [&](double *zb) {
338 AXOS_QP_MAP(be, hpr_primal, n,
339 (HprPrimal{P(x), P(qw), P(aty), P(c), P(lb), P(ub), P(xbar),
340 P(xhat), zb, S}));
341 };
342 auto first_half = [&](int j,
343 int halpern_y) { // w half step, y step, A^T ybar
344 AXOS_QP_SPMV(be, hpr_whalf, Qm, P(xhat),
345 (HprWHalf{
346 P(w), P(qw), P(xhat), P(whalf), P(qwhalf), P(vv), S, QD}));
347 AXOS_QP_SPMV(be, hpr_dual, A, P(vv),
348 (HprDual{P(y), P(ybar), P(l), P(u), P(y0), S, j, halpern_y}));
349 AXOS_QP_SPMV_AT(be, cm, hpr_aty, AT, P(ybar), P(sn),
350 (HprAty{P(aty), P(atybar), P(d)}));
351 };
352 // the free-variable order: A^T y (x, w steps), Q wbar (Q w, point v), A
353 // v (y)
354 auto free_iteration = [&](int j, int halpern) {
355 AXOS_QP_SPMV_AT(be, cm, hpr_free_xw, AT, P(y), P(sn),
356 (HprFreeXW{P(x), P(w), P(xbar), P(xhat), P(wbar), P(aty), P(qw),
357 P(c), P(lb), P(ub), P(x0), P(w0), S, QD, j, halpern}));
358 AXOS_QP_SPMV(be, hpr_free_q, Qm, P(wbar),
359 (HprFreeQ{P(qw), P(vv), halpern ? nullptr : P(qwbar), P(xhat),
360 P(qw0), S, j, halpern}));
361 AXOS_QP_SPMV(be, hpr_dual, A, P(vv),
362 (HprDual{P(y), P(ybar), P(l), P(u), P(y0), S, j, halpern}));
363 };
364 // L iterations without a check (offsets j = 0 .. L-1 from the scalars'
365 // t0)
366 auto block = [&](long L) {
367 if (free_mode) {
368 for (long j = 0; j < L; ++j)
369 free_iteration(static_cast<int>(j), 1);
370 return;
371 }
372 primal(nullptr);
373 for (long j = 0; j < L; ++j) {
374 first_half(static_cast<int>(j), 1);
375 AXOS_QP_SPMV(be, hpr_halpern, Qm, P(d),
376 (HprHalpern{P(x), P(w), P(qw), P(aty), P(xbar), P(xhat),
377 P(whalf), P(qwhalf), P(d), P(atybar), P(x0), P(w0),
378 P(qw0), P(aty0), P(c), P(lb), P(ub), S, QD,
379 static_cast<int>(j), j + 1 < L ? 1 : 0}));
380 }
381 };
382 // one iteration with the merit terms (slots 0-8) and the KKT residuals
383 // at T(u) = (xbar, ybar, atybar) (slots 9-17), with the bound duals z
384 // chosen as in evaluate_qp (the projection of Q x + c - A^T y onto
385 // their sign cone), so the check measures exactly what the solution is
386 // scored with
387 auto check_iteration = [&] {
388 if (free_mode) {
389 // xbar, wbar, Q wbar and ybar without the Halpern steps, then
390 // A^T ybar and d = A^T(ybar - y) for the merit and the KKT test
391 free_iteration(0, 0);
392 AXOS_QP_SPMV_AT(be, cm, hpr_aty, AT, P(ybar), P(sn),
393 (HprAty{P(aty), P(atybar), P(d)}));
394 } else {
395 primal(nullptr);
396 first_half(0, 0);
397 }
398 be.spmv(Qm, P(d), P(qd));
399 // (free order: wbar, Q wbar are whalf, qwhalf; no second half)
400 AXOS_QP_RED_TO(be, hpr_halpern_red, n,
401 (HprHalpernRed{P(x), P(w), P(qw), P(aty), P(wbar), P(qwbar),
402 P(xbar), free_mode ? P(wbar) : P(whalf),
403 free_mode ? P(qwbar) : P(qwhalf), P(d), P(qd), P(atybar),
404 P(x0), P(w0), P(qw0), P(aty0), S, QD, 0,
405 free_mode ? 0 : 1}),
406 0);
407 AXOS_QP_RED_TO(be, hpr_halpern_y_red, ml,
408 (HprHalpernYRed{P(y), P(ybar), P(y0), S, 0}), 8);
409 be.spmv(Qm, P(xbar), P(tn));
410 be.spmv(A, P(xbar), P(tm));
411 AXOS_QP_RED_TO(be, kkt_rows, ml,
412 (KktRows{P(tm), P(ybar), P(l), P(u), P(drinv)}), 9);
413 AXOS_QP_RED_TO(be, pd_kkt_cols, n,
414 (PdKktCols{
415 P(xbar), P(tn), P(atybar), P(c), P(lb), P(ub), P(dcinv)}),
416 12);
417 be.copy_results(18);
418 };

420 struct Kkt {
421 double eta_p = 0, eta_d = 0, eta_gap = 0, pobj = 0, dobj = 0;
422 double
423 err() const
424 { return std::max({eta_p, eta_d, eta_gap}); }
425 };
426 auto kkt_from = [&](const double *r) {
427 Kkt k;
428 k.pobj = 0.5 * r[15] + r[16] + q.lp.offset;
429 k.dobj = -0.5 * r[15] + r[11] + r[17] + q.lp.offset;
430 k.eta_p = r[9] / (1 + std::max(bmax, r[10]));
431 k.eta_d = r[12] / (1 + std::max({cmax, r[13], r[14]}));
432 k.eta_gap = std::abs(k.pobj - k.dobj) /
433 (1 + std::max(std::abs(k.pobj), std::abs(k.dobj)));
434 return k;
435 };
436 auto finish = [&](Status st, long iters, const Kkt &k) {
437 std::vector<double> hx, hy;
438 be.download(xbar, hx);
439 be.download(ybar, hy);
440 double drift =
441 0; // largest difference of the ranks' copies of x (checksum)
442 if (dist) {
443 double h[2] = {0, 0};
444 for (size_t j = 0; j < n; ++j)
445 h[0] += hx[j] * double(1 + j % 7);
446 h[1] = -h[0];
447 cm.max(h, 2);
448 drift = h[0] + h[1];
449 cm.bcast(hx.data(), n); // rank 0's copy
450 hy = cm.gather_rows(q.lp.A, hy);
451 }
452 for (size_t j = 0; j < n; ++j)
453 hx[j] *= sc.col[j];
454 for (size_t i = 0; i < m; ++i)
455 hy[i] *= sc.row[i];
456 QpSolution s = evaluate_qp(p, hx, hy);
457 s.status = st;
458 s.iterations = iters;
459 s.seconds = elapsed();
460 s.setup_seconds = setup_s;
461 s.method =
462 dist ? "hpr-qp (mpi, " + std::to_string(cm.size()) + " ranks)"
463 : "hpr-qp";
464 s.device = be.name();
465 if (opt.verbose && talk)
466 std::printf(
467 "[hpr-qp] %s after %ld iterations, %.3f s (setup %.3f s): "

490 be.name().c_str(), n, m, q.lp.A.nnz(), q.Q.nnz(), lamA, lamQ,
491 sigma, setup_s);

495 set_scalars(0);
496 const double tp = be.time_us(reps, [&] { primal(nullptr); });

513 P(whalf), P(qwhalf), P(d), P(atybar), P(x0), P(w0),

521 be.run_block(-1, check_iteration);
522 be.sync();

524 if (talk)

528 "| A^T ybar %.1f | "
529 "Q d + Halpern %.1f | plain Q x %.1f\n"
530 " iteration in a %d-iteration graph %.1f | check "
531 "iteration %.1f\n",
532 be.name().c_str(), n, m, q.lp.A.nnz(), q.Q.nnz(), tp, t1,
533 t2, t3, t4, tq, int(kMaxBlock), tb / kMaxBlock, tc);
534 return finish(Status::NotSolved, 0, Kkt{});

536
537 const long ce = std::max(1, opt.check_every);
538 // Checks: every 10 iterations up to 1000, then every ce; restart
539 // decisions at the multiples of ce. The first iteration of an epoch (t
540 // == 0) is also a check: its merit R0 is what the epoch's later merits
541 // are compared with.
542 auto is_check = [&](long i, long tt) {
543 return tt == 0 || i % ce == 0 ||
544 (i < 1000 && i % std::min<long>(10, ce) == 0);
545 };
546 long it = 0, t = 0;
547 // merits R = |T(u) - u|_M: R0 at the epoch start, Rsave at the previous
548 // check, Rw at the first restart check; the best one with its sigma
549 double R0 = 0, Rsave = HUGE_VAL, Rw = 0, Rbest = HUGE_VAL,
550 sigma_best = sigma;
551 bool first = true;
552 Kkt kkt;
553 int restarts = 0;
554 std::vector<double>
555 hd; // A^T(ybar - y0) on the host (diagonal-Q penalty update)
556 while (true) {
557 if (!is_check(it, t)) {
558 long L = 1;
559 while (!is_check(it + L, t + L))
560 ++L;
561 // in blocks of 64, 32, ..., 1 iterations: a few graphs serve
562 // every length
563 for (long len = kMaxBlock; len >= 1; len /= 2)
564 while (L >= len) {
565 set_scalars(t);
566 be.run_block(len, [&] { block(len); });
567 it += len;
568 t += len;
569 L -= len;
570 }
571 continue;
572 }
573
574 set_scalars(t);
575 be.run_block(-1, check_iteration);
576 be.sync();
577 double r[19];
578 std::copy(be.results(), be.results() + 18, r);
579 r[18] = elapsed();
580 if (dist) { // rows split: sums and maxima of the row terms over the
581 // ranks
582 double add[2] = {r[8], r[11]}, mx[3] = {r[9], r[10], r[18]};
583 cm.sum(add, 2);
584 cm.max(mx, 3);
585 r[8] = add[0], r[11] = add[1];
586 r[9] = mx[0], r[10] = mx[1], r[18] = mx[2];
587 cm.bcast(
588 r, 19); // rank 0's values: the same decisions on every rank
589 }
590 // r: 0 |ex|^2, 1 <ew,Q ew>, 2 <Q ew,d>, 3 <d,Qd>, 4 |Q ew|^2, 5 <Q
591 // ew,ex>,
592 // 6 <d,ex>, 7 sum d (Qd) / (1 + s q), 8 |ybar - y|^2, 9-17 KKT
593 // terms, 18 time (the slowest rank's)
594 const double M1 = sigma * (lamA * r[8] - 2.0 * r[2] +
595 (qdiag_mode ? r[4] : lamQ * r[1]));
596 const double M2 =
597 r[0] / sigma - 2.0 * r[5] + 2.0 * r[6] + std::max(M1, 0.0);
598 const double M3 = sigma * sigma *
599 (qdiag_mode ? r[7] : r[3] / (1.0 + sigma * lamQ));
600 const double R = std::sqrt(std::max(M2, 0.0) + std::max(M3, 0.0));
601 kkt = kkt_from(r);
602 if (opt.verbose >= 2 && talk)
603 std::printf("[hpr-qp] it %-8ld t %-7ld pobj % .10e dobj % .10e "
604 "eta_p %.2e eta_d %.2e "
605 "gap %.2e sigma %.3e R %.3e\n",
606 it, t, kkt.pobj, kkt.dobj, kkt.eta_p, kkt.eta_d,
607 kkt.eta_gap, sigma, R);
608 if (!std::isfinite(kkt.err()))
609 return finish(Status::NumericalError, it, kkt);
610 if (kkt.err() <= opt.tol)
611 return finish(Status::Optimal, it + 1, kkt);
612 if (it + 1 >= opt.max_iterations)
613 return finish(Status::IterationLimit, it + 1, kkt);
614 if (r[18] > opt.time_limit)
615 return finish(Status::TimeLimit, it + 1, kkt);
616
617 // Restart when the merit fell to 0.2 R0 (sufficient), or to 0.8 R0
618 // and grew since the last check (necessary), or the epoch is long:
619 // half of all iterations, a fifth once R fell below 0.1 of the
620 // first check's merit. The first restart check always restarts.
621 bool restart = false;
622 if (t == 0) {
623 R0 = R;
624 } else if (it % ce == 0) {
625 if (first) {
626 restart = true;
627 first = false;
628 Rw = Rbest = R;
629 sigma_best = sigma;
630 } else {
631 const bool suff = R <= 0.2 * R0;
632 const bool nec = R <= 0.8 * R0 && R > Rsave;
633 const double frac = R > 0.1 * Rw ? 0.5 : 0.2;
634 const bool lng = static_cast<double>(t) >=
635 frac * static_cast<double>(it);
636 restart = suff || nec || lng;
637 if (R < Rbest) {
638 Rbest = R;
639 sigma_best = sigma;
640 }
641 Rsave = R;
642 }
643 }
644 ++it;
645 ++t;
646 if (!restart) continue;
647
648 // ---- restart at T(u) with a re-balanced penalty
649 // ------------------------ sigma minimizes a s + b / s + s^2 g(s)
650 // (the merit of the epoch's displacement), g(s) = <d, Q d> / (1 +
651 // lq s) or, for diagonal Q, sum q_i d_i^2 / (1 + s q_i), with d =
652 // A^T(ybar - y0); blended with the best sigma so far, and balanced
653 // between the primal and dual residuals near convergence.
654 const auto th = AXOS_QP_RED(be, hpr_theta_n, n,
655 (HprThetaN{P(xbar), P(x0), P(wbar), P(w0), P(qwbar), P(qw0),
656 P(atybar), P(aty0), P(tn2)}));
657 double dy2 =
658 ml ? AXOS_QP_RED(be, diff_sq, ml, (DiffSq{P(ybar), P(y0)}))[0]
659 : 0.0;
660 cm.sum(&dy2, 1);
661 const double a = std::max(
662 lamA * dy2 - 2.0 * th[2] + (qdiag_mode ? th[3] : lamQ * th[1]),
663 1e-12);
664 const double b = std::max(th[0], 1e-12);
665 double s_est;
666 if (m == 0) {
667 s_est = std::sqrt(b / a);
668 } else if (qdiag_mode) {
669 be.download(tn2, hd);
670 std::vector<std::pair<double, double>> qd2; // (q_i, q_i d_i^2)
671 for (size_t j = 0; j < n; ++j)
672 if (hq[j] > 0 && hd[j] != 0)
673 qd2.emplace_back(hq[j], hq[j] * hd[j] * hd[j]);
674 s_est = golden_log([&](double s) {
675 double g = 0;
676 for (const auto &e : qd2)
677 g += e.second / (1.0 + s * e.first);
678 return a * s + b / s + s * s * g;
679 });
680 } else {
681 double cq = 0;
682 if (hasQ) {
683 be.spmv(Qm, P(tn2), P(tn));
684 cq = AXOS_QP_RED(be, dot2, n, (Dot2{P(tn2), P(tn)}))[0];
685 }
686 s_est = golden_log([&](double s) {
687 return a * s + b / s + s * s * cq / (1.0 + lamQ * s);
688 });
689 }
690 const double fact = std::exp(-0.05 * (R / Rbest));
691 double s_new = std::exp(
692 fact * std::log(s_est) + (1.0 - fact) * std::log(sigma_best));
693 const double t1 = std::max(
694 std::min(kkt.eta_d, kkt.eta_p), std::min(kkt.eta_gap, R));
695 if (t1 <= 9e-10 && kkt.eta_p > 0 && kkt.eta_d > 0) {
696 const double ratio = kkt.eta_d / kkt.eta_p;
697 s_new *= std::min(100.0,
698 std::max(1e-2, t1 > 5e-10 ? std::sqrt(ratio) : ratio));
699 }
700 if (std::isfinite(s_new) && s_new > 0) sigma = s_new;
701 cm.bcast(&sigma, 1);
702 set_qb();
703 copy(x, xbar, n);
704 copy(w, wbar, n);
705 copy(y, ybar, ml);
706 // recompute Q w and A^T y exactly (no drift from the recurrences)
707 be.spmv(Qm, P(w), P(qw));
708 be.spmv(AT, P(y), P(aty));
709 allreduce_sum(be, P(aty), n, cm);
710 copy(x0, x, n);
711 copy(w0, w, n);
712 copy(qw0, qw, n);
713 copy(aty0, aty, n);
714 copy(y0, y, ml);
715 t = 0;
716 Rsave = HUGE_VAL;
717 ++restarts;
718 if (opt.verbose >= 2 && talk)
719 std::printf("[hpr-qp] restart %d at it %ld: sigma %.3e "
720 "(estimate %.3e, a %.2e b %.2e)\n",
721 restarts, it, sigma, s_est, a, b);
722 }
723 }

726 // argmin over s in [1e-12, 1e12] of a function that is convex in s
727 // (so unimodal): golden section on log s.
728 template <class F>
729 static double
730 golden_log(F &&f)
731 {
732 double lo = std::log(1e-12), hi = std::log(1e12);
733 const double g = 0.5 * (std::sqrt(5.0) - 1.0);
734 double a = hi - g * (hi - lo), b = lo + g * (hi - lo);
735 double fa = f(std::exp(a)), fb = f(std::exp(b));
736 for (int k = 0; k < 200 && hi - lo > 1e-10; ++k) {
737 if (fa < fb) {
738 hi = b;
739 b = a;
740 fb = fa;
741 a = hi - g * (hi - lo);
742 fa = f(std::exp(a));
743 } else {
744 lo = a;
745 a = b;
746 fa = fb;
747 b = lo + g * (hi - lo);
748 fb = f(std::exp(b));
749 }
750 }
751 return std::exp(0.5 * (lo + hi));
752 }
