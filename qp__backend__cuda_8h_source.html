3// GPU backend of the first-order QP solvers. Host code only: needs the CUDA
4// runtime and driver (cudart, cuda) but no nvcc. The kernels (qp_kernels.cuh,
5// built on qp_ops.h) are compiled once per process at run time with NVRTC for
6// the device's architecture and launched through the driver API; vectors live
7// in the GPU memory pool. No cuBLAS or cuSPARSE: the CSR products are this
8// project's own kernels.
9//
10// NVRTC is loaded dynamically (nvrtc64_120_0.dll / libnvrtc.so.12, from the
11// library path), so it is not a link-time dependency. The kernel sources come
12// from the source tree: AXOS_SRC_DIR (environment variable, else the
13// compile-time macro, else derived from this file's path) must name the
14// directory containing solver/qp/.
15//
16// Same interface as qp_backend_cpu.h. All work is queued on one stream, so a
17// block of iterations can be captured as a CUDA graph (run_block) and replayed
18// with a single launch.

78// The subset of the NVRTC API used here, resolved at run time.
79struct Nvrtc {
80 typedef int (*Create)(void **, const char *, const char *, int,
81 const char *const *, const char *const *);
82 typedef int (*Compile)(void *, int, const char *const *);
83 typedef int (*Size)(void *, size_t *);
84 typedef int (*Get)(void *, char *);
85 typedef int (*Destroy)(void **);
86 typedef const char *(*ErrStr)(int);
87 Create create = nullptr;
88 Compile compile = nullptr;
89 Size cubin_size = nullptr, log_size = nullptr;
90 Get cubin = nullptr, log = nullptr;
91 Destroy destroy = nullptr;
92 ErrStr error_string = nullptr;
93
94 static const Nvrtc &
95 get()
96 {
97 static Nvrtc n = load();
98 return n;
99 }
100
101 private:
102 static Nvrtc
103 load()
104 {
105 Nvrtc n;
106#ifdef _WIN32
107 const char *names[] = {
108 "nvrtc64_120_0.dll", "nvrtc64_130_0.dll", "nvrtc64_112_0.dll"};
109 HMODULE h = nullptr;
110 for (const char *nm : names)
111 if ((h = LoadLibraryA(nm)) != nullptr) break;
112 auto sym = [&](const char *s) {
113 return reinterpret_cast<void *>(GetProcAddress(h, s));
114 };
115#else
116 const char *names[] = {
117 "libnvrtc.so.12", "libnvrtc.so.13", "libnvrtc.so"};
118 void *h = nullptr;
119 for (const char *nm : names)
120 if ((h = dlopen(nm, RTLD_NOW | RTLD_LOCAL)) != nullptr) break;
121 auto sym = [&](const char *s) { return dlsym(h, s); };
122#endif
123 if (!h)
124 throw std::runtime_error(
125 "NVRTC not found: put the nvrtc library "
126 "(pip: nvidia-cuda-nvrtc-cu12) on the library path");
127 n.create = reinterpret_cast<Create>(sym("nvrtcCreateProgram"));
128 n.compile = reinterpret_cast<Compile>(sym("nvrtcCompileProgram"));
129 n.cubin_size = reinterpret_cast<Size>(sym("nvrtcGetCUBINSize"));
130 n.cubin = reinterpret_cast<Get>(sym("nvrtcGetCUBIN"));
131 n.log_size = reinterpret_cast<Size>(sym("nvrtcGetProgramLogSize"));
132 n.log = reinterpret_cast<Get>(sym("nvrtcGetProgramLog"));
133 n.destroy = reinterpret_cast<Destroy>(sym("nvrtcDestroyProgram"));
134 n.error_string = reinterpret_cast<ErrStr>(sym("nvrtcGetErrorString"));
135 if (!n.create || !n.compile || !n.cubin_size || !n.cubin ||
136 !n.log_size || !n.log || !n.destroy)
137 throw std::runtime_error("NVRTC library lacks a required function");
138 return n;
139 }
140};

169// The compiled kernels, once per process (current device).
170class Module {
171 public:
172 static Module &
173 get()
174 {
175 static std::once_flag once;
176 static Module *m = nullptr;
177 std::call_once(once, [] { m = new Module(); }); // kept until exit
178 return *m;
179 }
180
181 CUfunction
182 fn(const char *name)
183 {
184 std::lock_guard<std::mutex> g(mu_);
185 auto it = fns_.find(name);
186 if (it != fns_.end()) return it->second;
187 CUfunction f;
188 check(cuModuleGetFunction(&f, mod_, name), name);
189 fns_.emplace(name, f);
190 return f;
191 }
192
193 std::string
194 arch() const
195 { return arch_; }
196 double
197 compile_seconds() const
198 { return compile_s_; }
199
200 private:
201 Module()
202 {
203 check(cudaFree(nullptr), "cudaFree(0) (runtime initialization)");
204 int dev = 0, major = 0, minor = 0;
205 check(cudaGetDevice(&dev), "cudaGetDevice");
206 check(cudaDeviceGetAttribute(
207 &major, cudaDevAttrComputeCapabilityMajor, dev),
208 "attribute");
209 check(cudaDeviceGetAttribute(
210 &minor, cudaDevAttrComputeCapabilityMinor, dev),
211 "attribute");
212 arch_ = "sm_" + std::to_string(major * 10 + minor);
213 const auto t0 = std::chrono::steady_clock::now();
214 const Nvrtc &nv = Nvrtc::get();
215 const std::string src = "#include \"solver/qp/qp_kernels.cuh\"\n";
216 const std::string opt_arch = "--gpu-architecture=" + arch_;
217 const std::string opt_inc = "-I" + source_dir();
218 const char *opts[] = {opt_arch.c_str(), "-std=c++17", opt_inc.c_str(),
219 "--fmad=true", "-lineinfo"};
220 void *prog = nullptr;
221 if (nv.create(&prog, src.c_str(), "axos_qp.cu", 0, nullptr, nullptr) !=
222 0)
223 throw std::runtime_error("nvrtcCreateProgram failed");
224 const int rc = nv.compile(prog, 5, opts);
225 size_t ls = 0;
226 nv.log_size(prog, &ls);
227 std::string log(ls, '\0');
228 if (ls) nv.log(prog, &log[0]);
229 if (rc != 0) {
230 nv.destroy(&prog);
231 throw std::runtime_error(
232 "NVRTC compilation of the QP kernels failed (" + opt_inc +
233 "):\n" + log);
234 }
235 size_t cs = 0;
236 nv.cubin_size(prog, &cs);
237 std::vector<char> bin(cs);
238 nv.cubin(prog, bin.data());
239 nv.destroy(&prog);
240 check(cuModuleLoadData(&mod_, bin.data()), "cuModuleLoadData");
241 compile_s_ =
242 std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
243 .count();
244 }
245
246 CUmodule mod_ = nullptr;
247 std::string arch_;
248 double compile_s_ = 0;
249 std::mutex mu_;
250 std::unordered_map<std::string, CUfunction> fns_;
251};

255class CudaBackend {
256 public:
257 static constexpr bool is_gpu = true;
258 using Vec = tensorET<1, double, Cuda::CudaStorage<double>>;
259 using IVec = tensorET<1, int32_t, Cuda::CudaStorage<int32_t>>;
260 // CSR on the device, T = tpr threads per row (from the mean row length). A
261 // row is summed in len / T sequential steps; to keep a few dense rows from
262 // serializing the whole product, a matrix whose longest row would take much
263 // longer than the target (ch = T * steps nonzeros, steps for about kThreads
264 // threads in flight) is split: the product runs over chunks of at most ch
265 // nonzeros (cp, ctpr threads each) into part, and a second kernel adds each
266 // row's chunks (rcp) before the epilogue.
267 struct Mat {
268 IVec rp, ci;
269 Vec v;
270 size_t nrows = 0, nnzs = 0;
271 int tpr = 1; // threads per row
272 bool split = false;
273 IVec cp, rcp;
274 Vec part;
275 size_t nchunks = 0;
276 int ctpr = 1; // threads per chunk
277 size_t
278 rows() const
279 { return nrows; }
280 size_t
281 nnz() const
282 { return nnzs; }
283 };
284
285 CudaBackend() : mod_(cuda_rt::Module::get())
286 {
287 cuda_rt::check(
288 cuStreamCreate(&stream_, CU_STREAM_DEFAULT), "cuStreamCreate");
289 part_ = Vec({size_t(kMaxBlocks) * 8}, 0.0);
290 out_ = Vec({size_t(kSlots)}, 0.0);
291 cuda_rt::check(cudaMallocHost(reinterpret_cast<void **>(&host_),
292 kSlots * sizeof(double)),
293 "cudaMallocHost");
294 int dev = 0;
295 cuda_rt::check(cudaGetDevice(&dev), "cudaGetDevice");
296 cudaDeviceProp prop;
297 cuda_rt::check(
298 cudaGetDeviceProperties(&prop, dev), "cudaGetDeviceProperties");
299 name_ = std::string("gpu ") + prop.name;
300 }
301
302 ~CudaBackend()
303 {
304 if (stream_) cuStreamSynchronize(stream_);
305 for (auto &g : graphs_)
306 cuGraphExecDestroy(g.second);
307 if (stream_) cuStreamDestroy(stream_);
308 if (host_) cudaFreeHost(host_);
309 if (stage_) cudaFreeHost(stage_);
310 }

312 CudaBackend &operator=(const CudaBackend &) = delete;

357 Mat
358 upload(const HostMatrix &A) const
359 {
360 Mat M;
361 M.nrows = A.rows();
362 M.nnzs = A.nnz();
363 const int32_t *hrp = A.row_ptr();
364 M.rp = ivec(hrp, A.rows() + 1);
365 M.ci = ivec(A.nnz() ? A.col_ind() : nullptr, A.nnz());
366 tensorET<1, double> v({std::max<size_t>(A.nnz(), 1)}, 0.0);
367 if (A.nnz()) std::memcpy(v.data, A.values(), A.nnz() * sizeof(double));
368 M.v = Vec(v);
369 M.tpr =
370 threads_for(A.rows() ? double(A.nnz()) / double(A.rows()) : 0.0);
371 int32_t longest = 0;
372 for (size_t i = 0; i < A.rows(); ++i)
373 longest = std::max(longest, hrp[i + 1] - hrp[i]);
374 const size_t steps =
375 std::min<size_t>(32, std::max<size_t>(4, A.nnz() / kThreads));
376 const int32_t ch = static_cast<int32_t>(steps) * M.tpr;
377 if (longest > 4 * ch) {
378 std::vector<int32_t> cp{0}, rcp{0};
379 for (size_t i = 0; i < A.rows(); ++i) {
380 for (int32_t b = hrp[i]; b < hrp[i + 1]; b += ch)
381 cp.push_back(std::min(b + ch, hrp[i + 1]));
382 rcp.push_back(static_cast<int32_t>(cp.size() - 1));
383 }
384 M.split = true;
385 M.nchunks = cp.size() - 1;
386 M.cp = ivec(cp.data(), cp.size());
387 M.rcp = ivec(rcp.data(), rcp.size());
388 M.part = Vec({std::max<size_t>(M.nchunks, 1)}, 0.0);
389 M.ctpr = threads_for(double(A.nnz()) / double(M.nchunks));
390 }
391 return M;
392 }

394 // s = (M x)_i, then f(i, s, a) for every row i (kernels "<name>_<T>").
395 template <class Args, class F>
396 void
397 spmv_epi(const char *name, const Mat &M, const double *x, const Args &a, F)
398 {
399 if (M.nrows == 0) return;
400 axos_qp::idx rows = static_cast<axos_qp::idx>(M.nrows);
401 Args copy = a;
402 const int *ci = M.ci.data;
403 const double *v = M.v.data;
404 if (!M.split) {
405 const int *rp = M.rp.data;
406 void *args[] = {&rows, &rp, &ci, &v, &x, &copy};
407 launch(fn(name, M.tpr), blocks(M.nrows * size_t(M.tpr)), args);
408 return;
409 }
410 axos_qp::idx nch = static_cast<axos_qp::idx>(M.nchunks);
411 const int *cp = M.cp.data, *rcp = M.rcp.data;
412 double *part = const_cast<double *>(M.part.data);
413 void *cargs[] = {&nch, &cp, &ci, &v, &x, &part};
414 launch(
415 fn("k_chunks", M.ctpr), blocks(M.nchunks * size_t(M.ctpr)), cargs);
416 const double *cpart = part;
417 void *fargs[] = {&rows, &rcp, &cpart, &copy};
418 launch(fn(name, 0), grid(M.nrows, 16384), fargs);
419 }

421 void
422 spmv(const Mat &M, const double *x, double *y)
423 { spmv_epi("k_store", M, x, axos_qp::Store{y}, 0); }

450 // Reduction into the result slots [slot, slot + Args::K) on the device
451 // (zeros when n == 0); nothing is copied back before copy_results().
452 template <class Args, class F>
453 void
454 reduce_to(const char *name, size_t n, const Args &a, F, int slot)
455 {
456 double *part = part_.data;
457 int nb = 0;
458 if (n) {
459 axos_qp::idx nn = static_cast<axos_qp::idx>(n);
460 Args copy = a;
461 void *args[] = {&nn, &copy, &part};
462 const unsigned b = grid(n, kMaxBlocks);
463 launch(mod_.fn(name), b, args);
464 nb = static_cast<int>(b);
465 }
466 int kk = Args::K;
467 unsigned mx = Args::kMax;
468 double *o = out_.data + slot;
469 void *fargs[] = {&nb, &kk, &mx, &part, &o};
470 launch(mod_.fn("k_finish"), 1, fargs);
471 }

473 // Queues the copy of result slots [0, n) to results() (pinned memory, so
474 // it can be part of a graph); valid after sync().
475 void
476 copy_results(int n)
477 {
478 cuda_rt::check(
479 cuMemcpyDtoHAsync(host_, reinterpret_cast<CUdeviceptr>(out_.data),
480 size_t(n) * sizeof(double), stream_),
481 "copy results");
482 }
483 const double *
484 results() const
485 { return host_; }
486 // the result slots in device memory (for ops that read a reduction)
487 const double *
488 results_dev() const
489 { return out_.data; }
490
491 template <class Args, class F>
492 std::array<double, 8>
493 reduce(const char *name, size_t n, const Args &a, F f)
494 {
495 reduce_to(name, n, a, f, 0);
496 copy_results(Args::K);
497 sync();
498 std::array<double, 8> out{};
499 for (int k = 0; k < Args::K; ++k)
500 out[k] = host_[k];
501 return out;
502 }

504 // Runs fn(), which must only queue work whose arguments are the same on
505 // every call with this key: captured into a CUDA graph on the first call
506 // and replayed with one launch afterwards.
507 template <class F>
508 void
509 run_block(long key, F &&fn)
510 {
511 if (!graphs_on_) { // work with host steps in between (MPI)
512 fn();
513 return;
514 }
515 auto it = graphs_.find(key);
516 if (it == graphs_.end()) {
517 cuda_rt::check(cuStreamBeginCapture(
518 stream_, CU_STREAM_CAPTURE_MODE_THREAD_LOCAL),
519 "cuStreamBeginCapture");
520 CUgraph g = nullptr;
521 try {
522 fn();
523 } catch (...) {
524 cuStreamEndCapture(stream_, &g);
525 if (g) cuGraphDestroy(g);
526 throw;
527 }
528 cuda_rt::check(
529 cuStreamEndCapture(stream_, &g), "cuStreamEndCapture");
530 CUgraphExec exec = nullptr;
531 const CUresult r = cuGraphInstantiate(&exec, g, 0);
532 cuGraphDestroy(g);
533 cuda_rt::check(r, "cuGraphInstantiate");
534 it = graphs_.emplace(key, exec).first;
535 }
536 cuda_rt::check(cuGraphLaunch(it->second, stream_), "cuGraphLaunch");
537 }
538
539 // drops the graph of a key whose buffers are about to be freed
540 void
541 forget(long key)
542 {
543 auto it = graphs_.find(key);
544 if (it == graphs_.end()) return;
545 cuGraphExecDestroy(it->second);
546 graphs_.erase(it);
547 }

558 // fn(h) on a host copy h of the n doubles at v, copied back to v after:
559 // ordered with the queued work, through pinned memory (a host-side
560 // exchange of a device vector, such as an MPI reduction)
561 template <class F>
562 void
563 through_host(double *v, size_t n, F &&fn)
564 {
565 if (n == 0) return;
566 if (n > stage_n_) {
567 sync();
568 if (stage_) cudaFreeHost(stage_);
569 stage_ = nullptr;
570 stage_n_ = 0;
571 cuda_rt::check(cudaMallocHost(reinterpret_cast<void **>(&stage_),
572 n * sizeof(double)),
573 "cudaMallocHost");
574 stage_n_ = n;
575 }
576 cuda_rt::check(
577 cuMemcpyDtoHAsync(stage_, reinterpret_cast<CUdeviceptr>(v),
578 n * sizeof(double), stream_),
579 "through_host (to host)");
580 sync();
581 fn(stage_);
582 // stage_ is reused only after a later copy on the same stream, so
583 // this one has finished by then
584 cuda_rt::check(cuMemcpyHtoDAsync(reinterpret_cast<CUdeviceptr>(v),
585 stage_, n * sizeof(double), stream_),
586 "through_host (to device)");
587 }

589 // Microseconds per call of fn() (which queues work), over reps calls
590 // timed with events on the backend stream.
591 template <class F>
592 double
593 time_us(int reps, F &&fn)
594 {
595 CUevent a = nullptr, b = nullptr;
596 cuda_rt::check(cuEventCreate(&a, CU_EVENT_DEFAULT), "cuEventCreate");
597 cuda_rt::check(cuEventCreate(&b, CU_EVENT_DEFAULT), "cuEventCreate");
598 fn(); // warm (graph capture, first launch)
599 cuda_rt::check(cuEventRecord(a, stream_), "cuEventRecord");
600 for (int r = 0; r < reps; ++r)
601 fn();
602 cuda_rt::check(cuEventRecord(b, stream_), "cuEventRecord");
603 cuda_rt::check(cuEventSynchronize(b), "cuEventSynchronize");
604 float ms = 0;
605 cuEventElapsedTime(&ms, a, b);
606 cuEventDestroy(a);
607 cuEventDestroy(b);
608 return 1e3 * double(ms) / reps;
609 }

611 private:
612 static constexpr unsigned kBlock = 256, kMaxBlocks = 1024;
613 static constexpr int kSlots = 64;
614 static constexpr size_t kThreads = 65536;
615
616 static int
617 threads_for(double mean)
618 {
619 int t = 1;
620 while (t < 32 && 2.0 * t <= mean)
621 t *= 2;
622 return t;
623 }
624
625 static IVec
626 ivec(const int32_t *h, size_t n)
627 {
628 tensorET<1, int32_t> t({std::max<size_t>(n, 1)}, 0);
629 if (n && h) std::memcpy(t.data, h, n * sizeof(int32_t));
630 return IVec(t);
631 }
632
633 static unsigned
634 grid(size_t n, unsigned cap)
635 {
636 const size_t g = (n + kBlock - 1) / kBlock;
637 return static_cast<unsigned>(g < cap ? (g ? g : 1) : cap);
638 }
639 static unsigned
640 blocks(size_t threads)
641 { return grid(threads, 0x7fffffffu); }
642
643 // "<name>_<t>" (t threads per row), or "<name>_fin" when t == 0
644 CUfunction
645 fn(const char *name, int t)
646 {
647 std::string s(name);
648 s += t ? "_" + std::to_string(t) : std::string("_fin");
649 return mod_.fn(s.c_str());
650 }
651
652 void
653 launch(CUfunction f, unsigned nblocks, void **args)
654 {
655 cuda_rt::check(cuLaunchKernel(f, nblocks, 1, 1, kBlock, 1, 1, 0,
656 stream_, args, nullptr),
657 "cuLaunchKernel");
658 }
