3// LP presolve with a postsolve stack that restores primal AND dual solutions.
4//
5// Reductions:
6// - empty rows (feasibility check, then removed); free rows
7// (both bounds infinite) are removed too
8// - empty columns (fixed at the bound the cost prefers; may prove the
9// problem dual infeasible / unbounded)
10// - fixed columns (lb == ub: substituted out of the rows and cost)
11// - singleton rows (turned into column bounds)
12// - duplicate / parallel rows (merged; their bounds intersected)
13// applied until nothing changes.
14//
15// postsolve() maps a solution of the reduced problem back to the original
16// problem: x from the stack, y for removed rows from the rules below, and
17// z = c - A^T y recomputed on the original matrix.
18// * singleton row i on column j: if the reduced problem wants x_j at a bound
19// that this row tightened (z_j > 0 at the lower bound, z_j < 0 at the upper
20// bound), the row takes the dual: y_i = z_j / a_ij, which zeroes z_j.
21// * duplicate rows: the merged row's dual goes to whichever original row
22// supplies the binding bound.

68 // Build the original-space solution from a reduced-space one. Also works
69 // when the reduced problem is empty (pass an empty LpSolution with
70 // status Optimal).
71 LpSolution
72 postsolve(const LpSolution &r) const
73 {
74 const size_t m = orig_.rows(), n = orig_.cols();
75 LpSolution s;
76 s.status = r.status;
77 s.iterations = r.iterations;
78 s.seconds = r.seconds;
79 s.x.assign(n, 0.0);
80 s.y.assign(m, 0.0);
81 for (size_t j = 0; j < col_map_.size(); ++j)
82 if (j < r.x.size()) s.x[col_map_[j]] = r.x[j];
83 for (size_t i = 0; i < row_map_.size(); ++i)
84 if (i < r.y.size()) s.y[row_map_[i]] = r.y[i];
85 for (auto &op : stack_) // primal values of removed columns
86 if (op.kind == Op::EmptyCol || op.kind == Op::FixedCol)
87 s.x[op.j] = op.value;
88
89 for (auto it = stack_.rbegin(); it != stack_.rend(); ++it) {
90 const Op &op = *it;
91 if (op.kind == Op::SingletonRow) {
92 double zj = orig_.c[op.j];
93 for (int k = At_.row_ptr()[op.j]; k < At_.row_ptr()[op.j + 1];
94 ++k)
95 zj -= At_.values()[k] * s.y[At_.col_ind()[k]];
96 if ((zj > 0 && op.tight_lb) || (zj < 0 && op.tight_ub))
97 s.y[op.i] = zj / op.a;
98 } else if (op.kind == Op::DuplicateRow) {
99 const double y = s.y[op.i];
100 if (y > 0) {
101 if (op.lb_i >= op.impl_lb_k) { /* row i binds */
102 } else {
103 s.y[op.k] = y / op.alpha;
104 s.y[op.i] = 0;
105 }
106 } else if (y < 0) {
107 if (op.ub_i <= op.impl_ub_k) { /* row i binds */
108 } else {
109 s.y[op.k] = y / op.alpha;
110 s.y[op.i] = 0;
111 }
112 }
113 }
114 }
115
116 // z = c - A^T y on the original matrix.
117 s.z = orig_.c;
118 for (size_t i = 0; i < m; ++i)
119 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1];
120 ++k)
121 s.z[orig_.A.col_ind()[k]] -= orig_.A.values()[k] * s.y[i];
122 LpSolution e = evaluate_solution(orig_, s.x, s.y);
123 e.status = s.status;
124 e.iterations = s.iterations;
125 e.seconds = s.seconds;
126 return e;
127 }

183 void
184 do_row(int i)
185 {
186 if (!row_on_[i]) return;
187 if (opt_.empty_rows && rlb_[i] == -kInf && rub_[i] == kInf) {
188 // a free row constrains nothing
189 Op op;
190 op.kind = Op::EmptyRow;
191 op.i = i;
192 stack_.push_back(op);
193 remove_row(i);
194 return;
195 }
196 if (row_cnt_[i] == 0 && opt_.empty_rows) {
197 if (rlb_[i] > tol_scale(rlb_[i], opt_.feas_tol) ||
198 rub_[i] < -tol_scale(rub_[i], opt_.feas_tol)) {
199 status_ = Status::Infeasible;
200 return;
201 }
202 Op op;
203 op.kind = Op::EmptyRow;
204 op.i = i;
205 stack_.push_back(op);
206 row_on_[i] = 0;
207 } else if (row_cnt_[i] == 1 && opt_.singleton_rows) {
208 int j = -1;
209 double a = 0;
210 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1];
211 ++k)
212 if (col_on_[orig_.A.col_ind()[k]]) {
213 j = orig_.A.col_ind()[k];
214 a = orig_.A.values()[k];
215 break;
216 }
217 if (j < 0 || a == 0) return;
218 double lo = a > 0 ? rlb_[i] / a : rub_[i] / a;
219 double hi = a > 0 ? rub_[i] / a : rlb_[i] / a;
220 Op op;
221 op.kind = Op::SingletonRow;
222 op.i = i;
223 op.j = j;
224 op.a = a;
225 if (lo > clb_[j]) {
226 clb_[j] = lo;
227 op.tight_lb = true;
228 }
229 if (hi < cub_[j]) {
230 cub_[j] = hi;
231 op.tight_ub = true;
232 }
233 if (clb_[j] > cub_[j]) {
234 if (clb_[j] - cub_[j] > tol_scale(cub_[j], opt_.feas_tol)) {
235 status_ = Status::Infeasible;
236 return;
237 }
238 clb_[j] = cub_[j] = 0.5 * (clb_[j] + cub_[j]);
239 }
240 stack_.push_back(op);
241 remove_row(i);
242 cq_.push_back(j);
243 }
244 }
245
246 void
247 do_col(int j)
248 {
249 if (!col_on_[j]) return;
250 if (col_cnt_[j] == 0 && opt_.empty_cols) {
251 double v;
252 if (cost_[j] > 0)
253 v = clb_[j];
254 else if (cost_[j] < 0)
255 v = cub_[j];
256 else
257 v = std::min(std::max(0.0, clb_[j]), cub_[j]);
258 if (std::isinf(v)) {
259 if (cost_[j] != 0) {
260 status_ = Status::Unbounded;
261 return;
262 }
263 v = std::isinf(clb_[j]) ? cub_[j] : clb_[j];
264 if (std::isinf(v)) v = 0;
265 }
266 Op op;
267 op.kind = Op::EmptyCol;
268 op.j = j;
269 op.value = v;
270 stack_.push_back(op);
271 offset_ += cost_[j] * v;
272 col_on_[j] = 0;
273 } else if (clb_[j] == cub_[j] && opt_.fixed_cols) {
274 double v = clb_[j];
275 Op op;
276 op.kind = Op::FixedCol;
277 op.j = j;
278 op.value = v;
279 stack_.push_back(op);
280 offset_ += cost_[j] * v;
281 for (int k = At_.row_ptr()[j]; k < At_.row_ptr()[j + 1]; ++k) {
282 int i = At_.col_ind()[k];
283 if (!row_on_[i]) continue;
284 double d = At_.values()[k] * v;
285 rlb_[i] -= d;
286 rub_[i] -= d;
287 }
288 remove_col(j);
289 }

292 bool
293 merge_duplicates()
294 {
295 // Group active rows (>= 2 entries) by a hash of their normalized
296 // pattern, then verify candidates exactly.
297 std::unordered_map<size_t, std::vector<int>> groups;
298 auto norm_of = [&](int i, int &first_col, double &first_val) {
299 first_col = -1;
300 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1];
301 ++k)
302 if (col_on_[orig_.A.col_ind()[k]]) {
303 first_col = orig_.A.col_ind()[k];
304 first_val = orig_.A.values()[k];
305 return;
306 }
307 };
308 for (size_t i = 0; i < orig_.rows(); ++i) {
309 if (!row_on_[i] || row_cnt_[i] < 2) continue;
310 int fc;
311 double fv = 1;
312 norm_of(static_cast<int>(i), fc, fv);
313 size_t h = 1469598103934665603ULL;
314 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1];
315 ++k) {
316 int j = orig_.A.col_ind()[k];
317 if (!col_on_[j]) continue;
318 double v = orig_.A.values()[k] / fv;
319 long long q = std::llround(v * 1e9);
320 h = (h ^ static_cast<size_t>(j)) * 1099511628211ULL;
321 h = (h ^ static_cast<size_t>(q)) * 1099511628211ULL;
322 }
323 groups[h].push_back(static_cast<int>(i));
324 }
325 bool changed = false;
326 for (auto &g : groups) {
327 auto &rows = g.second;
328 for (size_t a = 0; a + 1 < rows.size(); ++a) {
329 int i = rows[a];
330 if (!row_on_[i]) continue;
331 for (size_t b = a + 1; b < rows.size(); ++b) {
332 int k = rows[b];
333 if (!row_on_[k]) continue;
334 double alpha;
335 if (!parallel(i, k, alpha)) continue;
336 // a_k = alpha * a_i => a_i x in [lb_k/alpha, ub_k/alpha]
337 double ilo = alpha > 0 ? rlb_[k] / alpha : rub_[k] / alpha;
338 double ihi = alpha > 0 ? rub_[k] / alpha : rlb_[k] / alpha;
339 Op op;
340 op.kind = Op::DuplicateRow;
341 op.i = i;
342 op.k = k;
343 op.alpha = alpha;
344 op.lb_i = rlb_[i];
345 op.ub_i = rub_[i];
346 op.impl_lb_k = ilo;
347 op.impl_ub_k = ihi;
348 double nlb = std::max(rlb_[i], ilo),
349 nub = std::min(rub_[i], ihi);
350 if (nlb > nub) {
351 if (nlb - nub > tol_scale(nub, opt_.feas_tol)) {
352 status_ = Status::Infeasible;
353 return true;
354 }
355 nlb = nub = 0.5 * (nlb + nub);
356 }
357 rlb_[i] = nlb;
358 rub_[i] = nub;
359 stack_.push_back(op);
360 remove_row(k);
361 changed = true;
362 }
363 }
364 }
365 return changed;
366 }

368 bool
369 parallel(int i, int k, double &alpha)
370 {
371 const auto *rp = orig_.A.row_ptr();
372 const auto *ci = orig_.A.col_ind();
373 const double *v = orig_.A.values();
374 int p = rp[i], q = rp[k];
375 bool have = false;
376 while (true) {
377 while (p < rp[i + 1] && !col_on_[ci[p]])
378 ++p;
379 while (q < rp[k + 1] && !col_on_[ci[q]])
380 ++q;
381 if (p >= rp[i + 1] || q >= rp[k + 1]) break;
382 if (ci[p] != ci[q]) return false;
383 double r = v[q] / v[p];
384 if (!have) {
385 alpha = r;
386 have = true;
387 } else if (std::abs(r - alpha) > 1e-12 * std::abs(alpha))
388 return false;
389 ++p;
390 ++q;
391 }
392 while (p < rp[i + 1] && !col_on_[ci[p]])
393 ++p;
394 while (q < rp[k + 1] && !col_on_[ci[q]])
395 ++q;
396 return have && p >= rp[i + 1] && q >= rp[k + 1];
397 }

399 void
400 run()
401 {
402 const size_t m = orig_.rows(), n = orig_.cols();
403 At_ = orig_.A.transpose();
404 rlb_ = orig_.row_lb;
405 rub_ = orig_.row_ub;
406 clb_ = orig_.col_lb;
407 cub_ = orig_.col_ub;
408 cost_ = orig_.c;
409 offset_ = orig_.offset;
410 row_on_.assign(m, 1);
411 col_on_.assign(n, 1);
412 row_cnt_.resize(m);
413 col_cnt_.resize(n);
414 for (size_t i = 0; i < m; ++i)
415 row_cnt_[i] = orig_.A.row_ptr()[i + 1] - orig_.A.row_ptr()[i];
416 for (size_t j = 0; j < n; ++j)
417 col_cnt_[j] = At_.row_ptr()[j + 1] - At_.row_ptr()[j];
418 for (size_t i = 0; i < m; ++i)
419 rq_.push_back(static_cast<int>(i));
420 for (size_t j = 0; j < n; ++j)
421 cq_.push_back(static_cast<int>(j));
422
423 while (status_ == Status::NotSolved) {
424 while (status_ == Status::NotSolved &&
425 (!rq_.empty() || !cq_.empty())) {
426 if (!rq_.empty()) {
427 int i = rq_.back();
428 rq_.pop_back();
429 do_row(i);
430 } else {
431 int j = cq_.back();
432 cq_.pop_back();
433 do_col(j);
434 }
435 }
436 if (status_ != Status::NotSolved) break;
437 if (!(opt_.duplicate_rows && merge_duplicates())) break;
438 // merged rows / new bounds may enable more reductions
439 for (size_t i = 0; i < m; ++i)
440 if (row_on_[i] && row_cnt_[i] <= 1)
441 rq_.push_back(static_cast<int>(i));
442 }
443 build_reduced();
444 }

446 void
447 build_reduced()
448 {
449 const size_t m = orig_.rows(), n = orig_.cols();
450 std::vector<int> new_row(m, -1), new_col(n, -1);
451 for (size_t i = 0; i < m; ++i)
452 if (row_on_[i]) {
453 new_row[i] = (int)row_map_.size();
454 row_map_.push_back((int)i);
455 }
456 for (size_t j = 0; j < n; ++j)
457 if (col_on_[j]) {
458 new_col[j] = (int)col_map_.size();
459 col_map_.push_back((int)j);
460 }
461 Sparse::CooBuilder<double> b(row_map_.size(), col_map_.size());
462 for (size_t i = 0; i < m; ++i) {
463 if (!row_on_[i]) continue;
464 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1];
465 ++k)
466 if (col_on_[orig_.A.col_ind()[k]])
467 b.add(new_row[i], new_col[orig_.A.col_ind()[k]],
468 orig_.A.values()[k]);
469 }
470 reduced_.name = orig_.name + "_presolved";
471 reduced_.A = b.build();
472 for (int i : row_map_) {
473 reduced_.row_lb.push_back(rlb_[i]);
474 reduced_.row_ub.push_back(rub_[i]);
475 }
476 for (int j : col_map_) {
477 reduced_.c.push_back(cost_[j]);
478 reduced_.col_lb.push_back(clb_[j]);
479 reduced_.col_ub.push_back(cub_[j]);
480 }
481 reduced_.offset = offset_;
482 reduced_.maximize = orig_.maximize;
483 }
