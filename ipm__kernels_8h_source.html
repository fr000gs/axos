3// Elementwise / reduction functors for the interior-point solver (see ipm.h),
4// executed by Parallel<Backend> (pdlp_kernels.h / pdlp_cuda.h).
5//
6// The IPM works on z = (x, w) of length N = n + m, where w = A x are row
7// slacks. Bounds are lbz/ubz (concatenated column and row bounds). Per entry:
8// barrier on the lower bound bl = finite(lb) && lb != ub
9// barrier on the upper bound bu = finite(ub) && lb != ub
10// fixed lb == ub (an equality row's slack)
11// tl = z - lb and tu = ub - z are the slacks to the bounds, sl/su their duals.
12#pragma once

22AXOS_HD inline bool
23bl(double lb, double ub)
24{ return fin(lb) && lb != ub; }
25AXOS_HD inline bool
26bu(double lb, double ub)
27{ return fin(ub) && lb != ub; }
28AXOS_HD inline bool
29fixedv(double lb, double ub)
30{ return lb == ub; }

32constexpr double kBig = 1e300;
33
34// rd = c - A^T lam - sl + su (columns); lam - sl + su (row slacks, 0 if fixed)
35struct ResD {
36 double *rd;
37 const double *c, *aty, *lam, *sl, *su, *lb, *ub;
38 size_t n;
39 AXOS_HD void
40 operator()(size_t j) const
41 {
42 if (j < n) {
43 rd[j] = c[j] - aty[j] - sl[j] + su[j];
44 } else {
45 rd[j] = fixedv(lb[j], ub[j]) ? 0.0 : lam[j - n] - sl[j] + su[j];
46 }
47 }
48};
49
50// rp = w - A x
51struct ResP {
52 double *rp;
53 const double *w, *ax;
54 AXOS_HD void
55 operator()(size_t i) const
56 { rp[i] = w[i] - ax[i]; }
57};

59// |rd|^2, c^T x, dual objective, sum of complementarity products, count.
60struct Stats {
61 const double *rd, *z, *c, *lam, *tl, *tu, *sl, *su, *lb, *ub;
62 size_t n;
63 static constexpr int K = 5;
64 AXOS_HD void
65 operator()(size_t j, double *acc) const
66 {
67 acc[0] += rd[j] * rd[j];
68 if (j < n) acc[1] += c[j] * z[j];
69 if (bl(lb[j], ub[j])) {
70 acc[2] += lb[j] * sl[j];
71 acc[3] += tl[j] * sl[j];
72 acc[4] += 1.0;
73 }
74 if (bu(lb[j], ub[j])) {
75 acc[2] -= ub[j] * su[j];
76 acc[3] += tu[j] * su[j];
77 acc[4] += 1.0;
78 }
79 if (j >= n && fixedv(lb[j], ub[j])) acc[2] += lam[j - n] * lb[j];
80 }
81};

83struct SqNorm {
84 const double *a;
85 static constexpr int K = 1;
86 AXOS_HD void
87 operator()(size_t i, double *acc) const
88 { acc[0] += a[i] * a[i]; }

91// Theta^{-1} = sl/tl + su/tu (huge for fixed entries)
92struct BuildTheta {
93 double *hz;
94 const double *tl, *tu, *sl, *su, *lb, *ub;
95 AXOS_HD void
96 operator()(size_t j) const
97 {
98 if (fixedv(lb[j], ub[j])) {
99 hz[j] = kBig;
100 return;
101 }
102 double h = 0.0;
103 if (bl(lb[j], ub[j])) h += sl[j] / tl[j];
104 if (bu(lb[j], ub[j])) h += su[j] / tu[j];
105 hz[j] = h;
106 }
107};

109// KKT diagonal: -(Theta_x^{-1} + rho) for columns, Theta_w + delta for rows.
110struct ScatterKkt {
111 double *vals;
112 double *thw;
113 const int *dxpos, *dwpos;
114 const double *hz;
115 double rho, delta;
116 size_t n;
117 AXOS_HD void
118 operator()(size_t j) const
119 {
120 if (j < n) {
121 vals[dxpos[j]] = -(hz[j] + rho);
122 } else {
123 const size_t i = j - n;
124 const double th = hz[j] > 0.0 ? 1.0 / hz[j] : 1e12;
125 thw[i] = th;
126 vals[dwpos[i]] = th + delta;
127 }
128 }
129};

131// predictor complementarity right-hand sides (mu = 0)
132struct CompAff {
133 double *rcl, *rcu;
134 const double *tl, *tu, *sl, *su, *lb, *ub;
135 AXOS_HD void
136 operator()(size_t j) const
137 {
138 rcl[j] = bl(lb[j], ub[j]) ? -tl[j] * sl[j] : 0.0;
139 rcu[j] = bu(lb[j], ub[j]) ? -tu[j] * su[j] : 0.0;
140 }
141};
142
143// corrector: sigma*mu - t s - (second-order term of the predictor step)
144struct CompCorr {
145 double *rcl, *rcu;
146 const double *tl, *tu, *sl, *su, *lb, *ub, *dz, *dsl, *dsu;
147 double sigma_mu;
148 AXOS_HD void
149 operator()(size_t j) const
150 {
151 rcl[j] =
152 bl(lb[j], ub[j]) ? sigma_mu - tl[j] * sl[j] - dz[j] * dsl[j] : 0.0;
153 rcu[j] =
154 bu(lb[j], ub[j]) ? sigma_mu - tu[j] * su[j] + dz[j] * dsu[j] : 0.0;
155 }
156};
157
158// g = -rd + rcl/tl - rcu/tu
159struct BuildG {
160 double *g;
161 const double *rd, *rcl, *rcu, *tl, *tu, *lb, *ub;
162 AXOS_HD void
163 operator()(size_t j) const
164 {
165 double v = -rd[j];
166 if (bl(lb[j], ub[j])) v += rcl[j] / tl[j];
167 if (bu(lb[j], ub[j])) v -= rcu[j] / tu[j];
168 g[j] = v;
169 }
170};
171
172// reduced KKT right-hand side over (n + m): [-g_x ; rp + Theta_w g_w]
173struct PackRhs {
174 double *rhs;
175 const double *g, *rp, *thw;
176 size_t n;
177 AXOS_HD void
178 operator()(size_t j) const
179 {
180 if (j < n)
181 rhs[j] = -g[j];
182 else
183 rhs[j] = rp[j - n] + thw[j - n] * g[j];
184 }
185};

187// directions from the KKT solution
188struct Unpack {
189 double *dz, *dsl, *dsu, *dlam;
190 const double *sol, *g, *thw, *rcl, *rcu, *tl, *tu, *sl, *su, *lb, *ub;
191 size_t n;
192 AXOS_HD void
193 operator()(size_t j) const
194 {
195 double d;
196 if (j < n) {
197 d = sol[j];
198 } else {
199 const size_t i = j - n;
200 dlam[i] = sol[n + i];
201 d = fixedv(lb[j], ub[j]) ? 0.0 : thw[i] * (g[j] - sol[n + i]);
202 }
203 dz[j] = d;
204 dsl[j] = bl(lb[j], ub[j]) ? (rcl[j] - sl[j] * d) / tl[j] : 0.0;
205 dsu[j] = bu(lb[j], ub[j]) ? (rcu[j] + su[j] * d) / tu[j] : 0.0;
206 }
207};

209// largest primal step keeping tl, tu > 0
210struct StepP {
211 const double *tl, *tu, *dz, *lb, *ub;
212 AXOS_HD double
213 operator()(size_t j) const
214 {
215 double v = 1e300;
216 if (bl(lb[j], ub[j]) && dz[j] < 0) v = dmin(v, tl[j] / -dz[j]);
217 if (bu(lb[j], ub[j]) && dz[j] > 0) v = dmin(v, tu[j] / dz[j]);
218 return v;
219 }
220};
221
222// largest dual step keeping sl, su > 0
223struct StepD {
224 const double *sl, *su, *dsl, *dsu, *lb, *ub;
225 AXOS_HD double
226 operator()(size_t j) const
227 {
228 double v = 1e300;
229 if (bl(lb[j], ub[j]) && dsl[j] < 0) v = dmin(v, sl[j] / -dsl[j]);
230 if (bu(lb[j], ub[j]) && dsu[j] < 0) v = dmin(v, su[j] / -dsu[j]);
231 return v;
232 }
233};

235// sum of complementarity products after a trial step
236struct MuAff {
237 const double *tl, *tu, *sl, *su, *dz, *dsl, *dsu, *lb, *ub;
238 double ap, ad;
239 static constexpr int K = 1;
240 AXOS_HD void
241 operator()(size_t j, double *acc) const
242 {
243 if (bl(lb[j], ub[j]))
244 acc[0] += (tl[j] + ap * dz[j]) * (sl[j] + ad * dsl[j]);
245 if (bu(lb[j], ub[j]))
246 acc[0] += (tu[j] - ap * dz[j]) * (su[j] + ad * dsu[j]);
247 }
248};
249
250struct UpdatePrimalDual {
251 double *z, *tl, *tu, *sl, *su;
252 const double *dz, *dsl, *dsu, *lb, *ub;
253 double ap, ad;
254 AXOS_HD void
255 operator()(size_t j) const
256 {
257 const double zj = z[j] + ap * dz[j];
258 z[j] = zj;
259 if (bl(lb[j], ub[j])) {
260 tl[j] = dmax(zj - lb[j], 1e-300);
261 sl[j] += ad * dsl[j];
262 }
263 if (bu(lb[j], ub[j])) {
264 tu[j] = dmax(ub[j] - zj, 1e-300);
265 su[j] += ad * dsu[j];
266 }
267 }
268};

270// Keeps every complementarity pair centered: mu/(kappa t) <= s <= kappa mu/t
271// (the safeguard used by Ipopt), so products cannot collapse or explode
272// relative to mu.
273struct Recenter {
274 double *sl, *su;
275 const double *tl, *tu, *lb, *ub;
276 double mu, kappa;
277 AXOS_HD void
278 operator()(size_t j) const
279 {
280 if (bl(lb[j], ub[j]))
281 sl[j] = clampd(sl[j], mu / (kappa * tl[j]), kappa * mu / tl[j]);
282 if (bu(lb[j], ub[j]))
283 su[j] = clampd(su[j], mu / (kappa * tu[j]), kappa * mu / tu[j]);
284 }
285};

296// residual of the unregularized reduced KKT system
297// [-Theta_x^{-1} A^T; A Theta_w] v = rhs
298struct RefineRes {
299 double *res;
300 const double *rhs, *v, *tmpn, *tmpm, *hz, *thw;
301 size_t n;
302 AXOS_HD void
303 operator()(size_t j) const
304 {
305 if (j < n)
306 res[j] = rhs[j] - (-hz[j] * v[j] + tmpn[j]);
307 else
308 res[j] = rhs[j] - (tmpm[j - n] + thw[j - n] * v[j]);
309 }
310};
