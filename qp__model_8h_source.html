3// Convex quadratic programs:
4//
5// minimize 1/2 x^T Q x + c^T x + offset
6// subject to row_lb <= A x <= row_ub
7// col_lb <= x <= col_ub
8//
9// Q is symmetric positive semidefinite, stored as a full (both triangles) CSR
10// matrix; the linear part and all bounds are an LpProblem, so every LP is a QP
11// with an empty Q. Duals follow the LP convention of model.h: y_i >= 0 only at
12// a lower row bound, z_j >= 0 only at a lower column bound, and at an optimum
13// Q x + c - A^T y - z = 0.
14//
15// evaluate_qp() is the one yardstick used for every solver in the benchmarks:
16// the relative KKT residuals of HPR-QP / PDHCG (infinity norms),
17// eta_p = max(|Ax - P_K(Ax)|, |x - P_C(x)|) / (1 + max(|b|, |Ax|))
18// eta_d = |Q x + c - A^T y - z| / (1 + max(|c|, |A^T y|, |Q x|))
19// eta_gap = |P - D| / (1 + max(|P|, |D|))
20// with b = max(|row_lb|, |row_ub|) over finite entries, P the primal and D the
21// dual objective, and z the bound duals minimizing eta_d for the given (x, y)
22// (free for a boxed column, of one sign for a one-sided column, zero for a free
23// one).

35struct QpProblem {
36 LpProblem lp; // A, c, bounds, offset, names; lp.maximize is reported only
37 HostMatrix Q; // n x n, symmetric, both triangles stored; empty = LP
38
39 size_t
40 rows() const
41 { return lp.rows(); }
42 size_t
43 cols() const
44 { return lp.cols(); }
45 bool
46 has_quadratic() const
47 { return Q.nnz() > 0; }
48
49 std::string
50 validate() const
51 {
52 std::string e = lp.validate();
53 if (!e.empty()) return e;
54 if (Q.nnz() > 0 && (Q.rows() != cols() || Q.cols() != cols()))
55 return "Q must be cols() x cols()";
56 return Q.nnz() > 0 ? Q.validate() : std::string();
57 }
58
59 // y = Q x (host)
60 void
61 q_times(const std::vector<double> &x, std::vector<double> &y) const
62 {
63 y.assign(cols(), 0.0);
64 if (Q.nnz() == 0) return;
65 const auto *rp = Q.row_ptr();
66 const auto *ci = Q.col_ind();
67 const double *v = Q.values();
68 for (size_t i = 0; i < cols(); ++i) {
69 double s = 0;
70 for (int k = rp[i]; k < rp[i + 1]; ++k)
71 s += v[k] * x[ci[k]];
72 y[i] = s;
73 }
74 }
75
76 double
77 objective(const std::vector<double> &x) const
78 {
79 std::vector<double> qx;
80 q_times(x, qx);
81 double s = lp.offset;
82 for (size_t j = 0; j < cols(); ++j)
83 s += x[j] * (0.5 * qx[j] + lp.c[j]);
84 return s;
85 }
86};

88struct QpSolution {
89 Status status = Status::NotSolved;
90 std::vector<double> x, y, z;
91 double primal_objective = 0, dual_objective = 0;
92 double rel_primal = 0, rel_dual = 0, rel_gap = 0; // evaluate_qp metrics
93 long iterations = 0;
94 double seconds = 0; // solve time (after reading the file)
95 double setup_seconds =
96 0; // part of `seconds`: scaling, upload, spectral estimates
97 std::string method, device;
98
99 double
100 kkt() const
101 { return std::max({rel_primal, rel_dual, rel_gap}); }
102};

104// HprQp and Pdhcg: first-order, CPU or GPU; Ipm: interior point, CPU;
105// Auto: the interior point when its factorization is cheap enough
106// (ipm_auto_flops), else or when it fails HPR-QP (GPU for large problems).
107enum class QpMethod { HprQp, Pdhcg, Ipm, Auto };

111struct QpOptions {

113 bool use_gpu = false;

115 // is one) for problems with at least gpu_min_work = nnz(A) + nnz(Q) + n +
116 // m; below that, kernel launch latency outweighs the GPU's bandwidth.
117 bool auto_device = false;

122 // 9x at 3e4-1e5 on the RTX 3050 laptop GPU)

125 long max_iterations = 1000000000L;

129 bool scaling = true;

132 int power_iterations =

137 bool ipm_project = false; // IPM: stop after the first factorization when 25
138 // factorizations would not fit in time_limit
139 // (5 without it)
140 int verbose = 0; // 0 silent, 1 summary, 2 every check
141 // HPR-QP sweep order: 0 the sGS order (x, w, y, w); 2 the order for free
142 // variables (w, x, y: three products per iteration instead of four);
143 // 1 the second when more than 80% of the columns have no bounds.
144 int hpr_free_variant = 1;
145 bool profile =
146 false; // HPR-QP: time each step of an iteration after the setup,
147 // print the table and stop (the iterates are not a solve)
148 // HPR-QP over MPI ranks (qp_dist.h): every rank calls solve_qp with the
149 // same problem and options, holds a block of the constraint rows and
150 // gets the whole solution. nullptr or one rank: a single process.
151 const QpComm *comm = nullptr;

154// Relative KKT residuals of (x, y) for p (see the header comment); fills a
155// QpSolution with x, y, the chosen z and the objectives.
156inline QpSolution
157evaluate_qp(const QpProblem &p, const std::vector<double> &x,
158 const std::vector<double> &y)
159{
160 const LpProblem &q = p.lp;
161 const size_t m = p.rows(), n = p.cols();
162 QpSolution s;
163 s.x = x;
164 s.y = y;
165 s.z.assign(n, 0.0);
166 // A dual on an infinite row bound is infeasible: drop it (the same
167 // clean-up for every solver), so its effect shows in eta_d.
168 for (size_t i = 0; i < m; ++i) {
169 if (s.y[i] > 0 && !std::isfinite(q.row_lb[i])) s.y[i] = 0;
170 if (s.y[i] < 0 && !std::isfinite(q.row_ub[i])) s.y[i] = 0;
171 }
172 const std::vector<double> &yv = s.y;
173 std::vector<double> ax(m, 0.0), aty(n, 0.0), qx;
174 const auto *rp = q.A.row_ptr();
175 const auto *ci = q.A.col_ind();
176 const double *v = q.A.values();
177 for (size_t i = 0; i < m; ++i)
178 for (int k = rp[i]; k < rp[i + 1]; ++k) {
179 ax[i] += v[k] * x[ci[k]];
180 aty[ci[k]] += v[k] * yv[i];
181 }
182 p.q_times(x, qx);
183
184 double bmax = 0, axmax = 0, pviol = 0, dual = q.offset;
185 for (size_t i = 0; i < m; ++i) {
186 if (std::isfinite(q.row_lb[i]))
187 bmax = std::max(bmax, std::abs(q.row_lb[i]));
188 if (std::isfinite(q.row_ub[i]))
189 bmax = std::max(bmax, std::abs(q.row_ub[i]));
190 axmax = std::max(axmax, std::abs(ax[i]));
191 pviol = std::max({pviol, q.row_lb[i] - ax[i], ax[i] - q.row_ub[i]});
192 if (yv[i] > 0)
193 dual += yv[i] * q.row_lb[i];
194 else if (yv[i] < 0)
195 dual += yv[i] * q.row_ub[i];
196 }
197 double cmax = 0, atymax = 0, qxmax = 0, dres = 0, xqx = 0, cx = 0;
198 for (size_t j = 0; j < n; ++j) {
199 pviol = std::max({pviol, q.col_lb[j] - x[j], x[j] - q.col_ub[j]});
200 cmax = std::max(cmax, std::abs(q.c[j]));
201 atymax = std::max(atymax, std::abs(aty[j]));
202 qxmax = std::max(qxmax, std::abs(qx[j]));
203 const double g = qx[j] + q.c[j] - aty[j]; // = z_j at an optimum
204 const bool lo = std::isfinite(q.col_lb[j]),
205 hi = std::isfinite(q.col_ub[j]);
206 double zj = 0;
207 if (lo && hi)
208 zj = g;
209 else if (lo)
210 zj = std::max(g, 0.0);
211 else if (hi)
212 zj = std::min(g, 0.0);
213 s.z[j] = zj;
214 dres = std::max(dres, std::abs(g - zj));
215 if (zj > 0)
216 dual += zj * q.col_lb[j];
217 else if (zj < 0)
218 dual += zj * q.col_ub[j];
219 xqx += x[j] * qx[j];
220 cx += q.c[j] * x[j];
221 }
222 s.primal_objective = 0.5 * xqx + cx + q.offset;
223 s.dual_objective = dual - 0.5 * xqx;
224 s.rel_primal = std::max(pviol, 0.0) / (1 + std::max(bmax, axmax));
225 s.rel_dual = dres / (1 + std::max({cmax, atymax, qxmax}));
226 s.rel_gap = std::abs(s.primal_objective - s.dual_objective) /
227 (1 + std::max(std::abs(s.primal_objective),
228 std::abs(s.dual_objective)));
229 return s;
230}
