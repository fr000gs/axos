3// The AXOS solver API: one entry point for LP, QP and MILP.
4//
5// #include "solver/api.h"
6// using namespace AXOS::Solver;
7//
8// Model m = read_model("problem.mps"); // MPS, or QPS (QUADOBJ /
9// QMATRIX) Options o; o.time_limit = 60; // seconds
10// Result r = solve(m, o); // LP, QP or MILP, from the model
11// if (r.has_solution()) use(r.x, r.objective);
12//
13// The problem type follows from the model: MILP when a column is integer,
14// QP when the objective has a quadratic part, else LP (Options::type forces
15// one, e.g. the LP relaxation of a MILP). Methods (Options::method):
16//
17// LP auto dual simplex; HPR on the GPU (or CPU) if it fails or the
18// problem is very large
19// simplex bounded dual simplex (exact vertex solution)
20// ipm Mehrotra interior point
21// pdlp restarted PDHG (first order, CPU)
22// hpr Halpern Peaceman-Rachford (first order, GPU when available)
23// QP auto interior point when its factorization is cheap, else HPR-QP
24// hprqp HPR-QP (GPU or CPU)
25// pdhcg PDHCG (GPU or CPU)
26// ipm interior point (CPU)
27// MILP auto branch and cut (presolve, probing, cuts, heuristics; with
28// Device::Gpu / Auto the GPU probes, runs feasibility-jump
29// walkers and batched strong branching)
30//
31// Objective values are in the model's own sense (a MAX model reports its
32// maximum). Every returned solution is checked against the model:
33// Result::max_violation is its largest bound / row violation (relative to
34// 1 + |side|), integrality included for a MILP.
35//
36// MPI (build with -DAXOS_ENABLE_MPI): with Options::comm over several ranks,
37// every rank calls solve() with the same model and options; LPs and QPs are
38// solved by HPR (the "auto", "hpr" and "hprqp" methods) with the constraint
39// rows split over the ranks, one GPU per rank (qp_dist.h), and every rank
40// gets the whole solution. MILPs run on one process.

86// An optimization model: the linear part (constraints, bounds, costs,
87// integrality) in qp.lp, the quadratic objective part (if any) in qp.Q.
88struct Model {
89 QpProblem qp;
90
91 ProblemType
92 type() const
93 {
94 if (qp.lp.has_integers()) return ProblemType::MILP;
95 if (qp.has_quadratic()) return ProblemType::QP;
96 return ProblemType::LP;
97 }
98 const std::string &
99 name() const
100 { return qp.lp.name; }
101 size_t
102 rows() const
103 { return qp.rows(); }
104 size_t
105 cols() const
106 { return qp.cols(); }
107 size_t
108 nnz() const
109 { return qp.lp.A.nnz(); }
110 size_t
111 integers() const
112 {
113 size_t k = 0;
114 for (auto v : qp.lp.is_integer)
115 k += v ? 1 : 0;
116 return k;
117 }
118 bool
119 maximize() const
120 { return qp.lp.maximize; }
121};

135struct Options {
136 ProblemType type = ProblemType::Auto; // force LP / QP / MILP
137 std::string method = "auto";
138 Device device = Device::Auto;
139 double time_limit = 3600; // seconds
140 double tol = 1e-6; // LP / QP: relative KKT tolerance
141 double gap = 1e-4; // MILP: relative gap
142 int verbose = 0;
143 bool presolve = true; // MILP switches
144 bool cuts = true;
145 bool heuristics = true;
146 const QpComm *comm =
147 nullptr; // MPI ranks of a distributed LP / QP solve (qp_dist.h)
148};

150struct Result {
151 Status status = Status::NotSolved;
152 ProblemType type = ProblemType::Auto;
153 std::string method, device, message;
154 double objective =
155 std::numeric_limits<double>::quiet_NaN(); // of x, in the model's sense
156 double bound =
157 std::numeric_limits<double>::quiet_NaN(); // dual bound (MILP) / dual
158 // objective (LP, QP)
159 std::vector<double> x; // primal solution (columns), empty if none
160 std::vector<double> y; // row duals (LP, QP; minimization form)
161 double seconds = 0;
162 long iterations =
163 0; // simplex / IPM / first-order iterations (MILP: LP iterations)
164 long nodes = 0; // MILP
165 double max_violation = std::numeric_limits<double>::quiet_NaN();
166
167 bool
168 has_solution() const
169 { return !x.empty(); }
170 // relative gap between objective and bound (MILP convention)
171 double
172 gap() const
173 {
174 if (!has_solution() || std::isnan(bound))
175 return std::numeric_limits<double>::infinity();
176 const double d = std::abs(objective - bound);
177 if (d <= 1e-9) return 0;
178 return d / std::max(std::abs(objective), 1e-9);
179 }
180};

184inline double
185max_violation(
186 const LpProblem &p, const std::vector<double> &x, bool integrality)
187{
188 double v = 0;
189 for (size_t j = 0; j < p.cols(); ++j) {
190 if (x[j] < p.col_lb[j])
191 v = std::max(v, (p.col_lb[j] - x[j]) / (1 + std::abs(p.col_lb[j])));
192 if (x[j] > p.col_ub[j])
193 v = std::max(v, (x[j] - p.col_ub[j]) / (1 + std::abs(p.col_ub[j])));
194 if (integrality && !p.is_integer.empty() && p.is_integer[j])
195 v = std::max(v, std::abs(x[j] - std::round(x[j])));
196 }
197 const int32_t *rp = p.A.row_ptr(), *ci = p.A.col_ind();
198 const double *va = p.A.values();
199 for (size_t i = 0; i < p.rows(); ++i) {
200 double a = 0;
201 for (int32_t k = rp[i]; k < rp[i + 1]; ++k)
202 a += va[k] * x[ci[k]];
203 if (a < p.row_lb[i])
204 v = std::max(v, (p.row_lb[i] - a) / (1 + std::abs(p.row_lb[i])));
205 if (a > p.row_ub[i])
206 v = std::max(v, (a - p.row_ub[i]) / (1 + std::abs(p.row_ub[i])));
207 }
208 return v;
209}

215inline void
216from_qp(const QpSolution &s, double sense, Result &r)
217{
218 r.status = s.status;
219 if (!s.x.empty() &&
220 (s.status == Status::Optimal || s.status == Status::TimeLimit ||
221 s.status == Status::IterationLimit)) {
222 r.x = s.x;
223 r.y = s.y;
224 r.objective = sense * s.primal_objective;
225 r.bound = sense * s.dual_objective;
226 }
227 r.iterations = s.iterations;
228 r.method = s.method;
229 r.device = s.device;
230}

232inline Result
233solve_lp_model(const Model &m, const Options &o)
234{
235 Result r;
236 const LpProblem &p = m.qp.lp;
237 const double sense = p.maximize ? -1.0 : 1.0;
238 const std::string meth = o.method;
239 auto hpr = [&](double tl) {
240 QpProblem q = m.qp; // the reader's Q: n x n, empty
241 q.lp.is_integer.clear();
242 QpOptions qo;
243 qo.method = QpMethod::HprQp;
244 qo.use_gpu = gpu_wanted(o.device);
245 qo.tol = o.tol;
246 qo.time_limit = tl;
247 qo.verbose = o.verbose;
248 qo.comm = o.comm;
249 from_qp(solve_qp(q, qo), sense, r);
250 };
251 if (o.comm && o.comm->distributed()) { // over MPI ranks: HPR only
252 if (meth != "auto" && meth != "hpr")
253 throw std::invalid_argument(
254 "over MPI ranks an LP is solved by HPR (method auto or hpr)");
255 hpr(o.time_limit);
256 return r;
257 }
258 LpProblem q = p;
259 q.is_integer.clear();
260 SolverOptions so;
261 so.time_limit = o.time_limit;
262 so.verbose = o.verbose > 1;
263 so.set_tolerance(o.tol);
264 if (meth == "hpr") {
265 hpr(o.time_limit);
266 return r;
267 }
268 if (meth == "auto" && q.A.nnz() > 5000000 && gpu_wanted(o.device)) {
269 hpr(o.time_limit);
270 return r;
271 }
272 so.method = meth == "ipm" ? LpMethod::Ipm
273 : meth == "pdlp" ? LpMethod::Pdlp
274 : LpMethod::Simplex;
275 if (meth != "auto" && meth != "simplex" && meth != "ipm" && meth != "pdlp")
276 throw std::invalid_argument(
277 "unknown LP method '" + meth + "' (auto, simplex, ipm, pdlp, hpr)");
278 const auto t0 = std::chrono::steady_clock::now();
279 LpSolution s = solve_lp(q, so);
280 r.status = s.status;
281 r.iterations = s.iterations;
282 r.method = meth == "auto" ? "simplex" : meth;
283 r.device = "cpu";
284 if (!s.x.empty() && s.status == Status::Optimal) {
285 r.x = s.x;
286 r.y = s.y;
287 r.objective = sense * s.primal_objective;
288 r.bound = sense * s.dual_objective;
289 }
290 if (meth == "auto" && s.status != Status::Optimal &&
291 s.status != Status::Infeasible && s.status != Status::Unbounded) {
292 const double used =
293 std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
294 .count();
295 if (o.time_limit - used > 1) {
296 Result keep = r;
297 hpr(o.time_limit - used);
298 if (!r.has_solution())
299 r = keep;
300 else
301 r.method = "simplex+hpr";
302 }
303 }
304 return r;
305}

307inline Result
308solve_qp_model(const Model &m, const Options &o)
309{
310 Result r;
311 const double sense = m.qp.lp.maximize ? -1.0 : 1.0;
312 QpProblem q = m.qp;
313 q.lp.is_integer.clear();
314 QpOptions qo;
315 const std::string &meth = o.method;
316 if (meth == "auto")
317 qo.method = QpMethod::Auto;
318 else if (meth == "hprqp" || meth == "hpr")
319 qo.method = QpMethod::HprQp;
320 else if (meth == "pdhcg")
321 qo.method = QpMethod::Pdhcg;
322 else if (meth == "ipm")
323 qo.method = QpMethod::Ipm;
324 else
325 throw std::invalid_argument(
326 "unknown QP method '" + meth +
327 "' (auto, hprqp, pdhcg, ipm; hpr is a synonym for hprqp)");
328 qo.auto_device = o.device == Device::Auto;
329 qo.use_gpu =
330 o.device == Device::Gpu || (qo.auto_device && qp_gpu_available());
331 qo.tol = o.tol;
332 qo.time_limit = o.time_limit;
333 qo.verbose = o.verbose;
334 qo.comm = o.comm;
335 if (o.comm && o.comm->distributed() && qo.method != QpMethod::Auto &&
336 qo.method != QpMethod::HprQp)
337 throw std::invalid_argument(
338 "over MPI ranks a QP is solved by HPR-QP (method auto or hprqp)");
339 from_qp(solve_qp(q, qo), sense, r);
340 return r;
341}

343inline Result
344solve_milp_model(const Model &m, const Options &o)
345{
346 Result r;
347 if (o.method != "auto" && o.method != "bnb")
348 throw std::invalid_argument(
349 "unknown MILP method '" + o.method + "' (auto, bnb)");
350 const LpProblem &p = m.qp.lp;
351 const double sense = p.maximize ? -1.0 : 1.0;
352 MilpOptions mo;
353 mo.time_limit = o.time_limit;
354 mo.gap_rel = o.gap;
355 mo.verbose = o.verbose;
356 mo.presolve = o.presolve;
357 mo.cuts = o.cuts;
358 mo.heuristics = o.heuristics;
359#if defined(AXOS_ENABLE_CUDA)
360 mo.gpu = o.device == Device::Gpu ||
361 (o.device == Device::Auto && qp_gpu_available());
362#endif
363 const MilpSolution s = solve_milp(p, mo);
364 r.status = s.status;
365 r.iterations = s.lp_iterations;
366 r.nodes = s.nodes;
367 r.method = "branch-and-cut";
368 r.device = mo.gpu ? "cpu+gpu" : "cpu";
369 r.message = s.notes;
370 if (s.has_solution()) {
371 r.x = s.x;
372 r.objective = sense * s.objective;
373 }
374 if (std::isfinite(s.bound)) r.bound = sense * s.bound;
375 return r;
376}

380// Solver failures come back as status NumericalError with a Result::message;
381// only invalid arguments throw.
382inline Result
383solve(const Model &m, const Options &o = Options())
384{
385 const auto t0 = std::chrono::steady_clock::now();
386 ProblemType t = o.type == ProblemType::Auto ? m.type() : o.type;
387 if (t == ProblemType::QP && !m.qp.has_quadratic()) t = ProblemType::LP;
388 if (t == ProblemType::LP && m.qp.has_quadratic())
389 t = ProblemType::QP; // an LP of a QP is not defined
390 if (t == ProblemType::MILP && m.qp.has_quadratic())
391 throw std::invalid_argument("mixed-integer QP is not supported");
392 const bool dist = o.comm && o.comm->distributed();
393 if (t == ProblemType::MILP && dist)
394 throw std::invalid_argument(
395 "a MILP runs on one process (start it without mpiexec)");
396 Result r;
397 try {
398 if (t == ProblemType::MILP)
399 r = api_detail::solve_milp_model(m, o);
400 else if (t == ProblemType::QP)
401 r = api_detail::solve_qp_model(m, o);
402 else
403 r = api_detail::solve_lp_model(m, o);
404 } catch (const std::invalid_argument &) {
405 throw;
406 } catch (const std::exception &e) {
407 if (dist)
408 throw; // the other ranks may be waiting in a collective: the caller
409 // aborts
410 r = Result();
411 r.status = Status::NumericalError;
412 r.message = e.what();
413 }
414 r.type = t;
415 r.seconds =
416 std::chrono::duration<double>(std::chrono::steady_clock::now() - t0)
417 .count();
418 if (r.has_solution() && r.x.size() == m.cols())
419 r.max_violation =
420 api_detail::max_violation(m.qp.lp, r.x, t == ProblemType::MILP);
421 return r;
422}

424// The solution as "name value" lines (zero values omitted), after a header.
425inline void
426write_solution(std::ostream &out, const Model &m, const Result &r)
427{
428 out << "# AXOS " << kAxosVersion << " solution of " << m.name() << "\n";
429 out << "# status " << to_string(r.status) << "\n";
430 out << std::setprecision(17);
431 if (!r.has_solution()) return;
432 out << "# objective " << r.objective << "\n";
433 const auto &names = m.qp.lp.col_names;
434 for (size_t j = 0; j < r.x.size(); ++j)
435 if (r.x[j] != 0)
436 out << (j < names.size() ? names[j] : "x" + std::to_string(j))
437 << " " << r.x[j] << "\n";
438}

449inline std::string
450json_str(const std::string &v)
451{
452 std::string o = "\"";
453 for (char c : v) {
454 if (c == '"' || c == '\\') o += '\\';
455 if (c == '\n') {
456 o += "\\n";
457 continue;

459 o += c;

465// The result as one JSON object (with the solution when with_x).
466inline std::string
467result_json(const Model &m, const Result &r, bool with_x = true)
468{
469 using api_detail::json_num;
470 using api_detail::json_str;
471 std::ostringstream s;
472 s << "{\"solver\":\"AXOS " << kAxosVersion
473 << "\",\"model\":" << json_str(m.name())
474 << ",\"type\":" << json_str(to_string(r.type)) << ",\"rows\":" << m.rows()
475 << ",\"cols\":" << m.cols() << ",\"nnz\":" << m.nnz()
476 << ",\"integers\":" << m.integers()
477 << ",\"status\":" << json_str(to_string(r.status))
478 << ",\"objective\":" << json_num(r.objective)
479 << ",\"bound\":" << json_num(r.bound) << ",\"gap\":" << json_num(r.gap())
480 << ",\"seconds\":" << json_num(r.seconds)
481 << ",\"iterations\":" << r.iterations << ",\"nodes\":" << r.nodes
482 << ",\"method\":" << json_str(r.method)
483 << ",\"device\":" << json_str(r.device)
484 << ",\"max_violation\":" << json_num(r.max_violation)
485 << ",\"message\":" << json_str(r.message);
486 if (with_x && r.has_solution()) {
487 const auto &names = m.qp.lp.col_names;
488 s << ",\"x\":{";
489 bool first = true;
490 for (size_t j = 0; j < r.x.size(); ++j) {
491 if (r.x[j] == 0) continue;
492 s << (first ? "" : ",")
493 << json_str(j < names.size() ? names[j] : "x" + std::to_string(j))
494 << ":" << json_num(r.x[j]);
495 first = false;
496 }
497 s << "}";
498 }
499 s << "}";
500 return s.str();
501}
