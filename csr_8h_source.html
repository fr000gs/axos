3// Csr<T, Idx, Store>: compressed sparse row matrix.
4//
5// Invariants (checked by validate()):
6// - 0-based indices; row_ptr has rows+1 entries, non-decreasing,
7// row_ptr[0] == 0 and row_ptr[rows] == nnz
8// - column indices are sorted (strictly increasing) within each row, so
9// there are no duplicate entries
10// - the sparsity pattern is fixed after construction; values may change
11//
12// The three arrays live in Store<Idx> / Store<T> (Cpu::HostStorage or
13// Cuda::CudaStorage), so the matrix reuses the GPU pool and the storage
14// semantics of tensorET. Kernels live in sparse_cpu.h / sparse_cuda.h and are
15// selected by backend (Sparse::Kernels<Backend>).

72template <typename Store>
73inline constexpr bool is_host_store_v =
74 std::is_same_v<typename Store::backend_type, Cpu::Backend>;

76template <typename U, template <typename> class S>
77void
78host_to_store(S<U> &dst, const U *src, size_t n)
79{
80 if (n == 0) return;
81 if constexpr (is_host_store_v<S<U>>) {
82 std::memcpy(dst.data(), src, n * sizeof(U));
83 } else {
84#if defined(__CUDACC__) || defined(AXOS_ENABLE_CUDA)
85 cudaError_t e =
86 cudaMemcpy(dst.data(), src, n * sizeof(U), cudaMemcpyHostToDevice);
87 if (e != cudaSuccess)
88 throw std::runtime_error(
89 std::string("Csr upload failed: ") + cudaGetErrorString(e));
90#else
91 throw std::runtime_error("Csr: CUDA storage requires a CUDA build");
92#endif
93 }
94}

96template <typename U, template <typename> class S>
97void
98store_to_host(U *dst, const S<U> &src, size_t n)
99{
100 if (n == 0) return;
101 if constexpr (is_host_store_v<S<U>>) {
102 std::memcpy(dst, src.data(), n * sizeof(U));
103 } else {
104#if defined(__CUDACC__) || defined(AXOS_ENABLE_CUDA)
105 cudaError_t e =
106 cudaMemcpy(dst, src.data(), n * sizeof(U), cudaMemcpyDeviceToHost);
107 if (e != cudaSuccess)
108 throw std::runtime_error(
109 std::string("Csr download failed: ") + cudaGetErrorString(e));
110#else
111 throw std::runtime_error("Csr: CUDA storage requires a CUDA build");
112#endif
113 }
114}

116template <typename U, template <typename> class D, template <typename> class S>
117void
118store_to_store(D<U> &dst, const S<U> &src, size_t n)
119{
120 if (n == 0) return;
121 constexpr bool dh = is_host_store_v<D<U>>, sh = is_host_store_v<S<U>>;
122 if constexpr (dh && sh) {
123 std::memcpy(dst.data(), src.data(), n * sizeof(U));
124 } else {
125#if defined(__CUDACC__) || defined(AXOS_ENABLE_CUDA)
126 cudaMemcpyKind kind = dh ? cudaMemcpyDeviceToHost
127 : sh ? cudaMemcpyHostToDevice
128 : cudaMemcpyDeviceToDevice;
129 cudaError_t e = cudaMemcpy(dst.data(), src.data(), n * sizeof(U), kind);
130 if (e != cudaSuccess)
131 throw std::runtime_error(
132 std::string("Csr transfer failed: ") + cudaGetErrorString(e));
133#else
134 throw std::runtime_error("Csr: CUDA storage requires a CUDA build");
135#endif
136 }
137}

143class Csr {
144 static_assert(std::is_same_v<Idx, int32_t> || std::is_same_v<Idx, int64_t>,
145 "Csr index type must be int32_t or int64_t");

147 "Csr value type must be float, double, or std::complex of those");
148
149 public:
150 using value_type = T;
151 using index_type = Idx;
152 using backend_type = typename Store<T>::backend_type;

154 static constexpr bool on_host = detail::is_host_store_v<Store<T>>;

158 // Allocates arrays (uninitialized) for a rows x cols matrix with nnz
159 // stored entries. row_ptr is zero-filled only when nnz == 0.
160 Csr(size_t rows, size_t cols, size_t nnz)
161 : rows_(rows), cols_(cols), nnz_(nnz)
162 {
163 row_ptr_ = Store<Idx>(rows + 1);
164 col_ind_ = Store<Idx>(nnz);
165 vals_ = Store<T>(nnz);
166 if (nnz == 0) {
167 std::vector<Idx> z(rows + 1, 0);
168 detail::host_to_store(row_ptr_, z.data(), rows + 1);
169 }
170 }
171
172 // Upload from host arrays (row_ptr: rows+1, col_ind/vals: nnz).
173 Csr(size_t rows, size_t cols, const std::vector<Idx> &row_ptr,
174 const std::vector<Idx> &col_ind, const std::vector<T> &vals)
175 : Csr(rows, cols, vals.size())
176 {
177 if (row_ptr.size() != rows + 1 || col_ind.size() != vals.size())
178 throw std::invalid_argument("Csr: inconsistent array sizes");
179 detail::host_to_store(row_ptr_, row_ptr.data(), rows + 1);
180 detail::host_to_store(col_ind_, col_ind.data(), nnz_);
181 detail::host_to_store(vals_, vals.data(), nnz_);
182 }

314 // Structural / ordering invariants. Returns an empty string when valid.
315 std::string
316 validate() const
317 {
318 if (row_ptr_.data() == nullptr)
319 return (rows_ == 0 && nnz_ == 0) ? "" : "row_ptr is null";
320 std::vector<Idx> rp(rows_ + 1), ci(nnz_);
321 detail::store_to_host(rp.data(), row_ptr_, rows_ + 1);
322 detail::store_to_host(ci.data(), col_ind_, nnz_);
323 std::ostringstream err;
324 if (rp[0] != 0) err << "row_ptr[0] != 0; ";
325 if (static_cast<size_t>(rp[rows_]) != nnz_)
326 err << "row_ptr[rows] != nnz; ";
327 for (size_t i = 0; i < rows_; ++i) {
328 if (rp[i + 1] < rp[i]) {
329 err << "row_ptr decreases at row " << i << "; ";
330 break;
331 }
332 }
333 if (!err.str().empty()) return err.str();
334 for (size_t i = 0; i < rows_; ++i) {
335 for (Idx k = rp[i]; k < rp[i + 1]; ++k) {
336 if (ci[k] < 0 || static_cast<size_t>(ci[k]) >= cols_) {
337 err << "column index out of range in row " << i;
338 return err.str();
339 }
340 if (k > rp[i] && ci[k] <= ci[k - 1]) {
341 err << "columns not strictly increasing in row " << i;
342 return err.str();
343 }
344 }
345 }
346 return "";
347 }

349 // Dense copy, in the requested tensorET storage (host by default).
350 template <typename S = Cpu::HostStorage<T>>
351 tensorET<2, T, S>
352 to_dense() const
353 {
354 Csr<T, Idx, Cpu::HostStorage> h(*this);
355 tensorET<2, T> D({rows_, cols_}, T(0));
356 for (size_t i = 0; i < rows_; ++i)
357 for (Idx k = h.row_ptr()[i]; k < h.row_ptr()[i + 1]; ++k)
358 D.data[i * cols_ + h.col_ind()[k]] = h.values()[k];
359 if constexpr (std::is_same_v<S, Cpu::HostStorage<T>>) {
360 return D;
361 } else {
362 return tensorET<2, T, S>(D);
363 }
364 }

372 // Stack B's rows below this matrix's rows (same column count). Used to
373 // add cuts / constraints. Returns a new matrix (rebuilds the pattern).
374 Csr
375 append_rows(const Csr &B) const
376 {
377 if (B.cols_ != cols_)
378 throw std::invalid_argument("append_rows: column count mismatch");
379 Csr<T, Idx, Cpu::HostStorage> a(*this), b(B);
380 std::vector<Idx> rp(a.rows() + b.rows() + 1);
381 std::vector<Idx> ci;
382 std::vector<T> v;
383 ci.reserve(a.nnz() + b.nnz());
384 v.reserve(a.nnz() + b.nnz());
385 std::copy(a.row_ptr(), a.row_ptr() + a.rows() + 1, rp.begin());
386 ci.insert(ci.end(), a.col_ind(), a.col_ind() + a.nnz());
387 v.insert(v.end(), a.values(), a.values() + a.nnz());
388 for (size_t i = 0; i < b.rows(); ++i)
389 rp[a.rows() + 1 + i] =
390 static_cast<Idx>(a.nnz()) + b.row_ptr()[i + 1];
391 ci.insert(ci.end(), b.col_ind(), b.col_ind() + b.nnz());
392 v.insert(v.end(), b.values(), b.values() + b.nnz());
393 return Csr(a.rows() + b.rows(), cols_, rp, ci, v);
394 }

418// Triplet (COO) builder. Sorts entries, sums duplicates, optionally drops
419// zeros, and produces a host CSR satisfying the Csr invariants.
420template <typename T, typename Idx = int32_t> class CooBuilder {
421 public:
422 CooBuilder(size_t rows, size_t cols) : rows_(rows), cols_(cols) {}
423
424 void
425 reserve(size_t n)
426 {
427 i_.reserve(n);
428 j_.reserve(n);
429 v_.reserve(n);
430 }
431
432 void
433 add(size_t i, size_t j, T v)
434 {
435 if (i >= rows_ || j >= cols_)
436 throw std::out_of_range("CooBuilder::add: index out of range");
437 i_.push_back(static_cast<Idx>(i));
438 j_.push_back(static_cast<Idx>(j));
439 v_.push_back(v);
440 }
441
442 size_t
443 size() const
444 { return v_.size(); }
445
446 Csr<T, Idx, Cpu::HostStorage>
447 build(bool drop_zeros = false) const
448 {
449 const size_t n = v_.size();
450 // Counting sort by row, then sort each row by column.
451 std::vector<size_t> start(rows_ + 1, 0);
452 for (size_t k = 0; k < n; ++k)
453 start[i_[k] + 1]++;
454 for (size_t r = 0; r < rows_; ++r)
455 start[r + 1] += start[r];
456 std::vector<size_t> perm(n), fill(start.begin(), start.end() - 1);
457 for (size_t k = 0; k < n; ++k)
458 perm[fill[i_[k]]++] = k;
459
460 std::vector<Idx> rp(rows_ + 1, 0), ci;
461 std::vector<T> vals;
462 ci.reserve(n);
463 vals.reserve(n);
464 for (size_t r = 0; r < rows_; ++r) {
465 auto first = perm.begin() + start[r];
466 auto last = perm.begin() + start[r + 1];
467 std::stable_sort(
468 first, last, [&](size_t a, size_t b) { return j_[a] < j_[b]; });
469 size_t row_begin = ci.size();
470 for (auto it = first; it != last; ++it) {
471 if (ci.size() > row_begin && ci.back() == j_[*it]) {
472 vals.back() += v_[*it];
473 } else {
474 ci.push_back(j_[*it]);
475 vals.push_back(v_[*it]);
476 }
477 }
478 if (drop_zeros) {
479 size_t w = row_begin;
480 for (size_t k = row_begin; k < ci.size(); ++k) {
481 if (vals[k] != T(0)) {
482 ci[w] = ci[k];
483 vals[w] = vals[k];
484 ++w;
485 }
486 }
487 ci.resize(w);
488 vals.resize(w);
489 }
490 rp[r + 1] = static_cast<Idx>(ci.size());
491 }
492 return Csr<T, Idx, Cpu::HostStorage>(rows_, cols_, rp, ci, vals);
493 }
494
495 private:
496 size_t rows_, cols_;
497 std::vector<Idx> i_, j_;
498 std::vector<T> v_;
499};

501template <typename T, typename Idx, template <typename> class Store>
502template <typename S>
503Csr<T, Idx, Cpu::HostStorage>
504Csr<T, Idx, Store>::from_dense(
505 const tensorET<2, T, S> &A, real_of_t<T> drop_tol)
506{
507 tensorET<2, T> H(A); // host copy (no-op copy for host storage)
508 const size_t m = H.size(0), n = H.size(1);
509 CooBuilder<T, Idx> b(m, n);
510 for (size_t i = 0; i < m; ++i)
511 for (size_t j = 0; j < n; ++j) {
512 T v = H.data[i * n + j];
513 if (std::abs(v) > drop_tol) b.add(i, j, v);
514 }
515 return b.build();
516}
