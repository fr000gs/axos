3// GPU side of the MILP solver (host code; kernels in milp_kernels.cuh,
4// compiled at run time by NVRTC like the QP kernels, see qp_backend_cuda.h).
5//
6// DeviceProblem the constraint matrix as CSR and CSC, row sides and
7// integrality on the device
8// GpuPropagator batched domain propagation: one domain, or B probes /
9// children at once; double probing at the root
10// (probe_binaries: every binary at 0 and at 1, fixings
11// from infeasible probes, bounds both probes imply)
12// GpuFeasibilityJump feasibility jump with W walkers (thread blocks) from
13// different start points and seeds
14// GpuBatchLp Lagrangian bounds of up to 32 LP relaxations that
15// differ in column bounds, by batched PDHG from a warm
16// start (strong branching on the GPU)
17//
18// Everything runs on the legacy default stream: a download waits for the
19// kernels before it.

85 "attribute");

88 device_ = prop.name;
89 const auto t0 = std::chrono::steady_clock::now();
90 const qp::cuda_rt::Nvrtc &nv = qp::cuda_rt::Nvrtc::get();
91 const std::string src = "#include \"solver/milp/milp_kernels.cuh\"\n";

96 opt_arch.c_str(), "-std=c++17", opt_inc.c_str(), "--fmad=true"};

126 std::mutex mu_;
127 std::unordered_map<std::string, CUfunction> fns_;

147// A device array (grows, never shrinks).
148template <typename T> class DBuf {
149 public:
150 DBuf() = default;
151 explicit DBuf(size_t n) { resize(n); }
152 ~DBuf()
153 {
154 if (p_) cudaFree(p_);
155 }
156 DBuf(const DBuf &) = delete;
157 DBuf &operator=(const DBuf &) = delete;
158
159 void
160 resize(size_t n)
161 {
162 if (n > cap_ || !p_) {
163 if (p_) cudaFree(p_);
164 p_ = nullptr;
165 check(cudaMalloc(reinterpret_cast<void **>(&p_),
166 std::max<size_t>(n, 1) * sizeof(T)),
167 "cudaMalloc");
168 cap_ = std::max<size_t>(n, 1);
169 }
170 n_ = n;
171 }
172 void
173 upload(const T *h, size_t n)
174 {
175 resize(n);
176 if (n)
177 check(cudaMemcpy(p_, h, n * sizeof(T), cudaMemcpyHostToDevice),
178 "upload");
179 }
180 void
181 upload(const std::vector<T> &h)
182 { upload(h.data(), h.size()); }
183 void
184 download(T *h, size_t n, size_t offset = 0) const
185 {
186 if (n)
187 check(cudaMemcpy(
188 h, p_ + offset, n * sizeof(T), cudaMemcpyDeviceToHost),
189 "download");
190 }
191 void
192 download(std::vector<T> &h) const
193 {
194 h.resize(n_);
195 download(h.data(), n_);
196 }
197 void
198 zero()
199 {
200 if (n_) check(cudaMemset(p_, 0, n_ * sizeof(T)), "cudaMemset");
201 }
202 T *
203 get() const
204 { return p_; }
205 size_t
206 size() const
207 { return n_; }
208
209 private:
210 T *p_ = nullptr;
211 size_t n_ = 0, cap_ = 0;
212};

214// The constraint matrix (CSR and CSC), row sides and integrality.
215struct DeviceProblem {
216 int n = 0, m = 0;
217 DBuf<int32_t> rp, ci, cp, ri;
218 DBuf<double> va, cv, rlb, rub;
219 DBuf<unsigned char> isint;
220
221 explicit DeviceProblem(const LpProblem &p)
222 {
223 n = static_cast<int>(p.cols());
224 m = static_cast<int>(p.rows());
225 const size_t nnz = p.A.nnz();
226 rp.upload(p.A.row_ptr(), m + 1);
227 ci.upload(p.A.col_ind(), nnz);
228 va.upload(p.A.values(), nnz);
229 const HostMatrix At = p.A.transpose();
230 cp.upload(At.row_ptr(), n + 1);
231 ri.upload(At.col_ind(), nnz);
232 cv.upload(At.values(), nnz);
233 rlb.upload(p.row_lb);
234 rub.upload(p.row_ub);
235 std::vector<unsigned char> ii(n);
236 for (int j = 0; j < n; ++j)
237 ii[j] = is_int(p, j) ? 1 : 0;
238 isint.upload(ii);
239 }
240};

246 public:
247 explicit GpuPropagator(const DeviceProblem &d) : d_(d) {}

249 // Propagates B = fix_col.size() domains: the base bounds with column
250 // fix_col[b] restricted to [fix_lo[b], fix_hi[b]] (fix_col[b] < 0: the base
251 // domain). infeasible[b] reports an empty domain; the propagated bounds (B
252 // x n, batch-major) are downloaded when lb_out is given. Returns the rounds
253 // run.
254 int
255 run(const std::vector<double> &lb0, const std::vector<double> &ub0,
256 const std::vector<int> &fix_col, const std::vector<double> &fix_lo,
257 const std::vector<double> &fix_hi, int max_rounds,
258 std::vector<int> &infeasible, std::vector<double> *lb_out = nullptr,
259 std::vector<double> *ub_out = nullptr)
260 {
261 int n = d_.n, m = d_.m, B = static_cast<int>(fix_col.size());
262 lb0_.upload(lb0);
263 ub0_.upload(ub0);
264 fcol_.upload(fix_col);
265 flo_.upload(fix_lo);
266 fhi_.upload(fix_hi);
267 lb_.resize(size_t(B) * n);
268 ub_.resize(size_t(B) * n);
269 minact_.resize(size_t(B) * m);
270 maxact_.resize(size_t(B) * m);
271 ninfmin_.resize(size_t(B) * m);
272 ninfmax_.resize(size_t(B) * m);
273 active_.resize(B);
274 changed_.resize(B);
275 infeas_.resize(B);
276 {
277 int32_t *fc = fcol_.get();
278 double *a0 = lb0_.get(), *a1 = ub0_.get(), *a2 = flo_.get(),
279 *a3 = fhi_.get(), *l = lb_.get(), *u = ub_.get();
280 int *ac = active_.get(), *ch = changed_.get(), *inf = infeas_.get();
281 void *args[] = {
282 &n, &B, &a0, &a1, &fc, &a2, &a3, &l, &u, &ac, &ch, &inf};
283 launch("prop_init", blocks_for((long long)n * B), kBlock, args);
284 }
285 std::vector<int> act(B);
286 int rounds = 0;
287 for (; rounds < max_rounds; ++rounds) {
288 const int32_t *rp = d_.rp.get(), *ci = d_.ci.get(),
289 *cp = d_.cp.get(), *ri = d_.ri.get();
290 const double *va = d_.va.get(), *cv = d_.cv.get(),
291 *rl = d_.rlb.get(), *ru = d_.rub.get();
292 const unsigned char *ii = d_.isint.get();
293 double *l = lb_.get(), *u = ub_.get(), *mn = minact_.get(),
294 *mx = maxact_.get();
295 int *c0 = ninfmin_.get(), *c1 = ninfmax_.get(), *ac = active_.get(),
296 *ch = changed_.get(), *inf = infeas_.get();
297 {
298 void *args[] = {&m, &n, &B, &rp, &ci, &va, &rl, &ru, &l, &u,
299 &mn, &mx, &c0, &c1, &ac, &inf};
300 launch("prop_activity", blocks_for((long long)m * B * 32),
301 kBlock, args);
302 }
303 {
304 void *args[] = {&n, &m, &B, &cp, &ri, &cv, &rl, &ru, &mn, &mx,
305 &c0, &c1, &l, &u, &ii, &ac, &ch, &inf};
306 launch(
307 "prop_tighten", blocks_for((long long)n * B), kBlock, args);
308 }
309 {
310 void *args[] = {&B, &ac, &ch, &inf};
311 launch("prop_next", blocks_for(B), kBlock, args);
312 }
313 active_.download(act.data(), B);
314 bool any = false;
315 for (int v : act)
316 any = any || v;
317 if (!any) {
318 ++rounds;
319 break;
320 }
321 }
322 infeasible.resize(B);
323 infeas_.download(infeasible.data(), B);
324 if (lb_out) lb_.download(*lb_out);
325 if (ub_out) ub_.download(*ub_out);
326 return rounds;
327 }

329 // One domain: tightens lb/ub in place; false if proven empty.
330 bool
331 propagate(
332 std::vector<double> &lb, std::vector<double> &ub, int max_rounds = 1000)
333 {
334 std::vector<int> inf;
335 std::vector<double> l, u;
336 run(lb, ub, {-1}, {0.0}, {0.0}, max_rounds, inf, &l, &u);
337 if (inf[0]) return false;
338 lb = l;
339 ub = u;
340 return true;
341 }

396// Double probing of the binary columns on the GPU: `slots` probes at once
397// (each binary at 0 and at 1 in neighbouring slots), propagated with row /
398// column frontiers (milp_kernels.cuh, pb_*), so a probe costs what its
399// propagation touches. A slot that empties its domain fixes the binary the
400// other way; otherwise each column gets the weaker of the two probes' bounds.
401// All probes of a pass start from the same base domain; the merged bounds come
402// back once at the end.
403class GpuProber {
404 public:
405 GpuProber(const DeviceProblem &d, int slots = 256) : d_(d)
406 {
407 const long long words_n = (long long)d.n * slots,
408 words_m = (long long)d.m * slots;
409 // keep the slot arrays within about 1.5 GB of device memory
410 const double per_slot = 16.0 * d.n + 24.0 * d.m +
411 8.0 * (d.n + 2.0 * d.m) + 4.0 * (d.n + d.m);
412 B_ = std::max(2,
413 std::min(slots, static_cast<int>(1.5e9 / std::max(per_slot, 1.0))) &
414 ~1);
415 (void)words_n;
416 (void)words_m;
417 }
418
419 // Probes the binaries of lb/ub (base domain) until the deadline; on
420 // return lb/ub hold the merged bounds. Infeasible: some binary empties
421 // the domain both ways.
422 ProbeResult
423 run(const LpProblem &p, std::vector<double> &lb, std::vector<double> &ub,
424 double seconds)
425 {
426 const auto t0 = std::chrono::steady_clock::now();
427 auto elapsed = [&] {
428 return std::chrono::duration<double>(
429 std::chrono::steady_clock::now() - t0)
430 .count();
431 };
432 ProbeResult res;
433 int n = d_.n, m = d_.m, B = B_;
434 const size_t N = size_t(B) * n, M = size_t(B) * m;
435 std::vector<int> bins;
436 for (int j = 0; j < n; ++j)
437 if (is_int(p, j) && lb[j] == 0 && ub[j] == 1) bins.push_back(j);
438 if (bins.empty()) return res;
439 // base domain and its activities
440 blb_.upload(lb);
441 bub_.upload(ub);
442 bmin_.resize(m);
443 bmax_.resize(m);
444 bcmin_.resize(m);
445 bcmax_.resize(m);
446 {
447 int one = 1;
448 DBuf<int> act(1), inf(1);
449 const int on = 1, zero = 0;
450 act.upload(&on, 1);
451 inf.upload(&zero, 1);
452 const int32_t *rp = d_.rp.get(), *ci = d_.ci.get();
453 const double *va = d_.va.get(), *rl = d_.rlb.get(),
454 *ru = d_.rub.get();
455 double *l = blb_.get(), *u = bub_.get(), *mn = bmin_.get(),
456 *mx = bmax_.get();
457 int *c0 = bcmin_.get(), *c1 = bcmax_.get(), *ac = act.get(),
458 *in = inf.get();
459 void *args[] = {&m, &n, &one, &rp, &ci, &va, &rl, &ru, &l, &u, &mn,
460 &mx, &c0, &c1, &ac, &in};
461 launch(
462 "prop_activity", blocks_for((long long)m * 32), kBlock, args);
463 int bad = 0;
464 inf.download(&bad, 1);
465 if (bad) {
466 res.infeasible = true;
467 return res;
468 }
469 }
470 lb_.resize(N);
471 ub_.resize(N);
472 mn_.resize(M);
473 mx_.resize(M);
474 cmn_.resize(M);
475 cmx_.resize(M);
476 rbits_.resize(M / 32 + 1);
477 rbits_.zero();
478 trbits_.resize(M / 32 + 1);
479 trbits_.zero();
480 cbits_.resize(N / 32 + 1);
481 cbits_.zero();
482 tcbits_.resize(N / 32 + 1);
483 tcbits_.zero();
484 rfa_.resize(M);
485 rfb_.resize(M);
486 cf_.resize(N);
487 tc_.resize(N);
488 tr_.resize(M);
489 cnt_.resize(8);
490 cnt_.zero();
491 infeas_.resize(B);
492 infeas_.zero();
493 olb_.resize(n);
494 oub_.resize(n);
495 check(cudaMemcpy(olb_.get(), blb_.get(), n * sizeof(double),
496 cudaMemcpyDeviceToDevice),
497 "copy");
498 check(cudaMemcpy(oub_.get(), bub_.get(), n * sizeof(double),
499 cudaMemcpyDeviceToDevice),
500 "copy");
501 ProbeData pd{n, m, B, d_.rp.get(), d_.ci.get(), d_.va.get(),
502 d_.cp.get(), d_.ri.get(), d_.cv.get(), d_.rlb.get(), d_.rub.get(),
503 d_.isint.get(), blb_.get(), bub_.get(), bmin_.get(), bmax_.get(),
504 bcmin_.get(), bcmax_.get(), lb_.get(), ub_.get(), mn_.get(),
505 mx_.get(), cmn_.get(), cmx_.get(), rbits_.get(), cbits_.get(),
506 trbits_.get(), tcbits_.get(), cf_.get(), tc_.get(), tr_.get(),
507 cnt_.get(), infeas_.get(), olb_.get(), oub_.get()};
508 {
509 void *args[] = {&pd};
510 launch(
511 "pb_fill", blocks_for((long long)std::max(N, M)), kBlock, args);
512 }
513 std::vector<int> pcol(B);
514 std::vector<double> plo(B), phi(B);
515 std::vector<int> inf(B);
516 unsigned cnt[8];
517 const int pairs = B / 2;
518 for (size_t s = 0; s < bins.size() && elapsed() < seconds; s += pairs) {
519 const int K =
520 static_cast<int>(std::min<size_t>(pairs, bins.size() - s));
521 int Bk = 2 * K;
522 for (int k = 0; k < K; ++k) {
523 pcol[2 * k] = pcol[2 * k + 1] = bins[s + k];
524 plo[2 * k] = 0;
525 phi[2 * k] = 0;
526 plo[2 * k + 1] = 1;
527 phi[2 * k + 1] = 1;
528 }
529 pcol_.upload(pcol.data(), Bk);
530 plo_.upload(plo.data(), Bk);
531 phi_.upload(phi.data(), Bk);
532 pd.B = Bk; // slots beyond 2K stay at the base domain
533 unsigned *rf = rfa_.get(), *rf_next = rfb_.get();
534 {
535 int32_t *pc = pcol_.get();
536 double *lo = plo_.get(), *hi = phi_.get();
537 void *args[] = {&pd, &pc, &lo, &hi, &rf};
538 launch("pb_start", blocks_for(Bk), kBlock, args);
539 }
540 bool capped = false;
541 for (int round = 0;; ++round) {
542 cnt_.download(cnt, 5);
543 unsigned nrf = cnt[1];
544 if (nrf == 0) break;
545 if (round >= 40) {
546 capped = true;
547 break;
548 }
549 check(cudaMemset(cnt_.get() + 1, 0, 2 * sizeof(unsigned)),
550 "cudaMemset");
551 {
552 void *args[] = {&pd, &rf, &nrf};
553 launch("pb_rows", blocks_for((long long)nrf * 32), kBlock,
554 args);
555 }
556 cnt_.download(cnt, 5);
557 unsigned ncf = cnt[2];
558 if (ncf == 0) break;
559 {
560 void *args[] = {&pd, &ncf, &rf_next};
561 launch("pb_cols", blocks_for(ncf), kBlock, args);
562 }
563 std::swap(rf, rf_next);
564 }
565 res.rounds += 0;
566 cnt_.download(cnt, 5);
567 unsigned ntc = cnt[3], ntr = cnt[4];
568 infeas_.download(inf.data(), Bk);
569 res.probes += Bk;
570 bool both = false;
571 for (int k = 0; k < K; ++k) {
572 if (inf[2 * k] && inf[2 * k + 1])
573 both = true;
574 else if (inf[2 * k] || inf[2 * k + 1])
575 ++res.fixed;
576 }
577 if (both) {
578 res.infeasible = true;
579 break;
580 }
581 if (ntc) {
582 void *args[] = {&pd, &ntc};
583 launch("pb_merge", blocks_for(ntc), kBlock, args);
584 }
585 {
586 const unsigned nr = std::max(ntc, ntr);
587 void *args[] = {&pd, &ntc, &ntr};
588 if (nr) launch("pb_reset", blocks_for(nr), kBlock, args);
589 }
590 cnt_.zero();
591 infeas_.zero();
592 if (capped) {
593 rbits_.zero();
594 cbits_.zero();
595 }
596 }
597 if (!res.infeasible) {
598 std::vector<double> ol, ou;
599 olb_.download(ol);
600 oub_.download(ou);
601 for (int j = 0; j < n; ++j) {
602 if (ol[j] > lb[j] + 1e-9) {
603 lb[j] = ol[j];
604 ++res.tightened;
605 }
606 if (ou[j] < ub[j] - 1e-9) {
607 ub[j] = ou[j];
608 ++res.tightened;
609 }
610 if (lb[j] > ub[j] + 1e-9) res.infeasible = true;
611 }
612 }
613 res.seconds = elapsed();
614 return res;
615 }

623 int B_ = 256;
624 DBuf<double> blb_, bub_, bmin_, bmax_, lb_, ub_, mn_, mx_, olb_, oub_, plo_,
625 phi_;
626 DBuf<int> bcmin_, bcmax_, cmn_, cmx_, infeas_;
627 DBuf<int32_t> pcol_;
628 DBuf<unsigned> rbits_, cbits_, trbits_, tcbits_, rfa_, rfb_, cf_, tc_, tr_,
629 cnt_;
630};

668 // W walkers from the start points (cycled, with random perturbation beyond
669 // the first round) within lb/ub. True with x when one walker reached a
670 // point satisfying every row. Gives up at the time limit or when *stop is
671 // set (another thread).
672 bool
673 run(const std::vector<double> &lb, const std::vector<double> &ub,
674 const std::vector<std::vector<double>> &starts,
675 const std::vector<uint8_t> &isint, double time_limit, uint64_t seed,
676 std::vector<double> &x, long long *moves_out = nullptr,
677 const std::atomic<bool> *stop = nullptr)
678 {
679 const auto t0 = std::chrono::steady_clock::now();
680 auto elapsed = [&] {
681 return std::chrono::duration<double>(
682 std::chrono::steady_clock::now() - t0)
683 .count();
684 };
685 const int n = d_.n, m = d_.m, W = W_;
686 std::mt19937_64 rng(seed);
687 std::vector<double> xs(size_t(W) * n);
688 for (int w = 0; w < W; ++w) {
689 const std::vector<double> *s0 =
690 starts.empty() ? nullptr : &starts[w % starts.size()];
691 const bool perturb =
692 !starts.empty() && w >= static_cast<int>(starts.size());
693 for (int j = 0; j < n; ++j) {
694 double v = s0 ? (*s0)[j] : 0.0;
695 if (perturb && isint[j]) {
696 const double f = v - std::floor(v);
697 v = (std::uniform_real_distribution<double>(0, 1)(rng) < f)
698 ? std::ceil(v)
699 : std::floor(v);
700 }
701 v = std::min(std::max(v, lb[j]), ub[j]);
702 if (!std::isfinite(v) || std::abs(v) > 1e15)
703 v = std::abs(lb[j]) < 1e15
704 ? lb[j]
705 : (std::abs(ub[j]) < 1e15 ? ub[j] : 0.0);
706 if (isint[j])
707 v = std::min(
708 std::max(std::round(v), std::ceil(lb[j] - 1e-9)),
709 std::floor(ub[j] + 1e-9));
710 xs[size_t(w) * n + j] = v;
711 }
712 }
713 lb_.upload(lb);
714 ub_.upload(ub);
715 x_.upload(xs);
716 act_.resize(size_t(W) * m);
717 wt_.resize(size_t(W) * m);
718 vl_.resize(size_t(W) * m);
719 vpos_.resize(size_t(W) * m);
720 vcnt_.resize(W);
721 status_.resize(W);
722 status_.zero();
723 moves_.resize(W);
724 moves_.zero();
725 std::vector<unsigned long long> rs(W);
726 for (int w = 0; w < W; ++w)
727 rs[w] = (rng() | 1ull);
728 rng_.upload(rs);
729 FjData fd{n, m, d_.rp.get(), d_.ci.get(), d_.va.get(), d_.cp.get(),
730 d_.ri.get(), d_.cv.get(), d_.rlb.get(), d_.rub.get(), lb_.get(),
731 ub_.get(), d_.isint.get(), x_.get(), act_.get(), wt_.get(),
732 vl_.get(), vpos_.get(), vcnt_.get(), status_.get(), rng_.get(),
733 moves_.get()};
734 {
735 void *args[] = {&fd};
736 launch("fj_init", W, 128, args);
737 }
738 std::vector<int> st(W);
739 int steps = 64;
740 bool found = false;
741 int wfound = -1;
742 while (elapsed() < time_limit && !(stop && stop->load())) {
743 void *args[] = {&fd, &steps};
744 launch("fj_run", W, 128, args);
745 status_.download(st.data(), W);
746 for (int w = 0; w < W && !found; ++w)
747 if (st[w] == 1) {
748 found = true;
749 wfound = w;
750 }
751 if (found) break;
752 steps = std::min(steps * 2, 4096);
753 }
754 if (moves_out) {
755 std::vector<long long> mv(W);
756 moves_.download(mv.data(), W);
757 *moves_out = 0;
758 for (auto v : mv)
759 *moves_out += v;
760 }
761 if (!found) return false;
762 x.resize(n);
763 x_.download(x.data(), n, size_t(wfound) * n);
764 return true;
765 }

768 const DeviceProblem &d_;
769 int W_;
770 DBuf<double> lb_, ub_, x_, act_, wt_;
771 DBuf<int> vl_, vpos_, vcnt_, status_;
772 DBuf<unsigned long long> rng_;
773 DBuf<long long> moves_;

779// Lagrangian bounds of up to 32 LPs min c^T x, rl <= A x <= ru, l^b <= x <= u^b
780// by PDHG on the Ruiz-scaled problem from a common warm start.
781class GpuBatchLp {
782 public:
783 static constexpr int kMaxBatch = 32;
784
785 explicit GpuBatchLp(const LpProblem &lp)
786 : n_(static_cast<int>(lp.cols())), m_(static_cast<int>(lp.rows()))
787 {
788 offset_ = lp.offset;
789 // Ruiz equilibration: rows and columns to unit infinity norm
790 dr_.assign(m_, 1.0);
791 dc_.assign(n_, 1.0);
792 const int32_t *rp = lp.A.row_ptr(), *ci = lp.A.col_ind();
793 const double *va = lp.A.values();
794 const size_t nnz = lp.A.nnz();
795 std::vector<double> v(va, va + nnz);
796 for (int it = 0; it < 10; ++it) {
797 std::vector<double> rmax(m_, 0.0), cmax(n_, 0.0);
798 for (int i = 0; i < m_; ++i)
799 for (int32_t k = rp[i]; k < rp[i + 1]; ++k) {
800 const double a = std::abs(v[k]);
801 rmax[i] = std::max(rmax[i], a);
802 cmax[ci[k]] = std::max(cmax[ci[k]], a);
803 }
804 for (int i = 0; i < m_; ++i)
805 if (rmax[i] > 0) dr_[i] /= std::sqrt(rmax[i]);
806 for (int j = 0; j < n_; ++j)
807 if (cmax[j] > 0) dc_[j] /= std::sqrt(cmax[j]);
808 for (int i = 0; i < m_; ++i)
809 for (int32_t k = rp[i]; k < rp[i + 1]; ++k)
810 v[k] = va[k] * dr_[i] * dc_[ci[k]];
811 }
812 HostMatrix As(m_, n_, std::vector<int32_t>(rp, rp + m_ + 1),
813 std::vector<int32_t>(ci, ci + nnz), v);
814 const HostMatrix At = As.transpose();
815 rp_.upload(rp, m_ + 1);
816 ci_.upload(ci, nnz);
817 va_.upload(v);
818 cp_.upload(At.row_ptr(), n_ + 1);
819 ri_.upload(At.col_ind(), nnz);
820 cv_.upload(At.values(), nnz);
821 std::vector<double> c(n_), rl(m_), ru(m_);
822 for (int j = 0; j < n_; ++j)
823 c[j] = lp.c[j] * dc_[j];
824 for (int i = 0; i < m_; ++i) {
825 rl[i] =
826 std::abs(lp.row_lb[i]) < 1e20 ? lp.row_lb[i] * dr_[i] : -kInf;
827 ru[i] =
828 std::abs(lp.row_ub[i]) < 1e20 ? lp.row_ub[i] * dr_[i] : kInf;
829 }
830 c_.upload(c);
831 rl_.upload(rl);
832 ru_.upload(ru);
833 // ||A_s||_2 by power iteration on A^T A
834 std::vector<double> x(n_, 1.0), y(m_);
835 double norm = 1;
836 for (int it = 0; it < 40; ++it) {
837 double s = 0;
838 for (double t : x)
839 s += t * t;
840 s = std::sqrt(s);
841 if (s == 0) break;
842 for (double &t : x)
843 t /= s;
844 for (int i = 0; i < m_; ++i) {
845 double a = 0;
846 for (int32_t k = rp[i]; k < rp[i + 1]; ++k)
847 a += v[k] * x[ci[k]];
848 y[i] = a;
849 }
850 std::fill(x.begin(), x.end(), 0.0);
851 for (int i = 0; i < m_; ++i)
852 for (int32_t k = rp[i]; k < rp[i + 1]; ++k)
853 x[ci[k]] += v[k] * y[i];
854 double q = 0;
855 for (double t : x)
856 q += t * t;
857 norm = std::sqrt(std::sqrt(q)); // ||A^T A x|| -> sigma_max^2
858 }
859 // primal weight ||c|| / ||finite row sides||
860 double cn = 0, bn = 0;
861 for (double t : c)
862 cn += t * t;
863 for (int i = 0; i < m_; ++i) {
864 const double t = std::isfinite(rl[i])
865 ? rl[i]
866 : (std::isfinite(ru[i]) ? ru[i] : 0.0);
867 bn += t * t;
868 }
869 omega_ = (cn > 0 && bn > 0) ? std::sqrt(cn) / std::sqrt(bn) : 1.0;
870 eta_ = 0.95 / std::max(norm, 1e-12);
871 }
872
873 // Bounds of the LPs whose column bounds are lb/ub with column col[b]
874 // restricted to [lo[b], hi[b]], after `iters` PDHG iterations from (x0, y0)
875 // (original scale), evaluated every `every` iterations; the best Lagrangian
876 // bound of each LP (with the objective offset), -inf when every evaluation
877 // needed an infinite bound.
878 std::vector<double>
879 child_bounds(const std::vector<double> &x0, const std::vector<double> &y0,
880 const std::vector<double> &lb, const std::vector<double> &ub,
881 const std::vector<int> &col, const std::vector<double> &lo,
882 const std::vector<double> &hi, int iters, int every = 10)
883 {
884 int B = static_cast<int>(col.size()), n = n_, m = m_;
885 if (B > kMaxBatch)
886 throw std::invalid_argument("GpuBatchLp: at most 32 LPs per batch");
887 std::vector<double> L(size_t(n) * B), U(size_t(n) * B),
888 X(size_t(n) * B), Y(size_t(m) * B);
889 for (int j = 0; j < n; ++j)
890 for (int b = 0; b < B; ++b) {
891 double l = lb[j], u = ub[j];
892 if (col[b] == j) {
893 l = std::max(l, lo[b]);
894 u = std::min(u, hi[b]);
895 }
896 const size_t o = size_t(j) * B + b;
897 L[o] = std::abs(l) < 1e20 ? l / dc_[j] : -kInf;
898 U[o] = std::abs(u) < 1e20 ? u / dc_[j] : kInf;
899 X[o] = std::min(std::max(x0[j] / dc_[j], L[o]), U[o]);
900 }
901 for (int i = 0; i < m; ++i)
902 for (int b = 0; b < B; ++b)
903 Y[size_t(i) * B + b] = y0[i] / dr_[i];
904 L_.upload(L);
905 U_.upload(U);
906 X_.upload(X);
907 Xb_.resize(X.size());
908 Y_.upload(Y);
909 acc_.resize(B);
910 acc_.zero();
911 bad_.resize(B);
912 bad_.zero();
913 std::vector<double> best(B, -kInf);
914 best_.upload(best);
915 double tau = eta_ / omega_, sigma = eta_ * omega_;
916 const int32_t *rp = rp_.get(), *ci = ci_.get(), *cp = cp_.get(),
917 *ri = ri_.get();
918 const double *va = va_.get(), *cv = cv_.get(), *c = c_.get(),
919 *rl = rl_.get(), *ru = ru_.get();
920 double *Lp = L_.get(), *Up = U_.get(), *x = X_.get(), *xb = Xb_.get(),
921 *y = Y_.get(), *acc = acc_.get(), *bst = best_.get();
922 int *bad = bad_.get();
923 for (int k = 0; k <= iters; ++k) {
924 int eval = (k % every == 0 || k == iters) ? 1 : 0;
925 {
926 void *args[] = {&n, &B, &cp, &ri, &cv, &c, &Lp, &Up, &y, &x,
927 &xb, &tau, &eval, &acc, &bad};
928 launch(
929 "pd_primal", blocks_for((long long)n * 32), kBlock, args);
930 }
931 {
932 void *args[] = {&m, &B, &rp, &ci, &va, &rl, &ru, &xb, &y,
933 &sigma, &eval, &acc};
934 launch("pd_dual", blocks_for((long long)m * 32), kBlock, args);
935 }
936 if (eval) {
937 void *args[] = {&B, &acc, &bad, &bst};
938 launch("pd_take", 1, 32, args);
939 }
940 }
941 best_.download(best);
942 for (double &v : best)
943 if (v > -kInf) v += offset_;
944 return best;
945 }
946
947 private:
948 int n_, m_;
949 double offset_ = 0, omega_ = 1, eta_ = 1;
950 std::vector<double> dr_, dc_;
951 DBuf<int32_t> rp_, ci_, cp_, ri_;
952 DBuf<double> va_, cv_, c_, rl_, ru_, L_, U_, X_, Xb_, Y_, acc_, best_;
953 DBuf<int> bad_;
954};
