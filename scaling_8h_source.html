3// Diagonal scaling for LPs: Ruiz equilibration followed by Pock-Chambolle.
4//
5// With row factors r and column factors s, the scaled problem is
6// A~ = diag(r) A diag(s), c~ = s .* c, x~ = x ./ s,
7// lb~ = lb ./ s, ub~ = ub ./ s, row bounds~ = r .* row bounds.
8// Duals map back as y = r .* y~ and z = z~ ./ s; objective values (and the
9// offset) are unchanged.

24// Computes scaling factors for A (does not modify A).
25inline Scaling
26compute_scaling(
27 const HostMatrix &A0, int ruiz_iterations, double pock_chambolle_alpha)
28{
29 const size_t m = A0.rows(), n = A0.cols();
30 Scaling sc;
31 sc.row.assign(m, 1.0);
32 sc.col.assign(n, 1.0);
33 if (m == 0 || n == 0) return sc;
34 HostMatrix A(A0);
35 std::vector<double> rn(m), cn(n);
36
37 for (int it = 0; it < ruiz_iterations; ++it) {
38 Sparse::row_norms(A, rn.data(), Sparse::Norm::Linf);
39 Sparse::col_norms(A, cn.data(), Sparse::Norm::Linf);
40 for (size_t i = 0; i < m; ++i)
41 rn[i] = rn[i] > 0 ? 1.0 / std::sqrt(rn[i]) : 1.0;
42 for (size_t j = 0; j < n; ++j)
43 cn[j] = cn[j] > 0 ? 1.0 / std::sqrt(cn[j]) : 1.0;
44 Sparse::scale_rows_cols(A, rn.data(), cn.data());
45 for (size_t i = 0; i < m; ++i)
46 sc.row[i] *= rn[i];
47 for (size_t j = 0; j < n; ++j)
48 sc.col[j] *= cn[j];
49 }
50
51 if (pock_chambolle_alpha >= 0) {
52 // rows: l_{2-alpha} norms, columns: l_alpha norms (both l1 for alpha=1)
53 const double alpha = pock_chambolle_alpha;
54 const double pr = 2.0 - alpha, pc = alpha;
55 std::fill(rn.begin(), rn.end(), 0.0);
56 std::fill(cn.begin(), cn.end(), 0.0);
57 const auto *rp = A.row_ptr();
58 const auto *ci = A.col_ind();
59 const double *v = A.values();
60 for (size_t i = 0; i < m; ++i)
61 for (int k = rp[i]; k < rp[i + 1]; ++k) {
62 double a = std::abs(v[k]);
63 rn[i] += std::pow(a, pr);
64 cn[ci[k]] += std::pow(a, pc);
65 }
66 for (size_t i = 0; i < m; ++i)
67 rn[i] =
68 rn[i] > 0 ? 1.0 / std::sqrt(std::pow(rn[i], 1.0 / pr)) : 1.0;
69 for (size_t j = 0; j < n; ++j)
70 cn[j] = cn[j] > 0 ? 1.0 / std::sqrt(std::pow(
71 cn[j], 1.0 / std::max(pc, 1e-12)))
72 : 1.0;
73 for (size_t i = 0; i < m; ++i)
74 sc.row[i] *= rn[i];
75 for (size_t j = 0; j < n; ++j)
76 sc.col[j] *= cn[j];
77 }
78 return sc;
79}

81// Returns the scaled copy of `p`.
82inline LpProblem
83apply_scaling(const LpProblem &p, const Scaling &sc)
84{
85 LpProblem q = p;
86 HostMatrix A(p.A);
87 Sparse::scale_rows_cols(A, sc.row.data(), sc.col.data());
88 q.A = std::move(A);
89 for (size_t j = 0; j < p.cols(); ++j) {
90 q.c[j] = p.c[j] * sc.col[j];
91 q.col_lb[j] = p.col_lb[j] / sc.col[j];
92 q.col_ub[j] = p.col_ub[j] / sc.col[j];
93 }
94 for (size_t i = 0; i < p.rows(); ++i) {
95 q.row_lb[i] = p.row_lb[i] * sc.row[i];
96 q.row_ub[i] = p.row_ub[i] * sc.row[i];
97 }
98 return q;
99}

101// Maps a solution of the scaled problem back to the original variables.
102inline void
103unscale_solution(
104 std::vector<double> &x, std::vector<double> &y, const Scaling &sc)
105{
106 for (size_t j = 0; j < x.size(); ++j)
107 x[j] *= sc.col[j];
108 for (size_t i = 0; i < y.size(); ++i)
109 y[i] *= sc.row[i];
110}
