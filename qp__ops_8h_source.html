3// Per-element kernels of the first-order QP solvers. One definition serves
4// both backends: the CPU backend calls them inside OpenMP loops, the GPU
5// backend compiles this header at run time with NVRTC and wraps each
6// function in a grid-stride kernel (qp_kernels.cuh). Plain C++: no standard
7// headers (on the host, <cmath> for sqrt), no other library calls.
8//
9// map ops void f(idx i, const Args &a) writes outputs
10// reduce ops void f(idx i, const Args &a, double *acc) adds into acc[0..K)
11// (Args::K accumulators; bit k of Args::kMax set = max, else
12// sum; max-accumulators only see non-negative values, start at
13// 0)
14// epilogues void f(idx i, double s, const Args &a) s = (M v)_i, run
15// for every row i of a sparse product (backend spmv_epi)
16//
17// Notation (HPR-QP, Chen, Sun, Yuan, Zhang, Zhao 2025): state u = (y, w, x),
18// T(u) = (ybar, wbar, xbar); qw = Q w and aty = A^T y are carried along.

110// ---- HPR-QP iteration
111// --------------------------------------------------------- The scalars of a
112// block of iterations live in device memory (written by hpr_scalars once per
113// block), so a block's kernels have fixed arguments and can be replayed as a
114// CUDA graph: sc[kSigma .. kInvRho] and, for the block's iteration j (0 <= j <
115// kMaxBlock), the Halpern weights sc[kW + 2j] = (t+1)/(t+2) and sc[kW + 2j + 1]
116// = 1/(t+2) with t = t0 + j. No division is left in the per-element steps (FP64
117// division is slow on consumer GPUs, whose FP64 rate is 1/32 to 1/64 of FP32).
118//
119// Most steps are epilogues of a sparse product, f(i, s, a) with s = (M v)_i, so
120// an iteration is four kernels (Q xhat, A v, A^T ybar, Q d) plus, at the start
121// of a block, one map (hpr_primal).
122enum HprScalar {
123 kSigma = 0,
124 kA = 1,
125 kBq = 2,
126 kRho = 3,
127 kKappa = 4,
128 kInvRho = 5,
129 kW = 8,
130 kMaxBlock = 64,
131 kScalars = kW + 2 * kMaxBlock
132};
133
134// Factors of the two sGS half steps of w: whalf = a w + b xhat, and
135// wbar = whalf + kappa A^T(ybar - y). With the semi-proximal term s (lq Q -
136// Q^2) they are scalars (a = s lq / (1 + s lq), b = 1 / (1 + s lq), kappa = s /
137// (1 + s lq), in sc[]). A diagonal Q needs no proximal term: the w step is
138// exact, with lq replaced by q_i per element; then qb holds b_i = 1 / (1 + s
139// q_i) (hpr_qb, whenever s changes), a_i = 1 - b_i and kappa_i = s b_i.
140QP_HD inline void
141hpr_factors(const double *sc, const double *qb, idx i, double &a, double &b,
142 double &kappa)
143{
144 if (qb) {
145 b = qb[i];
146 a = 1.0 - b;
147 kappa = sc[kSigma] * b;
148 } else {
149 a = sc[kA];
150 b = sc[kBq];
151 kappa = sc[kKappa];
152 }
153}
154
155QP_HD inline void
156halpern_weights(const double *sc, int j, double &w1, double &w0)
157{
158 w1 = sc[kW + 2 * j];
159 w0 = sc[kW + 2 * j + 1];
160}

162// The scalars of the block that starts at iteration t0 (map over kMaxBlock).
163struct HprScalars {
164 double *sc;
165 double sigma, a, b, rho, kappa, t0;
166};
167QP_HD inline void
168hpr_scalars(idx i, const HprScalars &p)
169{
170 const double t = p.t0 + static_cast<double>(i);
171 p.sc[kW + 2 * i] = (t + 1.0) / (t + 2.0);
172 p.sc[kW + 2 * i + 1] = 1.0 / (t + 2.0);
173 if (i == 0) {
174 p.sc[kSigma] = p.sigma;
175 p.sc[kA] = p.a;
176 p.sc[kBq] = p.b;
177 p.sc[kRho] = p.rho;
178 p.sc[kKappa] = p.kappa;
179 p.sc[kInvRho] = 1.0 / p.rho;
180 }
181}

201// z and x: r = x + s (A^T y - Q w - c), xbar = P_C(r), xhat = 2 xbar - x,
202// zbar = (xbar - r) / s (only when zbar is set: the KKT check needs it).
203QP_HD inline void
204hpr_primal_at(idx i, double x, double qw, double aty, const double *c,
205 const double *lb, const double *ub, double sigma, double *xbar,
206 double *xhat, double *zbar)
207{
208 const double r = x + sigma * (aty - qw - c[i]);
209 const double xb = clampd(r, lb[i], ub[i]);
210 xbar[i] = xb;
211 xhat[i] = 2.0 * xb - x;
212 if (zbar) zbar[i] = (xb - r) / sigma;
213}
214struct HprPrimal {
215 const double *x, *qw, *aty, *c, *lb, *ub;
216 double *xbar, *xhat, *zbar;
217 const double *sc;
218};
219QP_HD inline void
220hpr_primal(idx i, const HprPrimal &a)
221{
222 hpr_primal_at(i, a.x[i], a.qw[i], a.aty[i], a.c, a.lb, a.ub, a.sc[kSigma],
223 a.xbar, a.xhat, a.zbar);
224}

226// Epilogue of s = (Q xhat)_i, first sGS half of w (restricted to range(Q)
227// through Q w only): whalf = a w + b xhat, Q whalf = a Q w + b s with a, b from
228// hpr_factors; v = xhat + s (Q w - Q whalf) is the point the y step multiplies
229// by A.
230struct HprWHalf {
231 const double *w, *qw, *xhat;
232 double *whalf, *qwhalf, *v;
233 const double *sc, *qb;
234};
235QP_HD inline void
236hpr_whalf(idx i, double s, const HprWHalf &p)
237{
238 double a, b, kappa;
239 hpr_factors(p.sc, p.qb, i, a, b, kappa);
240 const double qh = a * p.qw[i] + b * s;
241 p.whalf[i] = a * p.w[i] + b * p.xhat[i];
242 p.qwhalf[i] = qh;
243 p.v[i] = p.xhat[i] + p.sc[kSigma] * (p.qw[i] - qh);
244}

246// Epilogue of s = (A v)_i, the y step (linearized with the semi-proximal
247// term s (la I - A A^T)): r = s - rho y (rho = s la), ybar = (P_K(r) - r) /
248// rho; with halpern set, also the Halpern step y <- w1 (2 ybar - y) + w0 y0.
249struct HprDual {
250 double *y, *ybar;
251 const double *l, *u, *y0, *sc;
252 int j, halpern;
253};
254QP_HD inline void
255hpr_dual(idx i, double s, const HprDual &a)
256{
257 const double y = a.y[i];
258 const double r = s - a.sc[kRho] * y;
259 const double yb = (clampd(r, a.l[i], a.u[i]) - r) * a.sc[kInvRho];
260 a.ybar[i] = yb;
261 if (a.halpern) {
262 double w1, w0;
263 halpern_weights(a.sc, a.j, w1, w0);
264 a.y[i] = w1 * (2.0 * yb - y) + w0 * a.y0[i];
265 }
266}

280// Epilogue of s = (Q d)_i: second sGS half of w (wbar = whalf + kappa d),
281// the Halpern step v <- w1 (2 vbar - v) + w0 v0 for v in (x, w, Q w, A^T y),
282// and, with next set, the z/x step of the next iteration (same sigma).
283struct HprHalpern {
284 double *x, *w, *qw, *aty, *xbar, *xhat;
285 const double *whalf, *qwhalf, *d, *atybar, *x0, *w0, *qw0, *aty0, *c, *lb,
286 *ub, *sc, *qb;
287 int j, next;
288};
289QP_HD inline void
290hpr_halpern(idx i, double s, const HprHalpern &p)
291{
292 double w1, w0, a, b, kappa;
293 halpern_weights(p.sc, p.j, w1, w0);
294 hpr_factors(p.sc, p.qb, i, a, b, kappa);
295 const double wb = p.whalf[i] + kappa * p.d[i];
296 const double qwb = p.qwhalf[i] + kappa * s;
297 const double x = w1 * (2.0 * p.xbar[i] - p.x[i]) + w0 * p.x0[i];
298 const double qw = w1 * (2.0 * qwb - p.qw[i]) + w0 * p.qw0[i];
299 const double aty = w1 * (2.0 * p.atybar[i] - p.aty[i]) + w0 * p.aty0[i];
300 p.x[i] = x;
301 p.w[i] = w1 * (2.0 * wb - p.w[i]) + w0 * p.w0[i];
302 p.qw[i] = qw;
303 p.aty[i] = aty;
304 if (p.next)
305 hpr_primal_at(i, x, qw, aty, p.c, p.lb, p.ub, p.sc[kSigma], p.xbar,
306 p.xhat, nullptr);
307}

309// The same step at a convergence check, as a reduction over the n-sized parts
310// (qd = Q d stored); also stores wbar, Q wbar (needed at restarts) and
311// accumulates the terms of the merit ||T(u) - u||_M, with ex = xbar - x,
312// ew = wbar - w, eq = Q ew, d = A^T(ybar - y):
313// acc0 = |ex|^2, acc1 = <ew, eq>, acc2 = <eq, d>, acc3 = <d, Q d>,
314// acc4 = |eq|^2, acc5 = <eq, ex>, acc6 = <d, ex>,
315// acc7 = sum d_i (Q d)_i / (1 + s q_i) (diagonal Q only)
316// (second_half = 0: the free-variable variant below, whose wbar and Q wbar are
317// whalf and qwhalf as they come.)
318struct HprHalpernRed {
319 double *x, *w, *qw, *aty, *wbar, *qwbar;
320 const double *xbar, *whalf, *qwhalf, *d, *qd, *atybar, *x0, *w0, *qw0,
321 *aty0, *sc, *qb;
322 int j, second_half;
323 static constexpr int K = 8;
324 static constexpr unsigned kMax = 0;
325};
326QP_HD inline void
327hpr_halpern_red(idx i, const HprHalpernRed &p, double *acc)
328{
329 double w1, w0, a, b, kappa;
330 halpern_weights(p.sc, p.j, w1, w0);
331 hpr_factors(p.sc, p.qb, i, a, b, kappa);
332 if (!p.second_half) kappa = 0.0;
333 const double di = p.d[i], qdi = p.qd[i];
334 const double wb = p.whalf[i] + kappa * di;
335 const double qwb = p.qwhalf[i] + kappa * qdi;
336 const double ex = p.xbar[i] - p.x[i], ew = wb - p.w[i], eq = qwb - p.qw[i];
337 acc[0] += ex * ex;
338 acc[1] += ew * eq;
339 acc[2] += eq * di;
340 acc[3] += di * qdi;
341 acc[4] += eq * eq;
342 acc[5] += eq * ex;
343 acc[6] += di * ex;
344 if (p.qb) acc[7] += di * qdi * b;
345 p.wbar[i] = wb;
346 p.qwbar[i] = qwb;
347 p.x[i] = w1 * (2.0 * p.xbar[i] - p.x[i]) + w0 * p.x0[i];
348 p.w[i] = w1 * (2.0 * wb - p.w[i]) + w0 * p.w0[i];
349 p.qw[i] = w1 * (2.0 * qwb - p.qw[i]) + w0 * p.qw0[i];
350 p.aty[i] = w1 * (2.0 * p.atybar[i] - p.aty[i]) + w0 * p.aty0[i];
351}

353// ---- HPR-QP, variant for mostly free x --------------------------------------
354// When (nearly) no x has bounds, the authors' HPR-QP.jl sweeps w -> x -> y
355// without the second sGS half of w: three products per iteration instead of
356// four, A^T y (the x and w steps in its epilogue), Q wbar (the Q w update and
357// the point the y step multiplies by A) and A v (the y step, hpr_dual).
358
359// Epilogue of s = (A^T y)_i, with the fresh A^T y: z/x step, wbar = a w +
360// b xhat, and with halpern set the Halpern steps on x and w.
361struct HprFreeXW {
362 double *x, *w, *xbar, *xhat, *wbar, *aty;
363 const double *qw, *c, *lb, *ub, *x0, *w0, *sc, *qb;
364 int j, halpern;
365};
366QP_HD inline void
367hpr_free_xw(idx i, double s, const HprFreeXW &p)
368{
369 const double x = p.x[i];
370 const double r = x + p.sc[kSigma] * (s - p.qw[i] - p.c[i]);
371 const double xb = clampd(r, p.lb[i], p.ub[i]);
372 const double xh = 2.0 * xb - x;
373 double a, b, kappa;
374 hpr_factors(p.sc, p.qb, i, a, b, kappa);
375 const double wb = a * p.w[i] + b * xh;
376 p.aty[i] = s;
377 p.xbar[i] = xb;
378 p.xhat[i] = xh;
379 p.wbar[i] = wb;
380 if (p.halpern) {
381 double w1, w0;
382 halpern_weights(p.sc, p.j, w1, w0);
383 p.x[i] = w1 * xh + w0 * p.x0[i];
384 p.w[i] = w1 * (2.0 * wb - p.w[i]) + w0 * p.w0[i];
385 }
386}

388// Epilogue of s = (Q wbar)_i: v = xhat + s (Q w - Q wbar), the point of the
389// y step; Q wbar stored when qwbar is set; with halpern set the Halpern step
390// Q w <- w1 (2 Q wbar - Q w) + w0 Q w0.
391struct HprFreeQ {
392 double *qw, *v, *qwbar;
393 const double *xhat, *qw0, *sc;
394 int j, halpern;
395};
396QP_HD inline void
397hpr_free_q(idx i, double s, const HprFreeQ &p)
398{
399 const double qw = p.qw[i];
400 p.v[i] = p.xhat[i] + p.sc[kSigma] * (qw - s);
401 if (p.qwbar) p.qwbar[i] = s;
402 if (p.halpern) {
403 double w1, w0;
404 halpern_weights(p.sc, p.j, w1, w0);
405 p.qw[i] = w1 * (2.0 * s - qw) + w0 * p.qw0[i];
406 }
407}

427// Penalty update terms (differences between T(u) and the epoch's anchor):
428// acc0 = |xbar - x0|^2, acc1 = <wbar - w0, Q(wbar - w0)>,
429// acc2 = <Q(wbar - w0), A^T(ybar - y0)>, acc3 = |Q(wbar - w0)|^2;
430// writes dq = A^T(ybar - y0).
431struct HprThetaN {
432 const double *xbar, *x0, *wbar, *w0, *qwbar, *qw0, *atybar, *aty0;
433 double *dq;
434 static constexpr int K = 4;
435 static constexpr unsigned kMax = 0;
436};
437QP_HD inline void
438hpr_theta_n(idx i, const HprThetaN &p, double *acc)
439{
440 const double dx = p.xbar[i] - p.x0[i], dw = p.wbar[i] - p.w0[i];
441 const double dqw = p.qwbar[i] - p.qw0[i], da = p.atybar[i] - p.aty0[i];
442 acc[0] += dx * dx;
443 acc[1] += dw * dqw;
444 acc[2] += dqw * da;
445 acc[3] += dqw * dqw;
446 p.dq[i] = da;
447}

462// ---- KKT residuals at (xbar, ybar, zbar), in the original scale -----------
463// rows: acc0 = max |Ax - P_K(Ax)| / e, acc1 = max |Ax| / e,
464// acc2 = sum of ybar_i * (l_i if ybar_i > 0 else u_i)
465struct KktRows {
466 const double *ax, *y, *l, *u, *rinv;
467 static constexpr int K = 3;
468 static constexpr unsigned kMax = 3;
469};
470QP_HD inline void
471kkt_rows(idx i, const KktRows &a, double *acc)
472{
473 const double ax = a.ax[i];
474 acc[0] = dmax(acc[0], dabs(ax - clampd(ax, a.l[i], a.u[i])) * a.rinv[i]);
475 acc[1] = dmax(acc[1], dabs(ax) * a.rinv[i]);
476 const double y = a.y[i];
477 acc[2] += y > 0 ? a.l[i] * y : (y < 0 ? a.u[i] * y : 0.0);
478}

480// columns: acc0 = max |Q x + c - A^T y - z| / d, acc1 = max |Q x| / d,
481// acc2 = max |A^T y| / d, acc3 = x^T Q x, acc4 = c^T x,
482// acc5 = sum of z_j * (lb_j if z_j > 0 else ub_j)
483struct KktCols {
484 const double *x, *qx, *aty, *z, *c, *lb, *ub, *cinv;
485 static constexpr int K = 6;
486 static constexpr unsigned kMax = 7;
487};
488QP_HD inline void
489kkt_cols(idx i, const KktCols &a, double *acc)
490{
491 const double ci = a.cinv[i], qx = a.qx[i], aty = a.aty[i], z = a.z[i];
492 acc[0] = dmax(acc[0], dabs(qx + a.c[i] - aty - z) * ci);
493 acc[1] = dmax(acc[1], dabs(qx) * ci);
494 acc[2] = dmax(acc[2], dabs(aty) * ci);
495 acc[3] += a.x[i] * qx;
496 acc[4] += a.c[i] * a.x[i];
497 acc[5] += z > 0 ? a.lb[i] * z : (z < 0 ? a.ub[i] * z : 0.0);
498}

500// ---- PDHCG iteration
501// ------------------------------------------------------------ Primal
502// subproblem min_{x in C} 1/2 x^T Q x + (c - A^T y)^T x + |x - xk|^2 / (2
503// tau), gradient (Q + I/tau) x + c - A^T y - xk/tau. The inner iterate xt
504// starts at xk; qxt = Q xt is carried along (Q xk + sum of Q s).
505
506// Start of the inner solve: xt = xk, qxt = Q xk, g = Q xk + c - A^T y
507// (the gradient at xk), and for CG r = p = -g.
508struct PdStart {
509 const double *x, *qx, *c, *aty;
510 double *xt, *qxt, *g, *r, *pdir;
511};
512QP_HD inline void
513pd_start(idx i, const PdStart &a)
514{
515 const double g = a.qx[i] + a.c[i] - a.aty[i];
516 a.xt[i] = a.x[i];
517 a.qxt[i] = a.qx[i];
518 a.g[i] = g;
519 if (a.r) {
520 a.r[i] = -g;
521 a.pdir[i] = -g;
522 }
523}

525// Projected (BB) gradient step: xt+ = P_C(xt - alpha g), s = xt+ - xt.
526struct PdPgStep {
527 double *xt, *s;
528 const double *g, *lb, *ub;
529 double alpha;
530};
531QP_HD inline void
532pd_pg_step(idx i, const PdPgStep &a)
533{
534 const double xn = clampd(a.xt[i] - a.alpha * a.g[i], a.lb[i], a.ub[i]);
535 a.s[i] = xn - a.xt[i];
536 a.xt[i] = xn;
537}
538
539// After qs = Q s: g += qs + s / tau, qxt += qs;
540// acc0 = s.s, acc1 = s.(Q + I/tau)s, acc2 = max|s|, acc3 = max|xt|
541struct PdPgUpdate {
542 double *g, *qxt;
543 const double *s, *qs, *xt;
544 double inv_tau;
545 static constexpr int K = 4;
546 static constexpr unsigned kMax = 12;
547};
548QP_HD inline void
549pd_pg_update(idx i, const PdPgUpdate &a, double *acc)
550{
551 const double s = a.s[i], hs = a.qs[i] + s * a.inv_tau;
552 a.g[i] += hs;
553 a.qxt[i] += a.qs[i];
554 acc[0] += s * s;
555 acc[1] += s * hs;
556 acc[2] = dmax(acc[2], dabs(s));
557 acc[3] = dmax(acc[3], dabs(a.xt[i]));
558}

560// CG (no bounds): after qp = Q p: hp = qp + p / tau stored in hp;
561// acc0 = p.hp
562struct PdCgCurv {
563 const double *p, *qp;
564 double *hp;
565 double inv_tau;
566 static constexpr int K = 1;
567 static constexpr unsigned kMax = 0;
568};
569QP_HD inline void
570pd_cg_curv(idx i, const PdCgCurv &a, double *acc)
571{
572 const double h = a.qp[i] + a.p[i] * a.inv_tau;
573 a.hp[i] = h;
574 acc[0] += a.p[i] * h;
575}

577// xt += alpha p, qxt += alpha qp, r -= alpha hp;
578// acc0 = r.r, acc1 = max|alpha p|, acc2 = max|xt|
579struct PdCgStep {
580 double *xt, *qxt, *r;
581 const double *p, *qp, *hp;
582 double alpha;
583 static constexpr int K = 3;
584 static constexpr unsigned kMax = 6;
585};
586QP_HD inline void
587pd_cg_step(idx i, const PdCgStep &a, double *acc)
588{
589 const double dx = a.alpha * a.p[i];
590 a.xt[i] += dx;
591 a.qxt[i] += a.alpha * a.qp[i];
592 const double r = a.r[i] - a.alpha * a.hp[i];
593 a.r[i] = r;
594 acc[0] += r * r;
595 acc[1] = dmax(acc[1], dabs(dx));
596 acc[2] = dmax(acc[2], dabs(a.xt[i]));
597}

601 double *p;
602 const double *r;
603 double beta;
604};

606pd_cg_dir(idx i, const PdCgDir &a)

609// Dual step with extrapolation t = 2 A xt - A x (PDLP form):
610// yn = max(0, y + s(l - t)) + min(0, y + s(u - t))
611struct PdDual {
612 const double *y, *ax, *axn, *l, *u;
613 double *yn;
614 double sigma;
615};
616QP_HD inline void
617pd_dual(idx i, const PdDual &a)
618{
619 const double t = 2.0 * a.axn[i] - a.ax[i];
620 const double lo = a.y[i] + a.sigma * (a.l[i] - t);
621 const double hi = a.y[i] + a.sigma * (a.u[i] - t);
622 a.yn[i] = dmax(lo, 0.0) + dmin(hi, 0.0);
623}

625// Reflected Halpern step v <- w1 (2 vn - v) + w0 v0 on one vector.
626struct Halpern {
627 double *v;
628 const double *vn, *v0;
629 double w1, w0c;
630};
631QP_HD inline void
632halpern(idx i, const Halpern &a)
633{ a.v[i] = a.w1 * (2.0 * a.vn[i] - a.v[i]) + a.w0c * a.v0[i]; }

635// Halpern step on the n-sized PDHCG state (x, Q x, A^T y).
636struct PdHalpernN {
637 double *x, *qx, *aty;
638 const double *xn, *qxn, *atyn, *x0, *qx0, *aty0;
639 double w1, w0c;
640};
641QP_HD inline void
642pd_halpern_n(idx i, const PdHalpernN &a)
643{
644 a.x[i] = a.w1 * (2.0 * a.xn[i] - a.x[i]) + a.w0c * a.x0[i];
645 a.qx[i] = a.w1 * (2.0 * a.qxn[i] - a.qx[i]) + a.w0c * a.qx0[i];
646 a.aty[i] = a.w1 * (2.0 * a.atyn[i] - a.aty[i]) + a.w0c * a.aty0[i];
647}

649// KKT columns for a PDHG point: the bound duals z are the projection of
650// g = Q x + c - A^T y onto their sign cone (as in evaluate_qp); same
651// accumulators as KktCols.
652struct PdKktCols {
653 const double *x, *qx, *aty, *c, *lb, *ub, *cinv;
654 static constexpr int K = 6;
655 static constexpr unsigned kMax = 7;
656};
657QP_HD inline void
658pd_kkt_cols(idx i, const PdKktCols &a, double *acc)
659{
660 const double ci = a.cinv[i], qx = a.qx[i], aty = a.aty[i];
661 const double g = qx + a.c[i] - aty;
662 const bool lo = a.lb[i] > -1e300, hi = a.ub[i] < 1e300;
663 const double z =
664 (lo && hi) ? g : (lo ? dmax(g, 0.0) : (hi ? dmin(g, 0.0) : 0.0));
665 acc[0] = dmax(acc[0], dabs(g - z) * ci);
666 acc[1] = dmax(acc[1], dabs(qx) * ci);
667 acc[2] = dmax(acc[2], dabs(aty) * ci);
668 acc[3] += a.x[i] * qx;
669 acc[4] += a.c[i] * a.x[i];
670 acc[5] += z > 0 ? a.lb[i] * z : (z < 0 ? a.ub[i] * z : 0.0);
671}
