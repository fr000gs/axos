15// Presolve + solve + postsolve.
16template <template <typename> class Store = Cpu::HostStorage>
17LpSolution
18solve_lp(const LpProblem &p, const SolverOptions &opt = SolverOptions())
19{
20 auto t0 = std::chrono::steady_clock::now();
21 auto secs = [&] {
22 return std::chrono::duration<double>(
23 std::chrono::steady_clock::now() - t0)
24 .count();
25 };
26 // The interior-point method needs presolve (free rows, fixed columns).
27 const bool need_presolve = opt.presolve || opt.method != LpMethod::Pdlp ||
28 p.rows() == 0 || p.cols() == 0;
29 auto usable = [](const LpProblem &q, const LpSolution &s) {
30 if (s.x.size() != q.cols() || s.y.size() != q.rows()) return false;
31 for (double v : s.x)
32 if (!std::isfinite(v)) return false;
33 for (double v : s.y)
34 if (!std::isfinite(v)) return false;
35 return true;
36 };
37 // `start` (an interior / approximate solution of q) seeds the simplex by
38 // crossover
39 auto run = [&](const LpProblem &q, LpMethod m, const SolverOptions &o,
40 const LpSolution *start = nullptr) {
41 switch (m) {
42 case LpMethod::Ipm: {
43 LpSolution s = Ipm<Store>().solve(q, o);
44 if (o.crossover && s.status == Status::Optimal && usable(q, s)) {
45 SolverOptions oc = o;
46 oc.time_limit = std::max(1.0, o.time_limit - s.seconds);
47 LpSolution c = DualSimplex().solve(q, oc, nullptr, nullptr, &s);
48 if (c.status == Status::Optimal) return c;
49 }
50 return s;
51 }
52 case LpMethod::Simplex:
53 return DualSimplex().solve(q, o, nullptr, nullptr, start);
54 default:
55 return Pdlp<Store>().solve(q, o);
56 }
57 };
58 // Auto. A cheap interior-point factorization (predicted from the symbolic
59 // analysis) means interior point first, then simplex, then PDLP. Otherwise
60 // the dual simplex gets a slice of the predicted interior-point time (its
61 // typical win on network / sparse problems), then interior point, then
62 // PDLP. The time limit is shared.
63 auto run_auto = [&](const LpProblem &q) {
64 constexpr double kCheapFlops = 4e9; // Sum of squared column counts
65 constexpr double kFlopsPerSec =
66 4e10; // measured multifrontal rate on the CPU
67 auto remaining = [&] { return std::max(0.0, opt.time_limit - secs()); };
68 auto accept = [&](LpMethod m, const LpSolution &s) {
69 // Optimal, or a certificate from a method that can give one
70 return s.status == Status::Optimal ||
71 (m != LpMethod::Ipm && (s.status == Status::Infeasible ||
72 s.status == Status::Unbounded));
73 };
74 auto attempt = [&](LpMethod m, double budget, LpSolution &best,
75 double flops_limit = 0) {
76 SolverOptions o = opt;
77 o.time_limit = std::min(budget, remaining());
78 o.ipm_max_flops = flops_limit;
79 if (o.time_limit <= 0) return false;
80 LpSolution s = run(q, m, o);
81 if (accept(m, s)) {
82 best = s;
83 return true;
84 }
85 if (s.status != Status::NotSolved)
86 best = s;
87 else if (best.status == Status::NotSolved)
88 best.factor_flops = s.factor_flops;
89 return false;
90 };
91 LpSolution best;
92 if (attempt(LpMethod::Ipm, remaining(), best, kCheapFlops)) return best;
93 if (best.status == Status::NotSolved) { // IPM predicted to be expensive
94 const double t_ipm = 20.0 * best.factor_flops / kFlopsPerSec;
95 if (attempt(LpMethod::Simplex, std::max(2.0, 0.3 * t_ipm), best))
96 return best;
97 if (attempt(LpMethod::Ipm, 0.6 * remaining(), best)) return best;
98 attempt(LpMethod::Pdlp, remaining(), best);
99 return best;
100 }
101 if (attempt(LpMethod::Simplex, 0.5 * remaining(), best)) return best;
102 attempt(LpMethod::Pdlp, remaining(), best);
103 return best;
104 };
105 if (!need_presolve)
106 return opt.method == LpMethod::Auto ? run_auto(p)
107 : run(p, opt.method, opt);
108
109 Presolve pre(p);
110 if (pre.status() != Status::NotSolved) {
111 LpSolution s;
112 s.status = pre.status();
113 s.x.assign(p.cols(), 0.0);
114 s.y.assign(p.rows(), 0.0);
115 s.seconds = secs();
116 return s;
117 }
118 const LpProblem &r = pre.reduced();
119 LpSolution rs;
120 if (r.rows() > 0 && r.cols() > 0) {
121 rs = opt.method == LpMethod::Auto ? run_auto(r)
122 : run(r, opt.method, opt);
123 } else {
124 rs.status = Status::Optimal; // nothing left to optimize
125 }
126 LpSolution s = pre.postsolve(rs);
127 s.seconds = secs();
128 return s;
129}
