3// Diagonal preconditioning for QPs, as in HPR-QP and PDHCG: Ruiz
4// equilibration followed by Pock-Chambolle, applied to the symmetric KKT
5// matrix
6//
7// K = [ Q A^T ] with column factors d (n) and row factors e (m):
8// [ A 0 ] Q~ = D Q D, A~ = E A D, c~ = D c,
9// x-bounds~ = bounds / d, row bounds~ = e .* bounds.
10//
11// A solution maps back as x = D x~, y = E y~, z = z~ / d; objective values
12// are unchanged.

29// Row / column infinity (p = 0) or l1 (p = 1) norms of the scaled K blocks.
30inline void
31kkt_norms(const HostMatrix &Q, const HostMatrix &A,
32 const std::vector<double> &d, const std::vector<double> &e, bool l1,
33 std::vector<double> &cn, std::vector<double> &rn)
34{
35 const size_t n = d.size(), m = e.size();
36 cn.assign(n, 0.0);
37 rn.assign(m, 0.0);
38 auto acc = [l1](double &s, double v) { s = l1 ? s + v : std::max(s, v); };
39 if (Q.nnz()) {
40 const auto *rp = Q.row_ptr();
41 const auto *ci = Q.col_ind();
42 const double *v = Q.values();
43 for (size_t i = 0; i < n; ++i)
44 for (int k = rp[i]; k < rp[i + 1]; ++k)
45 acc(cn[i], std::abs(v[k]) * d[i] * d[ci[k]]);
46 }
47 const auto *rp = A.row_ptr();
48 const auto *ci = A.col_ind();
49 const double *v = A.values();
50 for (size_t i = 0; i < m; ++i)
51 for (int k = rp[i]; k < rp[i + 1]; ++k) {
52 const double a = std::abs(v[k]) * e[i] * d[ci[k]];
53 acc(rn[i], a);
54 acc(cn[ci[k]], a);
55 }
56}

60inline QpScaling
61compute_qp_scaling(
62 const QpProblem &p, int ruiz_iterations, double pock_chambolle_alpha)
63{
64 using namespace qp_scaling_detail;
65 const size_t n = p.cols(), m = p.rows();
66 QpScaling s;
67 s.col.assign(n, 1.0);
68 s.row.assign(m, 1.0);
69 std::vector<double> cn, rn;
70 for (int it = 0; it < ruiz_iterations; ++it) {
71 kkt_norms(p.Q, p.lp.A, s.col, s.row, false, cn, rn);
72 for (size_t j = 0; j < n; ++j)
73 if (cn[j] > 0) s.col[j] /= std::sqrt(cn[j]);
74 for (size_t i = 0; i < m; ++i)
75 if (rn[i] > 0) s.row[i] /= std::sqrt(rn[i]);
76 }
77 if (pock_chambolle_alpha >= 0) {
78 // alpha = 1: l1 norms of rows and columns (K is symmetric)
79 kkt_norms(p.Q, p.lp.A, s.col, s.row, true, cn, rn);
80 for (size_t j = 0; j < n; ++j)
81 if (cn[j] > 0) s.col[j] /= std::sqrt(cn[j]);
82 for (size_t i = 0; i < m; ++i)
83 if (rn[i] > 0) s.row[i] /= std::sqrt(rn[i]);
84 }
85 return s;
86}

88inline QpProblem
89apply_qp_scaling(const QpProblem &p, const QpScaling &s)
90{
91 QpProblem q = p;
92 const size_t n = p.cols(), m = p.rows();
93 {
94 double *v = q.lp.A.values_mut();
95 const auto *rp = q.lp.A.row_ptr();
96 const auto *ci = q.lp.A.col_ind();
97 for (size_t i = 0; i < m; ++i)
98 for (int k = rp[i]; k < rp[i + 1]; ++k)
99 v[k] *= s.row[i] * s.col[ci[k]];
100 }
101 if (q.Q.nnz()) {
102 double *v = q.Q.values_mut();
103 const auto *rp = q.Q.row_ptr();
104 const auto *ci = q.Q.col_ind();
105 for (size_t i = 0; i < n; ++i)
106 for (int k = rp[i]; k < rp[i + 1]; ++k)
107 v[k] *= s.col[i] * s.col[ci[k]];
108 }
109 for (size_t j = 0; j < n; ++j) {
110 q.lp.c[j] *= s.col[j];
111 q.lp.col_lb[j] /= s.col[j];
112 q.lp.col_ub[j] /= s.col[j];
113 }
114 for (size_t i = 0; i < m; ++i) {
115 q.lp.row_lb[i] *= s.row[i];
116 q.lp.row_ub[i] *= s.row[i];
117 }
118 return q;
119}
