3// Sparse LU of a simplex basis with product-form (eta) updates.
4//
5// factor() Gilbert-Peierls left-looking LU with threshold partial
6// pivoting: columns are ordered by increasing nonzero count
7// (singletons, i.e. slacks, go first) and among acceptable pivots
8// the row with the fewest original nonzeros is preferred.
9// Singular columns are reported so the caller can replace them
10// by slack columns.
11// ftran() solves B x = b (b indexed by row, x by basis position)
12// btran() solves B^T y = c (c indexed by basis position, y by row)
13// update() replaces the column at basis position r (eta file), so
14// B_new^{-1} = E B_old^{-1}
15//
16// All vectors are dense (length m); the triangular solves cost
17// O(nnz(L) + nnz(U) + nnz(etas)).

29// Sparse work vector: dense values plus the list of positions that may be
30// nonzero (each position appears at most once).
31struct SVec {
32 std::vector<double> v;
33 std::vector<int> idx;
34 std::vector<char> flag;
35
36 void
37 init(int n)
38 {
39 v.assign(n, 0.0);
40 flag.assign(n, 0);
41 idx.clear();
42 }
43 int
44 size() const
45 { return static_cast<int>(v.size()); }
46 void
47 clear()
48 {
49 for (int i : idx) {
50 v[i] = 0.0;
51 flag[i] = 0;
52 }
53 idx.clear();
54 }
55 // v[i] += a, tracking the pattern
56 void
57 add(int i, double a)
58 {
59 if (!flag[i]) {
60 flag[i] = 1;
61 idx.push_back(i);
62 }
63 v[i] += a;
64 }
65 void
66 set(int i, double a)
67 {
68 if (!flag[i]) {
69 flag[i] = 1;
70 idx.push_back(i);
71 }
72 v[i] = a;
73 }
74 // rebuild the pattern from the dense values
75 void
76 rebuild_pattern()
77 {
78 for (int i : idx)
79 flag[i] = 0;
80 idx.clear();
81 for (int i = 0; i < size(); ++i)
82 if (v[i] != 0.0) {
83 flag[i] = 1;
84 idx.push_back(i);
85 }
86 }
87};

89class BasisFactor {
90 public:
91 // A sparse column: parallel arrays of row indices and values.
92 struct Column {
93 const int *idx;
94 const double *val;
95 int nnz;

98 // Factor the m x m matrix whose k-th column is cols[k]. Returns the number
99 // of singular pivots; for each, singular_pos[i] is the basis position of
100 // the dependent column and free_row[i] a row left without a pivot (the
101 // caller substitutes the slack of that row).
102 int
103 factor(int m, const std::vector<Column> &cols,
104 std::vector<int> &singular_pos, std::vector<int> &free_row)
105 {
106 m_ = m;
107 n_updates_ = 0;
108 eta_ptr_.assign(1, 0);
109 eta_idx_.clear();
110 eta_val_.clear();
111 eta_pos_.clear();
112 eta_diag_.clear();
113 singular_pos.clear();
114 free_row.clear();
115
116 // row counts of the original matrix (sparsity tie-break)
117 std::vector<int> rowcnt(m, 0);
118 for (int k = 0; k < m; ++k)
119 for (int t = 0; t < cols[k].nnz; ++t)
120 rowcnt[cols[k].idx[t]]++;
121 // column order: fewest nonzeros first
122 q_.resize(m);
123 for (int k = 0; k < m; ++k)
124 q_[k] = k;
125 std::stable_sort(q_.begin(), q_.end(),
126 [&](int a, int b) { return cols[a].nnz < cols[b].nnz; });
127
128 lp_.assign(1, 0);
129 li_.clear();
130 lx_.clear();
131 up_.assign(1, 0);
132 ui_.clear();
133 ux_.clear();
134 udiag_.assign(m, 1.0);
135 prow_.assign(m, -1); // prow[k] = row pivoted at step k
136 std::vector<int> pinv(m, -1); // pinv[row] = step, -1 if not pivoted
137 std::vector<double> x(m, 0.0);
138 std::vector<int> xi; // nonzero pattern in topological order
139 std::vector<int> stack, pstack, mark(m, -1);
140 std::vector<int>
141 lcol_row; // L entries use ORIGINAL row ids until the end
142 int nsing = 0;
143 std::vector<int> failed_steps;
144 // L columns are stored by step; entries below the pivot in original
145 // rows
146 std::vector<int> lstart{0};
147 std::vector<int> l_orig_idx;
148 std::vector<double> l_val;
149
150 for (int k = 0; k < m; ++k) {
151 const Column &c = cols[q_[k]];
152 // ---- reach of the column pattern in the graph of L ----------
153 xi.clear();
154 for (int t = 0; t < c.nnz; ++t) {
155 int i = c.idx[t];
156 if (mark[i] == k) continue;
157 // iterative DFS from row i
158 stack.assign(1, i);
159 pstack.assign(1, 0);
160 mark[i] = k;
161 while (!stack.empty()) {
162 int r = stack.back();
163 int step = pinv[r];
164 int &pp = pstack.back();
165 bool pushed = false;
166 if (step >= 0) {
167 int beg = lstart[step], end = lstart[step + 1];
168 for (; beg + pp < end;) {
169 int ch = l_orig_idx[beg + pp];
170 ++pp;
171 if (mark[ch] != k) {
172 mark[ch] = k;
173 stack.push_back(ch);
174 pstack.push_back(0);
175 pushed = true;
176 break;
177 }
178 }
179 }
180 if (!pushed) {
181 xi.push_back(r);
182 stack.pop_back();
183 pstack.pop_back();
184 }
185 }
186 }
187 // xi is in reverse topological order (finish order): process
188 // reversed
189 for (int t = 0; t < c.nnz; ++t)
190 x[c.idx[t]] = c.val[t];
191 for (int t = static_cast<int>(xi.size()) - 1; t >= 0; --t) {
192 int r = xi[t];
193 int step = pinv[r];
194 if (step < 0) continue;
195 double xr = x[r];
196 if (xr == 0.0) continue;
197 for (int p = lstart[step]; p < lstart[step + 1]; ++p)
198 x[l_orig_idx[p]] -= l_val[p] * xr;
199 }
200 // ---- pivot selection among unpivoted rows ---------------------
201 double amax = 0;
202 for (int r : xi)
203 if (pinv[r] < 0) amax = std::max(amax, std::abs(x[r]));
204 int piv = -1;
205 if (amax > 1e-11) {
206 const double thresh = 0.1 * amax;
207 int best_cnt = 1 << 30;
208 double best_abs = 0;
209 for (int r : xi) {
210 if (pinv[r] >= 0) continue;
211 double a = std::abs(x[r]);
212 if (a < thresh) continue;
213 if (rowcnt[r] < best_cnt ||
214 (rowcnt[r] == best_cnt && a > best_abs)) {
215 best_cnt = rowcnt[r];
216 best_abs = a;
217 piv = r;
218 }
219 }
220 }
221 if (piv < 0) { // singular column
222 ++nsing;
223 failed_steps.push_back(k);
224 singular_pos.push_back(q_[k]);
225 for (int r : xi)
226 x[r] = 0.0;
227 for (int t = 0; t < c.nnz; ++t)
228 x[c.idx[t]] = 0.0;
229 // record an empty step; pinv stays -1 (row remains free)
230 lstart.push_back(static_cast<int>(l_orig_idx.size()));
231 up_.push_back(static_cast<int>(ui_.size()));
232 prow_[k] = -1;
233 udiag_[k] = 1.0;
234 continue;
235 }
236 const double pv = x[piv];
237 udiag_[k] = pv;
238 pinv[piv] = k;
239 prow_[k] = piv;
240 // U column: entries at already pivoted rows (steps < k)
241 for (int r : xi) {
242 if (r == piv) {
243 x[r] = 0.0;
244 continue;
245 }
246 if (pinv[r] >= 0 && pinv[r] < k) {
247 if (x[r] != 0.0) {
248 ui_.push_back(pinv[r]);
249 ux_.push_back(x[r]);
250 }
251 } else if (x[r] != 0.0) { // below the pivot: L entry
252 l_orig_idx.push_back(r);
253 l_val.push_back(x[r] / pv);
254 }
255 x[r] = 0.0;
256 }
257 for (int t = 0; t < c.nnz; ++t)
258 x[c.idx[t]] = 0.0;
259 up_.push_back(static_cast<int>(ui_.size()));
260 lstart.push_back(static_cast<int>(l_orig_idx.size()));
261 }
262 // ---- repair: pair every singular column with an unpivoted row -----
263 {
264 std::vector<int> unp;
265 for (int r = 0; r < m; ++r)
266 if (pinv[r] < 0) unp.push_back(r);
267 for (size_t t = 0; t < unp.size() && t < failed_steps.size(); ++t) {
268 free_row.push_back(unp[t]);
269 }
270 }
271 // ---- convert L row ids to pivot steps
272 // ---------------------------------
273 lp_ = lstart;
274 li_.resize(l_orig_idx.size());
275 for (size_t p = 0; p < l_orig_idx.size(); ++p)
276 li_[p] = pinv[l_orig_idx[p]];
277 lx_ = l_val;
278 // entries whose row is still unpivoted (only when singular) are dropped
279 if (nsing > 0) {
280 for (size_t p = 0; p < li_.size(); ++p)
281 if (li_[p] < 0) lx_[p] = 0.0, li_[p] = 0;
282 }
283 singular_ = nsing;
284 pinv_ = pinv;
285 qinv_.assign(m, 0);
286 for (int k = 0; k < m; ++k)
287 qinv_[q_[k]] = k;
288 mark_.assign(m, 0);
289 stamp_ = 0;
290 w_.assign(m, 0.0);
291 // row-wise copies of L and U so that BTRAN can skip zero entries
292 transpose(lp_, li_, lx_, lrp_, lci_, lrx_);
293 transpose(up_, ui_, ux_, urp_, uci_, urx_);
294 return nsing;
295 }

297 // Solve B x = b in place: on input b is indexed by row, on output by basis
298 // position. `work` must have m entries.
299 void
300 ftran(std::vector<double> &b) const
301 {
302 const int m = m_;
303 w_.assign(m, 0.0);
304 for (int k = 0; k < m; ++k)
305 w_[k] = prow_[k] >= 0 ? b[prow_[k]] : 0.0;
306 for (int k = 0; k < m; ++k) { // L
307 double yk = w_[k];
308 if (yk == 0.0) continue;
309 for (int p = lp_[k]; p < lp_[k + 1]; ++p)
310 w_[li_[p]] -= lx_[p] * yk;
311 }
312 for (int k = m - 1; k >= 0; --k) { // U (upper triangular in step order)
313 double zk = w_[k] / udiag_[k];
314 w_[k] = zk;
315 if (zk == 0.0) continue;
316 for (int p = up_[k]; p < up_[k + 1]; ++p)
317 w_[ui_[p]] -= ux_[p] * zk;
318 }
319 for (int k = 0; k < m; ++k)
320 b[q_[k]] = w_[k];
321 std::fill(
322 w_.begin(), w_.end(), 0.0); // the sparse solves expect w_ == 0
323 // eta file, oldest first
324 for (int e = 0; e < n_updates_; ++e) {
325 const int r = eta_pos_[e];
326 const double tr = b[r];
327 if (tr == 0.0) continue;
328 b[r] = tr * eta_diag_[e];
329 for (int p = eta_ptr_[e]; p < eta_ptr_[e + 1]; ++p)
330 b[eta_idx_[p]] += eta_val_[p] * tr;
331 }
332 }

334 // Solve B^T y = c: c indexed by basis position, y by row.
335 void
336 btran(std::vector<double> &c) const
337 {
338 const int m = m_;
339 for (int e = n_updates_ - 1; e >= 0; --e) { // eta file, newest first
340 const int r = eta_pos_[e];
341 double s = c[r] * eta_diag_[e];
342 for (int p = eta_ptr_[e]; p < eta_ptr_[e + 1]; ++p)
343 s += eta_val_[p] * c[eta_idx_[p]];
344 c[r] = s;
345 }
346 w_.assign(m, 0.0);
347 for (int k = 0; k < m; ++k)
348 w_[k] = c[q_[k]];
349 for (int i = 0; i < m; ++i) { // U^T z = w, row-wise sweep
350 const double zi = w_[i] / udiag_[i];
351 w_[i] = zi;
352 if (zi == 0.0) continue;
353 for (int p = urp_[i]; p < urp_[i + 1]; ++p)
354 w_[uci_[p]] -= urx_[p] * zi;
355 }
356 for (int i = m - 1; i >= 0; --i) { // L^T v = z, row-wise sweep
357 const double vi = w_[i];
358 if (vi == 0.0) continue;
359 for (int p = lrp_[i]; p < lrp_[i + 1]; ++p)
360 w_[lci_[p]] -= lrx_[p] * vi;
361 }
362 for (int k = 0; k < m; ++k)
363 if (prow_[k] >= 0) c[prow_[k]] = w_[k];
364 std::fill(w_.begin(), w_.end(), 0.0);
365 }

367 // ---- hypersparse solves
368 // --------------------------------------------------- Same operations as
369 // ftran/btran, on a sparse vector: the triangular solves visit only the
370 // nonzero pattern of the result (symbolic reach by DFS), and fall back to
371 // the dense loops when the vector is not sparse.
372
373 // Solve B x = b: b indexed by row on input, by basis position on output.
374 void
375 ftran(SVec &b) const
376 {
377 const int m = m_;
378 if (static_cast<int>(b.idx.size()) > 0.1 * m) {
379 ftran(b.v);
380 b.rebuild_pattern();
381 return;
382 }
383 // rows -> steps
384 starts_.clear();
385 for (int i : b.idx) {
386 const int k = pinv_[i];
387 const double val = b.v[i];
388 b.v[i] = 0.0;
389 b.flag[i] = 0;
390 if (k < 0 || val == 0.0) continue;
391 w_[k] = val;
392 starts_.push_back(k);
393 }
394 b.idx.clear();
395 reach(lp_, li_, starts_, order_); // L: edges step -> larger steps
396 for (int t = static_cast<int>(order_.size()) - 1; t >= 0; --t) {
397 const int k = order_[t];
398 const double yk = w_[k];
399 if (yk == 0.0) continue;
400 for (int p = lp_[k]; p < lp_[k + 1]; ++p)
401 w_[li_[p]] -= lx_[p] * yk;
402 }
403 starts_.assign(order_.begin(), order_.end());
404 reach(up_, ui_, starts_, order2_); // U: edges step -> smaller steps
405 for (int t = static_cast<int>(order2_.size()) - 1; t >= 0; --t) {
406 const int k = order2_[t];
407 const double zk = w_[k] / udiag_[k];
408 w_[k] = zk;
409 if (zk == 0.0) continue;
410 for (int p = up_[k]; p < up_[k + 1]; ++p)
411 w_[ui_[p]] -= ux_[p] * zk;
412 }
413 for (int k : order2_) {
414 const double val = w_[k];
415 w_[k] = 0.0;
416 if (val != 0.0) b.set(q_[k], val);
417 }
418 for (int e = 0; e < n_updates_; ++e) { // eta file, oldest first
419 const int r = eta_pos_[e];
420 const double tr = b.v[r];
421 if (tr == 0.0) continue;
422 b.v[r] = tr * eta_diag_[e];
423 for (int p = eta_ptr_[e]; p < eta_ptr_[e + 1]; ++p)
424 b.add(eta_idx_[p], eta_val_[p] * tr);
425 }
426 }

428 // Solve B^T y = c: c indexed by basis position on input, by row on output.
429 void
430 btran(SVec &c) const
431 {
432 const int m = m_;
433 if (static_cast<int>(c.idx.size()) > 0.1 * m && n_updates_ == 0) {
434 btran(c.v);
435 c.rebuild_pattern();
436 return;
437 }
438 for (int e = n_updates_ - 1; e >= 0; --e) { // eta file, newest first
439 const int r = eta_pos_[e];
440 double sum = c.v[r] * eta_diag_[e];
441 for (int p = eta_ptr_[e]; p < eta_ptr_[e + 1]; ++p)
442 sum += eta_val_[p] * c.v[eta_idx_[p]];
443 if (sum != 0.0 || c.v[r] != 0.0) c.set(r, sum);
444 }
445 if (static_cast<int>(c.idx.size()) > 0.1 * m) {
446 btran_lu_dense(c);
447 return;
448 }
449 starts_.clear();
450 for (int pos : c.idx) {
451 const double val = c.v[pos];
452 c.v[pos] = 0.0;
453 c.flag[pos] = 0;
454 if (val == 0.0) continue;
455 const int k = qinv_[pos];
456 w_[k] = val;
457 starts_.push_back(k);
458 }
459 c.idx.clear();
460 reach(urp_, uci_, starts_,
461 order_); // U^T: row-wise U, edges to larger steps
462 for (int t = static_cast<int>(order_.size()) - 1; t >= 0; --t) {
463 const int i = order_[t];
464 const double zi = w_[i] / udiag_[i];
465 w_[i] = zi;
466 if (zi == 0.0) continue;
467 for (int p = urp_[i]; p < urp_[i + 1]; ++p)
468 w_[uci_[p]] -= urx_[p] * zi;
469 }
470 starts_.assign(order_.begin(), order_.end());
471 reach(lrp_, lci_, starts_,
472 order2_); // L^T: row-wise L, edges to smaller steps
473 for (int t = static_cast<int>(order2_.size()) - 1; t >= 0; --t) {
474 const int i = order2_[t];
475 const double vi = w_[i];
476 if (vi == 0.0) continue;
477 for (int p = lrp_[i]; p < lrp_[i + 1]; ++p)
478 w_[lci_[p]] -= lrx_[p] * vi;
479 }
480 for (int k : order2_) {
481 const double val = w_[k];
482 w_[k] = 0.0;
483 if (val != 0.0 && prow_[k] >= 0) c.set(prow_[k], val);
484 }
485 }

487 // Column at basis position r is replaced; alpha = B_old^{-1} a_q (indexed
488 // by basis position). Returns false if the pivot is unusably small.
489 bool
490 update(int r, const std::vector<double> &alpha)
491 {
492 const double ar = alpha[r];
493 if (std::abs(ar) < 1e-11) return false;
494 eta_pos_.push_back(r);
495 eta_diag_.push_back(1.0 / ar);
496 for (int i = 0; i < m_; ++i) {
497 if (i == r) continue;
498 const double v = alpha[i];
499 if (v != 0.0 && std::abs(v) > 1e-14) {
500 eta_idx_.push_back(i);
501 eta_val_.push_back(-v / ar);
502 }
503 }
504 eta_ptr_.push_back(static_cast<int>(eta_idx_.size()));
505 ++n_updates_;
506 return true;
507 }
508
509 // Same as update(r, alpha) with a sparse alpha.
510 bool
511 update(int r, const SVec &alpha)
512 {
513 const double ar = alpha.v[r];
514 if (std::abs(ar) < 1e-11) return false;
515 eta_pos_.push_back(r);
516 eta_diag_.push_back(1.0 / ar);
517 for (int i : alpha.idx) {
518 if (i == r) continue;
519 const double v = alpha.v[i];
520 if (v != 0.0 && std::abs(v) > 1e-14) {
521 eta_idx_.push_back(i);
522 eta_val_.push_back(-v / ar);
523 }
524 }
525 eta_ptr_.push_back(static_cast<int>(eta_idx_.size()));
526 ++n_updates_;
527 return true;
528 }

545 // The L/U triangular part of btran on a dense-ish SVec (no eta file).
546 void
547 btran_lu_dense(SVec &c) const
548 {
549 const int m = m_;
550 w_.assign(m, 0.0);
551 for (int k = 0; k < m; ++k)
552 w_[k] = c.v[q_[k]];
553 for (int i = 0; i < m; ++i) {
554 const double zi = w_[i] / udiag_[i];
555 w_[i] = zi;
556 if (zi == 0.0) continue;
557 for (int p = urp_[i]; p < urp_[i + 1]; ++p)
558 w_[uci_[p]] -= urx_[p] * zi;
559 }
560 for (int i = m - 1; i >= 0; --i) {
561 const double vi = w_[i];
562 if (vi == 0.0) continue;
563 for (int p = lrp_[i]; p < lrp_[i + 1]; ++p)
564 w_[lci_[p]] -= lrx_[p] * vi;
565 }
566 c.clear();
567 for (int k = 0; k < m; ++k)
568 if (prow_[k] >= 0 && w_[k] != 0.0) c.set(prow_[k], w_[k]);
569 std::fill(w_.begin(), w_.end(), 0.0);
570 }

572 // Nodes reachable from `starts` in the graph with adjacency (ptr, idx), in
573 // DFS finishing order (so the reverse is a topological order).
574 void
575 reach(const std::vector<int> &ptr, const std::vector<int> &idx,
576 const std::vector<int> &starts, std::vector<int> &out) const
577 {
578 out.clear();
579 if (++stamp_ == 0x7fffffff) {
580 std::fill(mark_.begin(), mark_.end(), 0);
581 stamp_ = 1;
582 }
583 for (int s0 : starts) {
584 if (mark_[s0] == stamp_) continue;
585 mark_[s0] = stamp_;
586 stk_.assign(1, s0);
587 pst_.assign(1, ptr[s0]);
588 while (!stk_.empty()) {
589 const int node = stk_.back();
590 int &pp = pst_.back();
591 bool pushed = false;
592 while (pp < ptr[node + 1]) {
593 const int ch = idx[pp++];
594 if (mark_[ch] != stamp_) {
595 mark_[ch] = stamp_;
596 stk_.push_back(ch);
597 pst_.push_back(ptr[ch]);
598 pushed = true;
599 break;
600 }
601 }
602 if (!pushed) {
603 out.push_back(node);
604 stk_.pop_back();
605 pst_.pop_back();
606 }
607 }
608 }

611 // CSR transpose of a column-wise triangular factor (column k holds row
612 // indices idx[ptr[k]..ptr[k+1])).
613 void
614 transpose(const std::vector<int> &ptr, const std::vector<int> &idx,
615 const std::vector<double> &val, std::vector<int> &rp,
616 std::vector<int> &ci, std::vector<double> &rx) const
617 {
618 const int m = m_;
619 rp.assign(m + 1, 0);
620 for (size_t p = 0; p < idx.size(); ++p)
621 ++rp[idx[p] + 1];
622 for (int i = 0; i < m; ++i)
623 rp[i + 1] += rp[i];
624 ci.resize(idx.size());
625 rx.resize(idx.size());
626 std::vector<int> pos(rp.begin(), rp.end() - 1);
627 for (int k = 0; k < m; ++k)
628 for (int p = ptr[k]; p < ptr[k + 1]; ++p) {
629 const int q = pos[idx[p]]++;
630 ci[q] = k;
631 rx[q] = val[p];
632 }
633 }
634
635 int m_ = 0, n_updates_ = 0, singular_ = 0;
636 std::vector<int> q_, prow_;
637 std::vector<int> lp_, li_, up_, ui_;
638 std::vector<double> lx_, ux_, udiag_;
