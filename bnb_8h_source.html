3// Branch and cut for MILP (CPU orchestration; presolve and postsolve in
4// solve_milp.h).
5//
6// Root: domain propagation, double probing, feasibility jump (LP-free), the LP
7// relaxation by the dual simplex, rounds of Gomory mixed-integer cuts (cuts.h)
8// and c-MIR cuts from the model rows (cmir.h), then LP-based heuristics
9// (rounding with an LP repair of the continuous part, fix-and-propagate,
10// fractional diving, the feasibility pump when nothing was found yet, RENS,
11// RINS, feasibility jump with an objective cutoff row) and reduced-cost fixing.
12// RINS runs again in the tree.
13//
14// Tree: a node is the list of bound changes from the root. Processing one
15// applies them to the global bounds, propagates (propagate.h), and re-solves
16// the LP from the parent's basis with DualSimplex::resolve (dual feasible under
17// bound changes, so no phase 1). Nodes whose LP bound reaches the cutoff are
18// pruned; the cutoff is the incumbent less the gap tolerance, and incumbent - 1
19// when every solution has an integral objective. Branching: reliability
20// pseudocost - candidates with fewer than opt.reliability samples are strong
21// branched (both children, iteration-capped dual simplex); the product score of
22// the down and up gains picks the variable, and a child strong branching proves
23// infeasible is not created. Node selection: best bound, with depth-first
24// plunging into the child the pseudocosts favor while its bound stays close to
25// the best.
26//
27// GPU (opt.gpu, AXOS_ENABLE_CUDA; milp_cuda.h): the root probing runs on the
28// GPU (many probes per batch) and feasibility-jump walkers run on the GPU in a
29// second host thread next to the CPU feasibility jump and the root LP. For
30// models with at least opt.gpu_lp_min_nnz nonzeros the root LP is first solved
31// approximately by HPR on the GPU (tolerance 1e-4), and its point feeds
32// rounding, the GPU walkers and RENS. For LPs with at least opt.gpu_min_nnz
33// nonzeros the unreliable branching candidates are evaluated together by
34// batched PDHG, whose Lagrangian bounds are valid child bounds (pruning,
35// scores, pseudocosts). The tree itself stays on the CPU.

75 BranchAndBound(const LpProblem &p, const MilpOptions &opt,
76 std::chrono::steady_clock::time_point t0)
77 : p_(p), lp_(p), opt_(opt), prop_(p), fj_(p), cmir_(p), t0_(t0)

80 glb_ = p.col_lb;
81 gub_ = p.col_ub;

83 pc_sum_[d].assign(n, 0.0);
84 pc_n_[d].assign(n, 0);

90 if (p.c[j] == 0) continue;
91 if (!is_int(p, j) || p.c[j] != std::round(p.c[j]))
92 obj_integral_ = false;
93 }
94 sopt_.scaling = true;
95 sopt_.max_iterations = 50000000;
96 sopt_.verbose = opt.verbose >= 3;
97 }

103 MilpSolution
104 solve()
105 {
106 MilpSolution out;
107 const size_t n = p_.cols();
108 // ---- root
109 // ------------------------------------------------------------
110 gpu_init();
111 if (!prop_.propagate(glb_, gub_))
112 return finish(out, Status::Infeasible);
113 for (size_t j = 0; j < n; ++j)
114 if (glb_[j] > gub_[j]) return finish(out, Status::Infeasible);
115 if (opt_.probing && !root_probing())
116 return finish(out, Status::Infeasible);
117 if (opt_.heuristics) {
118 // launch GPU walker separately: races the CPU feasibility jump and
119 // keeps going during the root LP
120 start_gpu_fj(std::min(0.2 * opt_.time_limit, 60.0));
121 run_fj(nullptr, std::min(opt_.fj_effort * opt_.time_limit, 10.0),
122 "feasibility jump");
123 if (!best_x_.empty()) stop_gpu_fj();
124 }
125
126 LpSolution gpu_lp; // approximate root LP from the GPU (crossover start)
127 const bool have_gpu_lp = gpu_root_lp(gpu_lp);
128 set_time();
129 SimplexBasis basis;
130 LpSolution root = have_gpu_lp
131 ? spx_.solve(lp_, sopt_, nullptr, &basis, &gpu_lp)
132 : spx_.solve(lp_, sopt_, nullptr, &basis);
133 if (have_gpu_lp && root.status != Status::Optimal &&
134 root.status != Status::Infeasible &&
135 root.status != Status::Unbounded && !time_up()) {
136 set_time(); // crossover failed: from scratch
137 root = spx_.solve(lp_, sopt_, nullptr, &basis);
138 }
139 lp_iters_ += root.iterations;
140 if (root.status == Status::Infeasible)
141 return finish(out, Status::Infeasible);
142 if (root.status == Status::Unbounded)
143 return finish(
144 out, best_x_.empty() ? Status::Unbounded : Status::Optimal);
145 if (root.status != Status::Optimal)
146 return finish(out, Status::TimeLimit);
147 // the root LP: bounds from the global domain after propagation
148 root = spx_.resolve(glb_, gub_, &basis, &basis);
149 lp_iters_ += root.iterations;
150 if (root.status == Status::Infeasible)
151 return finish(out, Status::Infeasible);
152 if (root.status != Status::Optimal)
153 return finish(out, Status::TimeLimit);
154 if (opt_.verbose)
155 std::printf("[milp] root LP %.10g (%ld iterations, %.2f s)\n",
156 root.primal_objective, root.iterations, elapsed());
157
158 if (opt_.cuts) {
159 root = cut_loop(root, basis);
160 if (root.status == Status::Infeasible)
161 return finish(out, Status::Infeasible);
162 drop_inactive_cuts(root, basis);
163 }
164 out.root_bound = root.primal_objective;
165 out.cuts = ncuts_;
166 poll_gpu_fj();
167 gpu_batch_lp_init();
168 // reduced-cost fixing needs the simplex at the root LP: now, and again
169 // after the heuristics if they improved the incumbent
170 reduced_cost_fixing(root.primal_objective);
171 const double inc0 = best_obj_;
172 if (opt_.heuristics && !time_up()) {
173 lp_heuristics(root.x, glb_, gub_, basis);
174 if (!time_up()) dive(root.x, glb_, gub_, basis, 50);
175 if (best_x_.empty() && !time_up())
176 feasibility_pump(
177 root.x, basis, std::min(0.1 * opt_.time_limit, 30.0));
178 if (!time_up()) sub_mip(root.x, true);
179 if (!time_up()) rins(root.x);
180 if (!time_up())
181 fj_improve(root.x, std::min(0.08 * opt_.time_limit, 8.0));
182 }
183 if (best_obj_ < inc0 && !time_up()) {
184 set_time();
185 LpSolution r2 = spx_.resolve(glb_, gub_, &basis, &basis);
186 lp_iters_ += r2.iterations;
187 if (r2.status == Status::Optimal) {
188 root = r2;
189 reduced_cost_fixing(root.primal_objective);
190 }
191 }
192
193 if (opt_.verbose)
194 std::printf("[milp] root done: bound %.10g, %d cuts (%d c-MIR), "
195 "%ld LP iterations, %.2f s\n",
196 root.primal_objective, ncuts_, nmir_, lp_iters_, elapsed());
197
198 // ---- tree
199 // ------------------------------------------------------------
200 auto root_basis =
201 std::make_shared<const std::vector<VarStatus>>(basis.status);
202 Node rn;
203 rn.bound = root.primal_objective;
204 rn.basis = root_basis;
205 rn.x = std::make_shared<const std::vector<double>>(root.x);
206 rn.solved = true;
207 double root_obj = root.primal_objective;
208 // the root's LP is solved: branch it right away
209 branch(rn, root_obj, root.x, &root.y, glb_, gub_, basis);
210 long last_print = 0;
211 while (!open_.empty() || plunge_) {
212 if (time_up()) break;
213 if (nodes_ >= opt_.node_limit) break;
214 Node node;
215 if (plunge_) {
216 node = std::move(*plunge_);
217 plunge_.reset();
218 } else {
219 node = open_.top();
220 open_.pop();
221 }
222 if (node.bound >= cutoff()) continue;
223 process(node);
224 ++nodes_;
225 if ((nodes_ & 63) == 0) poll_gpu_fj();
226 if (opt_.heuristics && nodes_ % 200 == 0 && !time_up() && last_x_)
227 dive(*last_x_, last_lb_, last_ub_, last_basis_, 30);
228 if (opt_.heuristics && nodes_ >= next_rins_ && !time_up() &&
229 last_x_) {
230 bool ok = rins(*last_x_);
231 if (!ok && !time_up() && fj_time_ < 0.15 * opt_.time_limit) {
232 const double b = elapsed();
233 const double inc = best_obj_;
234 fj_improve(*last_x_, std::min(0.02 * opt_.time_limit, 2.0));
235 fj_time_ += elapsed() - b;
236 ok = best_obj_ < inc;
237 }
238 rins_fail_ = ok ? 0 : rins_fail_ + 1;
239 next_rins_ = nodes_ + (500L << std::min(rins_fail_, 6));
240 }
241 const double gb = global_bound();
242 if (!best_x_.empty() && closed(best_obj_, gb)) break;
243 if (opt_.verbose && nodes_ - last_print >= 1000) {
244 last_print = nodes_;
245 std::printf("[milp] %7ld nodes %7zu open best %.10g bound "
246 "%.10g %.1f s\n",
247 nodes_, open_.size(), best_obj_, gb, elapsed());
248 }
249 }
250 // optimal: the gap is closed (the tree may still hold nodes that cannot
251 // improve the incumbent by more than the tolerance); an exhausted tree
252 // with a lost node (LP failure) proves nothing beyond its bound
253 Status st;
254 if (!best_x_.empty() && closed(best_obj_, global_bound()))
255 st = Status::Optimal;
256 else if (open_.empty() && !plunge_)
257 st = best_x_.empty() && lost_bound_ == kInf
258 ? Status::Infeasible
259 : Status::NumericalError;
260 else
261 st = nodes_ >= opt_.node_limit ? Status::IterationLimit
262 : Status::TimeLimit;
263 return finish(out, st);
264 }

345 bool
346 closed(double primal, double dual) const
347 {
348 const double d = primal - dual;
349 return d <= opt_.gap_abs ||
350 d <= opt_.gap_rel * std::max(std::abs(primal), 1e-9);
351 }
352 // nodes with a bound at or above this cannot improve the incumbent enough
353 double
354 cutoff() const
355 {
356 if (best_x_.empty()) return kInf;
357 const double tol =
358 std::max(opt_.gap_abs, opt_.gap_rel * std::abs(best_obj_));
359 if (obj_integral_) return best_obj_ - 1.0 + 1e-6;
360 return best_obj_ - tol;
361 }
362 double
363 node_bound(double lp_obj) const
364 { return obj_integral_ ? std::ceil(lp_obj - 1e-6) : lp_obj; }
365 double
366 global_bound() const
367 {
368 double b = best_x_.empty() ? kInf : best_obj_;
369 if (!open_.empty()) b = std::min(b, open_.top().bound);
370 if (plunge_) b = std::min(b, plunge_->bound);
371 return std::min(b, lost_bound_);
372 }

374 MilpSolution
375 finish(MilpSolution &out, Status st)
376 {
377 stop_gpu_fj();
378 out.notes = notes_;
379#ifdef AXOS_ENABLE_CUDA
380 if (gpu_) {
381 out.notes += ";gpu_sb_batches=" + std::to_string(gpu_sb_calls_);
382 if (fj_moves_ >= 0)
383 out.notes += ";gpu_fj_moves=" + std::to_string(fj_moves_);
384 }
385#endif
386 out.status = st;
387 out.nodes = nodes_;
388 out.lp_iterations = lp_iters_;
389 out.seconds = elapsed();
390 out.first_solution_seconds = first_time_;
391 out.cuts = ncuts_;
392 if (!best_x_.empty()) {
393 out.x = best_x_;
394 out.objective = best_obj_;
395 out.incumbent_source = best_src_;
396 }
397 if (st == Status::Infeasible)
398 out.bound = kInf;
399 else
400 out.bound = std::min(global_bound(), out.objective);
401 return out;
402 }

431 // Double probing on the binaries (GPU batches, else CPU); false when it
432 // proves the problem infeasible.
433 bool
434 root_probing()
435 {
436 const double budget =
437 std::min(opt_.probing_time_frac * opt_.time_limit, 10.0);
438 int probes = 0, fixed = 0, tight = 0;
439 bool infeasible = false;
440 double secs = 0;
441 const char *dev = "cpu";
442#ifdef AXOS_ENABLE_CUDA
443 if (gpu_) {
444 try {
445 const gpu::ProbeResult r =
446 gpu::probe_binaries(*dev_, p_, glb_, gub_, budget);
447 probes = r.probes;
448 fixed = r.fixed;
449 tight = r.tightened;
450 infeasible = r.infeasible;
451 secs = r.seconds;
452 dev = "gpu";
453 } catch (const std::exception &e) {
454 if (opt_.verbose)
455 std::printf("[milp] GPU probing failed: %s\n", e.what());
456 gpu_ = false;
457 }
458 }
459#endif
460 if (std::string(dev) == "cpu") {
461 const ProbingStats r =
462 probe_binaries_cpu(prop_, p_, glb_, gub_, budget);
463 probes = r.probes;
464 fixed = r.fixed;
465 tight = r.tightened;
466 infeasible = r.infeasible;
467 secs = r.seconds;
468 }
469 notes_ += std::string(notes_.empty() ? "" : ";") + "probing_" + dev +
470 "=" + std::to_string(probes) + "/" + std::to_string(fixed) +
471 "/" + std::to_string(tight);
472 if (opt_.verbose)
473 std::printf("[milp] probing (%s): %d probes, %d binaries fixed, %d "
474 "bounds tightened, %.3f s\n",
475 dev, probes, fixed, tight, secs);
476 if (infeasible) return false;
477 if (fixed + tight > 0 && !prop_.propagate(glb_, gub_)) return false;
478 for (size_t j = 0; j < p_.cols(); ++j)
479 if (glb_[j] > gub_[j]) return false;
480 return true;
481 }

483 void
484 start_gpu_fj(double seconds, std::vector<std::vector<double>> starts = {})
485 {
486#ifdef AXOS_ENABLE_CUDA
487 if (!gpu_ || fj_thread_.joinable()) return;
488 std::vector<uint8_t> isint(p_.cols());
489 for (size_t j = 0; j < p_.cols(); ++j)
490 isint[j] = is_int(p_, j) ? 1 : 0;
491 fj_stop_ = false;
492 fj_thread_ =
493 std::thread([this, lb = glb_, ub = gub_, isint, seconds, starts]() {
494 try {
495 gpu::GpuFeasibilityJump g(*dev_, opt_.gpu_fj_walkers);
496 std::vector<double> x;
497 long long moves = 0;
498 const bool ok = g.run(lb, ub, starts, isint, seconds, 12345,
499 x, &moves, &fj_stop_);
500 std::lock_guard<std::mutex> lk(fj_mu_);
501 fj_moves_ = std::max<long long>(fj_moves_, 0) + moves;
502 if (ok) {
503 fj_x_ = x;
504 fj_found_ = true;
505 }
506 } catch (const std::exception &) {
507 }
508 });
509#else
510 (void)seconds;
511 (void)starts;
512#endif
513 }

543 // The root LP relaxation by HPR on the GPU (tolerance 1e-4), for models
544 // with at least opt.gpu_lp_min_nnz nonzeros. Its point is rounded, starts
545 // the GPU feasibility-jump walkers (when there is no incumbent yet) and
546 // RENS; returns true with out = (x, y) when it converged, as a crossover
547 // start for the dual simplex.
548 bool
549 gpu_root_lp(LpSolution &out)
550 {
551#ifdef AXOS_ENABLE_CUDA
552 if (!gpu_ || static_cast<long>(p_.A.nnz()) < opt_.gpu_lp_min_nnz ||
553 time_up())
554 return false;
555 const size_t n = p_.cols();
556 QpProblem q;
557 q.lp = p_;
558 q.lp.col_lb = glb_;
559 q.lp.col_ub = gub_;
560 q.lp.is_integer.clear();
561 q.Q = HostMatrix(n, n, std::vector<int32_t>(n + 1, 0),
562 std::vector<int32_t>(), std::vector<double>());
563 QpOptions qo;
564 qo.method = QpMethod::HprQp;
565 qo.use_gpu = true;
566 qo.tol = 1e-4;
567 qo.time_limit = std::min(0.1 * opt_.time_limit,
568 std::max(1.0, opt_.time_limit - elapsed() - 1.0));
569 stop_gpu_fj(); // the walkers would share the GPU with the LP
570 const double t = elapsed();
571 QpSolution s;
572 try {
573 s = solve_qp(q, qo);
574 } catch (const std::exception &e) {
575 if (opt_.verbose)
576 std::printf("[milp] GPU LP failed: %s\n", e.what());
577 return false;
578 }
579 notes_ += ";gpu_lp=" + std::string(to_string(s.status)) + "/" +
580 std::to_string(elapsed() - t).substr(0, 5) + "s";
581 if (opt_.verbose)
582 std::printf("[milp] GPU LP (HPR): %s, objective %.10g, %ld "
583 "iterations, %.2f s\n",
584 to_string(s.status), s.primal_objective, s.iterations,
585 elapsed() - t);
586 if (s.x.size() != n || s.y.size() != p_.rows()) return false;
587 if (opt_.heuristics) {
588 try_solution(s.x, "GPU LP rounding");
589 if (best_x_.empty())
590 start_gpu_fj(std::min(0.2 * opt_.time_limit, 60.0), {s.x});
591 if (!time_up()) sub_mip(s.x, true); // RENS around the GPU LP point
592 }
593 // crossover from this point measured slower than the cold dual simplex
594 // (academictimetablesmall 10 s vs 3.4 s, neos-950242 4.9 s vs 2.4 s):
595 // the point only seeds the heuristics
596 if (s.status != Status::Optimal || !opt_.gpu_lp_crossover) return false;
597 out = LpSolution();
598 out.x = s.x;
599 out.y = s.y;
600 out.status = s.status;
601 return true;
602#else
603 (void)out;
604 return false;
605#endif
606 }

625 bool
626 try_solution(const std::vector<double> &x, const char *src)
627 {
628 std::vector<double> xr = x;
629 for (size_t j = 0; j < xr.size(); ++j)
630 if (is_int(p_, j)) xr[j] = std::round(xr[j]);
631 const MilpCheck c = check_milp(p_, xr);
632 if (!c.ok(opt_.feas_tol, opt_.int_tol)) return false;
633 if (c.objective >= best_obj_ - 1e-9 * (1 + std::abs(best_obj_)))
634 return false;
635 best_obj_ = c.objective;
636 best_x_ = xr;
637 best_src_ = src;
638 if (first_time_ < 0) first_time_ = elapsed();
639 if (opt_.verbose)
640 std::printf("[milp] incumbent %.10g (%s, %.2f s, %ld nodes)\n",
641 best_obj_, src, elapsed(), nodes_);
642 return true;
643 }

652 // ---- root cuts
653 // -------------------------------------------------------------
654 LpSolution
655 cut_loop(LpSolution root, SimplexBasis &basis)
656 {
657 const size_t n = p_.cols();
658 std::vector<int> tidx, touched;
659 std::vector<double> tcoef, dense(n, 0.0);
660 double last = root.primal_objective;
661 int stall = 0;
662 const double t_end =
663 elapsed() + std::max(0.5, opt_.cut_time_frac * opt_.time_limit);
664 for (int round = 0;
665 round < opt_.cut_rounds && !time_up() && elapsed() < t_end;
666 ++round) {
667 std::vector<Cut> cuts;
668 Cut cut;
669 const int m = spx_.num_rows();
670 // fractional basic integers, most fractional first
671 std::vector<std::pair<double, int>> rows;
672 for (int r = 0; r < m; ++r) {
673 const int k = spx_.basic_at(r);
674 if (k >= static_cast<int>(n) || !is_int(lp_, k)) continue;
675 const double v = spx_.value_of(k), f = v - std::floor(v);
676 if (f > 0.01 && f < 0.99)
677 rows.emplace_back(-std::min(f, 1 - f), r);
678 }
679 std::sort(rows.begin(), rows.end());
680 for (const auto &pr : rows) {
681 if (cuts.size() >= 100) break;
682 if (gmi_cut(spx_, lp_, pr.second, glb_, gub_, root.x, cut, tidx,
683 tcoef, dense, touched))
684 cuts.push_back(cut);
685 }
686 const size_t ngmi = cuts.size();
687 {
688 const double left = std::max(0.0, t_end - elapsed());
689 const auto deadline = std::chrono::steady_clock::now() +
690 std::chrono::microseconds(
691 static_cast<long long>(0.5e6 * left));
692 cmir_.separate(root.x, glb_, gub_, cuts, 100, deadline);
693 }
694 const size_t nmir = cuts.size() - ngmi;
695 select_cuts(cuts, 150, dense);
696 nmir_ += static_cast<int>(nmir);
697 if (opt_.verbose >= 2)
698 std::printf(
699 "[milp] separated %zu GMI + %zu c-MIR, kept %zu\n", ngmi,
700 nmir, cuts.size());
701 if (cuts.empty()) break;
702 append_cuts(lp_, cuts);
703 ncuts_ += static_cast<int>(cuts.size());
704 // warm start: the old basis plus the new rows' slacks basic
705 for (size_t c = 0; c < cuts.size(); ++c)
706 basis.status.push_back(VarStatus::Basic);
707 set_time();
708 LpSolution s = spx_.solve(lp_, sopt_, &basis, &basis);
709 lp_iters_ += s.iterations;
710 if (s.status != Status::Optimal) break;
711 s = spx_.resolve(glb_, gub_, &basis, &basis);
712 lp_iters_ += s.iterations;
713 if (s.status == Status::Infeasible) {
714 root = s;
715 break;
716 }
717 if (s.status != Status::Optimal) break;
718 root = s;
719 if (opt_.verbose)
720 std::printf("[milp] cut round %d: %zu GMI cuts, LP %.10g (%ld "
721 "iterations, %.2f s)\n",
722 round + 1, cuts.size(), root.primal_objective, s.iterations,
723 elapsed());
724 const double gain = root.primal_objective - last;
725 last = root.primal_objective;
726 // a round counts as stalled below 0.1% of the objective (not the
727 // gap to the incumbent: an early incumbent can be far off)
728 const double min_gain = 1e-3 * std::max(1.0, std::abs(last));
729 if (gain <= min_gain) {
730 if (++stall >= 2) break;
731 } else {
732 stall = 0;
733 }
734 if (ncuts_ > 4 * static_cast<int>(p_.rows()) + 1000) break;
735 }
736 return root;
737 }

739 // Keeps at most max_keep cuts, most efficacious first, skipping cuts nearly
740 // parallel (cosine > 0.98) to one already kept. dense: scratch (n), zero.
741 static void
742 select_cuts(
743 std::vector<Cut> &cuts, size_t max_keep, std::vector<double> &dense)
744 {
745 std::sort(cuts.begin(), cuts.end(),
746 [](const Cut &a, const Cut &b) { return a.efficacy > b.efficacy; });
747 std::vector<Cut> kept;
748 std::vector<double> norm;
749 for (Cut &c : cuts) {
750 if (kept.size() >= max_keep) break;
751 double nc = 0;
752 for (size_t t = 0; t < c.idx.size(); ++t) {
753 dense[c.idx[t]] = c.coef[t];
754 nc += c.coef[t] * c.coef[t];
755 }
756 nc = std::sqrt(nc);
757 bool parallel = false;
758 const size_t from = kept.size() > 60 ? kept.size() - 60 : 0;
759 for (size_t k = from; k < kept.size() && !parallel; ++k) {
760 double dot = 0;
761 const Cut &o = kept[k];
762 for (size_t t = 0; t < o.idx.size(); ++t)
763 dot += dense[o.idx[t]] * o.coef[t];
764 parallel = dot > 0.98 * nc * norm[k];
765 }
766 for (int j : c.idx)
767 dense[j] = 0;
768 if (parallel) continue;
769 kept.push_back(std::move(c));
770 norm.push_back(nc);
771 }
772 cuts.swap(kept);
773 }

775 // Removes the cuts whose slack is basic at the root optimum: the basis
776 // stays optimal for the smaller LP (those rows have zero duals), and every
777 // node re-solves the smaller one. Restores the full LP if the re-solve
778 // fails.
779 void
780 drop_inactive_cuts(LpSolution &root, SimplexBasis &basis)
781 {
782 const size_t m0 = p_.rows(), m = lp_.rows(), n = lp_.cols();
783 if (m == m0 || basis.status.size() != n + m) return;
784 std::vector<char> keep(m, 1);
785 int dropped = 0;
786 for (size_t i = m0; i < m; ++i)
787 if (basis.status[n + i] == VarStatus::Basic) {
788 keep[i] = 0;
789 ++dropped;
790 }
791 if (!dropped) return;
792 const LpProblem full = lp_;
793 const SimplexBasis full_basis = basis;
794 remove_rows(lp_, keep);
795 SimplexBasis b;
796 b.status.assign(basis.status.begin(), basis.status.begin() + n);
797 for (size_t i = 0; i < m; ++i)
798 if (keep[i]) b.status.push_back(basis.status[n + i]);
799 set_time();
800 LpSolution s = spx_.solve(lp_, sopt_, &b, &b);
801 lp_iters_ += s.iterations;
802 if (s.status == Status::Optimal) {
803 s = spx_.resolve(glb_, gub_, &b, &b);
804 lp_iters_ += s.iterations;
805 }
806 if (s.status == Status::Optimal) {
807 if (opt_.verbose)
808 std::printf(
809 "[milp] dropped %d inactive cuts, %zu left, LP %.10g\n",
810 dropped, lp_.rows() - m0, s.primal_objective);
811 root = s;
812 basis = b;
813 return;
814 }
815 lp_ = full; // keep everything
816 basis = full_basis;
817 set_time();
818 LpSolution r = spx_.solve(lp_, sopt_, &basis, &basis);
819 lp_iters_ += r.iterations;
820 if (r.status == Status::Optimal)
821 r = spx_.resolve(glb_, gub_, &basis, &basis);
822 if (r.status == Status::Optimal) root = r;
823 }

825 // ---- LP-based heuristics
826 // ---------------------------------------------------- Fix the integers (at
827 // given values) and solve the LP of the continuous part; an integer-only
828 // problem only needs the feasibility check.
829 bool
830 repair(const std::vector<double> &lb0, const std::vector<double> &ub0,
831 const SimplexBasis &warm, const char *src)
832 {
833 bool cont = false;
834 for (size_t j = 0; j < p_.cols() && !cont; ++j)
835 if (!is_int(p_, j) && lb0[j] < ub0[j]) cont = true;
836 if (!cont) {
837 std::vector<double> x(lb0);
838 return try_solution(x, src);
839 }
840 set_time();
841 LpSolution s = spx_.resolve(lb0, ub0, &warm, nullptr);
842 lp_iters_ += s.iterations;
843 if (s.status != Status::Optimal) return false;
844 return try_solution(s.x, src);
845 }

847 void
848 lp_heuristics(const std::vector<double> &x, const std::vector<double> &lb,
849 const std::vector<double> &ub, const SimplexBasis &warm)
850 {
851 const size_t n = p_.cols();
852 // 1. simple rounding + LP repair
853 {
854 std::vector<double> l = lb, u = ub;
855 bool ok = true;
856 for (size_t j = 0; j < n && ok; ++j)
857 if (is_int(p_, j)) {
858 const double v =
859 std::min(std::max(std::round(x[j]), l[j]), u[j]);
860 l[j] = u[j] = v;
861 }
862 if (ok && prop_.propagate(l, u)) repair(l, u, warm, "rounding");
863 }
864 if (time_up()) return;
865 // 2. fix-and-propagate, most decided integers first
866 {
867 std::vector<std::pair<double, int>> ord;
868 for (size_t j = 0; j < n; ++j)
869 if (is_int(p_, j))
870 ord.emplace_back(
871 std::abs(x[j] - std::round(x[j])), static_cast<int>(j));
872 std::sort(ord.begin(), ord.end());
873 std::vector<int> order;
874 for (auto &pr : ord)
875 order.push_back(pr.second);
876 std::vector<double> l = lb, u = ub;
877 if (fix_and_propagate(
878 prop_, p_, order, x, l, u, opt_.time_limit, t0_))
879 repair(l, u, warm, "fix-and-propagate");
880 }
881 if (time_up()) return;
882 // 3. feasibility jump from the LP point
883 if (best_x_.empty())
884 run_fj(&x, std::min(0.02 * opt_.time_limit, 5.0),
885 "feasibility jump (LP start)");
886 }

888 // Fractional diving: fix the least fractional integer to its rounding,
889 // propagate, re-solve; one backtrack per level.
890 void
891 dive(const std::vector<double> &x0, const std::vector<double> &lb0,
892 const std::vector<double> &ub0, const SimplexBasis &warm0,
893 int max_depth)
894 {
895 std::vector<double> x = x0, lb = lb0, ub = ub0;
896 SimplexBasis b = warm0;
897 std::vector<int> changed(1);
898 for (int depth = 0; depth < max_depth && !time_up(); ++depth) {
899 int jb = -1;
900 double fb = 2;
901 for (size_t j = 0; j < x.size(); ++j) {
902 if (!is_int(p_, j) || lb[j] == ub[j]) continue;
903 const double f = std::abs(x[j] - std::round(x[j]));
904 if (f > opt_.int_tol && f < fb) {
905 fb = f;
906 jb = static_cast<int>(j);
907 }
908 }
909 if (jb < 0) {
910 try_solution(x, "diving");
911 return;
912 }
913 bool ok = false;
914 for (int attempt = 0; attempt < 2 && !ok; ++attempt) {
915 const double r = std::round(x[jb]);
916 const double v = attempt == 0 ? r : (x[jb] > r ? r + 1 : r - 1);
917 if (v < lb[jb] || v > ub[jb]) continue;
918 std::vector<double> l = lb, u = ub;
919 l[jb] = u[jb] = v;
920 changed[0] = jb;
921 if (!prop_.propagate(l, u, &changed, nullptr, 2.0)) continue;
922 set_time();
923 SimplexBasis nb;
924 LpSolution s = spx_.resolve(l, u, &b, &nb, 2000);
925 lp_iters_ += s.iterations;
926 if (s.status != Status::Optimal ||
927 s.primal_objective >= cutoff())
928 continue;
929 x = s.x;
930 lb.swap(l);
931 ub.swap(u);
932 b = std::move(nb);
933 ok = true;
934 }
935 if (!ok) return;
936 }

939 // Objective feasibility pump (M. Fischetti, F. Glover, A. Lodi, Math. Prog.
940 // 104 (2005); T. Achterberg, T. Berthold, Discrete Optimization 4 (2007)):
941 // round the integers of the LP point, then find the LP point closest (L1)
942 // to that rounding, with the original objective mixed in at a weight
943 // decaying by 0.9 per round, until the LP point is integral. The distance
944 // is linear in the binaries and in general integers at a bound; others are
945 // left out. A rounding met again flips the T integers farthest from it (T
946 // random in [10, 30)); a longer cycle perturbs at random. Each LP is
947 // warm-started by the primal simplex (only the costs change). Integral LP
948 // points go through try_solution.
949 bool
950 feasibility_pump(const std::vector<double> &x0, const SimplexBasis &basis0,
951 double seconds)
952 {
953 const size_t n = p_.cols();
954 std::vector<int> ints;
955 for (size_t j = 0; j < n; ++j)
956 if (is_int(p_, j) && glb_[j] < gub_[j])
957 ints.push_back(static_cast<int>(j));
958 if (ints.empty()) return false;
959 const double t0 = elapsed(),
960 deadline = std::min(t0 + seconds, opt_.time_limit);
961 double cn = 0;
962 for (size_t j = 0; j < n; ++j)
963 cn += p_.c[j] * p_.c[j];
964 cn = std::sqrt(cn);
965 const double dn = std::sqrt(static_cast<double>(ints.size()));
966 std::vector<double> x = x0, xr(n, 0.0), prev(n, kInf), c(n, 0.0);
967 std::vector<uint64_t> seen;
968 SimplexBasis b = basis0;
969 double alpha = 1.0;
970 std::mt19937_64 rng(seed_++);
971 bool found = false;
972 int it = 0;
973 for (; it < 500 && elapsed() < deadline && !found; ++it) {
974 // round, and stop at an integral LP point
975 double frac = 0;
976 for (int j : ints) {
977 xr[j] = std::min(std::max(std::round(x[j]), glb_[j]), gub_[j]);
978 frac += std::abs(x[j] - xr[j]);
979 }
980 if (frac < 1e-6 * ints.size()) {
981 found = try_solution(x, "feasibility pump");
982 break;
983 }
984 // cycles: the same rounding as last round, or one seen before
985 bool same = true;
986 for (int j : ints)
987 same = same && xr[j] == prev[j];
988 if (same) {
989 std::vector<std::pair<double, int>> away;
990 for (int j : ints)
991 away.emplace_back(-std::abs(x[j] - xr[j]), j);
992 std::sort(away.begin(), away.end());
993 const int T = 10 + static_cast<int>(rng() % 20);
994 for (int t = 0; t < T && t < static_cast<int>(away.size());
995 ++t) {
996 const int j = away[t].second;
997 if (away[t].first == 0) break;
998 xr[j] = xr[j] > x[j] ? xr[j] - 1 : xr[j] + 1;
999 xr[j] = std::min(std::max(xr[j], glb_[j]), gub_[j]);
1000 }
1001 } else {
1002 uint64_t h = 1469598103934665603ull;
1003 for (int j : ints)
1004 h = (h ^ static_cast<uint64_t>(xr[j] + 7)) *
1005 1099511628211ull;
1006 if (std::find(seen.begin(), seen.end(), h) != seen.end()) {
1007 std::uniform_real_distribution<double> u(-0.3, 0.7);
1008 for (int j : ints) {
1009 const double r =
1010 std::abs(x[j] - xr[j]) + std::max(u(rng), 0.0);
1011 if (r > 0.5)
1012 xr[j] = std::min(
1013 std::max(xr[j] > x[j] ? xr[j] - 1 : xr[j] + 1,
1014 glb_[j]),
1015 gub_[j]);
1016 }
1017 }
1018 seen.push_back(h);
1019 if (seen.size() > 100) seen.erase(seen.begin());
1020 }
1021 for (int j : ints)
1022 prev[j] = xr[j];
1023 // the distance objective (plus the decaying original objective)
1024 alpha *= 0.9;
1025 std::fill(c.begin(), c.end(), 0.0);
1026 for (int j : ints) {
1027 if (xr[j] <= glb_[j])
1028 c[j] = 1.0;
1029 else if (xr[j] >= gub_[j])
1030 c[j] = -1.0;
1031 }
1032 if (alpha > 1e-3 && cn > 0)
1033 for (size_t j = 0; j < n; ++j)
1034 c[j] = (1 - alpha) * c[j] + alpha * dn / cn * p_.c[j];
1035 set_time();
1036 LpSolution s = spx_.reoptimize(c, glb_, gub_, &b, &b, 20000);
1037 lp_iters_ += s.iterations;
1038 if (s.status != Status::Optimal) break;
1039 x = s.x;
1040 }
1041 spx_.restore_costs();
1042 if (opt_.verbose)
1043 std::printf("[milp] feasibility pump: %s after %d rounds, %.2f s\n",
1044 found ? "solution" : "nothing", it, elapsed() - t0);
1045 return found;
1046 }

1048 // Feasibility jump as an improvement heuristic: the model plus the cutoff
1049 // row c^T x <= incumbent - delta, searched from the LP point x; every
1050 // improving point tightens the cutoff and the search goes on (good at
1051 // moving a poor incumbent when too few integers agree with the LP point for
1052 // RINS).
1053 void
1054 fj_improve(const std::vector<double> &x, double seconds)
1055 {
1056 if (best_x_.empty() || !opt_.heuristics) return;
1057 const size_t n = p_.cols();
1058 Cut obj;
1059 for (size_t j = 0; j < n; ++j)
1060 if (p_.c[j] != 0) {
1061 obj.idx.push_back(static_cast<int>(j));
1062 obj.coef.push_back(-p_.c[j]);
1063 }
1064 if (obj.idx.empty()) return;
1065 const double t_end = std::min(elapsed() + seconds, opt_.time_limit);
1066 int found = 0;
1067 while (elapsed() < t_end) {
1068 const double delta =
1069 obj_integral_ ? 1.0 - 1e-6
1070 : std::max(opt_.gap_abs,
1071 1e-3 * std::max(1.0, std::abs(best_obj_)));
1072 obj.rhs = -(best_obj_ - delta - p_.offset);
1073 LpProblem pc = p_;
1074 append_cuts(pc, {obj});
1075 FeasibilityJump fjc(pc);
1076 std::vector<double> xs;
1077 if (!fjc.run(glb_, gub_, &x, t_end - elapsed(), xs, seed_++)) break;
1078 if (!try_solution(xs, "feasibility jump (cutoff)")) break;
1079 ++found;
1080 }
1081 if (opt_.verbose >= 2)
1082 std::printf("[milp] FJ with cutoff: %d improvements\n", found);
1083 }

1085 // RINS: fix the integer columns where the LP point x agrees with the
1086 // incumbent and solve the rest as a sub-MIP (presolved, node and time
1087 // limits, no RINS of its own) with a cutoff row c^T x <= incumbent - delta,
1088 // so only an improving solution comes back. Skipped when less than half the
1089 // integers would be fixed; at most a fifth of the time limit in all.
1090 bool
1091 rins(const std::vector<double> &x)
1092 { return sub_mip(x, false); }
1093
1094 // RINS (rens = false) or RENS (rens = true, T. Berthold, Math. Prog. Comp.
1095 // 6 (2014)): RENS fixes the integers integral at the LP point and restricts
1096 // the others to their floor and ceiling (no incumbent needed). With an
1097 // incumbent, a cutoff row keeps only improving solutions.
1098 bool
1099 sub_mip(const std::vector<double> &x, bool rens)
1100 {
1101 if (!opt_.rins || (!rens && best_x_.empty())) return false;
1102 const double budget = std::min(
1103 {0.05 * opt_.time_limit, 10.0, 0.2 * opt_.time_limit - rins_time_,
1104 opt_.time_limit - elapsed() - 0.5});
1105 if (budget < 0.2) return false;
1106 const size_t n = p_.cols();
1107 LpProblem sub = p_;
1108 sub.col_lb = glb_;
1109 sub.col_ub = gub_;
1110 size_t nint = 0, nfix = 0;
1111 for (size_t j = 0; j < n; ++j) {
1112 if (!is_int(p_, j)) continue;
1113 ++nint;
1114 if (glb_[j] == gub_[j]) {
1115 ++nfix;
1116 continue;
1117 }
1118 if (rens) {
1119 const double r = std::round(x[j]);
1120 if (std::abs(x[j] - r) < 1e-6) {
1121 sub.col_lb[j] = sub.col_ub[j] =
1122 std::min(std::max(r, glb_[j]), gub_[j]);
1123 ++nfix;
1124 } else {
1125 sub.col_lb[j] = std::max(glb_[j], std::floor(x[j]));
1126 sub.col_ub[j] = std::min(gub_[j], std::ceil(x[j]));
1127 }
1128 } else if (std::abs(x[j] - best_x_[j]) < 1e-6) {
1129 sub.col_lb[j] = sub.col_ub[j] = best_x_[j];
1130 ++nfix;
1131 }
1132 }
1133 if (nint == 0 || nfix < nint / 2 || nfix == nint) return false;
1134 if (!best_x_.empty()) {
1135 const double delta = obj_integral_
1136 ? 1.0 - 1e-6
1137 : std::max(opt_.gap_abs,
1138 opt_.gap_rel * std::abs(best_obj_));
1139 Cut obj;
1140 for (size_t j = 0; j < n; ++j)
1141 if (p_.c[j] != 0) {
1142 obj.idx.push_back(static_cast<int>(j));
1143 obj.coef.push_back(
1144 -p_.c[j]); // -c^T x >= -(best - delta - offset)
1145 }
1146 if (obj.idx.empty()) return false;
1147 obj.rhs = -(best_obj_ - delta - p_.offset);
1148 append_cuts(sub, {obj});
1149 }
1150 MilpOptions so;
1151 so.time_limit = budget;
1152 so.node_limit = 1000;
1153 so.gap_rel = opt_.gap_rel;
1154 so.gap_abs = opt_.gap_abs;
1155 so.rins = false;
1156 so.probing = false;
1157 so.gpu = false;
1158 so.verbose = 0;
1159 so.cut_rounds = 5;
1160 so.fj_effort = 0.02;
1161 const double t = elapsed();
1162 bool improved = false;
1163 try {
1164 const MilpSolution r = solve_milp(sub, so);
1165 if (r.has_solution())
1166 improved = try_solution(r.x, rens ? "RENS" : "RINS");
1167 } catch (const std::exception &) {
1168 }
1169 rins_time_ += elapsed() - t;
1170 if (opt_.verbose >= 2)
1171 std::printf("[milp] %s: fixed %zu of %zu integers, %s, %.2f s\n",
1172 rens ? "RENS" : "RINS", nfix, nint,
1173 improved ? "improved" : "no improvement", elapsed() - t);
1174 return improved;
1175 }

1177 // ---- reduced-cost fixing at the root -----------------------------------
1178 void
1179 reduced_cost_fixing(double root_obj)
1180 {
1181 if (best_x_.empty()) return;
1182 const double gap = best_obj_ - root_obj;
1183 if (gap <= 0) return;
1184 int fixed = 0;
1185 for (size_t j = 0; j < p_.cols(); ++j) {
1186 if (!is_int(p_, j) || glb_[j] == gub_[j]) continue;

1266 last_ub_ = ub;

1272 }

1320 return std::max(down, 1e-6) * std::max(up, 1e-6);
1321 };

1323 for (int j : frac) {
1324 const double f = x[j] - std::floor(x[j]);

1326 -score_of(pc_avg(0, j) * f, pc_avg(1, j) * (1 - f)), j);
1327 }

1330 double best_score = -cand[0].first;
1331 double best_down = obj,
1332 best_up = obj; // proven child bounds (strong branching)
1333 bool down_inf = false, up_inf = false;
1334 // reliability: strong branch unreliable candidates (a few, best first)
1335 int tried = 0, no_gain = 0;
1336 double sb_best = -1;
1337#ifdef AXOS_ENABLE_CUDA

1339 // up to 16 unreliable candidates (32 children) in one batched PDHG
1340 // run on the GPU; its bounds are valid, so they also prune

1345 if (std::min(pc_n_[0][c.second], pc_n_[1][c.second]) <
1346 opt_.reliability)
1347 sel.push_back(c.second);
1348 }
1349 for (int j : sel) {
1350 col.push_back(j);
1351 lo.push_back(-kInf);
1352 hi.push_back(std::floor(x[j]));
1353 col.push_back(j);
1354 lo.push_back(std::ceil(x[j]));
1355 hi.push_back(kInf);

1361 ++gpu_sb_calls_;

1368 }

1370 const int j = sel[k];

1372 const double od = bd[2 * k], ou = bd[2 * k + 1];
1373 const bool dinf = od > -kInf && node_bound(od) >= cutoff();

1384 pc_n_[1][j] += 1;

1388 best_down = std::max(obj, od);
1389 best_up = std::max(obj, ou);
1390 down_inf = dinf;
1391 up_inf = uinf;

1394 }

1396 if (sc > sb_best) {
1397 sb_best = sc;
1398 best = j;
1399 best_down = std::max(obj, od);
1400 best_up = std::max(obj, ou);
1401 }
1402 }
1403 if (!bd.empty()) tried = 1 << 30; // no CPU strong branching on top
1404 }
1405#endif
1406 for (const auto &c : cand) {
1407 if (tried >= 10 || no_gain >= 4 || time_up()) break;
1408 const int j = c.second;
1409 if (std::min(pc_n_[0][j], pc_n_[1][j]) >= opt_.reliability)
1410 continue;

1423 pc_n_[1][j] += 1;

1427 best_down = dbd;
1428 best_up = ubd;
1429 down_inf = dinf;
1430 up_inf = uinf;

1444 }

1470 const int pref =

1472 const double gb = global_bound();

1474 const int d = k == 0 ? pref : 1 - pref;
1475 if (!keep[d]) continue;
1476 const bool close = best_x_.empty() ||
1477 child[d].bound <= gb + 0.5 * (best_obj_ - gb);
1478 if (k == 0 && !plunge_ && close && child[d].depth < 1000)
1479 plunge_.reset(new Node(std::move(child[d])));
1480 else
1481 open_.push(std::move(child[d]));
1482 }
1483 }

1487 // objective when the child LP was solved to optimality, else -inf (an early
1488 // stop is dual feasible only for the shifted costs, so its objective is an
1489 // estimate, good for scores but not for pruning).
1490 void
1491 strong_branch(int j, double v, const std::vector<double> &lb,
1492 const std::vector<double> &ub, const SimplexBasis &basis, double &od,
1493 double &ou, bool &dinf, bool &uinf, double &dbound, double &ubound)

1496 set_time();

1499 spx_.resolve(l, u, &basis, nullptr, opt_.strong_branch_iters);

1502 od = (s.status == Status::Optimal || s.status == Status::IterationLimit)
1503 ? s.primal_objective
1504 : -kInf;
1505 dbound = s.status == Status::Optimal ? s.primal_objective : -kInf;
1506 if (s.status == Status::Optimal) try_integral(s.x);
1507 u[j] = ub[j];
1508 l[j] = std::ceil(v);
1509 s = spx_.resolve(l, u, &basis, nullptr, opt_.strong_branch_iters);
1510 lp_iters_ += s.iterations;
1511 uinf = s.status == Status::Infeasible;
1512 ou = (s.status == Status::Optimal || s.status == Status::IterationLimit)
1513 ? s.primal_objective
1514 : -kInf;
1515 ubound = s.status == Status::Optimal ? s.primal_objective : -kInf;
1516 if (s.status == Status::Optimal) try_integral(s.x);
1517 if (!dinf && node_bound(dbound) >= cutoff())
1518 dinf = true; // pruned: as good as infeasible
1519 if (!uinf && node_bound(ubound) >= cutoff()) uinf = true;
1520 }
