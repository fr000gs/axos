3// Sparse symmetric direct solver: A = P^T L D L^T P.
4//
5// analyze(A) ordering + symbolic factorization (once per sparsity pattern)
6// factorize(A) numeric factorization (repeat as the values change)
7// solve(b, x) solves A x = b
8//
9// A must be square and symmetric with full storage (both triangles); it is
10// read through row k's entries with permuted column index <= k, so only one
11// triangle's values matter. No numerical pivoting is done: Symmetry::SPD
12// requires all pivots D > 0 (Cholesky-equivalent); Symmetry::Symmetric only
13// requires nonzero pivots, which suffices for quasi-definite systems such as
14// interior-point KKT matrices.
15//
16// This header is the CPU implementation (up-looking LDL^T after T. Davis's
17// LDL, with a minimum-degree ordering). The CUDA specialization on top of
18// cuDSS is in cudss_solver.h.
19#pragma once

31enum class Symmetry { SPD, Symmetric };

33// Fill-reducing orderings. MinDegree is approximate minimum degree (AMD);
34// ExactMinDegree is a slow reference implementation. The cuDSS backend picks
35// its own default unless told otherwise: it maps MinDegree to its AMD,
36// NestedDissection to its nested dissection, and Natural / ExactMinDegree to
37// its default. On the CPU, NestedDissection falls back to MinDegree.
38enum class Ordering { Natural, MinDegree, ExactMinDegree, NestedDissection };
39
40// CPU factorization algorithm. Simplicial is the scalar up-looking LDL^T;
41// Supernodal is the multifrontal code (dense kernels, much faster once the
42// factor has a few hundred thousand entries); Auto picks by factor size.
43enum class Factorization { Auto, Simplicial, Supernodal };
44

46 template <typename> class Store = Cpu::HostStorage>

51// Exact minimum-degree ordering on the graph of A (both triangles). This is
52// a straightforward elimination-graph implementation: fine for the CPU
53// fallback, but its cost grows with the largest cliques it forms.
54template <typename Idx>
55std::vector<Idx>
56min_degree_order(size_t n, const Idx *rp, const Idx *ci)
57{
58 std::vector<std::vector<Idx>> adj(n);
59 for (size_t i = 0; i < n; ++i)
60 for (Idx k = rp[i]; k < rp[i + 1]; ++k) {
61 Idx j = ci[k];
62 if (static_cast<size_t>(j) != i) {
63 adj[i].push_back(j);
64 adj[j].push_back(static_cast<Idx>(i));
65 }
66 }
67 for (auto &a : adj) {
68 std::sort(a.begin(), a.end());
69 a.erase(std::unique(a.begin(), a.end()), a.end());
70 }
71 std::set<std::pair<size_t, Idx>> pq;
72 for (size_t v = 0; v < n; ++v)
73 pq.insert({adj[v].size(), static_cast<Idx>(v)});
74 std::vector<Idx> perm;
75 perm.reserve(n);
76 std::vector<Idx> merged;
77 while (!pq.empty()) {
78 Idx v = pq.begin()->second;
79 pq.erase(pq.begin());
80 perm.push_back(v);
81 std::vector<Idx> nb = std::move(adj[v]);
82 adj[v].clear();
83 for (Idx u : nb) {
84 pq.erase({adj[u].size(), u});
85 // adj[u] = (adj[u] union nb) minus {u, v}
86 merged.clear();
87 std::set_union(adj[u].begin(), adj[u].end(), nb.begin(), nb.end(),
88 std::back_inserter(merged));
89 merged.erase(std::remove_if(merged.begin(), merged.end(),
90 [&](Idx w) { return w == u || w == v; }),
91 merged.end());
92 adj[u] = merged;
93 pq.insert({adj[u].size(), u});
94 }
95 }
96 return perm;
97}

109 explicit SparseLdlt(Symmetry kind = Symmetry::SPD,
110 Ordering ord = Ordering::MinDegree,
111 Factorization fact = Factorization::Auto)
112 : kind_(kind), ord_(ord), fact_(fact)
113 {
114 }

116 // Ordering and symbolic factorization for A's sparsity pattern.
117 void
118 analyze(const matrix_type &A)
119 {
120 if (A.rows() != A.cols())
121 throw std::invalid_argument("SparseLdlt: matrix must be square");
122 n_ = A.rows();
123 nnz_ = A.nnz();
124 const Idx *rp = A.row_ptr(), *ci = A.col_ind();
125
126 perm_.resize(n_);
127 if ((ord_ == Ordering::MinDegree ||
128 ord_ == Ordering::NestedDissection) &&
129 n_ > 0)
130 perm_ = amd_order<Idx>(n_, rp, ci);
131 else if (ord_ == Ordering::ExactMinDegree && n_ > 0)
132 perm_ = detail::min_degree_order<Idx>(n_, rp, ci);
133 else
134 std::iota(perm_.begin(), perm_.end(), Idx(0));
135 pinv_.assign(n_, 0);
136 for (size_t k = 0; k < n_; ++k)
137 pinv_[perm_[k]] = static_cast<Idx>(k);
138
139 parent_.assign(n_, -1);
140 lnz_.assign(n_, 0);
141 flag_.assign(n_, 0);
142 for (size_t k = 0; k < n_; ++k) {
143 parent_[k] = -1;
144 flag_[k] = static_cast<Idx>(k);
145 lnz_[k] = 0;
146 const Idx kk = perm_[k];
147 for (Idx p = rp[kk]; p < rp[kk + 1]; ++p) {
148 Idx i = pinv_[ci[p]];
149 if (static_cast<size_t>(i) < k) {
150 for (; flag_[i] != static_cast<Idx>(k); i = parent_[i]) {
151 if (parent_[i] == -1) parent_[i] = static_cast<Idx>(k);
152 lnz_[i]++;
153 flag_[i] = static_cast<Idx>(k);
154 }
155 }
156 }
157 }
158 size_t total = 0;
159 double flops = 0;
160 for (size_t k = 0; k < n_; ++k) {
161 total += static_cast<size_t>(lnz_[k]);
162 flops +=
163 static_cast<double>(lnz_[k]) * static_cast<double>(lnz_[k]);
164 }
165 factor_flops_ = flops;
166 use_mf_ = fact_ == Factorization::Supernodal ||
167 (fact_ == Factorization::Auto && total > 200000);
168 if (use_mf_) {
169 mf_ = MultifrontalLdl<T, Idx>();
170 mf_.analyze(n_, rp, ci, perm_);
171 lp_.assign(n_ + 1, 0); // simplicial storage is not used
172 li_.clear();
173 lx_.clear();
174 d_.assign(n_, T(0));
175 factor_nnz_ = total;
176 } else {
177 lp_.assign(n_ + 1, 0);
178 for (size_t k = 0; k < n_; ++k)
179 lp_[k + 1] = lp_[k] + lnz_[k];
180 li_.assign(lp_[n_], 0);
181 lx_.assign(lp_[n_], T(0));
182 d_.assign(n_, T(0));
183 factor_nnz_ = total;
184 }
185 analyzed_ = true;
186 factored_ = false;
187 }

189 // Numeric factorization. A must have the pattern given to analyze().
190 // Returns false when a pivot fails (SPD: D <= 0, Symmetric: D == 0);
191 // failed_pivot() then gives the (permuted) index.
192 bool
193 factorize(const matrix_type &A)
194 {
195 if (!analyzed_) analyze(A);
196 if (A.rows() != n_ || A.nnz() != nnz_)
197 throw std::invalid_argument(
198 "SparseLdlt::factorize: pattern differs from analyze()");
199 if (use_mf_) {
200 n_reg_ = 0;
201 factored_ = mf_.factorize(A.values(),
202 kind_ == Symmetry::SPD
203 ? MultifrontalLdl<T, Idx>::Kind::SPD
204 : MultifrontalLdl<T, Idx>::Kind::Symmetric,
205 signs_.empty() ? nullptr : signs_.data(), pivot_eps_);
206 n_reg_ = mf_.regularized_pivots();
207 failed_ = factored_ ? -1 : mf_.failed_pivot();
208 return factored_;
209 }
210 const Idx *rp = A.row_ptr(), *ci = A.col_ind();
211 const T *ax = A.values();
212 std::vector<T> y(n_, T(0));
213 std::vector<Idx> pattern(n_), flag(n_), lnz(n_, 0);
214 failed_ = -1;
215 factored_ = false;
216 n_reg_ = 0;
217 for (size_t k = 0; k < n_; ++k) {
218 y[k] = T(0);
219 size_t top = n_;
220 flag[k] = static_cast<Idx>(k);
221 lnz[k] = 0;
222 const Idx kk = perm_[k];
223 for (Idx p = rp[kk]; p < rp[kk + 1]; ++p) {
224 Idx i = pinv_[ci[p]];
225 if (static_cast<size_t>(i) <= k) {
226 y[i] += ax[p];
227 size_t len = 0;
228 for (; flag[i] != static_cast<Idx>(k); i = parent_[i]) {
229 pattern[len++] = i;
230 flag[i] = static_cast<Idx>(k);
231 }
232 while (len > 0)
233 pattern[--top] = pattern[--len];
234 }
235 }
236 d_[k] = y[k];
237 y[k] = T(0);
238 for (; top < n_; ++top) {
239 const Idx i = pattern[top];
240 const T yi = y[i];
241 y[i] = T(0);
242 const Idx p2 = lp_[i] + lnz[i];
243 for (Idx p = lp_[i]; p < p2; ++p)
244 y[li_[p]] -= lx_[p] * yi;
245 const T lki = yi / d_[i];
246 d_[k] -= lki * yi;
247 li_[p2] = static_cast<Idx>(k);
248 lx_[p2] = lki;
249 lnz[i]++;
250 }
251 if (!signs_.empty() && signs_[kk] != 0) {
252 const T sg = static_cast<T>(signs_[kk]);
253 if (!(sg * d_[k] >= pivot_eps_)) {
254 d_[k] = sg * pivot_eps_;
255 ++n_reg_;
256 }
257 }
258 const bool bad =
259 (kind_ == Symmetry::SPD) ? !(d_[k] > T(0)) : (d_[k] == T(0));
260 if (bad || !std::isfinite(d_[k])) {
261 failed_ = static_cast<long>(k);
262 return false;
263 }
264 }
265 factored_ = true;
266 return true;
267 }

269 // Solves A x = b (host tensors).
270 void
271 solve(const vector_type &b, vector_type &x) const
272 {
273 if (!factored_)
274 throw std::runtime_error("SparseLdlt::solve: not factorized");
275 if (b.size() != n_ || x.size() != n_)
276 throw std::invalid_argument("SparseLdlt::solve: size mismatch");
277 if (use_mf_) {
278 mf_.solve(b.data, x.data); // b and x must not alias
279 return;
280 }
281 std::vector<T> w(n_);
282 for (size_t k = 0; k < n_; ++k)
283 w[k] = b.data[perm_[k]];
284 for (size_t j = 0; j < n_; ++j) // L w = w
285 for (Idx p = lp_[j]; p < lp_[j + 1]; ++p)
286 w[li_[p]] -= lx_[p] * w[j];
287 for (size_t j = 0; j < n_; ++j) // D w = w
288 w[j] /= d_[j];
289 for (size_t j = n_; j-- > 0;) // L^T w = w
290 for (Idx p = lp_[j]; p < lp_[j + 1]; ++p)
291 w[j] -= lx_[p] * w[li_[p]];
292 for (size_t k = 0; k < n_; ++k)
293 x.data[perm_[k]] = w[k];
294 }

296 // Dynamic pivot regularization (Symmetric kind): entry i of `signs` is the
297 // expected sign (+1 / -1, 0 = none) of the pivot of variable i. A pivot
298 // whose value times its sign is below `eps` is replaced by sign * eps, so
299 // only pivots that really need it are perturbed. regularized_pivots()
300 // reports how many were.
301 void
302 set_pivot_regularization(std::vector<signed char> signs, T eps)
303 {
304 signs_ = std::move(signs);
305 pivot_eps_ = eps;
306 }

328 "%.1f store %.1f ms (max front %zu, %zu supernodes)\n",

331 std::printf("[ldl] wall: %zu parallel subtrees %.1f ms, %zu top "
332 "nodes %.1f ms\n",
333 p.ntasks, p.wall_tasks, p.ntop, p.wall_top);
334 }

339 // Counts of positive / negative / zero pivots after factorize().
340 void
341 inertia(size_t &pos, size_t &neg, size_t &zero) const
342 {
343 pos = neg = zero = 0;
344 const std::vector<T> &d = use_mf_ ? mf_.pivots() : d_;
345 for (size_t k = 0; k < n_; ++k) {
346 if (d[k] > T(0))
347 ++pos;
348 else if (d[k] < T(0))
349 ++neg;
350 else
351 ++zero;
352 }
353 }

364 bool analyzed_ = false, factored_ = false;
365 long failed_ = -1;

370 std::vector<T> lx_, d_;
