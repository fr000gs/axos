44 // Round to nearest, ties to even; overflow gives inf, NaN stays NaN.
45 static std::uint16_t
46 from_float(float f)
47 {
48 std::uint32_t x;
49 std::memcpy(&x, &f, 4);
50 const std::uint32_t sign = (x >> 16) & 0x8000u;
51 x &= 0x7fffffffu;
52 if (x >= 0x7f800000u) // inf or NaN
53 return static_cast<std::uint16_t>(
54 sign | 0x7c00u | (x > 0x7f800000u ? 0x200u : 0u));
55 if (x >= 0x477ff000u) // rounds to >= 65520: overflow
56 return static_cast<std::uint16_t>(sign | 0x7c00u);
57 if (x < 0x38800000u) { // below the smallest normal half: subnormal
58 if (x < 0x33000000u) return static_cast<std::uint16_t>(sign);
59 const std::uint32_t e = x >> 23;
60 const std::uint32_t m = (x & 0x7fffffu) | 0x800000u;
61 const std::uint32_t shift = 126 - e; // 14..24
62 std::uint32_t r = m >> shift;
63 const std::uint32_t rem = m & ((1u << shift) - 1);
64 const std::uint32_t halfway = 1u << (shift - 1);
65 if (rem > halfway || (rem == halfway && (r & 1u))) ++r;
66 return static_cast<std::uint16_t>(sign | r);
67 }
68 std::uint32_t r = ((x >> 13) - (112u << 10));
69 const std::uint32_t rem = x & 0x1fffu;
70 if (rem > 0x1000u || (rem == 0x1000u && (r & 1u))) ++r;
71 return static_cast<std::uint16_t>(sign | r);
72 }
73
74 static float
75 to_float(std::uint16_t h)
76 {
77 const std::uint32_t sign = static_cast<std::uint32_t>(h & 0x8000u)
78 << 16;
79 std::uint32_t e = (h >> 10) & 0x1fu, m = h & 0x3ffu, x;
80 if (e == 0x1f) {
81 x = sign | 0x7f800000u | (m << 13);
82 } else if (e != 0) {
83 x = sign | ((e + 112u) << 23) | (m << 13);
84 } else if (m == 0) {
85 x = sign;
86 } else { // subnormal: normalize
87 e = 113;
88 while (!(m & 0x400u)) {
89 m <<= 1;
90 --e;
91 }
92 x = sign | (e << 23) | ((m & 0x3ffu) << 13);
93 }
94 float f;
95 std::memcpy(&f, &x, 4);
96 return f;
97 }
